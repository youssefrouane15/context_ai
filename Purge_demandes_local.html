1. Base fund-management

FICHIER NOUVEAUÉtape 0 — tables dépendantes
purge-01-dependances.sql
-- Étape 0 — Repérer les tables qui référencent cash_request (FK) : ce sont elles que TRUNCATE … CASCADE videra aussi.
-- Si cette requête ne renvoie rien, vérifie le schéma / le nom de table (onglet Tables de pgAdmin) et adapte les scripts suivants.
SELECT tc.table_schema || '.' || tc.table_name AS table_dependante,
       tc.constraint_name
FROM information_schema.table_constraints tc
JOIN information_schema.constraint_column_usage ccu ON ccu.constraint_name = tc.constraint_name
WHERE tc.constraint_type = 'FOREIGN KEY'
  AND ccu.table_schema = 'retail'
  AND ccu.table_name   = 'cash_request'
ORDER BY 1;
FICHIER NOUVEAUÉtape 1 — purge des demandes
purge-02-demandes.sql
-- Étape 1 — Purge des demandes et de tout ce qui en dépend (opérations, détails devises, dénominations saisies, verrous, décisions).
-- Les référentiels (denomination, currency, agences, zones, transporteurs) ne sont PAS touchés : ils n'ont pas de FK vers cash_request.
-- LOCAL UNIQUEMENT. Back arrêté de préférence (sinon une demande en cours de création peut réapparaître).
BEGIN;

TRUNCATE TABLE retail.cash_request CASCADE;

COMMIT;

-- Contrôle : tout doit être à 0 (ajoute ici les tables listées par l'étape 0 si tu veux les voir une par une)
SELECT (SELECT count(*) FROM retail.cash_request) AS demandes;
FICHIER NOUVEAUÉtape 2 — compteur de référence (optionnel)
purge-03-sequence.sql
-- Étape 2 (optionnel) — Repartir à -01 dans les références (20261005-1422-11 → 20261005-1422-01).
-- Liste les séquences du schéma : celle du compteur de référence s'y trouve si la numérotation est une séquence PG.
SELECT sequence_schema || '.' || sequence_name AS sequence, last_value
FROM information_schema.sequences s
JOIN pg_sequences p ON p.schemaname = s.sequence_schema AND p.sequencename = s.sequence_name
WHERE sequence_schema = 'retail';

-- Puis, pour la séquence trouvée (remplace <nom>) :
-- ALTER SEQUENCE retail.<nom> RESTART WITH 1;
-- Si la numérotation est calculée par max(ref)+1 par agence et par jour (pas de séquence), rien à faire : la purge suffit.
