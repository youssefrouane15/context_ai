// console-gab-hors-site.component.spec.ts — À AJOUTER (dans le describe existant, après les tests de formulaireValide)

  describe('validation ID GAB / numéro de compte (retour QA AER_LM-313)', () => {
    beforeEach(() => {
      component.formulaire = { ...component.formulaire, nom: 'CASA TEST', agenceId: 1, ville: 'CASABLANCA', prixUnitaire: '0' };
    });

    it('accepte un ID GAB de 9 chiffres et un compte de 16 chiffres', () => {
      component.formulaire.idGab = '130120017';
      component.formulaire.numeroCompte = '0112081010030370';
      expect(component.idGabValide).toBeTrue();
      expect(component.numeroCompteValide).toBeTrue();
      expect(component.formulaireValide).toBeTrue();
    });

    it('refuse un ID GAB de 8 chiffres ou avec une lettre', () => {
      component.formulaire.numeroCompte = '0112081010030370';
      component.formulaire.idGab = '13012001';
      expect(component.idGabValide).toBeFalse();
      component.formulaire.idGab = '13012001A';
      expect(component.idGabValide).toBeFalse();
      expect(component.formulaireValide).toBeFalse();
    });

    it('refuse un numéro de compte de 15 ou 17 chiffres', () => {
      component.formulaire.idGab = '130120017';
      component.formulaire.numeroCompte = '011208101003037';
      expect(component.numeroCompteValide).toBeFalse();
      component.formulaire.numeroCompte = '01120810100303701';
      expect(component.numeroCompteValide).toBeFalse();
      expect(component.formulaireValide).toBeFalse();
    });
  });
