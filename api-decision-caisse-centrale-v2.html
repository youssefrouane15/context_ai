1Le design réel — l'API transmet la décision, le worker écrit le statut

Agent Caisse Centrale │ PATCH /cash/{id}/complete-task (corpo : /corpo/cash/{id}/complete-task) │ { "decision": "TRAITEE" } ou { "decision": "ANNULEE", "motifAnnulation": "Stock insuffisant..." } ▼ Controller ──► UseCase.completerTacheCaisseCentrale(id, dto) │ 1. garde locale : demande existe (404) + statut == SUBMITTED (conflit) │ 2. garde métier : ANNULEE ⇒ motif ≥ 10 caractères (règle PO) │ 3. ProcessusWorkflowPort.rechercherTacheActive(businessKey = id) ── vide → conflit │ 4. ProcessusWorkflowPort.completerTache(taskId, { decisionCaisseCentrale, [motifAnnulation] }) ▼ Réponse 200 — la demande est ENCORE SUBMITTED dans la réponse (voir note asynchronisme) …pendant ce temps, côté moteur (existant, rien à coder) : Gateway XOR : ${decisionCaisseCentrale == 'TRAITEE'} ──► gdf-cloturer-demande ──► handler → changerStatut(VALIDATED) ${decisionCaisseCentrale == 'ANNULEE'} ──► gdf-annuler-demande ──► handler → cancel(id) → CANCELLED
Asynchronisme assumé : la réponse HTTP revient dès que la tâche est complétée ; le passage VALIDATED/CANCELLED est fait par le worker quelques centaines de ms plus tard (long polling). C'est exactement ce que tu as observé en Cockpit lors du test bout en bout. Conséquence front : après un « Traiter/Annuler » réussi, on recharge la liste — pas de statut final dans la réponse. Le message de l'ApiResponse le dit explicitement.
Pourquoi pas de saga ici : la saga protège une séquence écriture locale + appel distant. Ici il n'y a aucune écriture locale avant l'appel — si Camunda échoue, rien à compenser, la demande reste SUBMITTED et l'agent réessaie. Un échec moteur remonte en 502 via le handler (section 7).

    import java.util.List;
    import java.util.Optional;

    // --- à ajouter dans la classe, sous demarrerProcessus / correlerMessage / supprimerInstance ---

    /**
     * GET /task?processInstanceBusinessKey={businessKey}&active=true[&tenantIdIn={tenant}]
     * → l'id de la user task active de l'instance, ou empty.
     */
    public Optional<String> rechercherTacheActive(String businessKey) {
        List<?> taches = camundaRestClient.get()
                .uri(uriBuilder -> {
                    uriBuilder.path("/task")
                            .queryParam("processInstanceBusinessKey", businessKey)
                            .queryParam("active", "true");
                    if (properties.getTenantId() != null && !properties.getTenantId().isBlank()) {
                        uriBuilder.queryParam("tenantIdIn", properties.getTenantId());
                    }
                    return uriBuilder.build();
                })
                .retrieve()
                .body(List.class);

        if (taches == null || taches.isEmpty()) {
            return Optional.empty();
        }
        if (taches.size() > 1) {
            log.warn("{} user tasks actives pour businessKey={} — la première est retenue",
                    taches.size(), businessKey);
        }
        Map<?, ?> premiere = (Map<?, ?>) taches.get(0);
        return Optional.ofNullable((String) premiere.get("id"));
    }

    /**
     * POST /task/{taskId}/complete avec variables au format Camunda {value, type}.
     * La gateway en aval lit decisionCaisseCentrale pour router.
     */
    public void completerTache(String taskId, Map<String, Object> variables) {
        camundaRestClient.post()
                .uri("/task/{taskId}/complete", taskId)
                .contentType(MediaType.APPLICATION_JSON)
                .body(Map.of("variables", typerVariables(variables)))
                .retrieve()
                .toBodilessEntity();
        log.info("User task {} complétée ({} variable(s))", taskId, variables.size());
    }

    @Override
    public CashRequestDTO completerTacheCaisseCentrale(UUID id, DecisionCaisseCentraleDTO decision) {
        // Garde 1 — payload
        if (decision == null || decision.getDecision() == null) {
            throw new IllegalArgumentException("La décision (TRAITEE ou ANNULEE) est obligatoire");
        }
        boolean annulation = decision.getDecision() == DecisionCaisseCentrale.ANNULEE;
        if (annulation && (decision.getMotifAnnulation() == null
                || decision.getMotifAnnulation().trim().length() < 10)) {
            throw new IllegalArgumentException(
                "Le motif d'annulation est obligatoire (10 caractères minimum)");
        }

        // Garde 2 — cycle de vie local (la base reste source de vérité, comme la garde de soumission)
        CashRequest demande = cashDemandRepository.getCashRequestById(id);
        if (demande == null) {
            throw new RetailNotFoundException("La demande " + id + " est introuvable");
        }
        if (!RequestStatus.SUBMITTED.name().equals(demande.getRequestStatus())) {
            throw new RetailConflictException(
                "Seule une demande SUBMITTED peut être traitée par la Caisse Centrale (statut actuel : "
                + demande.getRequestStatus() + ")");
        }

        // Garde 3 — le jeton doit être sur la user task (businessKey = UUID, invariant du projet)
        String taskId = processusWorkflowPort.rechercherTacheActive(id.toString())
            .orElseThrow(() -> new RetailConflictException(
                "Aucune tâche de traitement active pour la demande " + id
                + " : le workflow n'a pas (encore) atteint l'étape Caisse Centrale"));

        // Complétion — contrat gateway. AUCUNE écriture locale : le worker écrira le statut.
        Map<String, Object> variables = new HashMap<>();
        variables.put("decisionCaisseCentrale", decision.getDecision().name());
        if (annulation) {
            variables.put("motifAnnulation", decision.getMotifAnnulation().trim());
        }
        processusWorkflowPort.completerTache(taskId, variables);

        return webMapper.toResponseDTO(demande);
    }
