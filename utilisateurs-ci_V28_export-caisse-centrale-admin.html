0. Principe

Le contrôle back est @PreAuthorize("hasAuthority('GDF_EXPORTER_DEMANDE')") sur GET /cash/export, et le front affiche le bouton sur habilitations().exporter (dérivé de la même permission dans /me). Cette permission est portée par le rôle « GDF Export » (AER_LM-288), rattaché aux profils Caisse Centrale et Support. Pour Caisse Centrale Admin on fait pareil : une ligne dans hab_profile_role, pas de nouvelle permission ni de rôle. Résolution par code / name, insert idempotent — convention des scripts V17-V27.

Deux identifiants à vérifier avant de poser le script (requêtes de contrôle ci-dessous) : le code du profil (CAISSE_CENTRALE_ADMIN supposé, V23 d'origine) et le name du rôle d'export (GDF Export supposé). Si l'un diffère, remplace la valeur dans les deux requêtes du V28.

set search_path to qf_admin;

-- 1. Code exact du profil Caisse Centrale Admin et ses rôles actuels
select p.id, p.code, p.name, string_agg(r.name, ', ' order by r.name) as roles
from hab_profiles p
left join hab_profile_role pr on pr.profile_id = p.id
left join hab_roles r on r.id = pr.role_id
where p.name ilike '%caisse centrale%' or p.code ilike '%CAISSE_CENTRALE%'
group by p.id, p.code, p.name order by p.id;

-- 2. Le rôle qui porte l'export (nom exact à reporter dans V28 si ≠ « GDF Export »)
select r.id, r.name, r.platform
from hab_roles r join hab_role_permission rp on rp.role_id = r.id
where rp.permission = 'GDF_EXPORTER_DEMANDE';

-- 3. Après déploiement : /me de l'utilisateur Caisse Centrale Admin doit lister GDF_EXPORTER_DEMANDE
--    (uid à renseigner)
select u.uid, p.code, rp.permission
from users u join hab_profiles p on p.id = u.profile_id
join hab_profile_role pr on pr.profile_id = p.id
join hab_role_permission rp on rp.role_id = pr.role_id
where u.uid = '<uid>' and rp.permission like 'GDF_%' order by 3;

-- V28 — droit « Export Excel » (GDF_EXPORTER_DEMANDE) pour le profil Caisse Centrale Admin.
-- Même mécanisme que la Caisse Centrale et le Support (AER_LM-288) : on rattache le rôle « GDF Export » au profil,
-- on ne duplique pas la permission sur un autre rôle. Idempotent, résolution par code/name (convention du merge develop).
-- >>> À RENSEIGNER : le code du profil tel qu'il est en base (voir requête de contrôle 1 ci-dessous).
insert into hab_profile_role (profile_id, role_id)
select p.id, r.id
from hab_profiles p
join hab_roles r on r.platform = 'GESTION_FONDS' and r.name = 'GDF Export'
where p.code = 'CAISSE_CENTRALE_ADMIN'
  and not exists (select 1 from hab_profile_role x where x.profile_id = p.id and x.role_id = r.id);

-- Contrôle : le profil porte désormais GDF_EXPORTER_DEMANDE (1 ligne attendue)
select p.code, r.name, rp.permission
from hab_profiles p
join hab_profile_role pr on pr.profile_id = p.id
join hab_roles r on r.id = pr.role_id
join hab_role_permission rp on rp.role_id = r.id
where p.code = 'CAISSE_CENTRALE_ADMIN' and rp.permission = 'GDF_EXPORTER_DEMANDE';
