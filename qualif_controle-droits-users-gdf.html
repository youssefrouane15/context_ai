1. Vue d'ensemble — une ligne par utilisateur

FICHIER NOUVEAUDroits GDF de chaque utilisateur de la liste
pgAdmin · base utilisateurs-ci qualif · schéma qf_admin
SELECT u.uid, u.actif, p.code AS profil,
       a.type AS type_orga, a.code AS agence, a.name AS agence_nom,
       string_agg(DISTINCT r.name, ', ')       FILTER (WHERE rp.permission LIKE 'GDF_%') AS roles_gdf,
       string_agg(DISTINCT rp.permission, ', ') FILTER (WHERE rp.permission LIKE 'GDF_%') AS permissions_gdf,
       bool_or(rp.permission = 'GDF_CREER_DEMANDE')     AS peut_creer,
       bool_or(rp.permission = 'GDF_CONSULTER_DEMANDE') AS peut_consulter,
       bool_or(rp.permission = 'GDF_TRAITER_DEMANDE')   AS peut_traiter,
       bool_or(rp.permission = 'GDF_TOUTES_AGENCES')    AS toutes_agences
FROM qf_admin.users u
LEFT JOIN qf_admin.hab_profiles p         ON p.id = u.profile_id
LEFT JOIN qf_admin.orga_unit a            ON a.id = u.organization_unit_id
LEFT JOIN qf_admin.hab_profile_role pr    ON pr.profile_id = p.id
LEFT JOIN qf_admin.hab_roles r            ON r.id = pr.role_id
LEFT JOIN qf_admin.hab_role_permission rp ON rp.role_id = r.id
WHERE u.uid IN ('a83378','408542','408493','a09227','338872','980599','795377','234631')   -- colle ici les uid de la liste
GROUP BY u.uid, u.actif, p.code, a.type, a.code, a.name
ORDER BY p.code, u.uid;

2. Détail d'un utilisateur — rôle par rôle

FICHIER NOUVEAUChaque rôle et chaque permission d'un uid
pgAdmin · base utilisateurs-ci qualif · schéma qf_admin
SELECT u.uid, p.code AS profil, r.id AS role_id, r.name AS role, r.platform, rp.permission
FROM qf_admin.users u
JOIN qf_admin.hab_profiles p              ON p.id = u.profile_id
JOIN qf_admin.hab_profile_role pr         ON pr.profile_id = p.id
JOIN qf_admin.hab_roles r                 ON r.id = pr.role_id
LEFT JOIN qf_admin.hab_role_permission rp ON rp.role_id = r.id
WHERE u.uid = 'a83378'                     -- uid à contrôler
ORDER BY r.platform, r.name, rp.permission;

Sert à la Dérogation ; pour la Gestion des fonds seules l'agence et la zone comptent. Colonne vide = rattachement non fait à l'injection.

SELECT u.uid, a.code AS agence, a.name AS agence_nom,
       g.name AS groupe, dg.uid AS directeur_groupe, z.name AS zone
FROM qf_admin.users u
LEFT JOIN qf_admin.orga_unit a  ON a.id = u.organization_unit_id
LEFT JOIN qf_admin.orga_unit g  ON g.id = a.parent_id
LEFT JOIN qf_admin.users dg     ON dg.id = g.manager_id
LEFT JOIN qf_admin.orga_unit z  ON z.id = g.parent_id
WHERE u.uid IN ('a83378','408542','408493','a09227','338872','980599','795377','234631')
ORDER BY u.uid;
