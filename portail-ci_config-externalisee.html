FICHIER NOUVEAUconfiguration.ts
portail-ci\src\app\core\config\configuration.ts

Même mécanique que fund-management-front (fonction pure, fetch, fusion dans environment). Le portail est l'hôte : document.baseURI suffit pour l'URL.

import { environment } from '../../../environments/environment';

/**
 * Fichier de configuration d'environnement, écrasé par le CD de l'intégrateur à chaque déploiement
 * (/var/www/html/config/app-config.json) : une seule image portail pour tous les environnements.
 * Copié dans dist/portail-ci/config par angular.json (assets) ; en local = src/config/app-config.json.
 * Remplace l'ancien assets/config.json (lu mais jamais appliqué).
 */
export const FICHIER_CONFIGURATION = 'config/app-config.json';

let chargement: Promise<void> | null = null;

/**
 * Charge config/app-config.json et fusionne ses clés dans `environment` AVANT tout appel HTTP.
 * fetch() plutôt que HttpClient : rien ne doit passer par les intercepteurs (qui lisent eux-mêmes `environment`).
 * Toute clé du fichier surcharge la valeur du build (apiUrl, fundManagementUrl aujourd'hui ; mockAuthEnabled, useHash…
 * demain si le CD en a besoin) — sauf `production`, décidée par le build.
 * - fichier absent en dev : valeurs d'environment.ts conservées (avertissement console) ;
 * - fichier absent en prod : erreur bloquante, le portail ne doit pas démarrer sur des URL vides.
 * Idempotent : un seul chargement quel que soit le nombre d'appelants.
 */
export function chargerConfiguration(): Promise<void> {
  return chargement ??= charger();
}

async function charger(): Promise<void> {
  const url = new URL(FICHIER_CONFIGURATION, document.baseURI).toString();
  let reponse: Response;
  try {
    reponse = await fetch(url, { cache: 'no-store' });
  } catch (erreur) {
    return signalerAbsence(`${url} injoignable (${erreur})`);
  }
  if (!reponse.ok) {
    return signalerAbsence(`${url} : HTTP ${reponse.status}`);
  }
  const { production: _ignore, ...valeurs } = (await reponse.json()) as Record<string, unknown>;
  Object.assign(environment, valeurs);
}

function signalerAbsence(detail: string): void {
  if (environment.production) {
    throw new Error(`Configuration manquante : ${detail}`);
  }
  console.warn(`[config] ${detail} — valeurs par défaut d'environment.ts conservées.`);
}

/** Remise à zéro pour les tests unitaires. */
export function reinitialiserConfiguration(): void {
  chargement = null;
}
FICHIER NOUVEAUconfiguration.spec.ts
portail-ci\src\app\core\config\configuration.spec.ts
import { environment } from '../../../environments/environment';
import { chargerConfiguration, reinitialiserConfiguration } from './configuration';

describe('chargerConfiguration', () => {
  const apiUrlInitiale = environment.apiUrl;

  beforeEach(() => reinitialiserConfiguration());
  afterEach(() => { (environment as any).apiUrl = apiUrlInitiale; });

  it('surcharge environment avec les clés du fichier', async () => {
    spyOn(window, 'fetch').and.resolveTo(new Response(JSON.stringify({ apiUrl: 'https://gw.recette' }), { status: 200 }));

    await chargerConfiguration();

    expect(environment.apiUrl).toBe('https://gw.recette');
    expect((window.fetch as jasmine.Spy).calls.mostRecent().args[0]).toContain('config/app-config.json');
  });

  it('ne charge qu\'une fois', async () => {
    spyOn(window, 'fetch').and.resolveTo(new Response('{}', { status: 200 }));

    await Promise.all([chargerConfiguration(), chargerConfiguration()]);

    expect(window.fetch).toHaveBeenCalledTimes(1);
  });

  it('conserve environment.ts hors production quand le fichier manque', async () => {
    spyOn(window, 'fetch').and.resolveTo(new Response('', { status: 404 }));
    spyOn(console, 'warn');

    await chargerConfiguration();

    expect(environment.apiUrl).toBe(apiUrlInitiale);
    expect(console.warn).toHaveBeenCalled();
  });
});
