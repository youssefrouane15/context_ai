-- 0. Contrôle : ces adresses ne doivent être portées par AUCUN autre utilisateur (email unique) → doit renvoyer 0 ligne

                       'hamza.marouane@externe.bnpparibas.com',

2. Affectation

FICHIER NOUVEAUÉtapes 1 à 3 — sauvegarde, mise à jour, contrôle
recette\emails-test-notifications.sql

L'étape 3 doit afficher 3 lignes avec les nouvelles adresses.

SET search_path TO qf_admin;

-- 1. Sauvegarde des emails d'origine (ne s'écrase pas si tu relances)
CREATE TABLE IF NOT EXISTS recette_users_email_backup AS
SELECT uid, email FROM users WHERE uid IN ('j51204', 'j53488', 'c60010');

-- 2. Affectation des adresses de test
UPDATE users u
SET email = d.email
FROM (values
    ('j51204', 'youssef.rouane@externe.bnpparibas.com'),   -- DA (1er de la liste)
    ('j53488', 'hamza.marouane@externe.bnpparibas.com'),   -- DA (3e de la liste)
    ('c60010', 'omar.ghandour@externe.bnpparibas.com')     -- Caisse Centrale
) AS d (uid, email)
WHERE u.uid = d.uid;

-- 3. Contrôle : 3 lignes avec profil et agence
SELECT u.uid, u.email, p.code AS profil, o.code AS agence
FROM users u
LEFT JOIN hab_profiles p ON p.id = u.profile_id
LEFT JOIN orga_unit o ON o.id = u.organization_unit_id
WHERE u.uid IN ('j51204', 'j53488', 'c60010');

3. Scénarios

Soumission / mise à jour / annulation : le mail part au DA de l'agence de la demande → créer la demande avec un utilisateur de l'agence de j51204 (mail à Youssef) ou de l'agence de j53488 (mail à Hamza).
Déclassement d'urgence : DA + chefs de caisse de l'agence → mêmes agences.
Demande urgente : le mail Caisse Centrale part à la variable QF_NOTIF_CAISSE_CENTRALE, pas à c60010 → y mettre omar.ghandour@externe.bnpparibas.com pour qu'Omar le reçoive.

4. Restauration

SET search_path TO qf_admin;

-- 4. Après les tests : remettre les adresses d'origine
