import org.springframework.stereotype.Service;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.util.List;

/**
 * AER_LM-357 — génère le fichier Excel des GAB hors site : mêmes informations que l'écran de la console,
 * plus le numéro de compte.
 */
@Service
public class GabHorsSiteExportServiceImpl implements GabHorsSiteExportService {

    static final String FEUILLE = "GAB Hors Site";
    static final List<String> EN_TETES = List.of("ID GAB", "Nom du GAB", "Numéro de compte", "Code agence", "Agence",
            "Ville", "Transporteur", "Prix unitaire (MAD)", "Type", "État");

    /** Largeur fixe (en 1/256 de caractère) : pas d'autoSizeColumn, qui exige des polices sur le serveur. */
    private static final int LARGEUR_COLONNE = 24 * 256;

    @Override
    public byte[] exporter(List<GabHorsSiteDTO> gabs) {
        try (Workbook classeur = new XSSFWorkbook(); ByteArrayOutputStream sortie = new ByteArrayOutputStream()) {
            Sheet feuille = classeur.createSheet(FEUILLE);
            ecrireEnTetes(classeur, feuille);
            int numeroLigne = 1;
            for (GabHorsSiteDTO gab : gabs) {
                ecrireLigne(feuille.createRow(numeroLigne++), gab);
            }
            classeur.write(sortie);
            return sortie.toByteArray();
        } catch (IOException e) {
            throw new IllegalStateException("L'extraction Excel des GAB hors site a échoué", e);
        }
    }

    private void ecrireEnTetes(Workbook classeur, Sheet feuille) {
        Font gras = classeur.createFont();
        gras.setBold(true);
        CellStyle style = classeur.createCellStyle();
        style.setFont(gras);

        Row ligne = feuille.createRow(0);
        for (int colonne = 0; colonne < EN_TETES.size(); colonne++) {
            Cell cellule = ligne.createCell(colonne);
            cellule.setCellValue(EN_TETES.get(colonne));
            cellule.setCellStyle(style);
            feuille.setColumnWidth(colonne, LARGEUR_COLONNE);
        }
    }

    private void ecrireLigne(Row ligne, GabHorsSiteDTO gab) {
        ligne.createCell(0).setCellValue(texte(gab.getIdGab()));
        ligne.createCell(1).setCellValue(texte(gab.getNom()));
        ligne.createCell(2).setCellValue(texte(gab.getNumeroCompte()));
        ligne.createCell(3).setCellValue(gab.getAgence() == null ? "" : texte(gab.getAgence().getCode()));
        ligne.createCell(4).setCellValue(gab.getAgence() == null ? "" : texte(gab.getAgence().getNom()));
        ligne.createCell(5).setCellValue(gab.getVille() == null ? "" : texte(gab.getVille().getNom()));
        ligne.createCell(6).setCellValue(gab.getTransporteur() == null ? "" : texte(gab.getTransporteur().getNom()));
        if (gab.getPrixUnitaire() != null) {
            ligne.createCell(7).setCellValue(gab.getPrixUnitaire().doubleValue());
        }
        ligne.createCell(8).setCellValue(gab.getType() == null ? "" : gab.getType().name());
        ligne.createCell(9).setCellValue(libelleEtat(gab.getEtat()));
    }

    /** Mêmes libellés que la colonne « État » de l'écran. */
    private static String libelleEtat(EtatGab etat) {
        if (etat == null) {
            return "";
        }
        return etat == EtatGab.EN_LIGNE ? "En ligne" : "Hors ligne";
    }

    private static String texte(String valeur) {
        return valeur == null ? "" : valeur;
    }
}
