/**
 * AER_LM-285 / 287 — en-tête commun des écrans de consultation (agence) et de traitement (Caisse Centrale),
 * au pixel des maquettes Agence_Modification-* et Caisse-Centrale_Traiter-* :
 * lien de retour, bannière « Consultation seule », carte avec pastille de statut + « Demande Réf: … » + « Soumise le … par … (…) »,
 * et à droite le bouton-badge « Mode Modification » (toujours visible quand la demande est modifiable ; « actif » une fois dedans)
 * et le badge / la case « DEMANDE URGENTE ». Composant de présentation pur.

@Component({
  selector: 'app-demande-entete',
  standalone: true,
  imports: [CommonModule, MatIconModule],
  templateUrl: './demande-entete.component.html',
  styleUrl: './demande-entete.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DemandeEnteteComponent {

  @Input({ required: true }) reference!: string;
  @Input({ required: true }) statut!: StatutDemande;

  @Input() urgent = false;
  @Input() urgentModifiable = false;
  @Output() urgentChange = new EventEmitter<boolean>();

  /** Modification en cours (route modification/:id) : le badge affiche « Mode Modification actif ». */
  @Input() modeModification = false;
  /** Consultation d'une demande modifiable : le badge « Mode Modification » est cliquable et fait entrer en modification. */
  @Input() modificationActivable = false;
  @Output() activerModification = new EventEmitter<void>();

  @Input() soumiseLe: string | null = null;
  @Input() soumisePar: string | null = null;
  @Input() soumiseParDetail: string | null = null;

  @Input() decisionLe: string | null = null;
  @Input() decisionPar: string | null = null;
  @Input() decisionParDetail: string | null = null;
  @Input() motifAnnulation: string | null = null;

  @Input() banniere: BanniereLecture | null = null;

  @Input() lienRetour = 'Retour à l\'Historique des Demandes';
  @Output() retour = new EventEmitter<void>();

  /** En modification, la pastille reste « EN ATTENTE » (maquette) : le verrou IN_MODIFICATION est un état technique, pas un statut métier pour l'agence. */
  get statutAffiche(): StatutDemande {
    return this.modeModification && this.statut === 'EN_MODIFICATION' ? 'SOUMISE' : this.statut;
  }
  get libelleStatut(): string { return libelleStatut(this.statutAffiche); }
  get classeStatut(): string { return classeStatut(this.statutAffiche); }
  get iconeStatut(): string {
    switch (this.statutAffiche) {
      case 'SOUMISE':         return 'pending';
      case 'TRAITEE':         return 'check_circle';
      case 'EN_MODIFICATION': return 'edit';
      case 'INITIEE':         return 'edit_note';
      default:                return 'cancel';
    }
  }
  get libelleDecision(): string { return this.statut === 'TRAITEE' ? 'Traitée' : 'Annulée'; }
  get badgeModeVisible(): boolean { return this.modeModification || this.modificationActivable; }

  basculerUrgent(evt: Event): void {
    this.urgentChange.emit((evt.target as HTMLInputElement).checked);
  }
}

/** Ajoute à 0 les valeurs faciales du référentiel absentes de la demande, pour une grille complète ; tri décroissant. */
function completer(lignes: LigneCoupure[], valeurs: number[] | undefined): LigneCoupure[] {
  if (!valeurs?.length) { return lignes; }
  const presentes = new Set(lignes.map(l => l.faceValue));
  const completes = [...lignes, ...valeurs.filter(v => !presentes.has(v)).map(v => ({ code: '', faceValue: v, quantite: 0, montant: 0 }))];
  return completes.sort((a, b) => b.faceValue - a.faceValue);
}

/** Valeurs faciales d'une liste du référentiel, quel que soit le type exact de ses éléments (faceValue numérique ou texte). */
export function valeursFaciales(liste: ReadonlyArray<{ faceValue?: number | string | null }> | null | undefined): number[] {
  return (liste ?? []).map(c => Number(c.faceValue)).filter(v => !Number.isNaN(v) && v > 0);
}

/** Coupures d'une alimentation (CASH_DESK ou ATM, sens SUPPLY), cumulées par valeur faciale, triées par valeur décroissante. */
function coupures(ops: OperationDossier[], type: 'CASH_DESK' | 'ATM'): LigneCoupure[] {
  const parValeur = new Map<number, LigneCoupure>();
  for (const op of ops) {
    if (op.operationType !== type || op.operationSens !== 'SUPPLY') { continue; }
    for (const detail of op.currencyDetails ?? []) {
      for (const ligne of detail.operationDenominations ?? []) {
        const code = ligne.denomination?.denominationCode ?? '';
        const faceValue = valeurFaciale(ligne.denomination?.faceValue, code);
        if (faceValue === null) { continue; }
        const quantite = Number(ligne.quantity ?? 0) || 0;
        const existante = parValeur.get(faceValue);
        if (existante) {
          existante.quantite += quantite;
          existante.montant = arrondi(existante.quantite * faceValue);
        } else {
          parValeur.set(faceValue, { code, faceValue, quantite, montant: arrondi(quantite * faceValue) });
        }
      }
    }
  }
  return [...parValeur.values()].sort((a, b) => b.faceValue - a.faceValue);
}

/** Valeur faciale : celle du back si elle est fournie, sinon lue dans le code (BILLET_200 → 200, PIECE_0_50 → 0.5). */
export function valeurFaciale(faceValue: number | string | null | undefined, code: string): number | null {
  const fournie = Number(faceValue);
  if (faceValue !== null && faceValue !== undefined && faceValue !== '' && !Number.isNaN(fournie)) { return fournie; }
  const m = /(\d+)(?:[._](\d+))?\s*$/.exec(code);
  if (!m) { return null; }
  return Number(m[2] ? `${m[1]}.${m[2]}` : m[1]);
}

/** Ramassages (CURRENCY / COLLECTION) ou commandes de devises (CURRENCY / SUPPLY) : une ligne par devise. */

  const lignes: { devise: string; montant: number }[] = [];

    if (op.operationType !== 'CURRENCY' || op.operationSens !== sens) { continue; }
    for (const detail of op.currencyDetails ?? []) {
      lignes.push({ devise: detail.currency?.isoCode ?? '?', montant: Number(detail.totalAmountByCurrency ?? 0) || 0 });
    }
  }
  return lignes;
}

    if (!this.peutDecider()) { return; }
    this.actions.acquitterErreur();
    this.decision.set(type);
  }

  fermer(): void { this.decision.set(null); }

  async confirmer(motif: string | undefined): Promise<void> {
    const v = this.vue();
    const type = this.decision();
    if (!v || !type) { return; }
    const ok = await this.actions.decider(v.id, type, motif);
    if (ok) {
      this.decision.set(null);
      this.snackBar.open(
        type === 'TRAITEE'
          ? `Demande ${v.reference} traitée : le statut sera mis à jour dans quelques secondes.`
          : `Demande ${v.reference} annulée : motif consigné.`,
        'Fermer', { duration: 5000 });
      setTimeout(() => void this.charger(), 1500);
    } else {
      this.snackBar.open(this.actions.erreur() ?? 'Erreur lors de la décision.', 'Fermer', { duration: 6000 });
    }
