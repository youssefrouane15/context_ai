import com.fasterxml.jackson.annotation.JsonInclude;
import lombok.AllArgsConstructor;
import lombok.Getter;
import lombok.NoArgsConstructor;

/**
 * Enveloppe standard de toute réponse HTTP du microservice (retail et corpo),
 * alignée sur le format du socle QlickFlow : { success, message, data, error }.
 */
@Getter
@NoArgsConstructor
@AllArgsConstructor
@JsonInclude(JsonInclude.Include.NON_NULL)
public class ApiResponse<T> {

    private Boolean success;
    private String message;
    private T data;
    private String error;

    public static <T> ApiResponse<T> success(T data) {
        return new ApiResponse<>(true, null, data, null);
    }

    public static <T> ApiResponse<T> success(T data, String message) {
        return new ApiResponse<>(true, message, data, null);
    }

    public static <T> ApiResponse<T> failure(String message, String error) {
        return new ApiResponse<>(false, message, null, error);
    }
}

3cancelOperations retail — analyse et design

Ce que fait la version d'Othmane (ta capture) : charge la demande, marque CANCELLED les opérations demandées, refuse (RuntimeException) si l'une est déjà annulée, et si toutes les opérations finissent annulées, annule la demande elle-même. La règle est bonne — c'est l'écriture qui date d'avant le refactoring. Trois écarts avec ta branche, et comment on les résout :

Chez OthmaneSur ta branche refactorée
Règle métier codée dans l'adapter JPARègle dans l'agrégat CashRequest (comme addOperation/validateRequest) — l'adapter ne fait que charger et persister
RuntimeException partout → 500Exceptions typées : demande inconnue → RetailNotFoundException (404), opération déjà annulée ou inconnue → RetailConflictException (409)
Corps de requête = un CashRequestDTO entier pour n'en lire que les ids d'opérationsUn DTO dédié CancelOperationsRequestDTO { operationIds } — le front n'envoie que ce qui compte, et @Valid refuse une liste vide en 400
Le contrat REST retenu : PATCH /cash/{id}/operations/cancel, corps { "operationIds": ["uuid1", "uuid2"] }, réponse 200 { success, data: la demande mise à jour }. Même verbe et même URL qu'Othmane (le front qu'il a peut-être commencé ne change pas d'adresse), seul le corps se simplifie.

5Étape B — La règle métier dans l'agrégat

À ajouter dans domain-core\…\CashRequest.java (retail), à côté de addOperation. Le domaine lève des exceptions du domaine (IllegalStateException/IllegalArgumentException, comme ses règles existantes) ; c'est le use case qui les traduira en exceptions typées HTTP :

domain-core\…\CashRequest.java — AJOUTER la méthodeMODIFIER
/**
 * Annule les opérations désignées. Si toutes les opérations de la demande
 * sont alors annulées, la demande passe elle-même à CANCELLED.
 * @throws IllegalArgumentException si un id ne correspond à aucune opération
 * @throws IllegalStateException    si une opération visée est déjà annulée
 *                                  ou si la demande est déjà annulée
 */
public void cancelOperations(Set<UUID> operationIds) {
    if (RequestStatus.CANCELLED.equals(this.requestStatus)) {
        throw new IllegalStateException("La demande est déjà annulée");
    }
    for (UUID operationId : operationIds) {
        CashOperation operation = this.operations.stream()
                .filter(op -> op.getId() != null && operationId.equals(op.getId().getValue()))
                .findFirst()
                .orElseThrow(() -> new IllegalArgumentException(
                        "Opération inconnue dans cette demande : " + operationId));
        if (OperationStatus.CANCELLED.equals(operation.getOperationStatus())) {
            throw new IllegalStateException("Opération déjà annulée : " + operationId);
        }
        operation.setOperationStatus(OperationStatus.CANCELLED);
    }
    boolean allCancelled = this.operations.stream()
            .allMatch(op -> OperationStatus.CANCELLED.equals(op.getOperationStatus()));
    if (allCancelled) {
        this.requestStatus = RequestStatus.CANCELLED;
    }
}

Imports à ajouter dans l'agrégat s'ils manquent : java.util.Set, java.util.UUID, et tes enums RequestStatus/OperationStatus.

Pourquoi vérifier « déjà annulée » avant de toucher quoi que ce soit : la boucle modifie les opérations une par une ; si la 3ᵉ échoue, les 2 premières sont déjà passées à CANCELLED en mémoire. Ce n'est pas grave (le use case ne persistera pas — l'exception remonte avant le save), mais la garde en tête de méthode rend l'intention explicite : tout ou rien.

@ExceptionHandler(CorpoNotFoundException.class)
public ResponseEntity<ApiResponse<Void>> handleNotFound(CorpoNotFoundException ex) {
    return ResponseEntity.status(HttpStatus.NOT_FOUND)
            .body(ApiResponse.failure(message("corpo.error.notfound", ex.getMessage()), "NOT_FOUND"));
}

@ExceptionHandler(CorpoConflictException.class)
public ResponseEntity<ApiResponse<Void>> handleConflict(CorpoConflictException ex) {
    return ResponseEntity.status(HttpStatus.CONFLICT)
            .body(ApiResponse.failure(message("corpo.error.conflict", ex.getMessage()), "CONFLICT"));
}

@ExceptionHandler(CorpoValidationException.class)
public ResponseEntity<ApiResponse<Void>> handleValidation(CorpoValidationException ex) {
    return ResponseEntity.status(HttpStatus.UNPROCESSABLE_ENTITY)
            .body(ApiResponse.failure(message("corpo.error.validation", ex.getMessage()), "VALIDATION_ERROR"));
}

@ExceptionHandler(MethodArgumentNotValidException.class)
public ResponseEntity<ApiResponse<Void>> handleBadRequest(MethodArgumentNotValidException ex) {
    String detail = ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + " : " + e.getDefaultMessage())
            .collect(Collectors.joining("; "));
    return ResponseEntity.badRequest()
            .body(ApiResponse.failure(message("corpo.error.badrequest", detail), "BAD_REQUEST"));
}

@ExceptionHandler(Exception.class)
public ResponseEntity<ApiResponse<Void>> handleTechnical(Exception ex) {
    log.error("Erreur technique non gérée (corpo)", ex);
    return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
            .body(ApiResponse.failure(message("corpo.error.technical"), "TECHNICAL_ERROR"));
}

class CashRequestCancelOperationsTest {

    private CashRequest demandeAvecDeuxOperations(UUID id1, UUID id2) {
        CashOperation op1 = new CashOperation();
        op1.setId(new CashOperationId(id1));
        op1.setOperationStatus(OperationStatus.INITIATED);

        CashRequest demande = new CashRequest();
        demande.setRequestStatus(RequestStatus.INITIATED);
        demande.setOperations(new ArrayList<>(List.of(op1, op2)));

    @Test
    void annuleUneOperationSansToucherLaDemande() {
        UUID id1 = UUID.randomUUID(), id2 = UUID.randomUUID();
        CashRequest demande = demandeAvecDeuxOperations(id1, id2);

        demande.cancelOperations(Set.of(id1));

        assertEquals(OperationStatus.CANCELLED, demande.getOperations().get(0).getOperationStatus());
        assertEquals(OperationStatus.INITIATED, demande.getOperations().get(1).getOperationStatus());
        assertEquals(RequestStatus.INITIATED, demande.getRequestStatus());
    }

    @Test
    void annuleLaDemandeQuandToutesLesOperationsSontAnnulees() {
        UUID id1 = UUID.randomUUID(), id2 = UUID.randomUUID();
        CashRequest demande = demandeAvecDeuxOperations(id1, id2);

        demande.cancelOperations(Set.of(id1, id2));

        assertEquals(RequestStatus.CANCELLED, demande.getRequestStatus());
    }

    @Test
    void refuseUneOperationDejaAnnulee() {
        UUID id1 = UUID.randomUUID(), id2 = UUID.randomUUID();
        CashRequest demande = demandeAvecDeuxOperations(id1, id2);
        demande.getOperations().get(0).setOperationStatus(OperationStatus.CANCELLED);
