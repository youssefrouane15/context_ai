/**
 * Test du use case corpo — sur les signatures REELLES de ton code :
 * CorpoDemandRepository.save/getById/findAll/update/cancelById et
 * CorpoDenominationRepository.findFaceValuesByCode() -> Map<String, BigDecimal>.

@ExtendWith(MockitoExtension.class)
class CorpoDemandUseCaseImplTest {

    @Mock private CorpoDemandRepository corpoDemandRepository;
    @Mock private CorpoDenominationRepository corpoDenominationRepository;
    @Mock private CorpoCashRequestWebMapper webMapper;
    @InjectMocks private CorpoDemandUseCaseImpl useCase;

    // ==================== helpers (construction pure, ZERO stub) ====================

    private CorpoCashRequest validDomain() {
        CorpoCashOperation operation = new CorpoCashOperation();
        // Si ta validation corpo exige le sens, pose-le ici avec TON enum
        // (celui du champ operationSens de CorpoCashOperation), ex :
        // operation.setOperationSens(CorpoOperationSens.RAMASSAGE);
        CorpoCashRequest domain = new CorpoCashRequest();
        domain.setUidUser("j43709");
        domain.setOperations(new ArrayList<>(List.of(operation)));
        return domain;
    }

    /** Le referentiel corpo est deja une map code -> valeur faciale : stub direct. */
    private void stubReferential() {
        Mockito.when(corpoDenominationRepository.findFaceValuesByCode())
                .thenReturn(Map.of("BILLET_200", new BigDecimal("200")));
    }

    // ==================== chemins de REJET ====================

    @Test
    void createShouldRejectInvalidRequestWithoutTouchingRepository() {
        stubReferential();
        Mockito.when(webMapper.toDomain(Mockito.any(CorpoCashRequestDTO.class)))
                .thenReturn(new CorpoCashRequest());          // sans operations => rejet

        assertThrows(CorpoValidationException.class, () -> useCase.create(new CorpoCashRequestDTO()));
        Mockito.verify(corpoDemandRepository, Mockito.never()).save(Mockito.any());
    }

    @Test
    void updateShouldRevalidateBeforeDelegating() {
        stubReferential();
        Mockito.when(webMapper.toDomain(Mockito.any(CorpoCashRequestDTO.class)))
                .thenReturn(new CorpoCashRequest());

        assertThrows(CorpoValidationException.class,
                () -> useCase.update(UUID.randomUUID(), new CorpoCashRequestDTO()));
        Mockito.verify(corpoDemandRepository, Mockito.never()).update(Mockito.any(), Mockito.any());
    }

    // ==================== chemins HEUREUX ====================

    @Test
    void createShouldValidateBeforeSaving() {
        stubReferential();
        CorpoCashRequest domain = validDomain();
        Mockito.when(webMapper.toDomain(Mockito.any(CorpoCashRequestDTO.class))).thenReturn(domain);
        Mockito.when(corpoDemandRepository.save(Mockito.any())).thenReturn(domain);
        Mockito.when(webMapper.toResponseDTO(domain)).thenReturn(new CorpoCashRequestDTO());

        useCase.create(new CorpoCashRequestDTO());

        // Preuve indirecte de l'ordre : le domaine capture au save porte les
        // defauts serveur que SEULE la validation pose.
        ArgumentCaptor<CorpoCashRequest> saved = ArgumentCaptor.forClass(CorpoCashRequest.class);
        Mockito.verify(corpoDemandRepository).save(saved.capture());
        CorpoCashOperation operation = saved.getValue().getOperations().get(0);
        assertEquals("INITIATED", operation.getOperationStatus());
        assertEquals(1, operation.getInstitutionCode());
        assertEquals("001", operation.getCashDeskNumber());
    }

    @Test
    void updateShouldValidateThenDelegateWithSameId() {
        stubReferential();
        UUID id = UUID.randomUUID();
        CorpoCashRequest domain = validDomain();
        Mockito.when(webMapper.toDomain(Mockito.any(CorpoCashRequestDTO.class))).thenReturn(domain);
        Mockito.when(corpoDemandRepository.update(id, domain)).thenReturn(domain);
        Mockito.when(webMapper.toResponseDTO(domain)).thenReturn(new CorpoCashRequestDTO());

        useCase.update(id, new CorpoCashRequestDTO());

        Mockito.verify(corpoDemandRepository).update(id, domain);
    }

    @Test
    void getByIdShouldDelegateAndMapBack() {
        UUID id = UUID.randomUUID();
        CorpoCashRequest domain = validDomain();
        CorpoCashRequestDTO expected = new CorpoCashRequestDTO();
        Mockito.when(corpoDemandRepository.getById(id)).thenReturn(domain);   // getById : TA signature
        Mockito.when(webMapper.toResponseDTO(domain)).thenReturn(expected);

        assertSame(expected, useCase.getById(id));
    }

    @Test
    void getAllShouldMapEachElement() {
        Mockito.when(corpoDemandRepository.findAll()).thenReturn(List.of(validDomain(), validDomain()));
        Mockito.when(webMapper.toResponseDTO(Mockito.any())).thenReturn(new CorpoCashRequestDTO());

        assertEquals(2, useCase.getAll().size());
        Mockito.verify(webMapper, Mockito.times(2)).toResponseDTO(Mockito.any());
    }

    @Test
    void cancelShouldDelegateAndMapBack() {
        UUID id = UUID.randomUUID();
        CorpoCashRequest domain = validDomain();
        Mockito.when(corpoDemandRepository.cancelById(id)).thenReturn(domain);
        Mockito.when(webMapper.toResponseDTO(domain)).thenReturn(new CorpoCashRequestDTO());

        assertNotNull(useCase.cancel(id));
        Mockito.verify(corpoDemandRepository).cancelById(id);
    }
}

/**
 * Version corrigee sur TON implementation reelle : constructeur a UN SEUL
 * argument (pas de mapper — l'adapter corpo construit directement la map
 * code -> valeur faciale, plus direct que la version retail).

class CorpoDenominationRepositoryImplTest {

    private final CorpoDenominationJpaRepository jpaRepository =
            Mockito.mock(CorpoDenominationJpaRepository.class);
    private final CorpoDenominationRepositoryImpl repository =
            new CorpoDenominationRepositoryImpl(jpaRepository);

    @Test
    void shouldBuildFaceValueMapKeyedByStableCode() {
        CorpoDenominationEntity billet200 = new CorpoDenominationEntity();
        billet200.setDenominationCode("BILLET_200");
        billet200.setFaceValue(new BigDecimal("200.00"));
        CorpoDenominationEntity centime50 = new CorpoDenominationEntity();
        centime50.setDenominationCode("CENTIME_50");
        centime50.setFaceValue(new BigDecimal("0.50"));
        Mockito.when(jpaRepository.findAll()).thenReturn(List.of(billet200, centime50));

        Map<String, BigDecimal> result = repository.findFaceValuesByCode();

        assertEquals(2, result.size());
        assertEquals(0, new BigDecimal("200.00").compareTo(result.get("BILLET_200")));
        assertEquals(0, new BigDecimal("0.50").compareTo(result.get("CENTIME_50")));
    }
}
