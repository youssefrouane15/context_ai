package com.bmci.workflow.fund.management.application.rest.security;

import com.bmci.workflow.fund.management.service.domain.retail.contexte.ContexteAppelant;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.stereotype.Component;

import java.time.Instant;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.stream.Stream;

/**
 * uid (JWT) → IdentiteAppelant, comme QfJwtAuthenticationConverter côté users : les permissions viennent de la base
 * users (profil → rôles → permissions), jamais du jeton. Cache mémoire par uid (qf.users.cache-duree, 5 min).
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class IdentiteAppelantResolver {

    static final String PERMISSION_TOUTES_AGENCES = "GDF_TOUTES_AGENCES";
    static final String TYPE_AGENCE = "AGENCE";

    private final UtilisateursInterneClient utilisateursInterneClient;
    private final QfUsersProperties properties;
    private final Map<String, Entree> cache = new ConcurrentHashMap<>();

    /** Vide si l'utilisateur est inconnu ou désactivé (→ 401, même règle que users). */
    public Optional<IdentiteAppelant> resoudre(String uid) {
        Entree entree = cache.get(uid);
        if (entree != null && entree.expiration().isAfter(Instant.now())) {
            return Optional.of(entree.identite());
        }
        Optional<IdentiteAppelant> identite = utilisateursInterneClient.trouverParUid(uid)
                .filter(UtilisateurInterneDTO::enabled)
                .map(this::versIdentite);
        identite.ifPresent(i -> cache.put(uid, new Entree(i, Instant.now().plus(properties.cacheDuree()))));
        return identite;
    }

    public void invalider(String uid) { cache.remove(uid); }

    private IdentiteAppelant versIdentite(UtilisateurInterneDTO utilisateur) {
        List<String> permissions = permissions(utilisateur);
        boolean toutesAgences = permissions.contains(PERMISSION_TOUTES_AGENCES);
        String branchCode = utilisateur.orgaUnit() != null && TYPE_AGENCE.equals(utilisateur.orgaUnit().type())
                ? utilisateur.orgaUnit().code() : null;
        List<GrantedAuthority> authorities = permissions.stream()
                .<GrantedAuthority>map(SimpleGrantedAuthority::new)
                .toList();
        log.debug("Auth >> uid={} agence={} toutesAgences={} permissions={}", utilisateur.uid(), branchCode, toutesAgences, permissions);
        return new IdentiteAppelant(new ContexteAppelant(utilisateur.uid(), branchCode, toutesAgences), authorities);
    }

    private static List<String> permissions(UtilisateurInterneDTO utilisateur) {
        if (utilisateur.profile() == null || utilisateur.profile().roles() == null) {
            return List.of();
        }
        return utilisateur.profile().roles().stream()
                .filter(Objects::nonNull)
                .flatMap(role -> role.permissions() == null ? Stream.empty() : role.permissions().stream())
                .map(UtilisateurInterneDTO.PermissionDTO::code)
                .filter(Objects::nonNull)
                .distinct()
                .sorted()
                .toList();
    }

    private record Entree(IdentiteAppelant identite, Instant expiration) {}
}

package com.bmci.workflow.fund.management.application.rest.security;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.convert.converter.Converter;
import org.springframework.http.HttpStatus;
import org.springframework.security.authentication.AbstractAuthenticationToken;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.stereotype.Component;
import org.springframework.web.server.ResponseStatusException;

/** Mode oauth : JWT validé par le resource server → uid (claim uid, sinon sub) → identité depuis users. */
@Slf4j
@Component
@RequiredArgsConstructor
public class QfJwtAuthenticationConverter implements Converter<Jwt, AbstractAuthenticationToken> {

    private final IdentiteAppelantResolver resolver;

    @Override
    public AbstractAuthenticationToken convert(Jwt jwt) {
        String uid = jwt.getClaimAsString("uid");
        if (uid == null) {
            uid = jwt.getSubject();
        }
        log.debug("Auth >> uid={}, claims={}", uid, jwt.getClaims().keySet());
        return resolver.resoudre(uid)
                .map(identite -> (AbstractAuthenticationToken) new AppelantAuthenticationToken(identite, jwt))
                .orElseThrow(() -> new ResponseStatusException(HttpStatus.UNAUTHORIZED, "User not found or disabled"));
    }
}

import com.nimbusds.jwt.SignedJWT;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpHeaders;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.Optional;

/**
 * Mode mock (local) — miroir de QfSimplifiedAuthFilter (users) : le Bearer relayé par la gateway est un JWT signé
 * avec une clé régénérée à chaque démarrage de la gateway, donc lu sans vérification de signature. uid = sub.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class QfSimplifiedAuthFilter extends OncePerRequestFilter {

    private final IdentiteAppelantResolver resolver;
    private final PublicEndpointMatcher publicEndpointMatcher;

    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        return publicEndpointMatcher.matches(request);
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {

        String authorization = request.getHeader(HttpHeaders.AUTHORIZATION);
        if (authorization == null || !authorization.startsWith("Bearer ")) {
            log.warn("Missing Authorization Bearer token for path={}", request.getServletPath());
            response.sendError(HttpServletResponse.SC_UNAUTHORIZED, "Missing Authorization Bearer token");
            return;
        }

        String token = authorization.substring("Bearer ".length());
        try {
            String uid = SignedJWT.parse(token).getJWTClaimsSet().getSubject();
            Optional<IdentiteAppelant> identite = uid == null || uid.isBlank()
                    ? Optional.empty()
                    : resolver.resoudre(uid.trim());
            if (identite.isEmpty()) {
                log.warn("Auth >> User not found or disabled, uid={}", uid);
                SecurityContextHolder.clearContext();
                response.sendError(HttpServletResponse.SC_UNAUTHORIZED, "User not found or disabled");
                return;
            }
            SecurityContextHolder.getContext().setAuthentication(new AppelantAuthenticationToken(identite.get(), token));
            log.debug("Mocked authentication success for uid={}", uid);
            filterChain.doFilter(request, response);
        } catch (java.text.ParseException e) {
            log.error("Failed to parse mocked JWT", e);
            SecurityContextHolder.clearContext();
            response.sendError(HttpServletResponse.SC_UNAUTHORIZED, "Invalid token");
        }
    }
}

    this.actions.acquitterErreur();

    this.decision.set(null);
  }

  async confirmerDecision(): Promise<void> {
    const d = this.decision();
    if (!d) { return; }
    if (d.type === 'ANNULEE' && this.motifAnnulation.invalid) {
      this.motifAnnulation.markAsTouched();
      return;
    }
    const resultat = await this.actions.decider(d.id, d.type,
      d.type === 'ANNULEE' ? this.motifAnnulation.value.trim() : undefined);
    if (resultat) {
      this.decision.set(null);
      this.snackBar.open(
        d.type === 'TRAITEE'
          ? `Demande ${d.reference} traitée : le statut sera mis à jour dans quelques secondes.`
          : `Annulation de ${d.reference} transmise : le statut sera mis à jour dans quelques secondes.`,
        'Fermer', { duration: 5000 });
      setTimeout(() => this.rafraichir(), 1500);   // le statut final est écrit par le worker (~1 s)
    } else {
      this.snackBar.open(this.actions.erreur() ?? 'Erreur lors de la décision.', 'Fermer', { duration: 6000 });
    }
  }

  private construireCriteres(statut: string, urgent: boolean, transporteur: string, zone: string,
                             recherche: string, page: number): DemandeCriteria {
    const c: DemandeCriteria = { page, size: this.tailleParDefaut };
    if (statut !== 'TOUS')        { c.statut = statut; }
    if (urgent)                   { c.urgent = true; }
    if (transporteur !== 'TOUS')  { c.prestataire = transporteur; }
    if (zone !== 'TOUTES')        { c.zone = zone; }
    if (recherche)                { c.recherche = recherche; }
    return c;
  }

  /** Photo de l'agence portée par la demande ; repli sur le référentiel pour les demandes antérieures à V4. */
  private versLigne(d: DemandeFonds, repli: AgenceReferentiel | null, decider: boolean): LigneCaisseCentrale {
    const statut = versStatutDemande(d.requestStatus, (d as DemandeFondsDetailBrute).cancellationOrigin);
    return {
      demande: d,
      reference: d.uniqueRequestRef ?? '–',
      urgente: d.urgent ?? (d.operations?.some(op => op.isUrgent === true) ?? false),
      agenceNom: d.agenceNom ?? repli?.nom ?? null,
      zone: d.zoneNom ?? repli?.zoneNom ?? null,
      transporteurId: d.prestataire ?? repli?.prestataireId ?? null,
      transporteur: d.prestataireNom ?? repli?.prestataireNom ?? null,
      date: (d as DemandeFondsDetailBrute).submittedAt ?? d.createdAt,
      contenu: this.chipsContenu(d),
      statut,
      libelleStatut: libelleStatut(statut),
      classeStatut: classeStatut(statut),
      decidable: decider && statut === 'SOUMISE'
    };
  }

  /** Mêmes chips que le listing agence : GAB / CAISSE / RAMASSAGE / DEVISES. */
  private chipsContenu(d: DemandeFonds): string[] {
    const chips = (d.operations ?? []).map(op => {
      if (op.operationSens === 'COLLECTION') { return 'RAMASSAGE'; }
      if (op.operationType === 'ATM')        { return 'GAB'; }
      if (op.operationType === 'CASH_DESK')  { return 'CAISSE'; }
      if (op.operationType === 'CURRENCY')   { return 'DEVISES'; }
      return op.operationType;
    });
    return [...new Set(chips)];
  }

  classeTransporteur(l: LigneCaisseCentrale): string {
    return l.transporteurId ? `transporteur transporteur--${l.transporteurId.toLowerCase()}` : 'transporteur';
  }
