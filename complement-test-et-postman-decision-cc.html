@ExtendWith(MockitoExtension.class)
class CompleterTacheCaisseCentraleTest {

    @Mock private CashDemandRepository cashDemandRepository;
    @Mock private DenominationRepository denominationRepository;
    @Mock private CashRequestWebMapper webMapper;
    @Mock private SagaOrchestrator sagaOrchestrator;
    @Mock private ChangerStatutDemandeStep changerStatutDemandeStep;
    @Mock private DemarrerProcessusStep demarrerProcessusStep;
    @Mock private ProcessusWorkflowPort processusWorkflowPort;

    @InjectMocks
    private CashDemandUseCaseImpl useCase;

    private CashRequest demandeAuStatut(UUID id, String statut) {
        CashRequest demande = new CashRequest();
        demande.setId(new CashRequestId(id));
        demande.setRequestStatus(statut);
        return demande;
    }

    @Test
    void traiteeDoitCompleterLaTacheSansEcrireLeStatut() {
        UUID id = UUID.randomUUID();
        CashRequest demande = demandeAuStatut(id, "SUBMITTED");
        when(cashDemandRepository.getCashRequestById(id)).thenReturn(demande);
        when(processusWorkflowPort.rechercherTacheActive(id.toString())).thenReturn(Optional.of("task-1"));
        when(webMapper.toResponseDTO(demande)).thenReturn(new CashRequestDTO());

        useCase.completerTacheCaisseCentrale(id,
            DecisionCaisseCentraleDTO.builder().decision(DecisionCaisseCentrale.TRAITEE).build());

        // La preuve du design : la variable part, le statut local NE BOUGE PAS, aucun save.
        @SuppressWarnings("unchecked")
        ArgumentCaptor<Map<String, Object>> variables =
            ArgumentCaptor.forClass((Class<Map<String, Object>>) (Class<?>) Map.class);
        verify(processusWorkflowPort).completerTache(eq("task-1"), variables.capture());
        assertEquals("TRAITEE", variables.getValue().get("decisionCaisseCentrale"));
        assertFalse(variables.getValue().containsKey("motifAnnulation"));
        assertEquals("SUBMITTED", demande.getRequestStatus());
        verify(cashDemandRepository, never()).save(any());
    }

    @Test
    void annuleeDoitTransmettreLeMotif() {
        UUID id = UUID.randomUUID();
        CashRequest demande = demandeAuStatut(id, "SUBMITTED");
        when(cashDemandRepository.getCashRequestById(id)).thenReturn(demande);
        when(processusWorkflowPort.rechercherTacheActive(id.toString())).thenReturn(Optional.of("task-2"));
        when(webMapper.toResponseDTO(demande)).thenReturn(new CashRequestDTO());

        useCase.completerTacheCaisseCentrale(id, DecisionCaisseCentraleDTO.builder()
            .decision(DecisionCaisseCentrale.ANNULEE)
            .motifAnnulation("Stock insuffisant en caisse centrale")
            .build());

        @SuppressWarnings("unchecked")
        ArgumentCaptor<Map<String, Object>> variables =
            ArgumentCaptor.forClass((Class<Map<String, Object>>) (Class<?>) Map.class);
        verify(processusWorkflowPort).completerTache(eq("task-2"), variables.capture());
        assertEquals("ANNULEE", variables.getValue().get("decisionCaisseCentrale"));
        assertEquals("Stock insuffisant en caisse centrale",
            variables.getValue().get("motifAnnulation"));
    }

    @Test
    void annuleeSansMotifValideDoitLever422SansRienAppeler() {

        assertThrows(RetailValidationException.class, () ->
