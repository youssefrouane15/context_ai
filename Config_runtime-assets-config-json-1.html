0. Principe

Les URL ne sont plus figées au build : un fichier assets/app.config.json est lu au démarrage (APP_INITIALIZER) et fusionné dans l'objet environment existant. Aucun fichier qui importe environment ne change (services HTTP, intercepteur, portail.ts, gardes) : ils lisent les mêmes clés, remplies avant que le premier composant soit créé. En recette / prod, l'exploitation remplace assets/app.config.json dans le dossier déployé — pas de rebuild.

C:\dev\fund-management-front\src\app\shared\config\configuration.ts

cache: 'no-store' : le navigateur ne garde pas une ancienne config après un changement côté serveur. Le chemin est relatif au <base href>, donc valable en local (./) comme derrière /qlickflow/fund_management/.

import { APP_INITIALIZER, Provider } from '@angular/core';
import { Environnement, environment } from '../../../environments/environment';

/** Fichier lu au démarrage, relatif au <base href> (fonctionne en local « ./ » comme sous /qlickflow/fund_management/). */
const FICHIER_CONFIGURATION = 'assets/app.config.json';

/** Clés que config.json a le droit de surcharger (`production` reste décidé par le build). */
type ConfigurationExterne = Partial<Omit<Environnement, 'production'>>;

/**
 * Charge assets/config.json et fusionne ses valeurs dans `environment` AVANT le bootstrap des composants.
 * fetch() plutôt que HttpClient : rien ne doit passer par les intercepteurs (qui lisent eux-mêmes `environment`).
 * - fichier absent en dev : on garde les valeurs d'environment.ts (avertissement console) ;
 * - fichier absent en prod : erreur bloquante, l'application ne doit pas démarrer avec des URL vides.
 */
export async function chargerConfiguration(): Promise<void> {
  let reponse: Response;
  try {
    reponse = await fetch(FICHIER_CONFIGURATION, { cache: 'no-store' });
  } catch (erreur) {
    return signalerAbsence(`${FICHIER_CONFIGURATION} injoignable (${erreur})`);
  }
  if (!reponse.ok) {
    return signalerAbsence(`${FICHIER_CONFIGURATION} : HTTP ${reponse.status}`);
  }
  const externe = (await reponse.json()) as ConfigurationExterne;
  const { production: _ignore, ...valeurs } = externe as Record<string, unknown>;   // production jamais surchargée
  Object.assign(environment, valeurs);
}

function signalerAbsence(detail: string): void {
  if (environment.production) {
    throw new Error(`Configuration manquante : ${detail}`);
  }
  console.warn(`[config] ${detail} — valeurs par défaut d'environment.ts conservées.`);
}

/** À ajouter aux providers de app.config.ts. */
export const CONFIGURATION_INITIALISEUR: Provider = {
  provide: APP_INITIALIZER,
  useFactory: () => chargerConfiguration,
  multi: true
};

import { chargerConfiguration } from './configuration';
import { environment } from '../../../environments/environment';

describe('chargerConfiguration', () => {
  const initial = { ...environment };
  afterEach(() => { Object.assign(environment, initial); });

  function simulerFetch(reponse: Partial<Response>): jasmine.Spy {
    return spyOn(window, 'fetch').and.resolveTo(reponse as Response);
  }

  it('fusionne les valeurs de assets/config.json dans environment', async () => {
    simulerFetch({ ok: true, status: 200, json: () => Promise.resolve({ gatewayUrl: 'https://gw.recette', portailUrl: 'https://portail.recette' }) });
    await chargerConfiguration();
    expect(environment.gatewayUrl).toBe('https://gw.recette');
    expect(environment.portailUrl).toBe('https://portail.recette');
    expect(environment.usersApi).toBe(initial.usersApi);          // clé absente du JSON : valeur par défaut conservée
  });

  it('ignore la clé production du JSON', async () => {
    simulerFetch({ ok: true, status: 200, json: () => Promise.resolve({ production: true }) });
    await chargerConfiguration();
    expect(environment.production).toBe(initial.production);
  });

  it('hors prod, garde les valeurs par défaut si le fichier manque', async () => {
    simulerFetch({ ok: false, status: 404 });
    spyOn(console, 'warn');
    await chargerConfiguration();
    expect(environment.gatewayUrl).toBe(initial.gatewayUrl);
    expect(console.warn).toHaveBeenCalled();
  });

  it('en prod, échoue si le fichier manque', async () => {
    environment.production = true;
    simulerFetch({ ok: false, status: 404 });
    await expectAsync(chargerConfiguration()).toBeRejectedWithError(/Configuration manquante/);
  });
});
