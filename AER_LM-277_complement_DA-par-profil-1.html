import com.bmci.q360.domain.retail.enums.EvenementDemande;

/**
 * AER_LM-277 — notification email au Directeur d'agence après une soumission, une mise à jour ou une annulation par l'agence.
 * Le DA est déduit côté fund-management : parmi les uid de profil DA connus (retail.user_profile), celui dont la fiche
 * utilisateurs-ci est rattachée à l'agence de la demande (demande.branchCode).
 * Best effort : aucune exception ne remonte au use case appelant — la demande est déjà enregistrée, l'email ne doit jamais la faire échouer.
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationDemandeService {

    static final String PROFIL_DA = "DA";

    private final UserProfileRepository userProfileRepository;
    private final DirecteurAgencePort directeurAgencePort;
    private final NotificationPort notificationPort;
    private final GabaritEmailDemande gabarit;

    public void notifier(CashRequestDTO demande, EvenementDemande evenement) {
        if (demande == null) { return; }
        try {
            Optional<DirecteurAgence> directeur = directeurDe(demande.getBranchCode());
            if (directeur.isEmpty() || directeur.get().email() == null || directeur.get().email().isBlank()) {
                log.warn("[notification] {} {} : aucun Directeur d'agence avec email pour l'agence {} — email non envoyé",
                        evenement, demande.getUniqueRequestRef(), demande.getBranchCode());
                return;
            }
            String sujet = "[QlickFlow] " + evenement.getLibelle() + " " + demande.getUniqueRequestRef()
                    + " — Agence " + demande.getBranchCode();
            String corps = gabarit.construire(demande, evenement, directeur.get().nomComplet());
            notificationPort.envoyer(new EmailNotification(directeur.get().email(), sujet, corps));
            log.info("[notification] {} {} → {}", evenement, demande.getUniqueRequestRef(), directeur.get().email());
        } catch (RuntimeException e) {
            log.error("[notification] {} {} : échec de l'envoi, demande non impactée — {}",
                    evenement, demande.getUniqueRequestRef(), e.getMessage());
        }
    }

    private Optional<DirecteurAgence> directeurDe(String branchCode) {
        if (branchCode == null || branchCode.isBlank()) { return Optional.empty(); }
        return userProfileRepository.uidsParProfil(PROFIL_DA).stream()
                .map(uid -> directeurAgencePort.siDirecteurDe(uid, branchCode))
                .flatMap(Optional::stream)
                .findFirst();
    }
}

package com.bmci.workflow.fund.management.external.users;

import com.bmci.q360.domain.retail.valueobjects.referentiel.DirecteurAgence;
import com.bmci.workflow.fund.management.external.users.dto.UtilisateurInterneDTO;
import com.bmci.workflow.fund.management.service.domain.ports.output.referentiel.DirecteurAgencePort;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.RestClient;

import java.util.Optional;

/**
 * AER_LM-277 — fiche utilisateur lue chez utilisateurs-ci (GET /api/v1/internal/users/{uid}, endpoint existant).
 * La déduction « est le Directeur d'agence de branchCode » se fait ici : enabled + profile.code = DA + orgaUnit AGENCE de même code.
 * Même style qu'UtilisateursAgenceAdapter : RestClient.create (pas un bean), tolérant — toute erreur → Optional.empty().
 */
@Component
@Slf4j
public class UtilisateursDirecteurAdapter implements DirecteurAgencePort {

    static final String PROFIL_DA = "DA";
    static final String TYPE_AGENCE = "AGENCE";

    private final RestClient client;

    public UtilisateursDirecteurAdapter(@Value("${qf.users.internal-base-url}") String baseUrl) {
        this.client = RestClient.create(baseUrl);
    }

    @Override
    public Optional<DirecteurAgence> siDirecteurDe(String uid, String branchCode) {
        if (uid == null || uid.isBlank() || branchCode == null || branchCode.isBlank()) { return Optional.empty(); }
        try {
            UtilisateurInterneDTO u = client.get()
                    .uri("/api/v1/internal/users/{uid}", uid)
                    .retrieve()
                    .body(UtilisateurInterneDTO.class);
            return estDirecteurDe(u, branchCode) ? Optional.of(versDirecteur(u)) : Optional.empty();
        } catch (HttpClientErrorException.NotFound e) {
            log.debug("[notification] {} inconnu chez utilisateurs-ci", uid);
            return Optional.empty();
        } catch (RuntimeException e) {
            log.warn("[notification] fiche utilisateur {} injoignable chez utilisateurs-ci : {}", uid, e.getMessage());
            return Optional.empty();
        }
    }

    static boolean estDirecteurDe(UtilisateurInterneDTO u, String branchCode) {
        return u != null
                && Boolean.TRUE.equals(u.enabled())
                && u.profile() != null && PROFIL_DA.equalsIgnoreCase(u.profile().code())
                && u.orgaUnit() != null && TYPE_AGENCE.equalsIgnoreCase(u.orgaUnit().type())
                && branchCode.equalsIgnoreCase(u.orgaUnit().code());
    }

    private static DirecteurAgence versDirecteur(UtilisateurInterneDTO u) {
        String nom = ((u.firstName() == null ? "" : u.firstName()) + " " + (u.lastName() == null ? "" : u.lastName())).trim();
        return new DirecteurAgence(u.uid(), u.email(), nom.isEmpty() ? u.uid() : nom);
    }
}
