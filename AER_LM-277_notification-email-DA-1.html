/**
 * AER_LM-277 — notification email au Directeur d'agence après une soumission, une mise à jour ou une annulation par l'agence.
 * Le DA = manager de l'agence de l'utilisateur qui agit (orgaUnit.manager chez utilisateurs-ci).
 * Best effort : aucune exception ne remonte au use case (la demande est déjà enregistrée, l'email ne doit jamais la faire échouer).
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationDemandeService {

    private final DirecteurAgencePort directeurAgencePort;
    private final NotificationPort notificationPort;
    private final GabaritEmailDemande gabarit;

    public void notifier(CashRequestDTO demande, EvenementDemande evenement, String uidAppelant) {
        if (demande == null) { return; }
        try {
            var directeur = directeurAgencePort.directeurDe(uidAppelant);
            if (directeur.isEmpty() || directeur.get().email() == null || directeur.get().email().isBlank()) {
                log.warn("[notification] {} {} : aucun Directeur d'agence (manager avec email) pour l'agence de {} — email non envoyé",
                        evenement, demande.getUniqueRequestRef(), uidAppelant);
                return;
            }
            String sujet = "[QlickFlow] " + evenement.libelle() + " " + demande.getUniqueRequestRef()
                    + " — Agence " + demande.getBranchCode();
            String corps = gabarit.construire(demande, evenement, directeur.get().nomComplet());
            notificationPort.envoyer(new EmailNotification(directeur.get().email(), sujet, corps));
            log.info("[notification] {} {} → {}", evenement, demande.getUniqueRequestRef(), directeur.get().email());
        } catch (RuntimeException e) {
            log.error("[notification] {} {} : échec de l'envoi, demande non impactée — {}",
                    evenement, demande.getUniqueRequestRef(), e.getMessage());
        }
    }
}

import com.bmci.q360.domain.retail.enums.OperationType;

/**
 * Construit le corps HTML de l'email (maquette Jira : bandeau header / footer, rappel agence - date - initiateur,
 * évènement, récap de la demande) à partir du gabarit templates/notification-demande.html et de ses ${cles}.
 * Pas de moteur de template : un remplacement de clés avec échappement HTML suffit, et aucune dépendance de plus.
 */
@Component
public class GabaritEmailDemande {

    private static final String GABARIT = "templates/notification-demande.html";
    private static final DateTimeFormatter DATE_HEURE = DateTimeFormatter.ofPattern("dd/MM/yyyy 'à' HH:mm");
    private static final DecimalFormat MONTANT = new DecimalFormat("#,##0.00", DecimalFormatSymbols.getInstance(Locale.FRANCE));

    private final String gabarit;

    public GabaritEmailDemande() {
        try {
            gabarit = new ClassPathResource(GABARIT).getContentAsString(StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new UncheckedIOException("Gabarit email introuvable : " + GABARIT, e);
        }
    }

    public String construire(CashRequestDTO d, EvenementDemande evenement, String nomDirecteur) {
        List<CashOperationDTO> ops = d.getOperations() == null ? List.of() : d.getOperations();
        boolean urgente = Boolean.TRUE.equals(d.getUrgent())
                || ops.stream().anyMatch(op -> Boolean.TRUE.equals(op.getIsUrgent()));
        LocalDateTime date = d.getUpdatedAt() != null ? d.getUpdatedAt() : d.getCreatedAt();

        Map<String, String> valeurs = Map.ofEntries(
                Map.entry("evenement", evenement.libelle()),
                Map.entry("verbe", evenement.verbe()),
                Map.entry("directeur", vide(nomDirecteur)),
                Map.entry("reference", vide(d.getUniqueRequestRef())),
                Map.entry("agence", vide(d.getBranchCode()) + (d.getAgenceNom() != null ? " - " + d.getAgenceNom() : "")),
                Map.entry("zone", d.getZoneNom() != null ? d.getZoneNom() : vide(d.getZoneCode())),
                Map.entry("transporteur", d.getPrestataireNom() != null ? d.getPrestataireNom() : vide(d.getPrestataire())),
                Map.entry("date", date != null ? DATE_HEURE.format(date) : "—"),
                Map.entry("initiateur", d.getAgentNom() != null ? d.getAgentNom() + (d.getAgentProfil() != null ? " (" + d.getAgentProfil() + ")" : "") : vide(d.getUidUser())),
                Map.entry("urgence", urgente ? "OUI" : "Non"),
                Map.entry("classeUrgence", urgente ? "urgent" : "normal"),
                Map.entry("totalCaisse", montant(total(ops, OperationType.CASH_DESK, RequestType.SUPPLY))),
                Map.entry("totalGab", montant(total(ops, OperationType.ATM, RequestType.SUPPLY))),
                Map.entry("lignesDevises", lignes(ops, RequestType.SUPPLY)),
                Map.entry("lignesRamassage", lignes(ops, RequestType.COLLECTION)),
                Map.entry("commentaire", d.getComment() != null && !d.getComment().isBlank() ? d.getComment() : "—"),
                Map.entry("blocAnnulation", evenement == EvenementDemande.ANNULATION
                        ? "<tr><th>Motif d'annulation</th><td>" + echapper(vide(d.getCancellationReason())) + "</td></tr>" : "")
        );

        String html = gabarit;
        for (Map.Entry<String, String> e : valeurs.entrySet()) {
            // Les blocs HTML construits ici (lignes*, blocAnnulation) sont déjà échappés champ par champ
            boolean brut = e.getKey().startsWith("lignes") || e.getKey().equals("blocAnnulation") || e.getKey().equals("classeUrgence");
            html = html.replace("${" + e.getKey() + "}", brut ? e.getValue() : echapper(e.getValue()));
        }
        return html;
    }

    private static BigDecimal total(List<CashOperationDTO> ops, OperationType type, RequestType sens) {
        BigDecimal somme = BigDecimal.ZERO;
        for (CashOperationDTO op : ops) {
            if (op.getOperationType() != type || op.getOperationSens() != sens) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (detail.getTotalAmountByCurrency() != null) somme = somme.add(detail.getTotalAmountByCurrency());
            }
        }
        return somme;
    }

    /** Une ligne &lt;tr&gt; par devise pour CURRENCY / sens donné (SUPPLY = commande, COLLECTION = ramassage). */
    private static String lignes(List<CashOperationDTO> ops, RequestType sens) {
        StringBuilder sb = new StringBuilder();
        for (CashOperationDTO op : ops) {
            if (op.getOperationType() != OperationType.CURRENCY || op.getOperationSens() != sens) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                String devise = detail.getCurrency() == null ? "—" : vide(detail.getCurrency().getIsoCode());
                sb.append("<tr><td>").append(echapper(devise)).append("</td><td class=\"montant\">")
                  .append(montant(detail.getTotalAmountByCurrency())).append(' ').append(echapper(devise)).append("</td></tr>");
            }
        }
        return sb.length() == 0 ? "<tr><td colspan=\"2\" class=\"vide\">Aucune</td></tr>" : sb.toString();
    }

    private static List<OperationCurrencyDetailDTO> details(CashOperationDTO op) {
        return op.getCurrencyDetails() == null ? List.of() : op.getCurrencyDetails();
    }

    private static String montant(BigDecimal m) { return MONTANT.format(m == null ? BigDecimal.ZERO : m); }
    private static String vide(String s) { return s == null ? "" : s; }

    static String echapper(String s) {
        if (s == null) return "";
        return s.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;").replace("\"", "&quot;");
    }
}

/**
 * AER_LM-277 — serveur email injoignable (infra) : « boîte aux lettres » locale.
 * Chaque email est (1) journalisé, (2) écrit en .html dans gdf.notification.mock.dossier, (3) gardé en mémoire (50 derniers)
 * pour être consulté via GET /notifications (NotificationMockController). Actif quand gdf.notification.mode=mock (défaut).
 */
@Component
@ConditionalOnProperty(name = "gdf.notification.mode", havingValue = "mock", matchIfMissing = true)
@Slf4j
public class NotificationMockAdapter implements NotificationPort {

    public record EmailEnregistre(String id, LocalDateTime date, String destinataire, String sujet, String corpsHtml) {}

    private static final int MAX_MEMOIRE = 50;
    private static final DateTimeFormatter HORODATAGE = DateTimeFormatter.ofPattern("yyyyMMdd-HHmmss-SSS");

    private final Path dossier;
    private final Deque<EmailEnregistre> boite = new ArrayDeque<>();

    public NotificationMockAdapter(@Value("${gdf.notification.mock.dossier:./notifications-mock}") String dossier) {
        this.dossier = Path.of(dossier);
    }

    @Override
    public synchronized void envoyer(EmailNotification email) {
        String id = HORODATAGE.format(LocalDateTime.now()) + "_" + nettoyer(email.sujet());
        EmailEnregistre enregistre = new EmailEnregistre(id, LocalDateTime.now(), email.destinataire(), email.sujet(), email.corpsHtml());
        boite.addFirst(enregistre);
        while (boite.size() > MAX_MEMOIRE) { boite.removeLast(); }
        try {
            Files.createDirectories(dossier);
            Files.writeString(dossier.resolve(id + ".html"), email.corpsHtml(), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new UncheckedIOException("Écriture de l'email mock impossible dans " + dossier, e);
        }
        log.info("[notification-mock] À : {} | Sujet : {} | Fichier : {}/{}.html", email.destinataire(), email.sujet(), dossier, id);
    }

    public synchronized List<EmailEnregistre> liste() { return new ArrayList<>(boite); }

    public synchronized Optional<EmailEnregistre> trouver(String id) {
        return boite.stream().filter(e -> e.id().equals(id)).findFirst();
    }

    private static String nettoyer(String s) {
        return s.replaceAll("[^A-Za-z0-9_-]+", "_").replaceAll("^_+|_+$", "");
    }
}

package com.bmci.workflow.fund.management.external.notification;

import com.bmci.workflow.fund.management.service.domain.ports.output.notification.EmailNotification;
import com.bmci.workflow.fund.management.service.domain.ports.output.notification.NotificationPort;
import jakarta.mail.MessagingException;
import jakarta.mail.internet.MimeMessage;
import lombok.RequiredArgsConstructor;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.mail.javamail.JavaMailSender;
import org.springframework.mail.javamail.MimeMessageHelper;
import org.springframework.stereotype.Component;

import java.nio.charset.StandardCharsets;

/**
 * Envoi réel — à activer quand l'infra aura ouvert le serveur mail : gdf.notification.mode=smtp + spring.mail.* + la dépendance
 * spring-boot-starter-mail dans fund-management-container. Aucun autre fichier ne change.
 */
@Component
@ConditionalOnProperty(name = "gdf.notification.mode", havingValue = "smtp")
@RequiredArgsConstructor
public class NotificationSmtpAdapter implements NotificationPort {

    private final JavaMailSender mailSender;

    @Value("${gdf.notification.expediteur:noreply-qlickflow@bmci.ma}")
    private String expediteur;

    @Override
    public void envoyer(EmailNotification email) {
        try {
            MimeMessage message = mailSender.createMimeMessage();
            MimeMessageHelper helper = new MimeMessageHelper(message, false, StandardCharsets.UTF_8.name());
            helper.setFrom(expediteur);
            helper.setTo(email.destinataire());
            helper.setSubject(email.sujet());
            helper.setText(email.corpsHtml(), true);
            mailSender.send(message);
        } catch (MessagingException e) {
            throw new IllegalStateException("Envoi SMTP impossible : " + e.getMessage(), e);
        }
    }
}
