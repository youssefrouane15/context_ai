Étape 0 · Le fix CORS — 2 fichiers, 2 minutes

Le proxy Angular CRÉER + l'URL relative MODIFIER

En clairLe navigateur interdit à une page servie sur le port 4200 d'appeler le port 8300 — deux « origines » différentes, c'est la règle CORS. Le contournement standard en dev : le front appelle son propre serveur (/cash sur 4200, même origine = autorisé), et le petit serveur de ng serve relaie discrètement vers 8300. Les serveurs entre eux ne connaissent pas le CORS. C'est exactement ce qu'on avait fait pour le portail.
proxy.conf.json — À LA RACINE du projet (à côté de package.json)
{
  "/cash": {
    "target": "http://localhost:8300",
    "secure": false,
    "changeOrigin": true,
    "logLevel": "debug"
  }
}
fundManagement-http.service.ts — MODIFIER la ligne apiUrl
  private readonly apiUrl = `/cash`;

Puis relance avec le proxy :

PowerShell
ng serve --proxy-config proxy.conf.json
VérificationDans l'onglet Network, l'appel devient http://localhost:4200/cash → statut 200 et tes vraies données. Le logLevel: debug affiche chaque relais dans le terminal ng serve. ⚠️ Le findAll() existant de ton collègue pointe en dur sur localhost:8080/cash — ne touche pas, c'est sa partie, mais signale-le-lui.

fund-management-list.component.ts
import { ChangeDetectionStrategy, Component, OnInit, computed } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormControl, ReactiveFormsModule } from '@angular/forms';
import { toSignal } from '@angular/core/rxjs-interop';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatSelectModule } from '@angular/material/select';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatCardModule } from '@angular/material/card';
import { MatSnackBar, MatSnackBarModule } from '@angular/material/snack-bar';
import { FunManagementFacade } from '../../../adapters/localState/fundMangement.facade';
import { DemandeFonds } from '../../../domain/domain-core/entities/operation.model';
import { FeedbackStateComponent } from '../../../../../shared/presenter/components/feedback-state/feedback-state.component';
import { InputComponent } from '../../../../../shared/presenter/components/input/input.component';

@Component({
  selector: 'app-fund-management-list',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule, MatFormFieldModule, MatSelectModule,
    MatButtonModule, MatIconModule, MatCardModule, MatSnackBarModule,
    FeedbackStateComponent, InputComponent],
  templateUrl: './fund-management-list.component.html',
  styleUrl: './fund-management-list.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class FundManagementListComponent implements OnInit {

  // ===== Filtres (maquette : période / agence / type / statut / référence) =====
  periodeControl   = new FormControl('7J',     { nonNullable: true });
  agenceControl    = new FormControl('TOUTES', { nonNullable: true });
  typeControl      = new FormControl('TOUS',   { nonNullable: true });
  statutControl    = new FormControl('TOUS',   { nonNullable: true });
  rechercheControl = new FormControl('',       { nonNullable: true });

  readonly periodes = [
    { code: '7J',   libelle: '7 derniers jours' },
    { code: '30J',  libelle: '30 derniers jours' },
    { code: 'TOUS', libelle: 'Toute la période' }
  ];
  readonly types   = ['TOUS', 'GAB', 'CAISSE', 'RAMASSAGE', 'DEVISES'];
  readonly statuts = [
    { code: 'TOUS',      libelle: 'Tous les statuts' },
    { code: 'INITIATED', libelle: 'En Attente' },
    { code: 'VALIDATED', libelle: 'Validé' },
    { code: 'CANCELLED', libelle: 'Archivé' }   // TODO confirmer le mapping avec Yassin
  ];

  private periode   = toSignal(this.periodeControl.valueChanges,   { initialValue: '7J' });
  private agence    = toSignal(this.agenceControl.valueChanges,    { initialValue: 'TOUTES' });
  private type      = toSignal(this.typeControl.valueChanges,      { initialValue: 'TOUS' });
  private statut    = toSignal(this.statutControl.valueChanges,    { initialValue: 'TOUS' });
  private recherche = toSignal(this.rechercheControl.valueChanges, { initialValue: '' });

  // Liste déroulante des agences, construite depuis les données
  agences = computed(() => {
    const codes = this.facade.demandes().map(d => d.branchCode).filter(Boolean);
    return ['TOUTES', ...new Set(codes)];
  });

  demandesFiltrees = computed(() => {
    const terme = this.recherche().trim().toLowerCase();
    const statut = this.statut();
    const type = this.type();
    const agence = this.agence();
    const limite = this.limitePeriode(this.periode());

    return this.facade.demandes().filter(d => {
      const matchStatut = statut === 'TOUS' || d.requestStatus === statut;
      const matchAgence = agence === 'TOUTES' || d.branchCode === agence;
      const matchType = type === 'TOUS' || this.chipsContenu(d).includes(type);
      const matchTerme = !terme
        || this.reference(d).toLowerCase().includes(terme)
        || d.id?.toLowerCase().includes(terme)
        || d.user?.toLowerCase().includes(terme);
      const dateStr = this.dateDemande(d);
      const matchPeriode = !limite || !dateStr || new Date(dateStr) >= limite;
      return matchStatut && matchAgence && matchType && matchTerme && matchPeriode;
    });
  });

  constructor(
    public facade: FunManagementFacade,
    private snackBar: MatSnackBar
  ) { }

  ngOnInit(): void {
    this.facade.loadDemandes();
  }

  rafraichir(): void {
    this.facade.loadDemandes();
  }

  // ===== Mapping API -> écran (aligné maquette) =====

  /** TODO backend : exposer uniqueRequestRef ; en attendant on fabrique un visuel */
  reference(d: DemandeFonds): string {
    const dateStr = this.dateDemande(d);
    const jour = dateStr ? dateStr.slice(0, 10).replace(/-/g, '') : '00000000';
    const suffixe = d.id ? d.id.slice(0, 2).toUpperCase() : '00';
    return `${jour}-${d.branchCode || '0000'}-${suffixe}`;
  }

  dateDemande(d: DemandeFonds): string | undefined {
    return d.operations?.[0]?.createdAt;   // TODO : date au niveau demande côté backend
  }

  estUrgente(d: DemandeFonds): boolean {
    return d.operations?.some(op => op.isUrgent === 'YES') ?? false;
  }

  /** Traduit les opérations en chips maquette : GAB / CAISSE / RAMASSAGE / DEVISES */
  chipsContenu(d: DemandeFonds): string[] {
    const chips = (d.operations ?? []).map(op => {
      if (op.operationSens === 'COLLECTION') { return 'RAMASSAGE'; }
      if (op.operationType === 'ATM')        { return 'GAB'; }
      if (op.operationType === 'CASH_DESK')  { return 'CAISSE'; }
      if (op.operationType === 'CURRENCY')   { return 'DEVISES'; }
      return op.operationType;
    });
    return [...new Set(chips)];
  }

  libelleStatut(s: string): string {
    const map: Record<string, string> = {
      INITIATED: 'En Attente', VALIDATED: 'Validé', CANCELLED: 'Archivé'
    };
    return map[s] ?? s;
  }

  async annuler(d: DemandeFonds): Promise<void> {
    if (d.requestStatus === 'CANCELLED') { return; }
    const ok = confirm(`Archiver la demande ${this.reference(d)} et toutes ses opérations ?`);
    if (!ok) { return; }

    try {
      await this.facade.annulerDemande(d.id);
      this.snackBar.open('Demande archivée.', 'Fermer', { duration: 4000 });
    } catch (e) {
      this.snackBar.open("Erreur lors de l'archivage de la demande.", 'Fermer', { duration: 4000 });
    }
  }

  private limitePeriode(code: string): Date | null {
    if (code === '7J')  { const d = new Date(); d.setDate(d.getDate() - 7);  return d; }
    if (code === '30J') { const d = new Date(); d.setDate(d.getDate() - 30); return d; }
    return null;
  }
}
