Fix — « Mode Modification » refusé sur une demande déjà en modification (verrou non libéré)

fund-management-front · 07/10/2026 · fichier complet

CauseFichierVérification

1. Ce qui se passe

Le bouton-badge « Mode Modification » appelle toujours PATCH /modification/start, qui pose le verrou (statut EN_MODIFICATION, lockedBy = toi). Si tu quittes l'écran sans passer par « Retour au listing » de la barre (menu latéral, bouton précédent, F5) — ou si ce retour a échoué silencieusement — le verrou reste posé. Au retour, le badge est actif (le verrou est à toi) mais le back refuse de reposer un verrou sur une demande déjà en modification.

Correctif, uniquement dans fund-management.component.ts :

  • modifier() : si la demande est EN_MODIFICATION et que le verrou est le tien → ouverture directe du mode modification (reprise), sans rappeler /modification/start.
  • abandonnerModification() : si la libération échoue, message d'erreur et on reste sur l'écran (avant : navigation quand même, verrou gardé).
  • ngOnDestroy : sortie du mode modification par un autre chemin → terminerModification(id, false) pour libérer le verrou (sinon la demande reste invisible de la Caisse Centrale jusqu'au timer du workflow).

  constructor(
    private fb: FormBuilder,
    public facade: FunManagementFacade,

    this.mode = this.route.snapshot.data['mode'] ?? 'creation';
    this.estCreation = this.mode === 'creation';
    this.estConsultation = this.mode === 'consultation';
    this.estModification = this.mode === 'modification';

    this.idDemande = this.route.snapshot.paramMap.get('id');

     * Les deux sont dans le MÊME effect, volontairement : les référentiels de coupures et le détail arrivent par deux
     * appels HTTP indépendants. Si le patch vivait dans un effect séparé, une arrivée tardive des coupures viderait

      const coupuresGab = this.facade.coupuresGab().filter(c => COUPURES_GAB.includes(Number(c.faceValue)));
      const detail = this.avecDetail ? this.facade.demandeDetail() : null;

      this.devisesDisponibles = this.facade.devises();

      this.gabCoupures.clear();
      this.caisseCoupures.clear();

      coupuresGab.forEach(c => this.gabCoupures.push(this.creerLigneCoupure(c)));
      coupuresCaisse.forEach(c => this.caisseCoupures.push(this.creerLigneCoupure(c)));

        this.form.disable({ emitEvent: false });

          ? urgent.enable({ emitEvent: false })
          : urgent.disable({ emitEvent: false });

      this.cdr.markForCheck();

        this.router.navigate(['consultation', this.idDemande]);

      if (erreur) {
        this.snackBar.open(erreur.message, 'Fermer', { duration: 6000 });
        this.facade.acquitterModification();
        return;

      const erreur = this.facade.erreurCreation();

  /**
   * Bouton-badge « Mode Modification » de l'en-tête : pose le verrou (la Caisse Centrale perd la main) puis ouvre le mode modification.
   * Brouillon : pas de verrou. Modification déjà commencée par l'utilisateur courant (verrou encore posé après un retour au listing,
   * un clic menu ou un F5) : on reprend directement, sans redemander le verrou — le back refuserait (« demande déjà en modification »).
   */
  async modifier(): Promise<void> {
    if (!this.idDemande) { return; }
    const d = this.demande();
    if (d?.statut === 'INITIEE' || this.verrouDetenuParMoi()) {
      this.router.navigate(['modification', this.idDemande], { relativeTo: this.route.parent });
      return;
    }
    const verrouillee = await this.actions.demarrerModification(this.idDemande);
    if (verrouillee) {
      this.router.navigate(['modification', this.idDemande], { relativeTo: this.route.parent });
    } else {
      this.snackBar.open(this.actions.erreur() ?? 'Impossible de passer en modification.', 'Fermer', { duration: 6000 });
    }
  }

  /** « Retour au listing » du mode modification : libère le verrou sans enregistrer (la demande redevient « En attente »). */
  async abandonnerModification(): Promise<void> {
    if (!this.idDemande) { return; }
    if (this.verrouDetenuParMoi()) {
      const libere = await this.actions.terminerModification(this.idDemande, false);
      if (!libere) {
        this.snackBar.open(this.actions.erreur() ?? 'Le verrou de modification n\'a pas pu être libéré.', 'Fermer', { duration: 6000 });
        return;                                                 // on reste sur l'écran : le verrou est encore posé
      }
      this.verrouLibere = true;
    }
    this.router.navigate(['consultation', this.idDemande], { relativeTo: this.route.parent });
  }

  /** Verrou de modification détenu par l'utilisateur courant (statut EN_MODIFICATION et lockedBy = uid). */
  private verrouDetenuParMoi(): boolean {
    const d = this.demande();
    const uid = this.contexteUtilisateur.contexte()?.uid;
    return d?.statut === 'EN_MODIFICATION' && !!uid && d.lockedBy === uid;
  }

  /** Bouton « Soumettre » en consultation d'un brouillon : soumet le contenu déjà enregistré. */
  async soumettreDepuisConsultation(): Promise<void> {
    if (!this.idDemande) { return; }
    const soumise = await this.actions.soumettre(this.idDemande, null);
    if (soumise) {
      this.snackBar.open(`Demande ${soumise.uniqueRequestRef} soumise.`, 'Fermer', { duration: 4000 });
      this.facade.loadDetail(this.idDemande);
    } else {
      this.snackBar.open(this.actions.erreur() ?? 'Erreur lors de la soumission.', 'Fermer', { duration: 6000 });
    }
  }

  ouvrirAnnulation(): void { this.annulationOuverte = true; this.motifAnnulation.reset(''); }
  fermerAnnulation(): void { this.annulationOuverte = false; }

  /** Bouton « Annuler la demande » : motif ≥ 10 caractères, msgAnnulerDemande côté back, statut écrit par le worker. */
  async annulerDemande(): Promise<void> {
    if (!this.idDemande || this.motifAnnulation.invalid) {
      this.motifAnnulation.markAsTouched();
      return;
    }
    const resultat = await this.actions.annuler(this.idDemande, this.motifAnnulation.value.trim());
    if (resultat) {
      this.annulationOuverte = false;
      this.snackBar.open("Annulation transmise : le statut sera mis à jour dans quelques secondes.", 'Fermer', { duration: 5000 });
      setTimeout(() => this.idDemande && this.facade.loadDetail(this.idDemande), 1500);
    } else {
      this.snackBar.open(this.actions.erreur() ?? "Erreur lors de l'annulation.", 'Fermer', { duration: 6000 });
    }
    this.cdr.markForCheck();
  }

  // ---------------------------------------------------------------------------
  // Soumission
  // ---------------------------------------------------------------------------
  /** Remise à zéro du formulaire (après une création réussie). */
  annuler(): void {
    this.form.reset({ urgent: false, comment: '' });

    [this.gabCoupures, this.caisseCoupures].forEach(arr =>
      arr.controls.forEach(row => row.patchValue({ quantite: 0 }))
    );

    this.ramassageLignes.clear();
    this.commandeDeviseLignes.clear();
    this.ramassageGroup.patchValue({ deviseCode: DEVISE_LOCALE, montant: null });
  }

  async submit(): Promise<void> {
    if (this.form.invalid) {
      const message = this.form.hasError('formulaireVide')
        ? 'Renseignez au moins une section (caisse, GAB, ramassage ou commande de devises).'
        : 'Veuillez corriger les champs en erreur.';
      this.snackBar.open(message, 'Fermer', { duration: 5000 });
      return;
    }
    const payload = this.construirePayload();

    if (this.estModification && this.idDemande) {
      this.facade.modifierDemande(this.idDemande, payload);      // PUT → l'effet (3) termine la modification
      return;
    }

    if (this.estCreation) {
      if (!this.idDemande) {
        this.snackBar.open("La demande n'a pas pu être initialisée : rechargez la page.", 'Fermer', { duration: 6000 });
        return;
      }
      this.submitting = true;
      const soumise = await this.actions.soumettre(this.idDemande, payload);   // contenu + Task_SoumettreDemande
      this.submitting = false;
      this.cdr.markForCheck();
      if (soumise) {
        this.snackBar.open(`Demande ${soumise.uniqueRequestRef} soumise.`, 'Fermer', { duration: 4000 });
        this.router.navigate(['consultation', soumise.id], { relativeTo: this.route.parent });
      } else {
        this.snackBar.open(this.actions.erreur() ?? 'Erreur lors de la soumission.', 'Fermer', { duration: 6000 });
      }
    }
  }

  // ---------------------------------------------------------------------------
  // Helpers privés
  // ---------------------------------------------------------------------------
  private creerLigneCoupure(d: Denomination): FormGroup {
    return this.fb.group({
      denominationId: [d.id],
      code:           [d.denominationCode],
      valeur:         [d.faceValue],
      libelle:        [d.denominationLabel],
      quantite:       [0, [Validators.min(0)]]
    });
  }

  private creerLigneDevise(deviseCode: string, montant: number): FormGroup {
    return this.fb.group({
      deviseCode: [deviseCode, Validators.required],
      montant: [montant, [Validators.required, Validators.min(0)]]
    });
  }

  private repartirCoupuresCaisse(): void {
    this.indexesBillets = [];
    this.indexesPieces = [];

    this.caisseCoupures.controls.forEach((row: AbstractControl, index: number) => {
      const valeur = Number(row.get('valeur')?.value ?? 0);
      (valeur >= SEUIL_BILLET ? this.indexesBillets : this.indexesPieces).push(index);
    });
  }

  /**
   * Applique le détail d'une demande existante au formulaire.
   * Inverse de construirePayload() : mêmes couples operationType / operationSens.
   */
  private appliquerDetail(d: DemandeFondsDetail): void {
    const operations = d.operations ?? [];
    const filtrer = (type: string, sens: string) =>
      operations.filter(o => o.operationType === type && o.operationSens === sens);

    this.form.patchValue(
      {
        urgent: operations.some(o => o.isUrgent),
        comment: d.comment ?? ''
      },
      { emitEvent: false }
    );

    this.appliquerQuantites(this.gabCoupures, filtrer('ATM', 'SUPPLY'));
    this.appliquerQuantites(this.caisseCoupures, filtrer('CASH_DESK', 'SUPPLY'));

    this.commandeDeviseLignes.clear({ emitEvent: false });
    filtrer('CURRENCY', 'SUPPLY').forEach(o =>
      (o.currencyDetails ?? []).forEach(cd =>
        this.commandeDeviseLignes.push(
          this.creerLigneDevise(cd.currency.isoCode, cd.totalAmountByCurrency),
          { emitEvent: false }
        )
      )
    );

    this.ramassageLignes.clear({ emitEvent: false });
    filtrer('CURRENCY', 'COLLECTION').forEach(o =>
      (o.currencyDetails ?? []).forEach(cd =>
        this.ramassageLignes.push(
          this.creerLigneDevise(cd.currency.isoCode, cd.totalAmountByCurrency),
          { emitEvent: false }
        )
      )
    );
  }

  /** Reporte les quantités par denominationCode, l'identifiant canonique des coupures. */
  private appliquerQuantites(coupures: FormArray, operations: OperationDetail[]): void {
    const quantites = new Map<string, number>();
    operations.forEach(o =>
      (o.currencyDetails ?? []).forEach(cd =>
        (cd.operationDenominations ?? []).forEach(od =>
          quantites.set(od.denomination.denominationCode, od.quantity)
        )
      )
    );

    coupures.controls.forEach(row => {
      const code = row.get('code')?.value as string;
      row.patchValue({ quantite: quantites.get(code) ?? 0 }, { emitEvent: false });
    });
  }

  private construirePayload(): DemandeFondsPayload {
    const urgent: boolean = this.form.get('urgent')?.value ?? false;
    const operations: OperationPayload[] = [];

    // GAB — SUPPLY ATM, ventilation MAD
    if (this.totalGab() > 0) {
      operations.push({
        operationType: 'ATM',
        operationSens: 'SUPPLY',
        isUrgent: urgent,
        cashDeskNumber: PARAMETRES_AGENCE.cashDeskNumber,
        currencyDetails: [this.detailMad(this.gabCoupures, this.totalGab())]
      });
    }

    if (this.totalCaisse() > 0) {
      operations.push({
        operationType: 'CASH_DESK',
        operationSens: 'SUPPLY',
        isUrgent: urgent,
        cashDeskNumber: PARAMETRES_AGENCE.cashDeskNumber,
        currencyDetails: [this.detailMad(this.caisseCoupures, this.totalCaisse())]
      });
    }

    this.commandeDeviseLignes.controls
      .filter(l => Number(l.get('montant')?.value ?? 0) > 0 && !!l.get('deviseCode')?.value)
      .forEach(l => operations.push({
        operationType: 'CURRENCY',
        operationSens: 'SUPPLY',
        isUrgent: urgent,
        cashDeskNumber: PARAMETRES_AGENCE.cashDeskNumber,
        currencyDetails: [{
          currency: { isoCode: l.get('deviseCode')!.value },
          totalAmountByCurrency: Number(l.get('montant')!.value),
          operationDenominations: []
        }]
      }));

    const ramassageParDevise = new Map<string, number>();
    this.ramassageLignes.controls.forEach(l => {
      const isoCode: string | null = l.get('deviseCode')?.value;
      const montant = Number(l.get('montant')?.value ?? 0);
      if (!isoCode || montant <= 0) { return; }
      ramassageParDevise.set(isoCode, (ramassageParDevise.get(isoCode) ?? 0) + montant);
    });
    if (ramassageParDevise.size > 0) {
      operations.push({
        operationType: 'CURRENCY',
        operationSens: 'COLLECTION',
        isUrgent: urgent,
        cashDeskNumber: PARAMETRES_AGENCE.cashDeskNumber,
        currencyDetails: Array.from(ramassageParDevise, ([isoCode, total]) => ({
          currency: { isoCode },
          totalAmountByCurrency: total,
          operationDenominations: []
        }))
      });
    }

    const ctx = this.contexteUtilisateur.contexte();
    return {
      branchCode: ctx?.agenceCode ?? '',
      user: ctx?.uid ?? '',
      establishment: PARAMETRES_AGENCE.establishment,
      operations,
      comment: (this.form.get('comment')?.value as string | null) ?? null
    };
  }

  private detailMad(coupures: FormArray, total: number) {
    return {
      currency: { isoCode: DEVISE_LOCALE },
      totalAmountByCurrency: total,
      operationDenominations: coupures.controls
        .filter(row => Number(row.get('quantite')?.value ?? 0) > 0)
        .map(row => ({
          denomination: { denominationCode: row.get('code')?.value as string },
          quantity: Number(row.get('quantite')?.value)
        }))
    };
  }
}

export function auMoinsUneSectionRenseigneeValidator(): ValidatorFn {
  return (form: AbstractControl): ValidationErrors | null => {
    const quantiteRenseignee = (chemin: string) =>
      ((form.get(chemin) as FormArray | null)?.controls ?? [])
        .some(row => Number(row.get('quantite')?.value ?? 0) > 0);

    const listeNonVide = (chemin: string) =>
      ((form.get(chemin) as FormArray | null)?.length ?? 0) > 0;

    const rempli =
      quantiteRenseignee('gab.coupures') ||
      quantiteRenseignee('caisse.coupures') ||
      listeNonVide('ramassage.lignes') ||
      listeNonVide('commandeDevises.lignes');

    return rempli ? null : { formulaireVide: true };
  };
}
