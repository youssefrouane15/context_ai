/**
 * Statuts métier de l'écran Agence (FR). Le back reste en EN (RequestStatus) :
 * le mapping se fait dans l'adapter HTTP via versStatutDemande().
 * AER_LM-267 : TRAITEMENT_EN_COURS et CREATED retirés (aucun équivalent back).
 */
export type StatutDemande =
  | 'INITIEE'
  | 'SOUMISE'
  | 'EN_MODIFICATION'
  | 'TRAITEE'
  | 'ANNULEE_AGENCE'
  | 'ANNULEE_CAISSE_CENTRALE';

export const STATUTS_FINAUX: readonly StatutDemande[] = ['TRAITEE', 'ANNULEE_AGENCE', 'ANNULEE_CAISSE_CENTRALE'];

export function versStatutDemande(requestStatus: string | null | undefined,
                                  cancellationOrigin?: string | null): StatutDemande {
  switch (requestStatus) {
    case 'INITIATED':       return 'INITIEE';
    case 'SUBMITTED':       return 'SOUMISE';
    case 'IN_MODIFICATION': return 'EN_MODIFICATION';
    case 'VALIDATED':       return 'TRAITEE';
    case 'CANCELLED':       return cancellationOrigin === 'CAISSE_CENTRALE' ? 'ANNULEE_CAISSE_CENTRALE' : 'ANNULEE_AGENCE';
    default:
      console.warn(`[statut-demande] requestStatus inconnu : ${requestStatus}`);
      return 'INITIEE';
  }
}

export function estStatutFinal(statut: StatutDemande): boolean {
  return STATUTS_FINAUX.includes(statut);
}

export function libelleStatut(statut: StatutDemande | string | null | undefined): string {
  switch (statut) {
    case 'INITIEE':                 return 'Brouillon';
    case 'SOUMISE':                 return 'En attente';
    case 'EN_MODIFICATION':         return 'En modification';
    case 'TRAITEE':                 return 'Traitée';
    case 'ANNULEE_AGENCE':          return 'Annulée (agence)';
    case 'ANNULEE_CAISSE_CENTRALE': return 'Annulée (Caisse Centrale)';
    default:                        return statut ?? '';
  }
}

/** Classe BEM du chip : statut--initiee, statut--soumise, statut--en-modification, statut--traitee, statut--annulee-agence, statut--annulee-caisse-centrale */
export function classeStatut(statut: StatutDemande | string | null | undefined): string {
  return statut ? `statut--${String(statut).toLowerCase().replace(/_/g, '-')}` : '';
}

/** Motif affiché dans le bandeau « Consultation seule », ou null si la demande est encore ouverte. */
export function motifDeVerrouillage(statut: StatutDemande, journeeCloturee: boolean): string | null {
  switch (statut) {
    case 'TRAITEE':                 return 'Demande traitée par la Caisse Centrale';
    case 'ANNULEE_AGENCE':          return "Demande annulée par l'agence";
    case 'ANNULEE_CAISSE_CENTRALE': return 'Demande annulée par la Caisse Centrale';
    default:
      return journeeCloturee
        ? "Cut-off de 16h00 dépassé : la demande n'est plus modifiable par l'agence. Toute intervention relève de la Caisse Centrale."
        : null;
  }
}

export function calculerCapacites(f: FaitsDemande): CapacitesDemande {
  const hab = f.habilitations;
  const finale = estStatutFinal(f.statut);
  const agenceOk = f.demandeDeMonAgence || hab.toutesAgences;
  const verrouAutre = f.verrouDetenuPar !== null && !f.verrouDetenuParUtilisateurCourant;
  const ouverte = !finale && !f.journeeCloturee && agenceOk && !verrouAutre;

  const peutEnregistrer = ouverte && hab.modifier
    && (f.statut === 'INITIEE' || (f.statut === 'EN_MODIFICATION' && f.verrouDetenuParUtilisateurCourant));
  const peutModifier = ouverte && hab.modifier && (f.statut === 'SOUMISE' || peutEnregistrer);
  const peutSoumettre = ouverte && hab.creer && f.statut === 'INITIEE';
  const peutAnnuler = ouverte && hab.annuler;

  let motif: string | null = motifDeVerrouillage(f.statut, f.journeeCloturee);
  if (!agenceOk) {
    motif = "Demande d'une autre agence : consultation seule";
  } else if (verrouAutre) {
    motif = `En cours de modification par ${f.verrouDetenuPar}`;
  } else if (!finale && !f.journeeCloturee && !hab.modifier && !hab.annuler && !hab.creer) {
    motif = 'Vos habilitations ne permettent que la consultation';
  }

  return {
    peutModifier,
    peutEnregistrer,
    peutSoumettre,
    peutAnnuler,
    peutModifierUrgence: peutEnregistrer && !f.cutoffUrgenceDepasse,
    motifVerrouillage: (peutModifier || peutAnnuler || peutSoumettre) ? null : (motif ?? 'Lecture seule')

    .filter(r => r.platform === PLATEFORME_GDF)

  private genererCsvDemande(d: DemandeFonds): string {
    const operations = d.operations ?? [];
    const types = [...new Set(
      operations.map(op => op.operationType === 'CURRENCY' ? 'DEVISE' : op.operationType)
    )].join(', ');
    const lignes = [

  private champCsv(valeur: unknown): string {

    const s = String(valeur);
    return /[;"\r\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;

  private versErreurApi(err: unknown): ErreurApi {
    if (err instanceof ErreurApi) { return err; }
    if (err instanceof HttpErrorResponse) {
      const corps = err.error as ApiResponse<Record<string, string>> | null;
      if (corps && typeof corps === 'object' && 'success' in corps) {
        return new ErreurApi(corps.error ?? String(err.status), corps.message ?? err.message, corps.data);
      }
      if (err.status === 0) {
        return new ErreurApi('0', 'Serveur injoignable. Vérifiez votre connexion ou le VPN.');
      }
      if (err.status === 401) {
        return new ErreurApi('401', 'Session expirée : reconnectez-vous.');

/**
 * Identité et habilitations GDF de l'utilisateur connecté, depuis GET /api/v1/public/users/me (UserDTO nu).
 * Chargé une fois par session d'écran ; remplace CONTEXTE_AGENCE en dur.
 */
@Injectable({ providedIn: 'root' })
export class ContexteUtilisateurService {

  private readonly http = inject(HttpClient);
  private readonly meUrl = `${environment.gatewayUrl}${environment.usersApi}/me`;

  readonly contexte = signal<ContexteGdf | null>(null);
  readonly chargement = signal(false);
  readonly erreur = signal<string | null>(null);

  readonly habilitations = computed(() => this.contexte()?.habilitations ?? HABILITATIONS_AUCUNE);
  readonly uid = computed(() => this.contexte()?.uid ?? null);
  readonly agenceCode = computed(() => this.contexte()?.agenceCode ?? null);

  private enCours: Promise<ContexteGdf | null> | null = null;

  /** Idempotent : un seul appel réseau, les appels concurrents partagent la même promesse. */
  charger(): Promise<ContexteGdf | null> {
    const deja = this.contexte();
    if (deja) { return Promise.resolve(deja); }
    if (this.enCours) { return this.enCours; }
    this.enCours = this.chargerDepuisMe();
    return this.enCours;
  }

  private async chargerDepuisMe(): Promise<ContexteGdf | null> {
    this.chargement.set(true);
    this.erreur.set(null);
    try {
      const me = await firstValueFrom(this.http.get<UtilisateurMe>(this.meUrl, { withCredentials: true }));
      const ctx = deriverContexteGdf(me);
      this.contexte.set(ctx);
      return ctx;
    } catch (err: unknown) {
      const status = (err as { status?: number } | null)?.status;
      this.erreur.set(status === 401 ? 'Session expirée : reconnectez-vous.' : "Impossible de charger l'utilisateur connecté.");
      return null;
    } finally {
      this.chargement.set(false);
      this.enCours = null;
    }
  }
