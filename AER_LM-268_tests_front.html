import { ComponentFixture, TestBed } from '@angular/core/testing';
import { FormsModule } from '@angular/forms';
import { NO_ERRORS_SCHEMA } from '@angular/core';
import { of, throwError } from 'rxjs';
import { ConsoleAgenceComponent } from './console-agence.component';
import { OrganizationService } from '../../../core/services/console/organization/organization.service';
import { ReferentielService } from '../../../core/services/console/referentiel/referentiel.service';
import { ApiResponse } from '../../../shared/interfaces/derogation.interface';
import { Page } from '../../../shared/interfaces/habilitation.interface';
import { ZoneDistributionDTO } from '../../../shared/interfaces/referentiel.interface';

describe('ConsoleAgenceComponent — AER_LM-268 (onglet Zone Transporteur)', () => {
  let fixture: ComponentFixture<ConsoleAgenceComponent>;
  let component: ConsoleAgenceComponent;
  let organizationService: jasmine.SpyObj<OrganizationService>;
  let referentielService: jasmine.SpyObj<ReferentielService>;

  const zoneCasa: ZoneDistributionDTO = { id: 1, code: 'CASA', nom: 'Zone Casablanca', actif: true };
  const zoneRabat: ZoneDistributionDTO = { id: 2, code: 'RABAT', nom: 'Zone Rabat-Salé', actif: true };
  const pageZones: Page<ZoneDistributionDTO> = { content: [zoneCasa, zoneRabat], totalPages: 1, totalElements: 2, size: 4, number: 0 };
  const reponseOk = { success: true, message: 'OK', data: null, error: null } as unknown as ApiResponse<string>;

  beforeEach(async () => {
    organizationService = jasmine.createSpyObj<OrganizationService>('OrganizationService', ['getAgencesPages']);
    referentielService = jasmine.createSpyObj<ReferentielService>('ReferentielService',
      ['getZonesDistributionPages', 'createZoneDistribution', 'updateZoneDistribution', 'toggleZoneDistributionActivation']);

    await TestBed.configureTestingModule({
      declarations: [ConsoleAgenceComponent],
      imports: [FormsModule],
      providers: [
        { provide: OrganizationService, useValue: organizationService },
        { provide: ReferentielService, useValue: referentielService }
      ],
      schemas: [NO_ERRORS_SCHEMA]
    }).compileComponents();

    fixture = TestBed.createComponent(ConsoleAgenceComponent);
    component = fixture.componentInstance;
    referentielService.getZonesDistributionPages.and.returnValue(of(pageZones));
  });

  // ---- setActiveTab / loadZonesTransporteur ----

  describe('setActiveTab(\'zonesTransporteur\')', () => {
    it('remet la page à 0, vide la recherche et charge la première page', () => {
      component.currentPageZonesTransporteur = 3;
      component.searchTerm = 'casa';

      component.setActiveTab('zonesTransporteur');

      expect(component.activeTab).toBe('zonesTransporteur');
      expect(component.searchTerm).toBe('');
      expect(referentielService.getZonesDistributionPages).toHaveBeenCalledWith(0, 4);
      expect(component.zonesTransporteur).toEqual([zoneCasa, zoneRabat]);
      expect(component.filteredZonesTransporteur).toEqual([zoneCasa, zoneRabat]);
      expect(component.totalPagesZonesTransporteur).toBe(1);
      expect(component.totalElementsZonesTransporteur).toBe(2);
      expect(component.isLoading).toBeFalse();
    });

    it('affiche un message et coupe le spinner en cas d\'erreur de chargement', () => {
      referentielService.getZonesDistributionPages.and.returnValue(throwError(() => new Error('KO')));

      component.setActiveTab('zonesTransporteur');

      expect(component.errorMessage).toBe('Erreur lors du chargement des zones transporteur');
      expect(component.isLoading).toBeFalse();
    });
  });

  // ---- pagination ----

  describe('onPageChangeZonesTransporteur', () => {
    it('convertit la page 1-based du composant pagination en page 0-based API', () => {
      component.onPageChangeZonesTransporteur(3);

      expect(component.currentPageZonesTransporteur).toBe(2);
      expect(referentielService.getZonesDistributionPages).toHaveBeenCalledWith(2, 4);
    });
  });

  // ---- filtre ----

  describe('applyFilterZonesTransporteur', () => {
    beforeEach(() => component.setActiveTab('zonesTransporteur'));

    it('filtre sur le code ou le libellé (insensible à la casse)', () => {
      component.searchTerm = 'rab';
      component.applyFilter();
      expect(component.filteredZonesTransporteur).toEqual([zoneRabat]);

      component.searchTerm = 'casablanca';
      component.applyFilter();
      expect(component.filteredZonesTransporteur).toEqual([zoneCasa]);
    });

    it('restaure la liste complète quand le terme est vide', () => {
      component.searchTerm = '   ';
      component.applyFilter();
      expect(component.filteredZonesTransporteur).toEqual([zoneCasa, zoneRabat]);
    });
  });

  // ---- toggle ----

  describe('toggleZoneTransporteurStatus', () => {
    beforeEach(() => component.setActiveTab('zonesTransporteur'));

    it('bascule de façon optimiste puis recharge la liste', () => {
      referentielService.toggleZoneDistributionActivation.and.returnValue(of(reponseOk));
      referentielService.getZonesDistributionPages.calls.reset();

      component.toggleZoneTransporteurStatus(zoneCasa);

      expect(referentielService.toggleZoneDistributionActivation).toHaveBeenCalledWith(1);
      expect(referentielService.getZonesDistributionPages).toHaveBeenCalledTimes(1);
    });

    it('revient à l\'état précédent si le serveur échoue', () => {
      referentielService.toggleZoneDistributionActivation.and.returnValue(throwError(() => new Error('KO')));

      component.toggleZoneTransporteurStatus(zoneCasa);

      const casa = component.zonesTransporteur.find(z => z.id === 1)!;
      const casaFiltree = component.filteredZonesTransporteur.find(z => z.id === 1)!;
      expect(casa.actif).toBeTrue();
      expect(casaFiltree.actif).toBeTrue();
    });
  });

  // ---- modale ----

  describe('modale Zone Transporteur', () => {
    it('openAddZoneTransporteurModal ouvre une modale vide en mode création', () => {
      component.openAddZoneTransporteurModal();

      expect(component.isAddZoneTransporteurModalOpen).toBeTrue();
      expect(component.isEditingZoneTransporteur).toBeFalse();
      expect(component.newZoneTransporteur).toEqual({});
      expect(component.errorMessage).toBeNull();
    });

    it('editZoneTransporteur pré-remplit code et libellé', () => {
      component.editZoneTransporteur(zoneRabat);

      expect(component.isEditingZoneTransporteur).toBeTrue();
      expect(component.editingZoneTransporteurId).toBe(2);
      expect(component.newZoneTransporteur).toEqual({ code: 'RABAT', nom: 'Zone Rabat-Salé' });
    });

    it('saveNewZoneTransporteur crée avec code en majuscules et libellé trimé, puis recharge et ferme', () => {
      referentielService.createZoneDistribution.and.returnValue(of({ id: 3, code: 'FES', nom: 'Zone Fès', actif: true }));
      component.openAddZoneTransporteurModal();
      component.newZoneTransporteur = { code: 'fes ', nom: ' Zone Fès ' };

      component.saveNewZoneTransporteur();

      expect(referentielService.createZoneDistribution).toHaveBeenCalledWith({ code: 'FES', nom: 'Zone Fès' });
      expect(referentielService.getZonesDistributionPages).toHaveBeenCalled();
      expect(component.isAddZoneTransporteurModalOpen).toBeFalse();
    });

    it('saveNewZoneTransporteur met à jour avec l\'id en mode édition', () => {
      referentielService.updateZoneDistribution.and.returnValue(of(reponseOk));
      component.editZoneTransporteur(zoneRabat);
      component.newZoneTransporteur.nom = 'Zone Rabat';

      component.saveNewZoneTransporteur();

      expect(referentielService.updateZoneDistribution).toHaveBeenCalledWith({ id: 2, code: 'RABAT', nom: 'Zone Rabat' });
      expect(referentielService.createZoneDistribution).not.toHaveBeenCalled();
      expect(component.isAddZoneTransporteurModalOpen).toBeFalse();
    });

    it('refuse un code ou un libellé vide sans appeler le serveur', () => {
      component.openAddZoneTransporteurModal();
      component.newZoneTransporteur = { code: 'FES', nom: '  ' };

      component.saveNewZoneTransporteur();

      expect(component.errorMessage).toBe('Veuillez remplir tous les champs obligatoires.');
      expect(referentielService.createZoneDistribution).not.toHaveBeenCalled();
    });

    it('affiche le message serveur (409) et laisse la modale ouverte', () => {
      referentielService.createZoneDistribution.and.returnValue(throwError(() => new Error('Une zone de distribution avec ce code existe déjà')));
      component.openAddZoneTransporteurModal();
      component.newZoneTransporteur = { code: 'CASA', nom: 'Doublon' };

      component.saveNewZoneTransporteur();

      expect(component.errorMessage).toBe('Une zone de distribution avec ce code existe déjà');
      expect(component.isAddZoneTransporteurModalOpen).toBeTrue();
    });

    it('closeAddZoneTransporteurModal réinitialise l\'état', () => {
      component.editZoneTransporteur(zoneRabat);

      component.closeAddZoneTransporteurModal();

      expect(component.isAddZoneTransporteurModalOpen).toBeFalse();
      expect(component.isEditingZoneTransporteur).toBeFalse();
      expect(component.editingZoneTransporteurId).toBe(0);
      expect(component.newZoneTransporteur).toEqual({});
    });
  });

  // ---- saisie du code ----

  describe('onCodeZoneTransporteurInput', () => {
    it('force les majuscules et retire les caractères non autorisés', () => {
      const input = document.createElement('input');
      input.value = 'fès meknes!';

      component.onCodeZoneTransporteurInput({ target: input } as unknown as Event);

      expect(input.value).toBe('FSMEKNES');
      expect(component.newZoneTransporteur.code).toBe('FSMEKNES');
    });
  });

  // ---- track by ----

  it('trackByZoneTransporteurId renvoie l\'id', () => {
    expect(component.trackByZoneTransporteurId(0, zoneCasa)).toBe(1);
  });
});
