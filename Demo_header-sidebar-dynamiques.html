import { environment } from '../../../environments/environment';

/**
 * Sorties vers le portail QlickFlow (shell). Un seul endroit pour les deux liens de sortie
 * (barre du haut et menu latéral), en attendant que le shell micro-frontend prenne la main.
 */

/** « Accueil » : dashboard du portail. */
export function allerAuPortail(): void {
  window.location.href = environment.portailUrl;
}

/** « Déconnexion » : page de connexion du portail. */
export function allerALaConnexion(): void {
  window.location.href = `${environment.portailUrl}/login`;
}

/** Initiales affichées dans l'avatar : « Youssef Rouane » → YR ; sans prénom/nom, les 2 premiers caractères de l'uid. */
export function initialesDe(nomComplet: string | null | undefined, uid: string): string {
  const nom = (nomComplet ?? '').trim();
  if (nom && nom !== uid) {
    const mots = nom.split(/\s+/).filter(Boolean);
    return mots.length >= 2
      ? (mots[0][0] + mots[mots.length - 1][0]).toUpperCase()
      : mots[0].slice(0, 2).toUpperCase();
  }
  return uid.slice(0, 2).toUpperCase();
}

import { NgIf } from '@angular/common';
import { ChangeDetectionStrategy, Component, HostListener, OnInit, computed, inject, signal } from '@angular/core';
import { toSignal } from '@angular/core/rxjs-interop';
import { NavigationEnd, Router } from '@angular/router';
import { filter, map, startWith } from 'rxjs';
import { ContexteUtilisateurService } from '../../../../features/fund-management/adapters/apiAdapters/contexte-utilisateur.service';
import { allerALaConnexion, allerAuPortail, initialesDe } from '../../portail';

const PREFIXE_CAISSE_CENTRALE = '/caisse-centrale';

/**
 * Barre du haut : titre selon l'écran (Vue Agence / Vue Caisse Centrale), « Accueil » vers le portail,
 * avatar aux initiales de l'utilisateur connecté (/me) avec menu « Connecté : uid » / « Déconnexion ».
 * Rien n'est en dur : tout vient de ContexteUtilisateurService et de l'URL courante.
 */
@Component({
  selector: 'app-header',
  standalone: true,
  imports: [NgIf],
  templateUrl: './header.component.html',
  styleUrl: './header.component.css',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class HeaderComponent implements OnInit {

  readonly contexteUtilisateur = inject(ContexteUtilisateurService);
  private readonly router = inject(Router);

  /** URL courante sous forme de signal : le titre bascule quand on entre / sort de l'écran Caisse Centrale. */
  private readonly url = toSignal(
    this.router.events.pipe(
      filter((e): e is NavigationEnd => e instanceof NavigationEnd),
      map(e => e.urlAfterRedirects),
      startWith(this.router.url)
    ),
    { requireSync: true }
  );

  readonly titre = computed(() =>
    this.url().includes(PREFIXE_CAISSE_CENTRALE) ? 'Gestion des fonds - Vue Caisse Centrale' : 'Gestion des fonds - Vue Agence');

  readonly initiales = computed(() => {
    const ctx = this.contexteUtilisateur.contexte();
    return ctx ? initialesDe(ctx.nomComplet, ctx.uid) : '…';
  });

  readonly menuOuvert = signal(false);
  readonly logoAbsent = signal(false);        // assets/logo-qlickflow.svg manquant → pas d'image cassée

  ngOnInit(): void {
    this.contexteUtilisateur.charger();        // idempotent
  }

  basculerMenu(evenement: Event): void {
    evenement.stopPropagation();
    this.menuOuvert.update(ouvert => !ouvert);
  }

  /** Un clic n'importe où ailleurs referme le menu. */
  @HostListener('document:click')
  fermerMenu(): void {
    this.menuOuvert.set(false);
  }

  navigateToHome(): void {
    allerAuPortail();
  }

  seDeconnecter(): void {
    allerALaConnexion();
  }
}
