import { Component, OnInit, computed, inject } from '@angular/core';
import { toSignal } from '@angular/core/rxjs-interop';
import { NavigationEnd, Router, RouterOutlet } from '@angular/router';
import { MatMenuModule } from '@angular/material/menu';
import { MatIconModule } from '@angular/material/icon';
import { filter, map, startWith } from 'rxjs';
import { SidebarComponent } from './shared/presenter/components/sidebar/sidebar.component';
import { ContexteUtilisateurService } from './features/fund-management/adapters/apiAdapters/contexte-utilisateur.service';
import { allerALaConnexion, initialesDe } from './shared/presenter/portail';

const PREFIXE_CAISSE_CENTRALE = '/caisse-centrale';

/**
 * Shell de l'application : menu latéral + barre du haut + router-outlet.
 * La barre du haut n'a plus rien en dur : titre selon le profil / l'écran, initiales et uid de /me, Déconnexion vers le portail.
 */
@Component({
  selector: 'app-root',
  standalone: true,
  imports: [RouterOutlet, MatMenuModule, MatIconModule, SidebarComponent],
  templateUrl: './app.component.html',
  styleUrl: './app.component.css'
})
export class AppComponent implements OnInit {

  readonly contexteUtilisateur = inject(ContexteUtilisateurService);
  private readonly router = inject(Router);

  /** URL courante sous forme de signal (valeur initiale synchrone grâce à startWith). */
  private readonly url = toSignal(
    this.router.events.pipe(
      filter((e): e is NavigationEnd => e instanceof NavigationEnd),
      map(e => e.urlAfterRedirects),
      startWith(this.router.url)
    ),
    { requireSync: true }
  );

  /**
   * « Vue Caisse Centrale » pour un profil toutes agences sans agence de rattachement (Caisse Centrale, Support, Super Admin)
   * ou dès que l'on est sur l'écran /caisse-centrale ; « Vue Agence » sinon.
   */
  readonly titre = computed(() => {
    const ctx = this.contexteUtilisateur.contexte();
    const profilCentral = !!ctx && !ctx.agenceCode && ctx.habilitations.toutesAgences;
    const ecranCentral = this.url().includes(PREFIXE_CAISSE_CENTRALE);
    return profilCentral || ecranCentral ? 'Gestion des fonds - Vue Caisse Centrale' : 'Gestion des fonds - Vue Agence';
  });

  /** « Youssef Rouane » → YR ; sans prénom/nom → 2 premiers caractères de l'uid ; « … » tant que /me n'a pas répondu. */
  readonly utilisateurInitiales = computed(() => {
    const ctx = this.contexteUtilisateur.contexte();
    return ctx ? initialesDe(ctx.nomComplet, ctx.uid) : '…';
  });

  readonly utilisateurUid = computed(() => this.contexteUtilisateur.contexte()?.uid ?? '…');
  readonly utilisateurNom = computed(() => this.contexteUtilisateur.contexte()?.nomComplet ?? '');

  ngOnInit(): void {
    this.contexteUtilisateur.charger();   // idempotent : partagé avec le menu latéral et les gardes de route
  }

  seDeconnecter(): void {
    allerALaConnexion();
  }
}
