/**
 * AER_LM-280 — nomenclature de la référence : AAAAMMJJ-CodeAgence-Séquence, éléments séparés par un tiret.
 * La séquence est sur 2 chiffres minimum (01, 02, …) ; au-delà de 99 elle continue sur 3 chiffres.
 */
public final class ReferenceDemande {

    private static final DateTimeFormatter JOUR = DateTimeFormatter.BASIC_ISO_DATE;

    private ReferenceDemande() {}

    public static String formater(LocalDate jour, String codeAgence, int numero) {
        if (codeAgence == null || codeAgence.isBlank()) {
            throw new RetailValidationException("Le code agence est obligatoire pour numéroter la demande");
        }
        if (numero <= 0) {
            throw new IllegalArgumentException("La séquence d'une référence commence à 1");
        }
        return "%s-%s-%02d".formatted(jour.format(JOUR), codeAgence.trim(), numero);
    }
}

    if (op.operationType === 'CASH_DESK')  { return 'CAISSE'; }

  });
  return [...new Set(chips)];
}
