0. Cause et correction

Ta console le dit : HttpMessageNotWritableException: No converter for [class [B] with preset Content-Type 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'.
Le fichier Excel est bien généré. C'est au moment de l'envoyer que Spring échoue : dans utilisateurs-ci, aucun convertisseur ne sait écrire un byte[] dans la réponse (dans fund-management, le même ResponseEntity<byte[]> fonctionne).
Correction : le contrôleur écrit lui-même le fichier dans la réponse HTTP. Plus aucun convertisseur n'entre en jeu, et la configuration web d'utilisateurs-ci n'est pas touchée.
2 fichiers : le contrôleur (4 éditions) et sa classe de test (à remplacer en entier). Rien côté front, service ni façade.

package com.bnpp.irb.qlickflow.admin.controller.publicapi.gab;

import com.bnpp.irb.qlickflow.admin.facade.GabHorsSiteFacade;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.Mock;
import org.mockito.Mockito;
import org.mockito.junit.jupiter.MockitoExtension;
import org.springframework.http.HttpHeaders;
import org.springframework.mock.web.MockHttpServletResponse;

import java.io.IOException;

import static org.junit.jupiter.api.Assertions.assertArrayEquals;
import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

@ExtendWith(MockitoExtension.class)
class GabHorsSiteControllerExportTest {

    @Mock GabHorsSiteFacade gabHorsSiteFacade;

    @InjectMocks GabHorsSiteController controller;

    @Test
    @DisplayName("GET /export : 200, fichier xlsx écrit dans la réponse, en pièce jointe horodatée")
    void export_ecritLeFichierEnPieceJointe() throws IOException {
        byte[] fichier = {1, 2, 3};
        Mockito.when(gabHorsSiteFacade.exporterGabsHorsSite()).thenReturn(fichier);
        MockHttpServletResponse reponse = new MockHttpServletResponse();

        controller.exportGabsHorsSite(reponse);

        assertEquals(200, reponse.getStatus());
        assertArrayEquals(fichier, reponse.getContentAsByteArray());
        assertEquals(GabHorsSiteController.CONTENT_TYPE_XLSX, reponse.getContentType());
        assertEquals(fichier.length, reponse.getContentLength());
        String disposition = reponse.getHeader(HttpHeaders.CONTENT_DISPOSITION);
        assertTrue(disposition.matches("attachment; filename=\"QlickFlow_GAB-Hors-Site_extraction_\\d{8}-\\d{6}\\.xlsx\""));
    }
}
