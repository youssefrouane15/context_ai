/**
 * AER_LM-279 — lecture d'une demande pour l'extraction. Tout accès aux DTO est ici.
 * Structure réelle (captures du 28/09) :
 *   CashOperationDTO { operationType: OperationType (ATM | CASH_DESK | CURRENCY), operationSens: RequestType (SUPPLY | COLLECTION),
 *                      isUrgent, currencyDetails: List<OperationCurrencyDetailDTO> }
 *   OperationCurrencyDetailDTO { currency: CurrencyDTO, totalAmountByCurrency, operationDenominations: List<OperationDenominationDTO> }
 *   OperationDenominationDTO { denomination: DenominationDTO, quantity }
 */
final class ExtractionDemande {

    enum Alimentation { CAISSE, GAB }
    enum Sens { RAMASSAGE, COMMANDE }
    record LigneDevise(String devise, BigDecimal montant) {}

    private static final Map<String, String> LIBELLES_STATUT = Map.of(
            "SUBMITTED", "En Attente", "IN_MODIFICATION", "En modification",
            "VALIDATED", "Traitée", "CANCELLED", "Annulée", "INITIATED", "Brouillon");

    private final CashRequestDTO d;

    ExtractionDemande(CashRequestDTO d) { this.d = d; }

    String reference()          { return d.getUniqueRequestRef(); }
    boolean urgente()           { return Boolean.TRUE.equals(d.getUrgent())
                                        || operations().stream().anyMatch(op -> Boolean.TRUE.equals(op.getIsUrgent())); }
    String codeAgence()         { return d.getBranchCode(); }
    String nomAgence()          { return d.getAgenceNom(); }
    String zone()               { return d.getZoneNom() != null ? d.getZoneNom() : d.getZoneCode(); }
    String transporteur()       { return d.getPrestataireNom() != null ? d.getPrestataireNom() : d.getPrestataire(); }
    BigDecimal coutDeplacement(){ return d.getCoutDeplacement(); }
    LocalDateTime dateDemande() { return d.getCreatedAt(); }
    String libelleStatut()      { return LIBELLES_STATUT.getOrDefault(d.getRequestStatus(), d.getRequestStatus()); }

    private List<CashOperationDTO> operations() { return d.getOperations() == null ? List.of() : d.getOperations(); }

    private static List<OperationCurrencyDetailDTO> details(CashOperationDTO op) {
        return op.getCurrencyDetails() == null ? List.of() : op.getCurrencyDetails();
    }

    boolean aUneAlimentation() {
        return operations().stream().anyMatch(op -> estAlimentation(op, Alimentation.CAISSE) || estAlimentation(op, Alimentation.GAB));
    }

    private static boolean estAlimentation(CashOperationDTO op, Alimentation type) {
        OperationType attendu = type == Alimentation.CAISSE ? OperationType.CASH_DESK : OperationType.ATM;
        return op.getOperationType() == attendu && op.getOperationSens() == RequestType.SUPPLY;
    }

    /** Quantités par valeur faciale (200, 100, … 0.05), cumulées sur toutes les opérations et devises du même type. */
    Map<BigDecimal, Integer> quantitesParFaceValue(Alimentation type, Map<String, BigDecimal> faceValuesByCode) {
        Map<BigDecimal, Integer> resultat = new HashMap<>();
        for (CashOperationDTO op : operations()) {
            if (!estAlimentation(op, type)) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (detail.getOperationDenominations() == null) continue;
                for (OperationDenominationDTO ligne : detail.getOperationDenominations()) {
                    if (ligne.getDenomination() == null || ligne.getQuantity() == null) continue;
                    BigDecimal faceValue = faceValuesByCode.get(ligne.getDenomination().getDenominationCode());
                    if (faceValue == null) continue;
                    resultat.merge(faceValue.stripTrailingZeros(), ligne.getQuantity(), Integer::sum);
                }
            }
        }
        return resultat;
    }

    /** Ramassages (CURRENCY / COLLECTION) ou commandes de devises (CURRENCY / SUPPLY) : une ligne par devise. */
    List<LigneDevise> lignesDevise(Sens sens) {
        RequestType sensAttendu = sens == Sens.RAMASSAGE ? RequestType.COLLECTION : RequestType.SUPPLY;
        List<LigneDevise> lignes = new ArrayList<>();
        for (CashOperationDTO op : operations()) {
            if (op.getOperationType() != OperationType.CURRENCY || op.getOperationSens() != sensAttendu) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                String devise = detail.getCurrency() == null ? null : detail.getCurrency().getIsoCode();
                lignes.add(new LigneDevise(devise, detail.getTotalAmountByCurrency()));
            }
        }
        return lignes;
    }
}

/**
 * AER_LM-279 — remplit le template « GesFond - Reporting Excel » (3 onglets) avec les demandes du périmètre.
 * Le template est la référence de mise en forme : en-têtes, groupes fusionnés, largeurs, et l'ordre des coupures
 * (lu dans la ligne d'en-tête, colonnes « Alimentation Caisse » puis « Alimentation GAB »). Le code ne dessine rien.
 */
@Component
public class ExportExcelDemandes {

    public static final String CONTENT_TYPE = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
    static final String TEMPLATE = "export/GesFond-Reporting.xlsx";
    static final DateTimeFormatter HORODATAGE = DateTimeFormatter.ofPattern("yyyyMMdd-HHmmss");
    static final DateTimeFormatter DATE_DEMANDE = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    /** Onglets du template, dans l'ordre. */
    static final int ONGLET_ALIMENTATIONS = 0, ONGLET_RAMASSAGES = 1, ONGLET_DEVISES = 2;
    /** Ligne des libellés de colonnes (0-based) et première ligne de données, identiques sur les 3 onglets. */
    static final int LIGNE_ENTETE = 1, PREMIERE_LIGNE = 2;
    /** Colonnes communes A→I : Référence, Urgence, Code Agence, Agence Émettrice, Zone, Transporteur, PU, Date Demande, Statut. */
    static final int NB_COLONNES_COMMUNES = 9;

    public static String nomDeFichier(LocalDateTime instant) {
        return "QlickFlow_GesFond_extraction_" + HORODATAGE.format(instant) + ".xlsx";
    }

    public byte[] generer(ExportDemandesDTO export) {
        try (InputStream template = new ClassPathResource(TEMPLATE).getInputStream();
             Workbook classeur = new XSSFWorkbook(template);
             ByteArrayOutputStream sortie = new ByteArrayOutputStream()) {

            remplirAlimentations(classeur.getSheetAt(ONGLET_ALIMENTATIONS), export);
            remplirLignesDevise(classeur.getSheetAt(ONGLET_RAMASSAGES), export.demandes(), ExtractionDemande.Sens.RAMASSAGE);
            remplirLignesDevise(classeur.getSheetAt(ONGLET_DEVISES), export.demandes(), ExtractionDemande.Sens.COMMANDE);

            classeur.write(sortie);
            return sortie.toByteArray();
        } catch (IOException e) {
            throw new UncheckedIOException("Génération de l'extraction Excel impossible", e);
        }
    }

    // ---------------------------------------------------------------- onglet 1 : une ligne par demande

    private void remplirAlimentations(Sheet onglet, ExportDemandesDTO export) {
        Row entete = onglet.getRow(LIGNE_ENTETE);
        // Valeurs faciales lues dans l'en-tête : J..U = caisse (200 → 0,05), V.. = GAB (200, 100, …)
        List<Integer> colonnesCaisse = new ArrayList<>(), colonnesGab = new ArrayList<>();
        Map<Integer, BigDecimal> faceValueParColonne = new HashMap<>();
        int debutGab = premiereColonneGab(onglet);
        for (int c = NB_COLONNES_COMMUNES; c < entete.getLastCellNum(); c++) {
            Cell cellule = entete.getCell(c);
            if (cellule == null || cellule.getCellType() != CellType.NUMERIC) continue;
            faceValueParColonne.put(c, BigDecimal.valueOf(cellule.getNumericCellValue()));
            (c < debutGab ? colonnesCaisse : colonnesGab).add(c);
        }
        Map<String, BigDecimal> faceValues = export.faceValuesByCode();

        int ligne = PREMIERE_LIGNE;
        Row modele = modeleDeLigne(onglet);
        for (CashRequestDTO d : export.demandes()) {
            ExtractionDemande x = new ExtractionDemande(d);
            if (!x.aUneAlimentation()) continue;
            Row r = ligne(onglet, ligne++, modele);
            ecrireColonnesCommunes(r, x);
            Map<BigDecimal, Integer> caisse = x.quantitesParFaceValue(ExtractionDemande.Alimentation.CAISSE, faceValues);
            Map<BigDecimal, Integer> gab    = x.quantitesParFaceValue(ExtractionDemande.Alimentation.GAB, faceValues);
            for (int c : colonnesCaisse) ecrireQuantite(r, c, caisse.get(faceValueParColonne.get(c).stripTrailingZeros()));
            for (int c : colonnesGab)    ecrireQuantite(r, c, gab.get(faceValueParColonne.get(c).stripTrailingZeros()));
        }
        supprimerLignesRestantes(onglet, ligne);
    }

    /** Colonne où commence le groupe « Alimentation GAB » (cellule fusionnée de la ligne 1) ; défaut : après les 12 coupures caisse. */
    private int premiereColonneGab(Sheet onglet) {
        Row groupes = onglet.getRow(0);
        if (groupes != null) {
            for (int c = NB_COLONNES_COMMUNES; c < groupes.getLastCellNum(); c++) {
                Cell cellule = groupes.getCell(c);
                if (cellule != null && cellule.getCellType() == CellType.STRING
                        && cellule.getStringCellValue().toLowerCase().contains("gab")) {
                    return c;
                }
            }
        }
        return NB_COLONNES_COMMUNES + 12;
    }

    // ---------------------------------------------------------------- onglets 2 et 3 : une ligne par devise

    private void remplirLignesDevise(Sheet onglet, List<CashRequestDTO> demandes, ExtractionDemande.Sens sens) {
        int ligne = PREMIERE_LIGNE;
        Row modele = modeleDeLigne(onglet);
        for (CashRequestDTO d : demandes) {
            ExtractionDemande x = new ExtractionDemande(d);
            for (ExtractionDemande.LigneDevise ld : x.lignesDevise(sens)) {
                Row r = ligne(onglet, ligne++, modele);
                ecrireColonnesCommunes(r, x);
                texte(r, NB_COLONNES_COMMUNES, ld.devise());
                nombre(r, NB_COLONNES_COMMUNES + 1, ld.montant());
            }
        }
        supprimerLignesRestantes(onglet, ligne);
    }

    // ---------------------------------------------------------------- écriture

    private void ecrireColonnesCommunes(Row r, ExtractionDemande x) {
        texte(r, 0, x.reference());
        texte(r, 1, x.urgente() ? "Oui" : "Non");
        texte(r, 2, x.codeAgence());
        texte(r, 3, x.nomAgence());
        texte(r, 4, x.zone());
        texte(r, 5, x.transporteur());
        nombre(r, 6, x.coutDeplacement());
        texte(r, 7, x.dateDemande() == null ? "" : DATE_DEMANDE.format(x.dateDemande()));
        texte(r, 8, x.libelleStatut());
    }

    private static void ecrireQuantite(Row r, int col, Integer quantite) {
        if (quantite != null && quantite > 0) nombre(r, col, BigDecimal.valueOf(quantite));
    }

    private static void texte(Row r, int col, String valeur) {
        Cell c = r.getCell(col, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
        c.setCellValue(valeur == null ? "" : valeur);
    }

    private static void nombre(Row r, int col, BigDecimal valeur) {
        Cell c = r.getCell(col, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
        if (valeur == null) c.setBlank(); else c.setCellValue(valeur.doubleValue());
    }

    /**
     * Le template porte une ou deux lignes d'exemple par onglet : tout ce qui suit la dernière ligne écrite est supprimé,
     * sinon un exemple resterait dans l'extraction (et compterait comme une demande).
     */
    private static void supprimerLignesRestantes(Sheet onglet, int aPartirDe) {
        for (int i = onglet.getLastRowNum(); i >= aPartirDe; i--) {
            Row r = onglet.getRow(i);
            if (r != null) onglet.removeRow(r);
        }
    }

    /** La ligne d'exemple du template (ligne 3) sert de modèle de styles, puis est réécrite par la première demande. */
    private static Row modeleDeLigne(Sheet onglet) {
        Row r = onglet.getRow(PREMIERE_LIGNE);
        return r != null ? r : onglet.createRow(PREMIERE_LIGNE);
    }

    private static Row ligne(Sheet onglet, int index, Row modele) {
        Row r = onglet.getRow(index);
        if (r == null) r = onglet.createRow(index);
        for (int c = 0; c < modele.getLastCellNum(); c++) {
            Cell source = modele.getCell(c);
            Cell cible = r.getCell(c, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
            cible.setBlank();
            CellStyle style = source != null ? source.getCellStyle() : null;
            if (style != null) cible.setCellStyle(style);
        }
        return r;
    }
}

    private static int valeurSousEntete(Sheet onglet, double faceValue, boolean caisse) {
        Row entete = onglet.getRow(1);
        int debutGab = -1;
        Row groupes = onglet.getRow(0);
        for (int c = 9; c < groupes.getLastCellNum(); c++) {
            if (groupes.getCell(c) != null && groupes.getCell(c).getCellType() == org.apache.poi.ss.usermodel.CellType.STRING
                    && groupes.getCell(c).getStringCellValue().toLowerCase().contains("gab")) { debutGab = c; break; }
        }
        for (int c = 9; c < entete.getLastCellNum(); c++) {
            var cellule = entete.getCell(c);
            if (cellule == null || cellule.getCellType() != org.apache.poi.ss.usermodel.CellType.NUMERIC) continue;
            boolean colonneCaisse = c < debutGab;
            if (cellule.getNumericCellValue() == faceValue && colonneCaisse == caisse) {
                var v = onglet.getRow(2).getCell(c);
                return v == null || v.getCellType() != org.apache.poi.ss.usermodel.CellType.NUMERIC ? 0 : (int) v.getNumericCellValue();
            }
        }
        throw new AssertionError("Colonne " + faceValue + " absente de l'en-tête du template");
    }

export function telechargerBlob(blob: Blob, nom: string): void {
  const url = URL.createObjectURL(blob);
