import { CommonModule } from '@angular/common';
import { ChangeDetectionStrategy, Component, EventEmitter, Input, Output } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { StatutDemande, classeStatut, libelleStatut } from '../../../domain/domain-core/entities/statut-demande';
import { BanniereLecture, iconeStatutEcran } from './demande-entete.regles';

/**
 * AER_LM-285 / 287 — en-tête commun des écrans de consultation (agence) et de traitement (Caisse Centrale) :
 * lien de retour, bannière de lecture seule, pastille de statut, référence, « Soumise le … par … »,
 * « Traitée / Annulée le … par … », motif d'annulation, bouton-badge « Mode Modification » (bascule) / « Mode Modification actif »,
 * badge / case « DEMANDE URGENTE ».
 * Composant de présentation pur : aucune injection, tout vient des entrées.
 */
@Component({
  selector: 'app-demande-entete',
  standalone: true,
  imports: [CommonModule, MatIconModule],
  templateUrl: './demande-entete.component.html',
  styleUrl: './demande-entete.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DemandeEnteteComponent {

  @Input({ required: true }) reference!: string;
  @Input({ required: true }) statut!: StatutDemande;

  /** Urgence : badge en lecture ; case à cocher si urgentModifiable (mode modification agence). */
  @Input() urgent = false;
  @Input() urgentModifiable = false;
  @Output() urgentChange = new EventEmitter<boolean>();

  /** Badge « Mode Modification actif » (modification en cours). */
  @Input() modeModification = false;
  /** Consultation d'une demande modifiable : le badge devient le bouton qui active le mode modification (maquette 287). */
  @Input() modificationActivable = false;
  @Output() activerModification = new EventEmitter<void>();

  @Input() soumiseLe: string | null = null;
  @Input() soumisePar: string | null = null;
  /** Entre parenthèses après le nom : « BackOffice » (profil, écran agence) ou « Agence 1001 - Nations Unies » (Caisse Centrale). */
  @Input() soumiseParDetail: string | null = null;

  @Input() decisionLe: string | null = null;
  @Input() decisionPar: string | null = null;
  @Input() decisionParDetail: string | null = null;
  @Input() motifAnnulation: string | null = null;

  @Input() banniere: BanniereLecture | null = null;

  @Input() lienRetour = 'Retour à l\'Historique des Demandes';
  @Output() retour = new EventEmitter<void>();

  get libelleStatut(): string { return libelleStatut(this.statut); }
  get classeStatut(): string { return classeStatut(this.statut); }
  get iconeStatut(): string { return iconeStatutEcran(this.statut); }
  get libelleDecision(): string { return this.statut === 'TRAITEE' ? 'Traitée' : 'Annulée'; }

  basculerUrgent(evt: Event): void {
    this.urgentChange.emit((evt.target as HTMLInputElement).checked);
  }
}
