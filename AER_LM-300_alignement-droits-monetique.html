-- V25 — AER_LM-300 : alignement des droits monétique sur la Jira.
--   Pôle opérations monétique : consultation + gestion des opérations monétiques
--   Caisse centrale            : consultation + traitement demande monétique
--   Support                    : consultation
-- Nouvelle permission GDF_CONSULTER_DEMANDE_MONETIQUE (enum Java Permission, seule source de vérité — pas de CHECK).
-- Le rôle « GDF Opérations monétique » perd GDF_CONSULTER_DEMANDE et GDF_TOUTES_AGENCES (droits agence attribués par erreur en V22).

-- 1. Consultation des demandes monétique : Pôle (rôle par nom), Caisse Centrale (11), Support (12)
insert into hab_role_permission (role_id, permission)
select r.id, 'GDF_CONSULTER_DEMANDE_MONETIQUE'
  from hab_roles r
 where (r.name = 'GDF Opérations monétique' or r.id in (11, 12))
   and not exists (select 1 from hab_role_permission x where x.role_id = r.id and x.permission = 'GDF_CONSULTER_DEMANDE_MONETIQUE');

-- 2. Retrait des droits agence du Pôle
delete from hab_role_permission x
 using hab_roles r
 where r.id = x.role_id
   and r.name = 'GDF Opérations monétique'
   and x.permission in ('GDF_CONSULTER_DEMANDE', 'GDF_TOUTES_AGENCES');
