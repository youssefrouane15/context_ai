3Port FundService + implémentation

Le contrat d'abord — même fichier que getDemandes() / annulerDemande() :

Puis l'implémentation dans le service HTTP. Pas d'appel réseau ici — la génération de fichier est de la technique navigateur, et c'est précisément le rôle de cette couche :

adapters\apiAdapters\fundManagement-http.service.ts — AJOUTERMODIFIER
telechargerDemande(d: DemandeFonds): void {
  const csv = this.genererCsvDemande(d);

  // BOM UTF-8 : indispensable pour les accents dans Excel
  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });

  const url = window.URL.createObjectURL(blob);
  const lien = document.createElement('a');
  lien.href = url;
  lien.download = `demande-fonds-${d.id}.csv`;
  lien.click();
  window.URL.revokeObjectURL(url);
}

private genererCsvDemande(d: DemandeFonds): string {
  const operations = d.operations ?? [];

  // Types d'opérations : même logique que les chips de la liste
  const types = [...new Set(
    operations.map(op => op.operationType === 'CURRENCY' ? 'DEVISE' : op.operationType)
  )].join(', ');

  const lignes = [
    ['Référence', 'Statut', "Types d'opérations", "Nombre d'opérations"].join(';'),
    [
      this.champCsv(d.id),
      this.champCsv(this.libelleStatutExport(d.requestStatus)),
      this.champCsv(types),
      this.champCsv(operations.length)
    ].join(';')
  ];

  return lignes.join('\r\n');
}

/** Échappement CSV : null-safe, guillemets doublés si nécessaire. */
private champCsv(valeur: unknown): string {
  if (valeur === null || valeur === undefined) { return ''; }
  const s = String(valeur);
  return /[;"\r\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
}

/** Statuts techniques → libellés FR (aligné sur libelleStatut du composant). */
private libelleStatutExport(s: string): string {
  const map: Record<string, string> = {
    INITIATED: 'En Attente',
    VALIDATED: 'Validé',
    CREATED: 'Validé',
    CANCELLED: 'Annulée'
  };
  return map[s] ?? s;
}
