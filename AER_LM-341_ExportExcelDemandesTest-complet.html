package com.bmci.workflow.fund.management.application.rest.retail.export;

import com.bmci.common.workflow.fund.management.application.rest.dto.CurrencyDTO;
import com.bmci.common.workflow.fund.management.application.rest.dto.DenominationDTO;
import com.bmci.q360.domain.retail.enums.OperationType;
import com.bmci.q360.domain.retail.enums.RequestType;
import com.bmci.q360.domain.retail.enums.TypeGab;
import com.bmci.workflow.fund.management.service.domain.retail.dto.*;
import org.apache.poi.ss.usermodel.Cell;
import org.apache.poi.ss.usermodel.CellType;
import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.ss.usermodel.Workbook;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;

import java.io.ByteArrayInputStream;

import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;

import static org.assertj.core.api.Assertions.assertThat;

class ExportExcelDemandesTest {

    private final ExportExcelDemandes export = new ExportExcelDemandes();

    private static final Map<String, BigDecimal> FACE_VALUES = Map.of("BILLET_200", new BigDecimal("200"), "BILLET_100", new BigDecimal("100"),
            "PIECE_10", new BigDecimal("10"), "PIECE_0_05", new BigDecimal("0.05"));

    // ---- fixtures
    private static CashRequestDTO demande(String ref, CashOperationDTO... ops) {
        return CashRequestDTO.builder().uniqueRequestRef(ref).branchCode("1001").agenceNom("Nations Unies").zoneNom("Zone Casablanca")
                .prestataireNom("Brink's").coutDeplacement(new BigDecimal("350.50")).createdAt(LocalDateTime.of(2026, 10, 26, 9, 0))
                .requestStatus("SUBMITTED").urgent(false).operations(List.of(ops)).build();
    }

    private static CashOperationDTO operation(OperationType type, RequestType sens, OperationCurrencyDetailDTO... details) {
        return CashOperationDTO.builder().operationType(type).operationSens(sens).currencyDetails(List.of(details)).build();
    }

    private static OperationCurrencyDetailDTO detail(String iso, String montant, OperationDenominationDTO... denoms) {
        return OperationCurrencyDetailDTO.builder().currency(CurrencyDTO.builder().isoCode(iso).build())
                .totalAmountByCurrency(new BigDecimal(montant)).operationDenominations(List.of(denoms)).build();
    }

    private static OperationDenominationDTO coupure(String code, int qte) {
        return OperationDenominationDTO.builder().denomination(DenominationDTO.builder().denominationCode(code).build()).quantity(qte).build();
    }

    private static CashRequestDTO demandeGab(String ref, CashOperationDTO... ops) {
        CashRequestDTO d = demande(ref, ops);
        d.setCanalDemande("MONETIQUE");
        d.setBranchCode("9000");
        return d;
    }

    private static CashOperationDTO operationGab(String gabId, String nom, TypeGab type, OperationDenominationDTO... cassettes) {
        OperationGabDTO gab = new OperationGabDTO();
        gab.setGabId(gabId); gab.setGabNom(nom); gab.setGabType(type); gab.setVille("CASABLANCA");
        gab.setAgenceCode("1120"); gab.setAgenceLibelle("CASA VAL D'ANFA"); gab.setNumeroCompte("0112081010030370");
        gab.setTransporteur("Brink's"); gab.setTarifTransporteur(new BigDecimal("450"));
        CashOperationDTO op = operation(OperationType.ATM, RequestType.SUPPLY, detail("MAD", "0", cassettes));
        op.setGab(gab);
        return op;
    }

    private static OperationDenominationDTO cassette(int numero, String code, int qte) {
        OperationDenominationDTO c = coupure(code, qte);
        c.setCassetteNumber(numero);
        return c;
    }

    private static Workbook lire(byte[] fichier) throws Exception { return new XSSFWorkbook(new ByteArrayInputStream(fichier)); }
    private static double nombre(Row r, int col) { return r.getCell(col).getNumericCellValue(); }
    private static String texte(Row r, int col) { Cell c = r.getCell(col); return c == null ? "" : c.getStringCellValue(); }

    @Test
    @DisplayName("Feuille MAD : une ligne par demande, quantités caisse / GAB dans les colonnes de la coupure, ramassage MAD en Z, formules de total recopiées")
    void feuilleMad() throws Exception {
        CashRequestDTO d = demande("20261026-1001-02",
                operation(OperationType.CASH_DESK, RequestType.SUPPLY, detail("MAD", "2100", coupure("BILLET_200", 10), coupure("PIECE_10", 10))),
                operation(OperationType.ATM, RequestType.SUPPLY, detail("MAD", "5000", coupure("BILLET_100", 50))),
                operation(OperationType.CASH_DESK, RequestType.COLLECTION, detail("MAD", "12000.50")));

        byte[] fichier = export.generer(new ExportDemandesDTO(List.of(d), FACE_VALUES));

        Sheet mad = lire(fichier).getSheetAt(ExportExcelDemandes.ONGLET_MAD);
        Row r = mad.getRow(ExportExcelDemandes.PREMIERE_LIGNE);
        assertThat(texte(r, 0)).isEqualTo("20261026-1001-02");
        assertThat(texte(r, 7)).isEqualTo("26/10/2026");
        assertThat(nombre(r, 6)).isEqualTo(350.50);
        assertThat(nombre(r, 9)).isEqualTo(10);                                   // J = 200 caisse
        assertThat(nombre(r, 13)).isEqualTo(10);                                  // N = 10 caisse
        assertThat(r.getCell(21).getCellType()).isEqualTo(CellType.FORMULA);     // V = Total Alim. Caisse
        assertThat(r.getCell(21).getCellFormula()).contains("J3");
        assertThat(nombre(r, 23)).isEqualTo(50);                                  // X = 100 GAB
        assertThat(r.getCell(24).getCellType()).isEqualTo(CellType.FORMULA);     // Y = Total Alim. GAB
        assertThat(nombre(r, 25)).isEqualTo(12000.50);                            // Z = ramassage MAD
        assertThat(r.getCell(25).getCellStyle().getDataFormatString()).isEqualTo(ExportExcelDemandes.FORMAT_MONTANT);
        Row ligneSuivante = mad.getRow(ExportExcelDemandes.PREMIERE_LIGNE + 1);   // ligne 4 : absente (template élagué) ou vide
        assertThat(ligneSuivante == null || ligneSuivante.getCell(0) == null || ligneSuivante.getCell(0).getCellType() == CellType.BLANK).isTrue();
        assertThat(r.getCell(19) == null || r.getCell(19).getCellType() == CellType.BLANK).isTrue();   // le « 1 » d'exemple en T3 a été écrasé
    }

    @Test
    @DisplayName("Feuille Devise : une ligne par devise triée par code, alimentation et ramassage côte à côte, MAD exclu")
    void feuilleDevise() throws Exception {
        CashRequestDTO d = demande("20261026-1001-03",
                operation(OperationType.CURRENCY, RequestType.SUPPLY, detail("USD", "1000"), detail("EUR", "500")),
                operation(OperationType.CURRENCY, RequestType.COLLECTION, detail("USD", "300"), detail("MAD", "999")));

        byte[] fichier = export.generer(new ExportDemandesDTO(List.of(d), FACE_VALUES));

        Sheet devise = lire(fichier).getSheetAt(ExportExcelDemandes.ONGLET_DEVISE);
        Row eur = devise.getRow(ExportExcelDemandes.PREMIERE_LIGNE), usd = devise.getRow(ExportExcelDemandes.PREMIERE_LIGNE + 1);
        assertThat(texte(eur, 0)).isEqualTo("20261026-1001-03");
        assertThat(texte(eur, 9)).isEqualTo("EUR"); assertThat(nombre(eur, 10)).isEqualTo(500); assertThat(texte(eur, 11)).isEmpty();
        assertThat(texte(usd, 9)).isEqualTo("USD"); assertThat(nombre(usd, 10)).isEqualTo(1000);
        assertThat(texte(usd, 11)).isEqualTo("USD"); assertThat(nombre(usd, 12)).isEqualTo(300);
        Row suivante = devise.getRow(ExportExcelDemandes.PREMIERE_LIGNE + 2);
        assertThat(suivante == null || suivante.getCell(0) == null || suivante.getCell(0).getCellType() == CellType.BLANK).isTrue();
    }

    @Test
    @DisplayName("Feuille GAB Hors Sites : demandes MONETIQUE seulement, une ligne par GAB, cassettes M-P, formule Q, absentes des feuilles MAD / Devise")
    void feuilleGab() throws Exception {
        CashRequestDTO gab = demandeGab("20261028-9000-01",
                operationGab("130120017", "CASA VAL D'ANFA", TypeGab.HORS_SITE,
                        cassette(1, "BILLET_100", 500), cassette(2, "BILLET_200", 300), cassette(4, "BILLET_100", 100)),
                operationGab("130092026", "CASA OASIS DRIVE", TypeGab.DRIVE, cassette(1, "BILLET_100", 50)));
        CashRequestDTO agence = demande("20261028-1001-01", operation(OperationType.CASH_DESK, RequestType.SUPPLY, detail("MAD", "200", coupure("BILLET_200", 1))));

        Workbook classeur = lire(export.generer(new ExportDemandesDTO(List.of(gab, agence), FACE_VALUES)));

        Sheet feuilleGab = classeur.getSheetAt(ExportExcelDemandes.ONGLET_GAB);
        Row r1 = feuilleGab.getRow(ExportExcelDemandes.GAB_PREMIERE_LIGNE), r2 = feuilleGab.getRow(ExportExcelDemandes.GAB_PREMIERE_LIGNE + 1);
        assertThat(texte(r1, 0)).isEqualTo("20261028-9000-01");
        assertThat(texte(r1, 7)).isEqualTo("130092026");                  // tri par ID GAB
        assertThat(texte(r1, 10)).isEqualTo("Drive");
        assertThat(nombre(r1, 12)).isEqualTo(50);                          // M = cassette 1
        assertThat(texte(r2, 7)).isEqualTo("130120017");
        assertThat(texte(r2, 10)).isEqualTo("Hors site");
        assertThat(nombre(r2, 6)).isEqualTo(450);                          // PU = tarif transporteur du GAB
        assertThat(nombre(r2, 12)).isEqualTo(500);
        assertThat(nombre(r2, 13)).isEqualTo(300);
        assertThat(r2.getCell(14) == null || r2.getCell(14).getCellType() == CellType.BLANK).isTrue();   // cassette 3 absente
        assertThat(nombre(r2, 15)).isEqualTo(100);
        assertThat(r2.getCell(16).getCellType()).isEqualTo(CellType.FORMULA);
        assertThat(r2.getCell(16).getCellFormula()).contains("M5").contains("$M$3");

        Sheet mad = classeur.getSheetAt(ExportExcelDemandes.ONGLET_MAD);
        assertThat(texte(mad.getRow(ExportExcelDemandes.PREMIERE_LIGNE), 0)).isEqualTo("20261028-1001-01");
        Row madSuivante = mad.getRow(ExportExcelDemandes.PREMIERE_LIGNE + 1);
        assertThat(madSuivante == null || madSuivante.getCell(0) == null || madSuivante.getCell(0).getCellType() == CellType.BLANK).isTrue();
    }

    @Test
    @DisplayName("Tri par référence sur les deux feuilles")
    void triParReference() throws Exception {
        CashRequestDTO b = demande("20261026-1001-02", operation(OperationType.CASH_DESK, RequestType.SUPPLY, detail("MAD", "200", coupure("BILLET_200", 1))));
        CashRequestDTO a = demande("20261026-1001-01", operation(OperationType.CASH_DESK, RequestType.SUPPLY, detail("MAD", "200", coupure("BILLET_200", 1))));

        Sheet mad = lire(export.generer(new ExportDemandesDTO(List.of(b, a), FACE_VALUES))).getSheetAt(ExportExcelDemandes.ONGLET_MAD);

        assertThat(texte(mad.getRow(2), 0)).isEqualTo("20261026-1001-01");
        assertThat(texte(mad.getRow(3), 0)).isEqualTo("20261026-1001-02");
    }
