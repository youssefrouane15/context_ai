1. Contexte et décisions

AER_LM-259 — « ETQ Administrateur, je veux associer à une agence son prestataire (Brink's, G4S), sa zone de rattachement (zone de distribution) et son coût fixe de déplacement (montant avec virgule). » Le front consomme le contrat livré par AER_LM-271 (agence) et AER_LM-270 / 269 (référentiels), sur le composant existant ConsoleAgenceComponent de portail-ci.

Ce que cette branche livre
  • Un nouveau modèle referentiel.interface.ts et un nouveau service ReferentielService (clone du patron OrganizationService).
  • La modale agence enrichie : select Zone transporteur, select Prestataire, champ Coût de déplacement (MAD), CODE en lecture seule en modification.
  • Aucune nouvelle route, aucun nouveau composant, aucune modification de module : tout reste dans l'onglet Agence(s) de /admin/agence.
Décisions prises (cohérentes avec l'analyse des parties 1 à 4)
  • Affectation (groupe) conservée dans la modale : la maquette ne la montre pas mais le champ est fonctionnel et attendu par CreateAgenceDTO.groupeId.
  • Pas de case « Actif » dans la modale : le toggle de la liste appelle déjà PATCH /toggle-activation.
  • Les trois nouveaux champs sont optionnels côté front, comme dans le contrat 271 (compatibilité écran 267). Ils ne sont envoyés que s'ils sont renseignés.
  • Le libellé écran est « Zone transporteur » (décision du 17/09) ; le code et l'API gardent zoneDistribution.
  • La zone courante d'une agence reste sélectionnable en modification même si elle est devenue inactive (sinon le select afficherait « vide » alors que la base a une valeur).
  • Boutons RETOUR / TERMINER et sous-titre existants conservés (cohérence avec les autres modales).
Prérequis backend
La branche feature/AER_LM-270 d'utilisateurs-ci (V14/V15, routes /referentiels/**, champs 271 sur AgenceDTO) doit être mergée sur develop-omega et déployée sur l'environnement visé. Gateway : la route qf-admin (Path=/api/v1/public/users/**) couvre déjà /referentiels/**, rien à faire.

import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { Observable, throwError } from 'rxjs';
import { catchError, map } from 'rxjs/operators';
import { ApiResponse } from '../../../../shared/interfaces/derogation.interface';
import { ApiConfigService } from '../../api-config.service';
import {
  PrestataireDTO,
  ZoneDistributionDTO
} from '../../../../shared/interfaces/referentiel.interface';

/**
 * AER_LM-259 — Accès aux référentiels de la console (zones de distribution, prestataires).
 * Même patron que OrganizationService : ApiResponse<T> déballé via map(response => response.data).
 */
@Injectable({
  providedIn: 'root'
})
export class ReferentielService {

  private readonly baseUrl: string;

  constructor(
    private http: HttpClient,
    private apiConfig: ApiConfigService
  ) {
    this.baseUrl = this.apiConfig.buildConsoleApiUrl() + '/referentiels';
  }

  // ========== ZONES DE DISTRIBUTION (libellé écran : « Zone Transporteur ») ==========

  // Récupérer les zones de distribution actives (pour le select du modal agence)
  getZonesDistributionActives(): Observable<ZoneDistributionDTO[]> {
    return this.http.get<ApiResponse<ZoneDistributionDTO[]>>(
      `${this.baseUrl}/zones-distribution/actives`
    ).pipe(
      map(response => response.data),
      catchError(this.handleError)
    );
  }

  // ========== PRESTATAIRES (enum côté serveur : BRINKS, G4S) ==========

  // Récupérer la liste des prestataires (pour le select du modal agence)
  getPrestataires(): Observable<PrestataireDTO[]> {
    return this.http.get<ApiResponse<PrestataireDTO[]>>(
      `${this.baseUrl}/prestataires`
    ).pipe(
      map(response => response.data),
      catchError(this.handleError)
    );
  }

  // ========== GESTION DES ERREURS ==========

  private handleError(error: HttpErrorResponse): Observable<never> {
    let errorMessage = 'Une erreur inconnue est survenue';

    if (error.error instanceof ErrorEvent) {
      // Erreur côté client
      errorMessage = `Erreur: ${error.error.message}`;
    } else if (error.error?.message) {
      errorMessage = error.error.message;
    } else {
      // Erreur côté serveur
      errorMessage = `Erreur ${error.status}: ${error.message}`;
    }

    console.error('Erreur API ReferentielService:', errorMessage);
    return throwError(() => new Error(errorMessage));
  }
}

À AJOUTERHelpers — après onCodeAgenceInput() (ligne 575), avant le bloc « // ========== HELPERS ==========»
  // ---- AER_LM-259 : saisie du coût de déplacement (chiffres, virgule ou point uniquement)
  onCoutDeplacementInput(event: Event): void {

  // ---- AER_LM-259 : chargement des selects du modal agence.
  // En modification, la zone courante reste sélectionnable même si elle est devenue inactive.
  private chargerReferentielsAgence(zoneCourante: ZoneDistributionDTO | null): void {
    this.referentielService.getZonesDistributionActives().subscribe({
      next: (zones) => {
        this.modalZonesTransporteur = zoneCourante && !zones.some(z => z.id === zoneCourante.id)

      },
      error: (error) => {

      }
    });
    this.referentielService.getPrestataires().subscribe({
      next: (prestataires) => {
        this.prestataires = prestataires;
      },
      error: (error) => {
        console.error('Erreur chargement prestataires:', error);
      }
    });
  }

  private resetConfigurationAgence(): void {
    this.modalZonesTransporteur = [];
    this.prestataires = [];
    this.selectedZoneTransporteurId = 0;
    this.selectedPrestataireId = '';
    this.coutDeplacementSaisi = '';
  }

  /** '' → undefined (non renseigné) ; saisie invalide → null ; sinon le montant (virgule acceptée, 2 décimales max). */
  private parserCoutDeplacement(saisie: string): number | undefined | null {
    const valeur = saisie.trim().replace(',', '.');
    if (!valeur) {
      return undefined;
    }
    if (!/^\d+(\.\d{1,2})?$/.test(valeur)) {
      return null;
    }
    return Number(valeur);
  }

  private formaterCoutDeplacement(cout: number | null | undefined): string {
    return cout === null || cout === undefined ? '' : String(cout).replace('.', ',');
  }
