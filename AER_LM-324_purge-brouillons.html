0. Périmètre

Règle retenue : les demandes sont créées et traitées le jour même ; un brouillon (INITIATED) encore présent à minuit n'a plus de raison d'être. Chaque nuit à 00:00, tous les brouillons sont supprimés : l'instance Camunda (bloquée sur Task_SoumettreDemande) d'abord, puis la demande et sa grappe en base. Pas de rétention, pas de ShedLock, pas d'outbox, pas d'endpoint : un @Scheduled Spring et un use case.

package com.bmci.workflow.fund.management.service.domain.retail.nettoyage;

import com.bmci.q360.domain.retail.enums.RequestStatus;
import com.bmci.workflow.fund.management.service.domain.retail.ports.input.NettoyerBrouillonsUseCase;
import com.bmci.workflow.fund.management.service.domain.retail.ports.output.repository.CashDemandRepository;
import com.bmci.workflow.fund.management.service.domain.retail.ports.output.workflow.ProcessusWorkflowPort; // même import que DemarrerProcessusStep
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.UUID;

/**
 * AER_LM-324 : les demandes sont créées et traitées le jour même ; un brouillon (INITIATED)
 * encore présent à minuit n'a plus de raison d'être. Pour chacun : suppression de l'instance
 * Camunda (bloquée sur Task_SoumettreDemande) puis de la demande et de sa grappe en base.
 * Une erreur sur une demande ne bloque pas les autres : elle est journalisée et la demande
 * sera reprise à la purge suivante.
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class NettoyageBrouillonsService implements NettoyerBrouillonsUseCase {

    static final String MOTIF_SUPPRESSION = "Purge quotidienne des brouillons (AER_LM-324)";

    private final CashDemandRepository cashDemandRepository;
    private final ProcessusWorkflowPort processusWorkflowPort;

    @Override
    public int nettoyer() {
        List<UUID> brouillons = cashDemandRepository.findIdsParStatut(RequestStatus.INITIATED);
        if (brouillons.isEmpty()) {
            log.info("Purge des brouillons : aucun brouillon à supprimer");
            return 0;
        }
        int supprimes = 0;
        for (UUID id : brouillons) {
            try {
                processusWorkflowPort.supprimerInstance(id.toString(), MOTIF_SUPPRESSION);
                cashDemandRepository.supprimer(id);
                supprimes++;
            } catch (RuntimeException e) {
                log.warn("Purge des brouillons : demande {} conservée, reprise à la prochaine purge ({})", id, e.getMessage());
            }
        }
        log.info("Purge des brouillons : {} supprimé(s) sur {} brouillon(s)", supprimes, brouillons.size());
        return supprimes;
    }
}

package com.bmci.workflow.fund.management.application.scheduler;

import com.bmci.workflow.fund.management.service.domain.retail.ports.input.NettoyerBrouillonsUseCase;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Scheduled;

/**
 * AER_LM-324 : adapter primaire planifié — déclenche la purge des brouillons chaque nuit.
 * Heure et fuseau en yaml (gdf.nettoyage-brouillons.*).
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class NettoyageBrouillonsScheduler {

    private final NettoyerBrouillonsUseCase nettoyerBrouillonsUseCase;

    @Scheduled(cron = "${gdf.nettoyage-brouillons.cron}", zone = "${gdf.nettoyage-brouillons.zone}")
    public void purgerBrouillons() {
        log.info("Purge des brouillons : démarrage");
        nettoyerBrouillonsUseCase.nettoyer();
    }
}

import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.InjectMocks;
import org.mockito.InOrder;
import org.mockito.Mock;
import org.mockito.junit.jupiter.MockitoExtension;

import java.util.List;
import java.util.UUID;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.mockito.ArgumentMatchers.anyString;
import static org.mockito.Mockito.*;

@ExtendWith(MockitoExtension.class)
class NettoyageBrouillonsServiceTest {

    @Mock private CashDemandRepository cashDemandRepository;
    @Mock private ProcessusWorkflowPort processusWorkflowPort;
    @InjectMocks private NettoyageBrouillonsService service;

    @Test
    @DisplayName("Aucun brouillon : rien n'est supprimé")
    void aucunBrouillon() {
        when(cashDemandRepository.findIdsParStatut(RequestStatus.INITIATED)).thenReturn(List.of());

        assertEquals(0, service.nettoyer());

        verifyNoInteractions(processusWorkflowPort);
        verify(cashDemandRepository, never()).supprimer(any());
    }

    @Test
    @DisplayName("Chaque brouillon : instance Camunda supprimée avant la demande")
    void supprimeInstancePuisDemande() {
        UUID a = UUID.randomUUID();
        UUID b = UUID.randomUUID();
        when(cashDemandRepository.findIdsParStatut(RequestStatus.INITIATED)).thenReturn(List.of(a, b));

        assertEquals(2, service.nettoyer());

        InOrder ordre = inOrder(processusWorkflowPort, cashDemandRepository);
        ordre.verify(processusWorkflowPort).supprimerInstance(eq(a.toString()), eq(NettoyageBrouillonsService.MOTIF_SUPPRESSION));
        ordre.verify(cashDemandRepository).supprimer(a);
        ordre.verify(processusWorkflowPort).supprimerInstance(eq(b.toString()), eq(NettoyageBrouillonsService.MOTIF_SUPPRESSION));
        ordre.verify(cashDemandRepository).supprimer(b);
    }

    @Test
    @DisplayName("Erreur moteur sur un brouillon : il est conservé, les autres sont purgés")
    void erreurSurUneDemandeNeBloquePasLesAutres() {
        UUID enErreur = UUID.randomUUID();
        UUID ok = UUID.randomUUID();
        when(cashDemandRepository.findIdsParStatut(RequestStatus.INITIATED)).thenReturn(List.of(enErreur, ok));
        doThrow(new IllegalStateException("moteur indisponible"))
                .when(processusWorkflowPort).supprimerInstance(eq(enErreur.toString()), anyString());

        assertEquals(1, service.nettoyer());

        verify(cashDemandRepository, never()).supprimer(enErreur);
        verify(cashDemandRepository).supprimer(ok);
    }
}

3. Contrôles après purge :
   base   : la requête du 1 renvoie 0 ligne ; les opérations/coupures liées sont parties avec (cascade de la grappe, déjà utilisée par la compensation de saga)
   moteur : GET /engine-rest/process-instance?businessKey=<id> → []
   les demandes SUBMITTED / IN_MODIFICATION / traitées / annulées sont intactes :
            select request_status, count(*) from retail.cash_request group by 1;

4. Cas d'erreur : arrêter le moteur Camunda, créer un brouillon, laisser passer une purge →
   log WARN "demande <id> conservée, reprise à la prochaine purge", la ligne reste en base ;
   relancer le moteur → purge suivante : supprimée.

5. Remettre le cron à "0 0 0 * * *" avant de committer le yaml.
