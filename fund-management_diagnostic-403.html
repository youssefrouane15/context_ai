Ta capture : anciens uids en codes courts (C, CC, SUP, AGT, RSP), nouveaux en codes longs. Si un garde compare profile_code à CAISSE_CENTRALE/CHEF_CAISSE, les anciens sont refusés. INSERT INTO user_profile (id, creation_date, modification_date, uid, profile_code, profile_label, profile_type)

package com.bmci.workflow.fund.management.application.rest.security;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

/**
 * SONDE TEMPORAIRE (diagnostic 403 local, 02/11) — à SUPPRIMER avant commit.
 * Loggue en WARN (jamais filtré par logback) : la requête arrive-t-elle ? avec quelles authorities ? quel statut sort ?
 * Ordre HIGHEST_PRECEDENCE : premier filtre de la chaîne, avant Spring Security.
 */
@Slf4j
@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class SondeRequeteFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        log.warn("SONDE >> {} {} origin={} auth={}", request.getMethod(), request.getRequestURI(),
                request.getHeader("Origin"), request.getHeader("Authorization") != null ? "Bearer présent" : "ABSENT");
        try {
            chain.doFilter(request, response);
        } finally {
            Authentication a = SecurityContextHolder.getContext().getAuthentication();
            log.warn("SONDE << {} {} status={} principal={} authorities={}", request.getMethod(), request.getRequestURI(),
                    response.getStatus(), a == null ? "null" : a.getName(), a == null ? "[]" : a.getAuthorities());
        }
    }
}

Lecture des deux lignes SONDE pour GET /cash :
  - aucune ligne "SONDE >>"              → la requête n'entre pas dans Tomcat : gateway → autre port / autre process / autre hôte (vérifier QF_MS_FUND_MANAGEMENT_URL)
  - ">>" avec auth=ABSENT                → la gateway ne relaie pas le Bearer (QfMockTokenRelayFilter / cookie) ; mais le filtre répondrait 401, pas 403
  - "<<" status=403 authorities=[]       → filtre passé mais résolution users vide → GET http://localhost:8091/api/v1/internal/users/g10002 en Postman
  - "<<" status=403 authorities=[GDF_…]  → @PreAuthorize OK, c'est un garde du use case (RetailForbiddenException) → le corps JSON de la réponse (§1) donne le motif
  - "<<" status=200                      → fund-management est bon, le 403 est fabriqué par la gateway après coup (CORS/DedupeResponseHeader) → logs gateway
