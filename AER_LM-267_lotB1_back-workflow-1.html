package com.bmci.workflow.fund.management.service.domain.retail;

import com.bmci.q360.domain.retail.entity.CashRequest;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;

/**
 * Règle du cut-off 16h00, calculée localement (même règle que dateCutoff16h côté worker).
 * Jour d'effet = jour de soumission, ou J+1 si soumise après 16h00.
 * Une demande jamais soumise (brouillon) n'est pas concernée.
 */
@Component
public class CutoffPolicy {

    public static final LocalTime HEURE_CUTOFF = LocalTime.of(16, 0);

    public boolean journeeCloturee(CashRequest demande) {
        LocalDateTime soumission = demande.getSubmittedAt();
        if (soumission == null) {
            return false;
        }
        return LocalDateTime.now().isAfter(dateCutoff16h(soumission));
    }

    public LocalDateTime dateCutoff16h(LocalDateTime soumission) {
        LocalDate jourEffet = soumission.toLocalTime().isAfter(HEURE_CUTOFF)
                ? soumission.toLocalDate().plusDays(1)
                : soumission.toLocalDate();
        return jourEffet.atTime(HEURE_CUTOFF);
    }
}

        sagaOrchestrator.executer("fin-modification", ctx, List.of(
                new LibererVerrouStep(cashDemandRepository, uid),
                new CompleterTacheStep(processusWorkflowPort, TASK_MODIFIER,
                        c -> Map.of("modificationConfirmee", confirmee, "urgent", estUrgente(c.getDemande())))));
        return enrichir(ctx.getDemande());
    }

        RequestStatus statut = demande.getRequestStatus();
        if (statut != RequestStatus.INITIATED && statut != RequestStatus.SUBMITTED

        verifierJourneeOuverte(demande);
        Map<String, Object> variables = new HashMap<>();
        variables.put("annulePar", contexte.uidOu(demande.getUidUser()));
        variables.put("motifAnnulation", annulation.getMotif().trim());
        // Contrat gateway : aucune écriture locale, le worker gdf-annuler-demande écrira CANCELLED.
        processusWorkflowPort.correlerMessage(MSG_ANNULER, id.toString(), variables);
        return enrichir(demande);
    }

    @Override

        // Garde 1 — payload

            throw new RetailValidationException("La décision (TRAITEE ou ANNULEE) est obligatoire");
        }
        boolean annulation = decision.getDecision() == DecisionCaisseCentrale.ANNULEE;

            throw new RetailValidationException(

        }

        // Garde 2 — cycle de vie local (IN_MODIFICATION est refusé ici : la Caisse Centrale n'a pas la main)
        CashRequest demande = cashDemandRepository.getCashRequestById(id);
        if (demande.getRequestStatus() != RequestStatus.SUBMITTED) {
            throw new RetailConflictException(
                    "Seule une demande SUBMITTED peut être traitée par la Caisse Centrale (statut actuel : "
                            + demande.getRequestStatus() + ")");
        }

        // Garde 3 — le jeton doit être sur la user task Caisse Centrale
        String taskId = processusWorkflowPort.rechercherTacheActive(id.toString(), TASK_CAISSE_CENTRALE)
                .orElseThrow(() -> new RetailConflictException(
                        "Aucune tâche de traitement active pour la demande " + id
                                + " : le workflow n'a pas (encore) atteint l'étape Caisse Centrale"));

        Map<String, Object> variables = new HashMap<>();
        variables.put("decisionCaisseCentrale", decision.getDecision().name());
        if (annulation) {
            variables.put("motifAnnulation", decision.getMotifAnnulation().trim());
        }
        processusWorkflowPort.completerTache(taskId, variables);

        return enrichir(demande);

    private void verifierMemeAgence(CashRequest demande, ContexteAppelant contexte) {
        if (!contexte.estIdentifie()) {
            log.warn("Contexte appelant transitoire : garde d'agence désactivée pour la demande {}", demande.getId());
            return;
        }
        if (!contexte.toutesAgences() && !Objects.equals(demande.getBranchCode(), contexte.branchCode())) {
            throw new RetailForbiddenException("La demande appartient à l'agence " + demande.getBranchCode()
                    + ", pas à celle de l'appelant");
        }
    }

    private void verifierDetenteurVerrou(CashRequest demande, ContexteAppelant contexte) {
        if (!contexte.estIdentifie()) {
            log.warn("Contexte appelant transitoire : garde de verrou désactivée pour la demande {}", demande.getId());
            return;
        }
        if (!Objects.equals(demande.getLockedBy(), contexte.uid())) {
            throw new RetailForbiddenException("La demande est en cours de modification par " + demande.getLockedBy());
        }
    }

    private void verifierJourneeOuverte(CashRequest demande) {
        if (cutoffPolicy.journeeCloturee(demande)) {
            throw new RetailConflictException(
                    "Journée clôturée (cut-off 16h00) : la demande n'est plus modifiable par l'agence");
        }
    }

    private CashRequestDTO enrichir(CashRequest demande) {
        CashRequestDTO dto = webMapper.toResponseDTO(demande);
        dto.setJourneeCloturee(cutoffPolicy.journeeCloturee(demande));
        return dto;
    }

    private static boolean estUrgente(CashRequest demande) {
        return demande != null && demande.getOperations() != null
                && demande.getOperations().stream().anyMatch(op -> Boolean.TRUE.equals(op.getIsUrgent()));
    }

    private Map<String, BigDecimal> faceValuesByCode() {
        return denominationRepository.findAllActiveDenominations().stream()
                .collect(Collectors.toMap(Denomination::getDenominationCode, Denomination::getFaceValue));
    }
