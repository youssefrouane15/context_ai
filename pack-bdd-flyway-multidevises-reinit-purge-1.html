-- ============================================================================
-- V1 — Schema RETAIL complet (fresh start) + SEED MULTI-DEVISES.
-- Structure = entites JPA retail (source de verite) + alignement corpo :
-- denomination_code (cle stable), unique_request_ref + sequence (ref RT-).

CREATE SCHEMA IF NOT EXISTS retail;

-- ============ Referentiels ============

CREATE TABLE retail.currency (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    iso_code           varchar(3)   NOT NULL UNIQUE,
    currency_name      varchar(100) NOT NULL
);

CREATE TABLE retail.denomination (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    denomination_code  varchar(30)  NOT NULL UNIQUE,
    item_type          varchar(10)  NOT NULL,
    face_value         numeric(10,2) NOT NULL,
    denomination_label varchar(100) NOT NULL,
    CONSTRAINT chk_retail_item_type CHECK (item_type IN ('BANKNOTE', 'COIN'))
);

CREATE TABLE retail.establishment (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    establishment_code varchar(3)   NOT NULL UNIQUE,
    establishment_name varchar(100) NOT NULL
);

CREATE TABLE retail.user_profile (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    uid                varchar(255),
    profile_code       varchar(50),
    profile_label      varchar(100),
    profile_type       varchar(10)
);

-- ============ Transactionnel ============

CREATE TABLE retail.cash_request (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    unique_request_ref varchar(20)  NOT NULL UNIQUE,
    branch_code        varchar(100) NOT NULL,
    request_status     varchar(30)  NOT NULL DEFAULT 'INITIATED',
    user_profile_id    uuid         NOT NULL,
    establishment_id   uuid         NOT NULL,
    CONSTRAINT fk_retail_request_user FOREIGN KEY (user_profile_id) REFERENCES retail.user_profile (id),
    CONSTRAINT fk_retail_request_etab FOREIGN KEY (establishment_id) REFERENCES retail.establishment (id),
    CONSTRAINT chk_retail_request_status CHECK (request_status IN ('INITIATED', 'VALIDATED', 'CANCELLED'))
);

CREATE TABLE retail.cash_operation (
    id                      uuid PRIMARY KEY,
    creation_date           timestamp,
    modification_date       timestamp,
    cash_request_id         uuid          NOT NULL,
    operation_type          varchar(20)   NOT NULL,
    operation_sens          varchar(50)   NOT NULL,
    institution_code        integer       NOT NULL DEFAULT 1,
    cash_desk_number        varchar(3)    NOT NULL DEFAULT '001',
    is_urgent               varchar(3)    NOT NULL DEFAULT 'NO',
    target_cash_desk_number varchar(3),
    total_amount            numeric(15,2) NOT NULL DEFAULT 0.00,
    created_at              timestamp     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at              timestamp     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    operation_status        varchar(30)   NOT NULL DEFAULT 'INITIATED',
    establishment           varchar(100)  NOT NULL,
    CONSTRAINT fk_retail_operation_request FOREIGN KEY (cash_request_id)
        REFERENCES retail.cash_request (id) ON DELETE CASCADE,
    CONSTRAINT chk_retail_operation_sens CHECK (operation_sens IN ('SUPPLY', 'COLLECTION')),
    CONSTRAINT chk_retail_is_urgent CHECK (is_urgent IN ('YES', 'NO'))
);

CREATE TABLE retail.operation_currency (
    id                        uuid PRIMARY KEY,
    creation_date             timestamp,
    modification_date         timestamp,
    cash_operation_id         uuid          NOT NULL,
    currency_id               uuid          NOT NULL,
    total_amount_by_curerency numeric(18,2) NOT NULL DEFAULT 0.00,
    CONSTRAINT fk_retail_oc_operation FOREIGN KEY (cash_operation_id)
        REFERENCES retail.cash_operation (id) ON DELETE CASCADE,
    CONSTRAINT fk_retail_oc_currency FOREIGN KEY (currency_id) REFERENCES retail.currency (id)
);

CREATE TABLE retail.operation_denomination (
    id                           uuid PRIMARY KEY,
    creation_date                timestamp,
    modification_date            timestamp,
    operation_currency_detail_id uuid,
    denomination_id              uuid,
    quantity                     integer NOT NULL DEFAULT 0,
    CONSTRAINT fk_retail_od_detail FOREIGN KEY (operation_currency_detail_id)
        REFERENCES retail.operation_currency (id) ON DELETE CASCADE,
    CONSTRAINT fk_retail_od_denomination FOREIGN KEY (denomination_id) REFERENCES retail.denomination (id),
    CONSTRAINT chk_retail_quantity CHECK (quantity >= 0),
    CONSTRAINT uq_retail_od UNIQUE (operation_currency_detail_id, denomination_id)
);

CREATE SEQUENCE retail.request_ref_seq START WITH 1 INCREMENT BY 1;

3V2 — Schéma corpo (multi-devises, miroir + colonnes Jira)

Structure miroir du retail + spécificités corpo (identité client sur l'entête, FKs demandeur/agence NULLABLES, CHECK ALIMENTATION/RAMASSAGE, séquence corpo_request_ref_seq) — mais seed MONO-devise : le corpo travaille en MAD, 11 coupures dont les centimes pour le rendu monnaie. Le multi-devises est une capacité RETAIL (opérations de change des agences) ; côté corpo, une devise étrangère reste donc un cas 422 — le cahier corpo (ER-08 : EUR inconnu) reste valide tel quel.

REMPLACERfund-management-container\\src\\main\\resources\\db\\migration\\V2__create_corpo_schema.sqlMono-devise MAD (11 coupures) — structure inchangée ; application via réinit complète UNIQUEMENT
-- ============================================================================
-- V2 — Schema CORPO : duplication du schema retail + colonnes Jira
-- (identite client sur l'entete, FKs demandeur/agence NULLABLES).
-- SEED MONO-DEVISE : le corpo travaille en MAD (11 coupures, centimes
-- compris pour le rendu monnaie) — le multi-devises est cote RETAIL.
-- ATTENTION : cette V2 remplace la precedente => uniquement via la
-- REINITIALISATION COMPLETE (les 2 DROP SCHEMA), jamais sur un historique
-- existant (regle du checksum).
-- ============================================================================

CREATE SCHEMA IF NOT EXISTS corpo;

-- ============ Referentiels ============

CREATE TABLE corpo.currency (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    iso_code           varchar(3)   NOT NULL UNIQUE,
    currency_name      varchar(100) NOT NULL
);

CREATE TABLE corpo.denomination (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    denomination_code  varchar(30)  NOT NULL UNIQUE,
    item_type          varchar(10)  NOT NULL,
    face_value         numeric(10,2) NOT NULL,
    denomination_label varchar(100) NOT NULL,
    CONSTRAINT chk_corpo_item_type CHECK (item_type IN ('BANKNOTE', 'COIN'))
);

CREATE TABLE corpo.establishment (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    establishment_code varchar(3)   NOT NULL UNIQUE,
    establishment_name varchar(100) NOT NULL
);

CREATE TABLE corpo.user_profile (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    uid                varchar(255),
    profile_code       varchar(50),
    profile_label      varchar(100),
    profile_type       varchar(10)
);

-- ============ Transactionnel ============

CREATE TABLE corpo.cash_request (
    id                 uuid PRIMARY KEY,
    creation_date      timestamp,
    modification_date  timestamp,
    unique_request_ref varchar(20)  NOT NULL UNIQUE,       -- CO-AAAA-NNNNNN (sequence)
    branch_code        varchar(100),
    request_status     varchar(30)  NOT NULL DEFAULT 'INITIATED',
    client_ref         varchar(50),                        -- identite CLIENT (Jira)
    city               varchar(100),
    address_complement varchar(255),
    account_number     varchar(34),
    user_profile_id    uuid,                               -- NULLABLE : l'identite client remplace
    establishment_id   uuid,                               -- NULLABLE
    CONSTRAINT fk_corpo_request_user FOREIGN KEY (user_profile_id) REFERENCES corpo.user_profile (id),
    CONSTRAINT fk_corpo_request_etab FOREIGN KEY (establishment_id) REFERENCES corpo.establishment (id),
    CONSTRAINT chk_corpo_request_status CHECK (request_status IN ('INITIATED', 'VALIDATED', 'CANCELLED'))
);

CREATE TABLE corpo.cash_operation (
    id                      uuid PRIMARY KEY,
    creation_date           timestamp,
    modification_date       timestamp,
    cash_request_id         uuid          NOT NULL,
    operation_type          varchar(20),
    operation_sens          varchar(50)   NOT NULL,
    institution_code        integer       NOT NULL DEFAULT 1,
    cash_desk_number        varchar(3)    NOT NULL DEFAULT '001',
    is_urgent               varchar(3)    NOT NULL DEFAULT 'NO',
    target_cash_desk_number varchar(3),
    total_amount            numeric(15,2) NOT NULL DEFAULT 0.00,
    created_at              timestamp     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at              timestamp     NOT NULL DEFAULT CURRENT_TIMESTAMP,
    operation_status        varchar(30)   NOT NULL DEFAULT 'INITIATED',
    establishment           varchar(100),
    CONSTRAINT fk_corpo_operation_request FOREIGN KEY (cash_request_id)
        REFERENCES corpo.cash_request (id) ON DELETE CASCADE,
    CONSTRAINT chk_corpo_operation_sens CHECK (operation_sens IN ('ALIMENTATION', 'RAMASSAGE')),
    CONSTRAINT chk_corpo_is_urgent CHECK (is_urgent IN ('YES', 'NO'))
);

CREATE TABLE corpo.operation_currency (
    id                        uuid PRIMARY KEY,
    creation_date             timestamp,
    modification_date         timestamp,
    cash_operation_id         uuid          NOT NULL,
    currency_id               uuid          NOT NULL,
    total_amount_by_curerency numeric(18,2) NOT NULL DEFAULT 0.00,
    CONSTRAINT fk_corpo_oc_operation FOREIGN KEY (cash_operation_id)
        REFERENCES corpo.cash_operation (id) ON DELETE CASCADE,
    CONSTRAINT fk_corpo_oc_currency FOREIGN KEY (currency_id) REFERENCES corpo.currency (id)
);

CREATE TABLE corpo.operation_denomination (
    id                           uuid PRIMARY KEY,
    creation_date                timestamp,
    modification_date            timestamp,
    operation_currency_detail_id uuid,
    denomination_id              uuid,
    quantity                     integer NOT NULL DEFAULT 0,
    CONSTRAINT fk_corpo_od_detail FOREIGN KEY (operation_currency_detail_id)
        REFERENCES corpo.operation_currency (id) ON DELETE CASCADE,
    CONSTRAINT fk_corpo_od_denomination FOREIGN KEY (denomination_id) REFERENCES corpo.denomination (id),
    CONSTRAINT chk_corpo_quantity CHECK (quantity >= 0),
    CONSTRAINT uq_corpo_od UNIQUE (operation_currency_detail_id, denomination_id)
);

CREATE SEQUENCE corpo.corpo_request_ref_seq START WITH 1 INCREMENT BY 1;

-- REINITIALISATION COMPLETE — app ARRETEE, pgAdmin, puis redemarrer.
-- L'historique Flyway vit DANS retail : supprimer les 2 schemas supprime

-- Redemarrer l'application, puis verifier :
SELECT version, description, success FROM retail.flyway_schema_history ORDER BY installed_rank;
-- attendu : 2 lignes (1, 2), success = true

SELECT (SELECT count(*) FROM retail.currency)     AS devises_retail,   -- 3 (MAD, EUR, USD)
       (SELECT count(*) FROM retail.denomination) AS coupures_retail,  -- 24
       (SELECT count(*) FROM corpo.currency)      AS devises_corpo,    -- 1 (MAD)
       (SELECT count(*) FROM corpo.denomination)  AS coupures_corpo,   -- 11
       (SELECT count(*) FROM retail.establishment) AS agences,         -- 2

5Purge des données (entre deux campagnes)

REMPLACERpgAdmin — app démarrée, avant chaque campagne APILe premier POST après purge doit redonner RT-2026-000001 (retail) et CO-2026-000001 (corpo)
-- ============================================================================
-- PURGE DES DONNEES DE TEST — entre deux campagnes API, app DEMARREE.
-- Vide le transactionnel des DEUX perimetres ; schemas, seeds (multi-devises
-- retail, MAD corpo) et historique conserves. Les references RT-/CO- repartent a 000001.
-- ============================================================================

TRUNCATE TABLE retail.operation_denomination,
               retail.operation_currency,
               retail.cash_operation,
               retail.cash_request;

TRUNCATE TABLE corpo.operation_denomination,
               corpo.operation_currency,
               corpo.cash_operation,
               corpo.cash_request;

ALTER SEQUENCE retail.request_ref_seq      RESTART WITH 1;
ALTER SEQUENCE corpo.corpo_request_ref_seq RESTART WITH 1;

-- Verification : transactionnel a zero, referentiels intacts
SELECT (SELECT count(*) FROM retail.cash_request) AS demandes_retail,  -- 0
       (SELECT count(*) FROM corpo.cash_request)  AS demandes_corpo,   -- 0
       (SELECT count(*) FROM retail.denomination) AS coupures_retail,  -- 24
       (SELECT count(*) FROM corpo.denomination)  AS coupures_corpo;   -- 11
