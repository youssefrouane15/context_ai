1. Objectif et principe

Les deux cards du dashboard de portail-ci — Dérogation (route interne /demande-derogation) et Gestion des fonds (lien externe fundManagementUrl) — ne sont affichées que si l'utilisateur connecté porte la permission correspondante. Aucun rôle, aucun code profil n'est testé côté front : uniquement des codes de permission, exactement comme SideMenuComponent.checkPermissions() le fait déjà avec AuthService.hasPermission().

CardPermission requisePlateformeQui la porte aujourd'hui
DérogationDTRP_CHOIX_WFDTRPRôles DTRP (V7) → profils CONSEILLER, DA, DIE, LMR, APAC_COMPTA, DZ
Gestion des fondsGDF_CONSULTER_DEMANDEGESTION_FONDSRôles 10 / 11 / 12 (V16) → profils DA, CHEF_CAISSE, CAISSE_CENTRALE, SUPPORT
Le Super Admin n'a ni l'une ni l'autre
Le profil 7 SA (h11111) n'a que les rôles 8 / 9 (CONSOLE_ADMIN). Sans le script V17, il ne verrait plus aucune card. V17 lui rattache le rôle 12 (GDF) et un rôle 13 dédié ne portant que DTRP_CHOIX_WF. Le profil AU (Admin User, rôle 9) ne verra aucune des deux cards : c'est le comportement attendu d'un gating par permission.
Ce que ce gating fait et ne fait pas
Il masque la card. La protection réelle reste côté serveur : @PreAuthorize sur utilisateurs-ci pour DTRP, et la sécurité de fund-management (261 / 266) pour Gestion des fonds. Taper l'URL à la main n'est pas bloqué par ce changement — c'est le rôle des guards de route et du backend.

-- =====================================================================
-- V17 : Super Admin — accès aux cards « Dérogation » et « Gestion des fonds »
-- ---------------------------------------------------------------------
-- Les deux cards du dashboard portail-ci sont désormais conditionnées par
-- permission (front) :
--   - card Dérogation        -> DTRP_CHOIX_WF          (plateforme DTRP, existe déjà)
--   - card Gestion des fonds -> GDF_CONSULTER_DEMANDE   (plateforme GESTION_FONDS, V16)
--
-- Le profil 7 SA (h11111) ne porte que les rôles 8 / 9 (CONSOLE_ADMIN) :
--   1) rôle 12 « GDF Support » (GDF_CONSULTER_DEMANDE + GDF_TOUTES_AGENCES) rattaché au SA
--   2) nouveau rôle 13 « DTRP Accès Dérogation » (plateforme DTRP) portant uniquement
--      DTRP_CHOIX_WF, rattaché au SA — un rôle dédié plutôt qu&#x27;un rôle métier
--      existant (1 à 7), pour ne pas donner au SA des droits d&#x27;initiation /
--      validation qu&#x27;il n'a pas aujourd'hui.
--
-- Aucune modification d&#x27;enum Java, de sécurité ou de gateway : les authorities
-- sont construites depuis la base (profil -> rôles -> permissions).
-- Idempotent : chaque insertion est ignorée si la ligne existe déjà.
-- =====================================================================

-- ---------- 1) Consultation Gestion des Fonds ----------

INSERT INTO hab_profile_role (profile_id, role_id)
SELECT 7, 12
WHERE EXISTS (SELECT 1 FROM hab_profiles WHERE id = 7)
  AND EXISTS (SELECT 1 FROM hab_roles    WHERE id = 12)
  AND NOT EXISTS (
        SELECT 1 FROM hab_profile_role
        WHERE profile_id = 7 AND role_id = 12
      );

-- ---------- 2) Accès à la card Dérogation ----------

INSERT INTO hab_roles (id, name, description, actif, created_at, updated_at, platform)
SELECT 13, &#x27;DTRP Accès Dérogation', 'Accès à la card Dérogation tarifaire du portail (choix du workflow)', true, NOW(), NOW(), 'DTRP'
WHERE NOT EXISTS (SELECT 1 FROM hab_roles WHERE id = 13);

INSERT INTO hab_role_permission (role_id, permission)
SELECT 13, &#x27;DTRP_CHOIX_WF'
WHERE NOT EXISTS (
        SELECT 1 FROM hab_role_permission
        WHERE role_id = 13 AND permission = &#x27;DTRP_CHOIX_WF'
      );

INSERT INTO hab_profile_role (profile_id, role_id)
SELECT 7, 13
WHERE EXISTS (SELECT 1 FROM hab_profiles WHERE id = 7)
  AND NOT EXISTS (
        SELECT 1 FROM hab_profile_role
        WHERE profile_id = 7 AND role_id = 13
      );

-- Réalignement de la séquence après insertion avec id explicite (même geste que V6 / V16)
SELECT setval(&#x27;hab_roles_id_seq', (SELECT MAX(id) FROM hab_roles));

import { ComponentFixture, TestBed } from '@angular/core/testing';
import { NO_ERRORS_SCHEMA } from '@angular/core';
import { DashboardComponent } from './dashboard.component';
import { AuthService } from '<même chemin que dans dashboard.component.ts>';
import { Permission } from '<même chemin que dans dashboard.component.ts>';

describe('DashboardComponent — accès aux cards par permission', () => {
  let fixture: ComponentFixture<DashboardComponent>;
  let component: DashboardComponent;
  let authService: jasmine.SpyObj<AuthService>;

  beforeEach(async () => {
    authService = jasmine.createSpyObj<AuthService>('AuthService', ['hasPermission']);

    await TestBed.configureTestingModule({
      declarations: [DashboardComponent],
      providers: [{ provide: AuthService, useValue: authService }],
      schemas: [NO_ERRORS_SCHEMA]
    }).compileComponents();

    fixture = TestBed.createComponent(DashboardComponent);
    component = fixture.componentInstance;
  });

  function stubPermissions(derogation: boolean, fonds: boolean): void {
    authService.hasPermission.and.callFake((codes: string[]) => {
      if (codes.includes(Permission.DTRP_CHOIX_WF)) { return derogation; }
      if (codes.includes(Permission.GDF_CONSULTER_DEMANDE)) { return fonds; }
      return false;
    });
  }

  it('interroge AuthService avec DTRP_CHOIX_WF et GDF_CONSULTER_DEMANDE', () => {
    stubPermissions(true, true);

    component.checkCardPermissions();

    expect(authService.hasPermission).toHaveBeenCalledWith([Permission.DTRP_CHOIX_WF]);
    expect(authService.hasPermission).toHaveBeenCalledWith([Permission.GDF_CONSULTER_DEMANDE]);
  });

  it('affiche les deux cards quand les deux permissions sont portées', () => {
    stubPermissions(true, true);

    fixture.detectChanges();

    const html: HTMLElement = fixture.nativeElement;
    expect(html.querySelector('#derogation')).not.toBeNull();
    expect(html.querySelector('#fund-management')).not.toBeNull();
    expect(html.querySelector('.no-service')).toBeNull();
  });

  it("masque la card Dérogation sans DTRP_CHOIX_WF", () => {
    stubPermissions(false, true);

    fixture.detectChanges();

    const html: HTMLElement = fixture.nativeElement;
    expect(html.querySelector('#derogation')).toBeNull();
    expect(html.querySelector('#fund-management')).not.toBeNull();
  });

  it("masque la card Gestion des fonds sans GDF_CONSULTER_DEMANDE", () => {
    stubPermissions(true, false);

    fixture.detectChanges();

    const html: HTMLElement = fixture.nativeElement;
    expect(html.querySelector('#derogation')).not.toBeNull();
    expect(html.querySelector('#fund-management')).toBeNull();
  });

  it('affiche le message « aucun service » sans aucune des deux permissions', () => {
    stubPermissions(false, false);

    fixture.detectChanges();

    const html: HTMLElement = fixture.nativeElement;
    expect(html.querySelector('#derogation')).toBeNull();
    expect(html.querySelector('#fund-management')).toBeNull();
    expect(html.querySelector('.no-service')?.textContent).toContain('Aucun service');
  });
});
