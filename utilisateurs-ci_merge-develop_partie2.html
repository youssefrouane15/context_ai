-- =====================================================================================
-- V19 (ex-V16, AER_LM-261) : plateforme GESTION_FONDS — rôles, profils et permissions GDF
-- Merge develop 30/10 : plus aucun id en dur (develop insère par nextval, incrément 1 sur
-- hab_profiles_seq → collision garantie sur les ids 9-11). Rôles et profils sont résolus
-- par name / code, inserts idempotents (NOT EXISTS), aucun setval.

-- -------------------------------------------------------------
-- 1. CHECK hérités de V1 (noms générés par Hibernate, on ne les suppose pas :
--    on supprime tout CHECK portant sur platform / permission, quel que soit son nom)
--    - platform   : recréé avec les 5 plateformes (DC vient de develop V14)
--    - permission : supprimé, l'enum Java Permission est la seule source de vérité
-- -------------------------------------------------------------
do $$
declare
r record;
begin
for r in
select t.relname, c.conname
from pg_constraint c
         join pg_class     t on t.oid = c.conrelid
         join pg_namespace n on n.oid = t.relnamespace
where c.contype = 'c'
  and n.nspname = current_schema()
  and (
        (t.relname in ('hab_roles', 'hab_roles_aud')
            and pg_get_constraintdef(c.oid) ilike '%platform%')
     or (t.relname in ('hab_role_permission', 'hab_role_permission_aud')
            and pg_get_constraintdef(c.oid) ilike '%permission%')
    )
    loop
        execute format('alter table %I drop constraint %I', r.relname, r.conname);
end loop;
end $$;

ALTER TABLE hab_roles DROP CONSTRAINT IF EXISTS hab_roles_platform_check;
ALTER TABLE hab_roles
    ADD CONSTRAINT hab_roles_platform_check
        CHECK (platform IN ('CONSOLE_ADMIN', 'DTRP', 'ML_CHAB', 'DC', 'GESTION_FONDS'));

ALTER TABLE hab_roles_aud DROP CONSTRAINT IF EXISTS hab_roles_aud_platform_check;
ALTER TABLE hab_roles_aud
    ADD CONSTRAINT hab_roles_aud_platform_check
        CHECK (platform IN ('CONSOLE_ADMIN', 'DTRP', 'ML_CHAB', 'DC', 'GESTION_FONDS'));

ALTER TABLE hab_role_permission     DROP CONSTRAINT IF EXISTS hab_role_permission_permission_check;
ALTER TABLE hab_role_permission_aud DROP CONSTRAINT IF EXISTS hab_role_permission_aud_permission_check;

-- 2. Rôles GDF (plateforme GESTION_FONDS)

SELECT nextval('hab_roles_seq'), r.name, r.description, TRUE, NOW(), NOW(), 'GESTION_FONDS'
FROM (VALUES
          ('GDF Agence',          'Rôle Gestion des Fonds - demandes de l''agence (DA, Chef de caisse)'),
          ('GDF Caisse Centrale', 'Rôle Gestion des Fonds - traitement des demandes de toutes les agences'),
          ('GDF Support',         'Rôle Gestion des Fonds - consultation de toutes les agences')
     ) AS r(name, description)
WHERE NOT EXISTS (SELECT 1 FROM hab_roles x WHERE x.name = r.name);

INSERT INTO hab_role_permission (role_id, permission)
SELECT r.id, p.permission
FROM hab_roles r
         JOIN (VALUES
                   ('GDF Agence',          'GDF_CONSULTER_DEMANDE'),
                   ('GDF Agence',          'GDF_CREER_DEMANDE'),
                   ('GDF Agence',          'GDF_MODIFIER_DEMANDE'),
                   ('GDF Agence',          'GDF_ANNULER_DEMANDE'),
                   ('GDF Caisse Centrale', 'GDF_CONSULTER_DEMANDE'),
                   ('GDF Caisse Centrale', 'GDF_TRAITER_DEMANDE'),
                   ('GDF Caisse Centrale', 'GDF_ANNULER_DEMANDE'),
                   ('GDF Caisse Centrale', 'GDF_TOUTES_AGENCES'),
                   ('GDF Support',         'GDF_CONSULTER_DEMANDE'),
                   ('GDF Support',         'GDF_TOUTES_AGENCES')
              ) AS p(role_name, permission) ON p.role_name = r.name
WHERE r.platform = 'GESTION_FONDS'
  AND NOT EXISTS (SELECT 1 FROM hab_role_permission x WHERE x.role_id = r.id AND x.permission = p.permission);

-- 3. Profils GDF - STANDARD_USER comme DA / CONSEILLER (mêmes colonnes et même ordre que V8)
-- -------------------------------------------------------------

FROM (VALUES
          ('CHEF DE CAISSE',  'CHEF_CAISSE',     'Profil Chef de caisse'),
          ('CAISSE CENTRALE', 'CAISSE_CENTRALE', 'Profil Caisse Centrale'),
          ('SUPPORT',         'SUPPORT',         'Profil Support')
     ) AS p(name, code, description)
WHERE NOT EXISTS (SELECT 1 FROM hab_profiles x WHERE x.code = p.code);

-- -------------------------------------------------------------
-- 4. Rattachement profil -> rôle (le profil DA existant reçoit le rôle GDF Agence)
-- -------------------------------------------------------------
