select u.uid, u.first_name, u.last_name, p.code as profil, o.code as agence, o.name as agence_nom, u.actif
from users u
join hab_profiles p on p.id = u.profile_id
left join orga_unit o on o.id = u.organization_unit_id
where u.uid in ('j51204','j52317','j53488','j54061','j55729','c60010','c60011','s70005')
order by p.code, o.code, u.uid;
