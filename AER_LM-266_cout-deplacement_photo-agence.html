/**
 * Adapter du port ReferentielAgencePort : GET {users}/api/v1/internal/orga/agences/{code} (endpoint interne, public côté users).
 * Tolérant : agence inconnue, users arrêté ou réponse invalide → Optional.empty() + WARN, la demande est créée sans photo.
 */
@Slf4j
@Component
public class UtilisateursAgenceAdapter implements ReferentielAgencePort {

    private final RestClient restClient;

    public UtilisateursAgenceAdapter(@Value("${qf.users.internal-base-url}") String baseUrl) {
        this.restClient = RestClient.create(baseUrl);
    }

    @Override
    public Optional<AgenceReferentiel> trouver(String codeAgence) {
        if (codeAgence == null || codeAgence.isBlank()) {
            return Optional.empty();
        }
        try {
            AgenceInterneDTO dto = restClient.get()
                    .uri("/api/v1/internal/orga/agences/{code}", codeAgence)
                    .retrieve()
                    .body(AgenceInterneDTO.class);
            if (dto == null) {
                return Optional.empty();
            }
            return Optional.of(new AgenceReferentiel(
                    dto.code(), dto.nom(),
                    dto.zoneDistribution() == null ? null : dto.zoneDistribution().code(),
                    dto.zoneDistribution() == null ? null : dto.zoneDistribution().nom(),
                    dto.prestataire() == null ? null : dto.prestataire().id(),
                    dto.prestataire() == null ? null : dto.prestataire().nom(),
                    dto.coutDeplacement()));
        } catch (RestClientException e) {
            log.warn("Référentiel agences (users) indisponible pour l'agence {} : {}", codeAgence, e.getMessage());
            return Optional.empty();
        }
    }

    /** Forme de AgenceDTO (utilisateurs-ci) — champs inconnus ignorés. */
    @JsonIgnoreProperties(ignoreUnknown = true)
    record AgenceInterneDTO(String code, String nom, ZoneDTO zoneDistribution, PrestataireDTO prestataire,
                            BigDecimal coutDeplacement) {
        @JsonIgnoreProperties(ignoreUnknown = true)
        record ZoneDTO(String code, String nom) {}
        @JsonIgnoreProperties(ignoreUnknown = true)
        record PrestataireDTO(String id, String nom) {}
    }
}

/** AgenceDTO de utilisateurs-ci (GET /api/v1/public/users/orga/agences, AER_LM-271). Champs utiles à l'écran Caisse Centrale. */
export interface AgenceDTO {
  id: number;
  code: string;
  nom: string;
  actif?: boolean;
  zoneDistribution?: { id: number; code: string; nom: string } | null;
  prestataire?: { id: string; nom: string } | null;
  coutDeplacement?: number | null;   // numeric(12,2) côté users
}

/** Vue écran : ce que la Caisse Centrale voit d'une agence émettrice (sert aux listes déroulantes et au repli d'affichage). */
export interface AgenceReferentiel {
  code: string;
  nom: string;
  zoneCode: string | null;
  zoneNom: string | null;
  prestataireId: string | null;    // BRINKS | G4S (enum Prestataire côté users)
  prestataireNom: string | null;   // « Brink's », « G4S »
  coutDeplacement: number | null;  // MAD
}

export function versAgenceReferentiel(a: AgenceDTO): AgenceReferentiel {
  return {
    code: a.code,
    nom: a.nom,
    zoneCode: a.zoneDistribution?.code ?? null,
    zoneNom: a.zoneDistribution?.nom ?? null,
    prestataireId: a.prestataire?.id ?? null,
    prestataireNom: a.prestataire?.nom ?? null,
    coutDeplacement: a.coutDeplacement ?? null
  };
}

/** « 350,50 MAD » — même affichage que la liste des agences de la console (259). Null si non renseigné. */
export function libelleCoutDeplacement(valeur: number | null | undefined): string | null {
  if (valeur === null || valeur === undefined) {
    return null;
  }
  return `${valeur.toLocaleString('fr-FR', { minimumFractionDigits: 2, maximumFractionDigits: 2 })} MAD`;
}
