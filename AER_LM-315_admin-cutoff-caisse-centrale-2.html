Posée sur la route parent admin (app-routing), la garde reçoit l'URL cible complète dans state.url : comparaison exacte avec /admin/parametres, donc /admin/parametres/gab-hors-site (313) reste réservé aux profils admin.

import { Injectable } from '@angular/core';
import { CanActivate, Router, ActivatedRouteSnapshot, RouterStateSnapshot } from '@angular/router';
import { AuthService } from './auth.service';

/** AER_LM-315 : seule page console ouverte hors profils admin, sur permission GDF_ADMINISTRER_CUTOFF. */
export const ROUTE_ADMIN_CUTOFF = '/admin/parametres';
export const PERMISSION_ADMINISTRER_CUTOFF = 'GDF_ADMINISTRER_CUTOFF';

@Injectable({
  providedIn: 'root',
})
export class AdminGuard implements CanActivate {
  constructor(
    private router: Router,
    private authService: AuthService) {}

  canActivate(route: ActivatedRouteSnapshot, state: RouterStateSnapshot): boolean {

    const user = this.authService.currentUser;
    const isAdmin = ['SA', 'AU', 'AP'].includes(user?.profile?.code || '');

    // AER_LM-315 : un profil non admin porteur de GDF_ADMINISTRER_CUTOFF n'accède qu'à l'écran Paramètres (cut-off)
    const urlSansParametres = state.url.split('?')[0];
    const accesCutoff = urlSansParametres === ROUTE_ADMIN_CUTOFF
      && this.authService.hasPermission([PERMISSION_ADMINISTRER_CUTOFF]);

    if (!isAdmin && !accesCutoff) {
      this.router.navigate(['/']);
      return false;
    }

    return true;
  }
}
