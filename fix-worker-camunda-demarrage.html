package com.bmci.workflow.fund.management.external.camunda.worker;

import com.bmci.workflow.fund.management.external.camunda.config.CamundaProperties;
import jakarta.annotation.PreDestroy;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.camunda.bpm.client.ExternalTaskClient;
import org.camunda.bpm.client.task.ExternalTaskHandler;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.event.EventListener;
import org.springframework.stereotype.Component;

/**
 * Démarre le worker external tasks quand l'application est prête, et l'arrête proprement.
 * Le client n'est PAS un @Bean : il n'est construit que si camunda.worker.enabled=true,
 * ce qui permet aux tests (et à tout environnement sans moteur) de démarrer sans Camunda.
 * 3 retries espacés de 10 s par tâche en échec, puis incident visible en Cockpit.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class GdfWorkerConfig {

    private static final int RETRIES = 3;
    private static final long RETRY_TIMEOUT_MS = 10_000;

    private final CamundaProperties properties;
    private final InitialiserDemandeHandler initialiserDemandeHandler;
    private final ReporterJourSuivantHandler reporterJourSuivantHandler;
    private final NormaliserUrgenceHandler normaliserUrgenceHandler;
    private final CloturerJourneeHandler cloturerJourneeHandler;
    private final CloturerDemandeHandler cloturerDemandeHandler;
    private final AnnulerDemandeHandler annulerDemandeHandler;

    private ExternalTaskClient client;

    @EventListener(ApplicationReadyEvent.class)
    public void demarrerAbonnements() {
        if (!properties.getWorker().isEnabled()) {
            log.warn("Worker Camunda désactivé (camunda.worker.enabled=false)");
            return;
        }
        client = ExternalTaskClient.create()
                .baseUrl(properties.getBaseUrl())
                .workerId(properties.getWorker().getWorkerId())
                .lockDuration(properties.getWorker().getLockDurationMs())
                .asyncResponseTimeout(properties.getWorker().getAsyncResponseTimeoutMs())
                .disableBackoffStrategy()
                .build();

        abonner(GdfTopics.INITIALISER_DEMANDE,    initialiserDemandeHandler);
        abonner(GdfTopics.REPORTER_JOUR_SUIVANT,  reporterJourSuivantHandler);
        abonner(GdfTopics.NORMALISER_URGENCE,     normaliserUrgenceHandler);
        abonner(GdfTopics.CLOTURER_JOURNEE,       cloturerJourneeHandler);
        abonner(GdfTopics.CLOTURER_DEMANDE,       cloturerDemandeHandler);
        abonner(GdfTopics.ANNULER_DEMANDE,        annulerDemandeHandler);
        abonner(GdfTopics.ECLATER_BESOINS_GAB,    new TacheTraceeHandler(GdfTopics.ECLATER_BESOINS_GAB));
        abonner(GdfTopics.APPLIQUER_MODIFICATION, new TacheTraceeHandler(GdfTopics.APPLIQUER_MODIFICATION));
        abonner(GdfTopics.HISTORISER_COMMENTAIRE, new TacheTraceeHandler(GdfTopics.HISTORISER_COMMENTAIRE));
        log.info("Worker Camunda abonné aux 9 topics gdf-* sur {}", properties.getBaseUrl());
    }

    @PreDestroy
    public void arreter() {
        if (client != null) {
            log.info("Arrêt du worker Camunda");
            client.stop();
        }
    }

    private void abonner(String topic, ExternalTaskHandler handler) {
        client.subscribe(topic)
                .handler((task, service) -> {
                    try {
                        handler.execute(task, service);
                    } catch (Exception erreur) {
                        log.error("[{}] échec pour la demande {} : {}", topic, task.getBusinessKey(), erreur.getMessage(), erreur);
                        int restants = task.getRetries() == null ? RETRIES : task.getRetries() - 1;
                        service.handleFailure(task, erreur.getMessage(), erreur.toString(), restants, RETRY_TIMEOUT_MS);
                    }
                })
                .open();
    }
}
Différences vs la version précédente : plus de méthode @Bean(destroyMethod = "stop") externalTaskClient() (supprimée), @Configuration → @Component, champ client + @PreDestroy arreter(), et abonner(topic, handler) n'a plus le paramètre client.
