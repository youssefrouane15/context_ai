Choix techniques

  • Aucun fixture.detectChanges() : ngOnInit n'est pas déclenché, les méthodes sont appelées directement ; getAgencesPages est stubbé pour les rechargements post-sauvegarde.
  • Pas de stub « global » consommé par un seul test : les retours des spies sont posés dans beforeEach uniquement quand tous les tests du bloc les utilisent (même discipline que côté Java).

import { TestBed } from '@angular/core/testing';
import { HttpClientTestingModule, HttpTestingController } from '@angular/common/http/testing';
import { ReferentielService } from './referentiel.service';
import { ApiConfigService } from '../../api-config.service';
import { PrestataireDTO, ZoneDistributionDTO } from '../../../../shared/interfaces/referentiel.interface';

describe('ReferentielService', () => {
  const BASE = 'http://localhost:9090/api/v1/public/users';
  let service: ReferentielService;
  let httpMock: HttpTestingController;

  beforeEach(() => {
    TestBed.configureTestingModule({
      imports: [HttpClientTestingModule],
      providers: [
        ReferentielService,
        { provide: ApiConfigService, useValue: { buildConsoleApiUrl: () => BASE } }
      ]
    });
    service = TestBed.inject(ReferentielService);
    httpMock = TestBed.inject(HttpTestingController);
  });

  afterEach(() => httpMock.verify());

  // ---- getZonesDistributionActives ----

  describe('getZonesDistributionActives', () => {
    it('appelle GET /referentiels/zones-distribution/actives et déballe data', () => {
      const zones: ZoneDistributionDTO[] = [
        { id: 1, code: 'CASA', nom: 'Zone Casablanca', actif: true },
        { id: 2, code: 'RABAT', nom: 'Zone Rabat-Salé', actif: true }
      ];
      let resultat: ZoneDistributionDTO[] = [];

      service.getZonesDistributionActives().subscribe(z => resultat = z);

      const req = httpMock.expectOne(`${BASE}/referentiels/zones-distribution/actives`);
      expect(req.request.method).toBe('GET');
      req.flush({ success: true, message: null, data: zones, error: null });
      expect(resultat).toEqual(zones);
    });

    it('remonte le message du serveur (ApiResponse.error) en Error', () => {
      let message = '';

      service.getZonesDistributionActives().subscribe({ error: (e: Error) => message = e.message });

      httpMock.expectOne(`${BASE}/referentiels/zones-distribution/actives`)
        .flush({ success: false, message: 'Accès refusé', data: null, error: 'FORBIDDEN' },
               { status: 403, statusText: 'Forbidden' });
      expect(message).toBe('Accès refusé');
    });
  });

  // ---- getPrestataires ----

  describe('getPrestataires', () => {
    it('appelle GET /referentiels/prestataires et déballe data', () => {
      const prestataires: PrestataireDTO[] = [
        { id: 'BRINKS', nom: "Brink's" },
        { id: 'G4S', nom: 'G4S' }
      ];
      let resultat: PrestataireDTO[] = [];

      service.getPrestataires().subscribe(p => resultat = p);

      const req = httpMock.expectOne(`${BASE}/referentiels/prestataires`);
      expect(req.request.method).toBe('GET');
      req.flush({ success: true, message: null, data: prestataires, error: null });
      expect(resultat).toEqual(prestataires);
    });

    it('produit un message générique quand le serveur ne renvoie pas de message', () => {
      let message = '';

      service.getPrestataires().subscribe({ error: (e: Error) => message = e.message });

      httpMock.expectOne(`${BASE}/referentiels/prestataires`)
        .flush(null, { status: 500, statusText: 'Server Error' });
      expect(message).toContain('Erreur 500');
    });
  });
});

import { ComponentFixture, TestBed } from '@angular/core/testing';
import { FormsModule } from '@angular/forms';
import { NO_ERRORS_SCHEMA } from '@angular/core';
import { of, throwError } from 'rxjs';
import { ConsoleAgenceComponent } from './console-agence.component';
import { OrganizationService } from '../../../core/services/console/organization/organization.service';
import { ReferentielService } from '../../../core/services/console/referentiel/referentiel.service';
import { AgenceDTO, GroupeDTO, Page } from '../../../shared/interfaces/habilitation.interface';
import { PrestataireDTO, ZoneDistributionDTO } from '../../../shared/interfaces/referentiel.interface';

describe('ConsoleAgenceComponent — AER_LM-259 (configuration agence)', () => {
  let fixture: ComponentFixture<ConsoleAgenceComponent>;
  let component: ConsoleAgenceComponent;
  let organizationService: jasmine.SpyObj<OrganizationService>;
  let referentielService: jasmine.SpyObj<ReferentielService>;

  const zoneCasa: ZoneDistributionDTO = { id: 1, code: 'CASA', nom: 'Zone Casablanca', actif: true };
  const zoneRabat: ZoneDistributionDTO = { id: 2, code: 'RABAT', nom: 'Zone Rabat-Salé', actif: true };
  const zoneFesInactive: ZoneDistributionDTO = { id: 3, code: 'FES', nom: 'Zone Fès', actif: false };
  const prestataires: PrestataireDTO[] = [{ id: 'BRINKS', nom: "Brink's" }, { id: 'G4S', nom: 'G4S' }];
  const groupe: GroupeDTO = { id: 7, nom: 'Groupe Centre', actif: true, zone: null, manager: null };
  const pageVide: Page<AgenceDTO> = { content: [], totalPages: 1, totalElements: 0, size: 4, number: 0 };

  const agenceConfiguree: AgenceDTO = {
    id: 12, nom: 'Nations Unies', code: '00001', actif: true,
    groupe: { id: 7, code: null, nom: 'Groupe Centre', type: 'GROUPE', actif: true, manager: null, parent: null },
    manager: null,
    zoneDistribution: zoneCasa,
    prestataire: { id: 'BRINKS', nom: "Brink's" },
    coutDeplacement: 350.5
  };

  beforeEach(async () => {
    organizationService = jasmine.createSpyObj<OrganizationService>('OrganizationService',
      ['getAgencesPages', 'getAllGroupes', 'createAgence', 'updateAgence']);
    referentielService = jasmine.createSpyObj<ReferentielService>('ReferentielService',
      ['getZonesDistributionActives', 'getPrestataires']);

    await TestBed.configureTestingModule({
      declarations: [ConsoleAgenceComponent],
      imports: [FormsModule],
      providers: [
        { provide: OrganizationService, useValue: organizationService },
        { provide: ReferentielService, useValue: referentielService }
      ],
      schemas: [NO_ERRORS_SCHEMA]
    }).compileComponents();

    fixture = TestBed.createComponent(ConsoleAgenceComponent);
    component = fixture.componentInstance;
    organizationService.getAgencesPages.and.returnValue(of(pageVide));
    organizationService.getAllGroupes.and.returnValue(of([groupe]));
    referentielService.getZonesDistributionActives.and.returnValue(of([zoneCasa, zoneRabat]));
    referentielService.getPrestataires.and.returnValue(of(prestataires));
  });

  // ---- openAddAgenceModal ----

  describe('openAddAgenceModal', () => {
    it('charge les zones actives et les prestataires et réinitialise la configuration', () => {
      component.selectedZoneTransporteurId = 9;
      component.coutDeplacementSaisi = '99';

      component.openAddAgenceModal();

      expect(referentielService.getZonesDistributionActives).toHaveBeenCalledTimes(1);
      expect(referentielService.getPrestataires).toHaveBeenCalledTimes(1);
      expect(component.modalZonesTransporteur).toEqual([zoneCasa, zoneRabat]);
      expect(component.prestataires).toEqual(prestataires);
      expect(component.selectedZoneTransporteurId).toBe(0);
      expect(component.selectedPrestataireId).toBe('');
      expect(component.coutDeplacementSaisi).toBe('');
      expect(component.isAddAgenceModalOpen).toBeTrue();
    });

    it('laisse la modale ouverte si un référentiel est en erreur', () => {
      referentielService.getPrestataires.and.returnValue(throwError(() => new Error('KO')));

      component.openAddAgenceModal();

      expect(component.isAddAgenceModalOpen).toBeTrue();
      expect(component.modalZonesTransporteur).toEqual([zoneCasa, zoneRabat]);
      expect(component.prestataires).toEqual([]);
    });
  });

  // ---- editAgence ----

  describe('editAgence', () => {
    it('pré-remplit zone, prestataire et coût (virgule) depuis AgenceDTO', () => {
      component.editAgence(agenceConfiguree);

      expect(component.isEditingAgence).toBeTrue();
      expect(component.editingAgenceId).toBe(12);
      expect(component.selectedZoneTransporteurId).toBe(1);
      expect(component.selectedPrestataireId).toBe('BRINKS');
      expect(component.coutDeplacementSaisi).toBe('350,5');
    });

    it('conserve la zone courante dans le select même si elle est devenue inactive', () => {
      const agence: AgenceDTO = { ...agenceConfiguree, zoneDistribution: zoneFesInactive };

      component.editAgence(agence);

      expect(component.modalZonesTransporteur).toEqual([zoneCasa, zoneRabat, zoneFesInactive]);
      expect(component.selectedZoneTransporteurId).toBe(3);
    });

    it('laisse les champs vides pour une agence non configurée (compatibilité 267)', () => {
      const agence: AgenceDTO = { ...agenceConfiguree, zoneDistribution: null, prestataire: null, coutDeplacement: null };

      component.editAgence(agence);

      expect(component.selectedZoneTransporteurId).toBe(0);
      expect(component.selectedPrestataireId).toBe('');
      expect(component.coutDeplacementSaisi).toBe('');
      expect(component.modalZonesTransporteur).toEqual([zoneCasa, zoneRabat]);
    });
  });

  // ---- saveNewAgence ----

  describe('saveNewAgence', () => {
    beforeEach(() => {
      organizationService.createAgence.and.returnValue(of(agenceConfiguree));
      organizationService.updateAgence.and.returnValue(of(agenceConfiguree));
    });

    it('envoie zoneDistributionId, prestataire et coutDeplacement (virgule convertie) à la création', () => {
      component.openAddAgenceModal();
      component.newAgence = { nom: ' Nations Unies ', code: '00001' };
      component.selectedGroupeId = 7;
      component.selectedZoneTransporteurId = 1;
      component.selectedPrestataireId = 'G4S';
      component.coutDeplacementSaisi = '350,5';

      component.saveNewAgence();

      expect(organizationService.createAgence).toHaveBeenCalledWith({
        nom: 'Nations Unies', code: '00001', groupeId: 7,
        zoneDistributionId: 1, prestataire: 'G4S', coutDeplacement: 350.5
      });
      expect(component.isAddAgenceModalOpen).toBeFalse();
      expect(organizationService.getAgencesPages).toHaveBeenCalled();
    });

    it("n'envoie pas les champs non renseignés (undefined → absents du JSON)", () => {
      component.openAddAgenceModal();
      component.newAgence = { nom: 'Agence Test', code: '00002' };
      component.selectedGroupeId = 7;

      component.saveNewAgence();

      const envoye = organizationService.createAgence.calls.mostRecent().args[0];
      expect(envoye.zoneDistributionId).toBeUndefined();
      expect(envoye.prestataire).toBeUndefined();
      expect(envoye.coutDeplacement).toBeUndefined();
      expect(JSON.stringify(envoye)).not.toContain('coutDeplacement');
    });

    it('utilise updateAgence avec l\'id en modification', () => {
      component.editAgence(agenceConfiguree);
      component.selectedPrestataireId = 'G4S';

      component.saveNewAgence();

      expect(organizationService.updateAgence).toHaveBeenCalledWith(jasmine.objectContaining({
        id: 12, code: '00001', prestataire: 'G4S', zoneDistributionId: 1, coutDeplacement: 350.5
      }));
      expect(organizationService.createAgence).not.toHaveBeenCalled();
    });

    it('refuse un coût invalide sans appeler le serveur', () => {
      component.openAddAgenceModal();
      component.newAgence = { nom: 'Agence Test', code: '00002' };
      component.coutDeplacementSaisi = '12,345';

      component.saveNewAgence();

      expect(component.errorMessage).toContain('coût de déplacement');
      expect(organizationService.createAgence).not.toHaveBeenCalled();
      expect(component.isAddAgenceModalOpen).toBeTrue();
    });

    it('garde la validation existante nom / code', () => {
      component.openAddAgenceModal();
      component.newAgence = { nom: '', code: '00002' };

      component.saveNewAgence();

      expect(component.errorMessage).toBe('Veuillez remplir tous les champs obligatoires.');
      expect(organizationService.createAgence).not.toHaveBeenCalled();
    });

    it('affiche le message serveur en cas d\'erreur (409 doublon)', () => {
      organizationService.createAgence.and.returnValue(throwError(() => new Error('Une agence avec ce code existe déjà')));
      component.openAddAgenceModal();
      component.newAgence = { nom: 'Agence Test', code: '00001' };

      component.saveNewAgence();

      expect(component.errorMessage).toBe('Une agence avec ce code existe déjà');
      expect(component.isAddAgenceModalOpen).toBeTrue();
    });
  });

  // ---- getCoutDeplacementAffiche ----

  describe('getCoutDeplacementAffiche', () => {
    it('formate le coût avec deux décimales, virgule et devise', () => {
      expect(component.getCoutDeplacementAffiche(agenceConfiguree)).toBe('350,50 MAD');
      expect(component.getCoutDeplacementAffiche({ ...agenceConfiguree, coutDeplacement: 1200 })).toBe('1200,00 MAD');
    });

    it("affiche un tiret quand le coût n'est pas renseigné", () => {
      expect(component.getCoutDeplacementAffiche({ ...agenceConfiguree, coutDeplacement: null })).toBe('-');
    });
  });

  // ---- onCoutDeplacementInput ----

  describe('onCoutDeplacementInput', () => {
    it('ne conserve que chiffres, virgule et point', () => {
      const input = document.createElement('input');
      input.value = '3a5,0b€';

      component.onCoutDeplacementInput({ target: input } as unknown as Event);

      expect(input.value).toBe('35,0');
      expect(component.coutDeplacementSaisi).toBe('35,0');
    });
  });

  // ---- closeAddAgenceModal ----

  describe('closeAddAgenceModal', () => {
    it('réinitialise la configuration', () => {
      component.editAgence(agenceConfiguree);

      component.closeAddAgenceModal();

      expect(component.selectedZoneTransporteurId).toBe(0);
      expect(component.selectedPrestataireId).toBe('');
      expect(component.coutDeplacementSaisi).toBe('');
      expect(component.modalZonesTransporteur).toEqual([]);
      expect(component.prestataires).toEqual([]);
    });
  });
});
