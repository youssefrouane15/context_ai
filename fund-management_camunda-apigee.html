0. Analyse

Flux : fund-management → Apigee (Bearer) → Camunda. Le token vient de POST …/auth/oauth/v2/token en client_credentials (Basic client/secret, form-urlencoded, scope non envoyé comme dans Postman), dure 599 s.

Fund-management appelle le moteur par deux canaux, les deux sont équipés :

  1. camundaRestClient (start, correlate, complete-task, history) → ApigeeBearerInterceptor : pose le Bearer, un rejeu sur 401 avec token renouvelé.
  2. ExternalTaskClient du worker (fetchAndLock / complete / failure) → ApigeeWorkerInterceptor via addInterceptor du builder Camunda.
Au milieu, ApigeeTokenProvider : un token en cache, thread-safe, renouvelé quand il reste moins de 60 s (marge configurable) ou après un 401. Aligné sur Oauth2TokenRefreshInterceptor d'utilisateurs-ci (Basic, form, 401 → refresh) mais en RestClient, sans OkHttp ni Gson : fund-management n'a pas ces dépendances.

Activation par propriété camunda.apigee.enabled (défaut false) : en local rien ne change (aucun bean Apigee créé, ObjectProvider vide). Aucun BPMN, aucun Flyway, aucune exception nouvelle : un échec de l'appel token est une RestClientException → 502 par le handler existant.

2 points à confirmer avec le tech lead avant déploiement (le code ne les suppose pas) :
1. Le chemin de Camunda derrière Apigee (QF_CAMUNDA_BASE_URL) — requête de contrôle en §4.
2. L'hôte s'appelle morocco-mtls.apigee… : si Apigee exige un certificat client (mTLS) en plus du Bearer, il faudra un keystore monté sur le RestClient et sur le client worker — hors de ce doc tant que ce n'est pas confirmé (le test Postman de §4 le dira : 200 = pas de mTLS côté client).

package com.bmci.workflow.fund.management.external.camunda.auth;

import com.bmci.workflow.fund.management.external.camunda.config.CamundaProperties;
import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import com.fasterxml.jackson.annotation.JsonProperty;
import lombok.extern.slf4j.Slf4j;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;

import java.time.Clock;
import java.time.Instant;

/**
 * Fournit le Bearer Apigee pour tous les appels au moteur Camunda (RestClient + worker).
 * Un seul token en cache, partagé par tous les threads ; renouvelé avant expiration (marge configurable)
 * ou après un 401 (token révoqué côté Apigee). Même contrat que Oauth2TokenRefreshInterceptor d'utilisateurs-ci,
 * sans OkHttp : fund-management parle RestClient.
 * Les erreurs de l'appel token sont des RestClientException → 502 via le handler existant (moteur injoignable).
 */
@Slf4j
@Component
@ConditionalOnProperty(prefix = "camunda.apigee", name = "enabled", havingValue = "true")
public class ApigeeTokenProvider {

    private final CamundaProperties.Apigee apigee;
    private final RestClient tokenClient;
    private final Clock clock;
    private volatile Jeton jeton;

    public ApigeeTokenProvider(CamundaProperties properties) {
        this(properties.getApigee(), RestClient.builder().build(), Clock.systemUTC());
    }

    /** Constructeur de test (serveur mock + horloge pilotée). */
    ApigeeTokenProvider(CamundaProperties.Apigee apigee, RestClient tokenClient, Clock clock) {
        this.apigee = apigee;
        this.tokenClient = tokenClient;
        this.clock = clock;
    }

    /** Token valide : celui du cache s'il lui reste plus que la marge, sinon un nouveau. */
    public String token() {
        Jeton courant = jeton;
        if (courant != null && courant.valide(clock.instant(), apigee.getRefreshMarginSeconds())) {
            return courant.accessToken();
        }
        return renouveler();
    }

    /** Après un 401 : on n'invalide le cache que s'il porte encore le token rejeté (un autre thread a pu le renouveler). */
    public String renouvelerApresRejet(String tokenRejete) {
        synchronized (this) {
            Jeton courant = jeton;
            if (courant != null && courant.accessToken().equals(tokenRejete)) {
                jeton = null;
            }
        }
        return token();
    }

    private synchronized String renouveler() {
        Jeton courant = jeton;
        if (courant != null && courant.valide(clock.instant(), apigee.getRefreshMarginSeconds())) {
            return courant.accessToken();          // renouvelé par un autre thread pendant l'attente du verrou
        }
        MultiValueMap<String, String> formulaire = new LinkedMultiValueMap<>();
        formulaire.add("grant_type", "client_credentials");

        ReponseToken reponse = tokenClient.post()
                .uri(apigee.getTokenUrl())
                .headers(h -> h.setBasicAuth(apigee.getClientId(), apigee.getClientSecret()))
                .contentType(MediaType.APPLICATION_FORM_URLENCODED)
                .accept(MediaType.APPLICATION_JSON)
                .body(formulaire)
                .retrieve()
                .body(ReponseToken.class);

        if (reponse == null || reponse.accessToken() == null || reponse.accessToken().isBlank()) {
            throw new RestClientException("Apigee : réponse token sans access_token");
        }
        long dureeSecondes = Long.parseLong(reponse.expiresIn().trim());      // Apigee renvoie "599" (chaîne)
        jeton = new Jeton(reponse.accessToken(), clock.instant().plusSeconds(dureeSecondes));
        log.info("Token Apigee renouvelé (valide {} s)", dureeSecondes);
        return jeton.accessToken();
    }

    record Jeton(String accessToken, Instant expireA) {
        boolean valide(Instant maintenant, long margeSecondes) {
            return maintenant.plusSeconds(margeSecondes).isBefore(expireA);
        }
    }

    @JsonIgnoreProperties(ignoreUnknown = true)
    record ReponseToken(@JsonProperty("access_token") String accessToken,
                        @JsonProperty("expires_in") String expiresIn) {
    }
}

package com.bmci.workflow.fund.management.external.camunda.auth;

import lombok.RequiredArgsConstructor;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.http.HttpRequest;
import org.springframework.http.client.ClientHttpRequestExecution;
import org.springframework.http.client.ClientHttpRequestInterceptor;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.stereotype.Component;

import java.io.IOException;

/**
 * Pose le Bearer Apigee sur chaque appel du RestClient Camunda (start, correlate, task complete, history…).
 * Sur 401 : un seul rejeu avec un token renouvelé — au-delà, la réponse remonte telle quelle.
 */
@Component
@ConditionalOnProperty(prefix = "camunda.apigee", name = "enabled", havingValue = "true")
@RequiredArgsConstructor
public class ApigeeBearerInterceptor implements ClientHttpRequestInterceptor {

    private static final int NON_AUTORISE = 401;

    private final ApigeeTokenProvider tokens;

    @Override
    public ClientHttpResponse intercept(HttpRequest request, byte[] body, ClientHttpRequestExecution execution) throws IOException {
        String token = tokens.token();
        ClientHttpResponse reponse = executer(request, body, execution, token);
        if (reponse.getStatusCode().value() != NON_AUTORISE) {
            return reponse;
        }
        reponse.close();
        return executer(request, body, execution, tokens.renouvelerApresRejet(token));
    }

    private static ClientHttpResponse executer(HttpRequest request, byte[] body, ClientHttpRequestExecution execution, String token) throws IOException {
        request.getHeaders().setBearerAuth(token);
        return execution.execute(request, body);
    }
}

package com.bmci.workflow.fund.management.external.camunda.auth;

import lombok.RequiredArgsConstructor;
import org.camunda.bpm.client.interceptor.ClientRequestContext;
import org.camunda.bpm.client.interceptor.ClientRequestInterceptor;

/**
 * Pose le Bearer Apigee sur chaque requête du worker (fetchAndLock, complete, failure, extendLock).
 * Pas de bean : construit dans GdfWorkerConfig uniquement quand ApigeeTokenProvider est présent.
 * Pas de rejeu sur 401 ici (le client Camunda ne l'expose pas) : le renouvellement anticipé (marge) couvre le cas nominal,
 * et le long polling repart de lui-même à l'itération suivante avec un token neuf.
 */
@RequiredArgsConstructor
public class ApigeeWorkerInterceptor implements ClientRequestInterceptor {

    private final ApigeeTokenProvider tokens;

    @Override
    public void intercept(ClientRequestContext requestContext) {
        requestContext.addHeader("Authorization", "Bearer " + tokens.token());
    }
}

package com.bmci.workflow.fund.management.external.camunda.auth;

import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.http.HttpStatus;
import org.springframework.http.client.ClientHttpRequestExecution;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.mock.http.client.MockClientHttpRequest;
import org.springframework.mock.http.client.MockClientHttpResponse;

import java.io.IOException;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertSame;

class ApigeeBearerInterceptorTest {

    private static final byte[] CORPS = "{}".getBytes();

    private ApigeeTokenProvider tokens;
    private ClientHttpRequestExecution execution;
    private MockClientHttpRequest requete;
    private ApigeeBearerInterceptor interceptor;

    @BeforeEach
    void setUp() {
        tokens = Mockito.mock(ApigeeTokenProvider.class);
        execution = Mockito.mock(ClientHttpRequestExecution.class);
        requete = new MockClientHttpRequest(HttpMethod.POST, "/process-definition/key/gestion-des-fonds/start");
        interceptor = new ApigeeBearerInterceptor(tokens);
    }

    @Test
    void poseLeBearerEtRendLaReponseSur200() throws IOException {
        Mockito.when(tokens.token()).thenReturn("abc");
        ClientHttpResponse ok = new MockClientHttpResponse(new byte[0], HttpStatus.OK);
        Mockito.when(execution.execute(requete, CORPS)).thenReturn(ok);

        ClientHttpResponse resultat = interceptor.intercept(requete, CORPS, execution);

        assertSame(ok, resultat);
        assertEquals("Bearer abc", requete.getHeaders().getFirst(HttpHeaders.AUTHORIZATION));
        Mockito.verify(tokens, Mockito.never()).renouvelerApresRejet(Mockito.anyString());
    }

    @Test
    void rejoueUneSeuleFoisAvecUnNouveauTokenSur401() throws IOException {
        Mockito.when(tokens.token()).thenReturn("abc");
        Mockito.when(tokens.renouvelerApresRejet("abc")).thenReturn("def");
        ClientHttpResponse nonAutorise = new MockClientHttpResponse(new byte[0], HttpStatus.UNAUTHORIZED);
        ClientHttpResponse ok = new MockClientHttpResponse(new byte[0], HttpStatus.OK);
        Mockito.when(execution.execute(requete, CORPS)).thenReturn(nonAutorise, ok);

        ClientHttpResponse resultat = interceptor.intercept(requete, CORPS, execution);

        assertSame(ok, resultat);
        assertEquals("Bearer def", requete.getHeaders().getFirst(HttpHeaders.AUTHORIZATION));
        Mockito.verify(execution, Mockito.times(2)).execute(requete, CORPS);
    }

    @Test
    void deux401DeSuite_rendLaSecondeReponseSansBoucler() throws IOException {
        Mockito.when(tokens.token()).thenReturn("abc");
        Mockito.when(tokens.renouvelerApresRejet("abc")).thenReturn("def");
        ClientHttpResponse premier = new MockClientHttpResponse(new byte[0], HttpStatus.UNAUTHORIZED);
        ClientHttpResponse second = new MockClientHttpResponse(new byte[0], HttpStatus.UNAUTHORIZED);
        Mockito.when(execution.execute(requete, CORPS)).thenReturn(premier, second);

        assertSame(second, interceptor.intercept(requete, CORPS, execution));
        Mockito.verify(execution, Mockito.times(2)).execute(requete, CORPS);
    }
}
