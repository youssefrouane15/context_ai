/**

 *   1. GET /api/v1/internal/users/{uid}            → l'unité d'organisation (id) du créateur de la demande, contrôlée = AGENCE de code branchCode
 *   2. GET /api/v1/internal/users/by-orga-unit?orgaUnitId=  → les utilisateurs actifs de cette agence
 * puis filtre : enabled, email renseigné, une permission GDF_CREER_DEMANDE dans les rôles du profil (DA, Chef de caisse).
 * Même style qu'UtilisateursDirecteurAdapter : RestClient.create (pas un bean), tolérant — toute erreur → liste vide.
 */
@Component
@Slf4j
public class UtilisateursGestionDemandesAdapter implements UtilisateursGestionDemandesPort {

    static final String PERMISSION_GESTION_DEMANDES = "GDF_CREER_DEMANDE";
    static final String TYPE_AGENCE = "AGENCE";

    private final RestClient client;

    public UtilisateursGestionDemandesAdapter(@Value("${qf.users.internal-base-url}") String baseUrl) {
        this.client = RestClient.create(baseUrl);
    }

    @Override
    public List<UtilisateurAgence> utilisateursGestionDemandes(String uidReference, String branchCode) {
        if (uidReference == null || uidReference.isBlank() || branchCode == null || branchCode.isBlank()) { return List.of(); }
        try {
            UtilisateurInterneDTO reference = client.get()
                    .uri("/api/v1/internal/users/{uid}", uidReference)
                    .retrieve()
                    .body(UtilisateurInterneDTO.class);
            if (reference == null || reference.orgaUnit() == null || reference.orgaUnit().id() == null
                    || !TYPE_AGENCE.equals(reference.orgaUnit().type()) || !branchCode.equals(reference.orgaUnit().code())) {
                log.warn("[notification] {} n'est pas rattaché à l'agence {} : destinataires non résolus", uidReference, branchCode);
                return List.of();
            }
            Long orgaUnitId = reference.orgaUnit().id();
            List<UtilisateurInterneDTO> utilisateurs = client.get()
                    .uri(b -> b.path("/api/v1/internal/users/by-orga-unit").queryParam("orgaUnitId", orgaUnitId).build())
                    .retrieve()
                    .body(new ParameterizedTypeReference<List<UtilisateurInterneDTO>>() { });
            if (utilisateurs == null) { return List.of(); }
            return utilisateurs.stream()
                    .filter(u -> Boolean.TRUE.equals(u.enabled()))
                    .filter(u -> u.email() != null && !u.email().isBlank())
                    .filter(UtilisateursGestionDemandesAdapter::gereLesDemandes)
                    .map(u -> new UtilisateurAgence(u.uid(), u.email(), nomComplet(u)))
                    .toList();
        } catch (RestClientException e) {
            log.warn("[notification] utilisateurs-ci injoignable pour l'agence {} : {}", branchCode, e.getMessage());
            return List.of();
        }
    }

    private static boolean gereLesDemandes(UtilisateurInterneDTO u) {
        if (u.profile() == null || u.profile().roles() == null) { return false; }
        return u.profile().roles().stream()
                .filter(r -> r != null && r.permissions() != null)
                .flatMap(r -> r.permissions().stream())
                .map(PermissionInterneDTO::code)
                .anyMatch(PERMISSION_GESTION_DEMANDES::equals);
    }

    public void notifier(CashRequestDTO demande, EvenementDemande evenement) {
        if (demande == null) { return; }
        try {
            List<UtilisateurAgence> destinataires = destinatairesDe(demande, evenement);
            if (destinataires.isEmpty()) {
                log.warn("[notification] {} {} : aucun destinataire avec email pour l'agence {} — email non envoyé",
                        evenement, demande.getUniqueRequestRef(), demande.getBranchCode());
                return;
            }
            String sujet = "[QlickFlow] " + evenement.getLibelle() + " " + demande.getUniqueRequestRef()
                    + " — Agence " + demande.getBranchCode();
            for (UtilisateurAgence destinataire : destinataires) {
                String corps = gabarit.construire(demande, evenement, destinataire.nomComplet());
                notificationPort.envoyer(new EmailNotification(destinataire.email(), sujet, corps));
                log.info("[notification] {} {} → {}", evenement, demande.getUniqueRequestRef(), destinataire.email());
            }
        } catch (RuntimeException e) {
            log.error("[notification] {} {} : échec de l'envoi, demande non impactée — {}",
                    evenement, demande.getUniqueRequestRef(), e.getMessage());
        }
    }

    private List<UtilisateurAgence> destinatairesDe(CashRequestDTO demande, EvenementDemande evenement) {
        if (evenement == EvenementDemande.DECLASSEMENT_URGENCE) {
            List<UtilisateurAgence> gestionnaires =
                    utilisateursGestionDemandesPort.utilisateursGestionDemandes(demande.getUser(), demande.getBranchCode());
            if (!gestionnaires.isEmpty()) { return gestionnaires; }
            log.warn("[notification] {} {} : aucun utilisateur gestion demandes trouvé pour l'agence {}, repli sur le Directeur d'agence",
                    evenement, demande.getUniqueRequestRef(), demande.getBranchCode());
        }
        return directeurDe(demande.getBranchCode())
                .filter(da -> da.email() != null && !da.email().isBlank())
                .map(da -> List.of(new UtilisateurAgence(da.uid(), da.email(), da.nomComplet())))
                .orElse(List.of());
    }

    private Optional<DirecteurAgence> directeurDe(String branchCode) {

        return userProfileRepository.uidsParProfil(PROFIL_DA).stream()
                .map(uid -> directeurAgencePort.siDirecteurDe(uid, branchCode))
                .flatMap(Optional::stream)
                .findFirst();
    }
