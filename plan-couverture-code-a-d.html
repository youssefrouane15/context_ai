import com.bmci.q360.domain.retail.entity.CashOperation;

import com.bmci.q360.domain.entity.enums.RequestType;

import org.mapstruct.factory.Mappers;

import java.time.LocalDateTime;

/**
 * ACTION A — Le web mapper est une FRONTIERE DE SECURITE : ces tests prouvent
 * que les champs pilotes serveur (id, ref, statuts, dates) sont IGNORES en
 * entree, meme si un client malicieux les envoie. Impl MapStruct REELLE.
 */
class CashRequestWebMapperTest {

    private final CashRequestWebMapper mapper = Mappers.getMapper(CashRequestWebMapper.class);

    // ==================== entree : les champs serveur sont ignores ====================

    @Test
    void toDomainShouldIgnoreServerControlledFieldsEvenIfClientSendsThem() {
        // Un client "malicieux" envoie id, ref et statut deja remplis
        CashRequestDTO dto = CashRequestDTO.builder()
                .id(UUID.randomUUID())                       // pilote serveur
                .uniqueRequestRef("RT-9999-999999")          // pilote serveur (sequence)
                .requestStatus("VALIDATED")                  // pilote serveur !
                .user("j43709")
                .establishment("001")
                .branchCode("AG-CASA-01")
                .operations(List.of(CashOperationDTO.builder()
                        .id(UUID.randomUUID())               // pilote serveur
                        .operationSens(RequestType.COLLECTION)
                        .operationStatus("CANCELLED")        // pilote serveur !
                        .createdAt(LocalDateTime.now())      // pilote serveur
                        .updatedAt(LocalDateTime.now())      // pilote serveur
                        .isUrgent("YES")
                        .build()))
                .build();

        CashRequest domain = mapper.toDomain(dto);

        // La frontiere tient : rien de tout ca n'est passe
        assertNull(domain.getId(), "l'id client doit être ignoré");
        assertNull(domain.getUniqueRequestRef(), "la réf client doit être ignorée");
        assertNull(domain.getRequestStatus(), "le statut client doit être ignoré");
        CashOperation operation = domain.getOperations().get(0);
        assertNull(operation.getId());
        assertNull(operation.getOperationStatus(), "le statut d'opération client doit être ignoré");
        assertNull(operation.getCreatedAt());
        assertNull(operation.getUpdatedAt());

        // Les champs legitimes passent
        assertEquals("j43709", domain.getUidUser());          // user -> uidUser
        assertEquals("001", domain.getEstablishment());
        assertEquals(UrgencyFlag.YES, operation.getIsUrgent());
        assertEquals(RequestType.COLLECTION, operation.getOperationSens());
    }

    // ==================== sortie : l'aller-retour restitue tout ====================

    @Test
    void toResponseDtoShouldExposeServerFields() {
        CashRequest domain = new CashRequest();
        domain.setId(new CashRequestId(UUID.randomUUID()));
        domain.setUniqueRequestRef("RT-2026-000001");
        domain.setRequestStatus(RequestStatus.INITIATED);
        domain.setUidUser("j43709");
        domain.setEstablishment("001");

        CashRequestDTO dto = mapper.toResponseDTO(domain);

        assertEquals(domain.getId().getValue(), dto.getId());
        assertEquals("RT-2026-000001", dto.getUniqueRequestRef());
        assertEquals("INITIATED", dto.getRequestStatus());
        assertEquals("j43709", dto.getUser());                // uidUser -> user
    }

    // ==================== VO <-> UUID : null-safe des deux cotes ====================

    @Test
    void voConvertersShouldBeNullSafe() {
        assertNull(mapper.toCashRequestId(null));
        assertNull(mapper.fromCashRequestId(null));
        UUID raw = UUID.randomUUID();
        assertEquals(raw, mapper.fromCashRequestId(mapper.toCashRequestId(raw)));
    }
}

BLes chemins heureux des use cases

L'astuce du test createShouldValidateBeforeSaving : on ne peut pas « voir » l'ordre validation→save avec des mocks classiques (la validation est une méthode du domaine réel, pas un mock). La preuve passe par un ArgumentCaptor : on capture le domaine au moment du save() et on vérifie qu'il porte déjà les défauts serveur — que seule la validation pose. S'ils y sont, elle a forcément tourné avant.

        CashRequest domain = new CashRequest();
        domain.setUidUser("j43709");
        domain.setEstablishment("001");

    @Test
    void createShouldValidateBeforeSaving() {
        CashRequest domain = validDomain();
        Mockito.when(webMapper.toDomain(Mockito.any(CashRequestDTO.class))).thenReturn(domain);
        Mockito.when(cashDemandRepository.save(Mockito.any())).thenReturn(domain);
        Mockito.when(webMapper.toResponseDTO(domain)).thenReturn(new CashRequestDTO());

        useCase.create(new CashRequestDTO());

        // La PREUVE que validateAndCompute a tourne AVANT le save : le domaine
        // capture porte les defauts serveur que seule la validation pose.
        ArgumentCaptor<CashRequest> saved = ArgumentCaptor.forClass(CashRequest.class);
        Mockito.verify(cashDemandRepository).save(saved.capture());
        CashOperation operation = saved.getValue().getOperations().get(0);
        assertEquals(UrgencyFlag.NO, operation.getIsUrgent());          // defaut pose
        assertEquals("INITIATED", operation.getOperationStatus());      // defaut pose
        assertEquals(RequestStatus.INITIATED, saved.getValue().getRequestStatus());
        Mockito.verify(webMapper).toResponseDTO(domain);                // mapping retour
    }

    void updateShouldValidateThenDelegateWithSameId() {

class CorpoDenominationRepositoryImplTest {

            jpaRepository, Mappers.getMapper(CorpoCashRequestPersistenceMapper.class));

    void shouldMapEntitiesToDomainWithStableCode() {
        CorpoDenominationEntity entity = new CorpoDenominationEntity();
        entity.setDenominationCode("CENTIME_50");
        entity.setFaceValue(new BigDecimal("0.50"));
        entity.setDenominationLabel("Pièce de 50 centimes");
        entity.setItemType("COIN");

        List<CorpoDenomination> result = repository.findAllActiveDenominations();

        assertEquals("CENTIME_50", result.get(0).getDenominationCode());
        assertEquals(0, new BigDecimal("0.50").compareTo(result.get(0).getFaceValue()));
    }

    @Test
    void shouldComputePerCurrencyAndGlobalTotalsAcrossTwoCurrencies() {
        Map<String, BigDecimal> referentiel = Map.of(
                "BILLET_200", new BigDecimal("200"),
                "BILLET_EUR_100", new BigDecimal("100"));

        OperationCurrencyDetail detailMad = detail("MAD", "BILLET_200", 10);       // 2000
        OperationCurrencyDetail detailEur = detail("EUR", "BILLET_EUR_100", 5);    // 500
        CashOperation operation = new CashOperation();
        operation.setOperationSens(RequestType.SUPPLY);
        operation.setOperationType(OperationType.CASH_DESK);
        operation.setCurrencyDetails(List.of(detailMad, detailEur));

        requestWith(operation).validateAndCompute(referentiel);

        assertEquals(0, new BigDecimal("2000").compareTo(detailMad.getTotalAmountByCurrency()));
        assertEquals(0, new BigDecimal("500").compareTo(detailEur.getTotalAmountByCurrency()));
        assertEquals(0, new BigDecimal("2500").compareTo(operation.getTotalAmount()));  // global = somme
    }

    private OperationCurrencyDetail detail(String iso, String code, int quantity) {
        Denomination denomination = new Denomination();
        denomination.setDenominationCode(code);
        Currency currency = new Currency();
        currency.setIsoCode(iso);
        OperationCurrencyDetail detail = new OperationCurrencyDetail();
        detail.setCurrency(currency);
        detail.setOperationDenominations(List.of(new OperationDenomination(denomination, quantity)));
        return detail;
    }
