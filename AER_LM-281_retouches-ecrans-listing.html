/** Caisse Centrale : seulement En attente / Traitée / Annulée — pas de brouillon, pas de modification en cours
 *  (le back exclut aussi ces deux statuts de son listing, quel que soit le filtre). */
export const STATUTS_CAISSE_CENTRALE: readonly OptionFiltre[] =
  STATUTS_AGENCE.filter(s => s.code !== 'INITIATED' && s.code !== 'IN_MODIFICATION');

/** Icône Material de la pastille de statut (maquettes 281). */
export function iconeStatut(statut: StatutDemande): string {
  switch (statut) {
    case 'INITIEE':                 return 'edit_note';
    case 'SOUMISE':                 return 'pending';
    case 'EN_MODIFICATION':         return 'edit';
    case 'TRAITEE':                 return 'check_circle';
    case 'ANNULEE_AGENCE':
    case 'ANNULEE_CAISSE_CENTRALE': return 'cancel';
  }
}
