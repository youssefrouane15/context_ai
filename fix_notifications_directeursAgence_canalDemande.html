package com.bmci.workflow.fund.management.external.users;

import com.bmci.q360.domain.retail.valueobjects.referentiel.UtilisateurAgence;
import com.bmci.workflow.fund.management.external.users.dto.PermissionInterneDTO;
import com.bmci.workflow.fund.management.external.users.dto.UtilisateurInterneDTO;
import com.bmci.workflow.fund.management.service.domain.retail.ports.output.referentiel.UtilisateursGestionDemandesPort;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientException;

import java.util.List;
import java.util.function.Predicate;

/**
 * AER_LM-333 — utilisateurs « gestion demandes » d'une agence, lus chez utilisateurs-ci avec deux endpoints internes existants :
 *   1. GET /api/v1/internal/users/{uid}                   → l'unité d'organisation (id) du créateur de la demande, contrôlée = AGENCE de code branchCode
 *   2. GET /api/v1/internal/users/by-orga-unit?orgaUnitId= → les utilisateurs actifs de cette agence
 * puis filtre : enabled, email renseigné, puis
 *   - utilisateursGestionDemandes : une permission GDF_CREER_DEMANDE dans les rôles du profil (DA, Chef de caisse) ;
 *   - directeursAgence            : le profil DA (notifications soumission / mise à jour / annulation).
 * Même style qu'UtilisateursDirecteurAdapter : RestClient.create (pas un bean), tolérant — toute erreur → liste vide.
 */
@Component
@Slf4j
public class UtilisateursGestionDemandesAdapter implements UtilisateursGestionDemandesPort {

    static final String PERMISSION_GESTION_DEMANDES = "GDF_CREER_DEMANDE";
    static final String PROFIL_DA = "DA";
    static final String TYPE_AGENCE = "AGENCE";

    private final RestClient client;

    public UtilisateursGestionDemandesAdapter(@Value("${qf.users.internal-base-url}") String baseUrl) {
        this.client = RestClient.create(baseUrl);
    }

    @Override
    public List<UtilisateurAgence> utilisateursGestionDemandes(String uidReference, String branchCode) {
        return utilisateursAgence(uidReference, branchCode, UtilisateursGestionDemandesAdapter::gereLesDemandes);
    }

    @Override
    public List<UtilisateurAgence> directeursAgence(String uidReference, String branchCode) {
        return utilisateursAgence(uidReference, branchCode, UtilisateursGestionDemandesAdapter::estDirecteur);
    }

    /** Utilisateurs actifs avec email de l'agence de uidReference (contrôlée = branchCode), retenus par le filtre. */
    private List<UtilisateurAgence> utilisateursAgence(String uidReference, String branchCode, Predicate<UtilisateurInterneDTO> retenu) {
        if (uidReference == null || uidReference.isBlank() || branchCode == null || branchCode.isBlank()) { return List.of(); }
        try {
            UtilisateurInterneDTO reference = client.get()
                    .uri("/api/v1/internal/users/{uid}", uidReference)
                    .retrieve()
                    .body(UtilisateurInterneDTO.class);
            if (reference == null || reference.orgaUnit() == null || reference.orgaUnit().id() == null
                    || !TYPE_AGENCE.equals(reference.orgaUnit().type()) || !branchCode.equals(reference.orgaUnit().code())) {
                log.warn("[notification] {} n'est pas rattaché à l'agence {} : destinataires non résolus", uidReference, branchCode);
                return List.of();
            }
            Long orgaUnitId = reference.orgaUnit().id();
            List<UtilisateurInterneDTO> utilisateurs = client.get()
                    .uri(b -> b.path("/api/v1/internal/users/by-orga-unit").queryParam("orgaUnitId", orgaUnitId).build())
                    .retrieve()
                    .body(new ParameterizedTypeReference<List<UtilisateurInterneDTO>>() { });
            if (utilisateurs == null) { return List.of(); }
            return utilisateurs.stream()
                    .filter(u -> Boolean.TRUE.equals(u.enabled()))
                    .filter(u -> u.email() != null && !u.email().isBlank())
                    .filter(retenu)
                    .map(u -> new UtilisateurAgence(u.uid(), u.email(), nomComplet(u)))
                    .toList();
        } catch (RestClientException e) {
            log.warn("[notification] utilisateurs-ci injoignable pour l'agence {} : {}", branchCode, e.getMessage());
            return List.of();
        }
    }

    private static boolean gereLesDemandes(UtilisateurInterneDTO u) {
        if (u.profile() == null || u.profile().roles() == null) { return false; }
        return u.profile().roles().stream()
                .filter(r -> r != null && r.permissions() != null)
                .flatMap(r -> r.permissions().stream())
                .map(PermissionInterneDTO::code)
                .anyMatch(PERMISSION_GESTION_DEMANDES::equals);
    }

    /** Profil Directeur d'agence (code « DA » du référentiel utilisateurs-ci). */
    static boolean estDirecteur(UtilisateurInterneDTO u) {
        return u.profile() != null && PROFIL_DA.equals(u.profile().code());
    }

    private static String nomComplet(UtilisateurInterneDTO u) {
        String nom = ((u.firstName() == null ? "" : u.firstName()) + " " + (u.lastName() == null ? "" : u.lastName())).trim();
        return nom.isEmpty() ? u.uid() : nom;
    }
}
