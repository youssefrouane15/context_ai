package com.bmci.workflow.fund.management.application.rest.logging;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.slf4j.spi.LoggingEventBuilder;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

/**
 * AER_LM-350 — journal d'accès HTTP au format ECS, identique à celui d'api-gateway-ci :
 * une ligne par requête, logger HTTP_ACCESS, message « HTTP request completed ».
 * Placé après le filtre d'observation (trace.id / span.id présents) et avant Spring Security
 * (les 401 / 403 sont tracés aussi).
 */
@Component
@Order(Ordered.HIGHEST_PRECEDENCE + 10)
public class HttpAccessLogFilter extends OncePerRequestFilter {

    static final String LOGGER_NAME = "HTTP_ACCESS";
    static final String MESSAGE = "HTTP request completed";
    static final String CHEMIN_ACTUATOR = "/actuator";

    /** Attribut de requête posé par UserLogContextFilter une fois l'appelant authentifié. */
    static final String ATTRIBUT_UTILISATEUR = HttpAccessLogFilter.class.getName() + ".utilisateur";

    private static final int ERREUR_CLIENT = 400;
    private static final int ERREUR_SERVEUR = 500;

    private static final Logger ACCESS_LOG = LoggerFactory.getLogger(LOGGER_NAME);

    /** Les sondes de santé (toutes les quelques secondes) ne sont pas tracées. */
    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        String chemin = request.getRequestURI().substring(request.getContextPath().length());
        return chemin.startsWith(CHEMIN_ACTUATOR);
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {
        long debut = System.nanoTime();
        boolean termine = false;
        try {
            filterChain.doFilter(request, response);
            termine = true;
        } finally {
            // exception non gérée : le conteneur répondra 500, le statut n'est pas encore posé sur la réponse
            int statut = termine ? response.getStatus() : Math.max(response.getStatus(), ERREUR_SERVEUR);
            tracer(request, statut, System.nanoTime() - debut);
        }
    }

    private void tracer(HttpServletRequest request, int statut, long dureeNanos) {
        LoggingEventBuilder evenement = evenementPour(statut)
                .setMessage(MESSAGE)
                .addKeyValue("event.category", "web")
                .addKeyValue("event.type", "access")
                .addKeyValue("event.action", "http-server-request")
                .addKeyValue("event.outcome", statut < ERREUR_CLIENT ? "success" : "failure")
                .addKeyValue("event.duration", dureeNanos)
                .addKeyValue("http.request.method", request.getMethod())
                .addKeyValue("http.response.status_code", statut)
                .addKeyValue("url.path", request.getRequestURI())
                .addKeyValue("client.address", request.getRemoteAddr());
        Object utilisateur = request.getAttribute(ATTRIBUT_UTILISATEUR);
        if (utilisateur != null) {
            evenement = evenement.addKeyValue("user.name", utilisateur);
        }
        evenement.log();
    }

    /** 2xx / 3xx : INFO — 4xx : WARN — 5xx : ERROR. */
    private static LoggingEventBuilder evenementPour(int statut) {
        if (statut >= ERREUR_SERVEUR) {
            return ACCESS_LOG.atError();
        }
        if (statut >= ERREUR_CLIENT) {
            return ACCESS_LOG.atWarn();
        }
        return ACCESS_LOG.atInfo();
    }
}

package com.bmci.workflow.fund.management.application.rest.logging;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.slf4j.MDC;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.security.authentication.AnonymousAuthenticationToken;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

/**
 * AER_LM-350 — pose l'uid de l'appelant authentifié dans le contexte de log (MDC « uid », écrit « user.name »
 * par logback-spring.xml) : chaque ligne émise pendant la requête dit qui agit.
 * Placé après Spring Security (dernier filtre) : l'authentification est déjà faite.
 */
@Component
@Order(Ordered.LOWEST_PRECEDENCE)
public class UserLogContextFilter extends OncePerRequestFilter {

    static final String MDC_UID = "uid";

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
            throws ServletException, IOException {
        String uid = uidAuthentifie();
        if (uid == null) {
            filterChain.doFilter(request, response);
            return;
        }
        request.setAttribute(HttpAccessLogFilter.ATTRIBUT_UTILISATEUR, uid);
        MDC.put(MDC_UID, uid);
        try {
            filterChain.doFilter(request, response);
        } finally {
            MDC.remove(MDC_UID);
        }
    }

    private static String uidAuthentifie() {
        Authentication authentication = SecurityContextHolder.getContext().getAuthentication();
        if (authentication == null || !authentication.isAuthenticated()
                || authentication instanceof AnonymousAuthenticationToken) {
            return null;
        }
        return authentication.getName();
    }
}
