package com.bmci.workflow.fund.management.service.domain.common;

import java.time.LocalDateTime;

/**
 * Critères de recherche d'une liste de demandes (retail et corpo).
 * Tous les champs sont optionnels sauf page/size. Les valeurs vides sont normalisées en null
 * dès la construction : le reste de la chaîne n'a plus à tester isBlank().
 */
public record DemandeSearchCriteria(
        String statut,
        String agence,
        String operationType,
        String operationSens,
        String recherche,
        LocalDateTime dateDebut,
        int page,
        int size) {

    public static final int SIZE_DEFAUT = 20;
    public static final int SIZE_MAX = 100;

    public DemandeSearchCriteria {
        statut = vide(statut);
        agence = vide(agence);
        operationType = vide(operationType);
        operationSens = vide(operationSens);
        recherche = vide(recherche);
        page = Math.max(page, 0);
        size = size <= 0 ? SIZE_DEFAUT : Math.min(size, SIZE_MAX);
    }

    private static String vide(String s) {
        return (s == null || s.isBlank()) ? null : s.trim();
    }
}

package com.bmci.workflow.fund.management.dataaccess.common.spec;

import com.bmci.workflow.fund.management.service.domain.common.DemandeSearchCriteria;
import jakarta.persistence.criteria.Join;
import jakarta.persistence.criteria.JoinType;
import jakarta.persistence.criteria.Predicate;
import org.springframework.data.jpa.domain.Specification;

import java.util.ArrayList;
import java.util.List;

/**
 * Construit la Specification JPA d'une recherche de demandes.
 * Générique : s'applique à CashRequestEntity (retail) et CorpoCashRequestEntity (corpo),
 * qui portent les mêmes attributs. Aucun fetch ici : la pagination reste exacte,
 * l'arbre est rechargé dans un second appel (findAllWithTreeByIdIn).
 */
public final class DemandeSpecifications {

    private DemandeSpecifications() {}

    public static <E> Specification<E> of(DemandeSearchCriteria c) {
        return (root, query, cb) -> {
            List<Predicate> predicates = new ArrayList<>();

            if (c.statut() != null) {
                predicates.add(cb.equal(root.get("requestStatus"), c.statut()));
            }
            if (c.agence() != null) {
                predicates.add(cb.equal(root.get("branchCode"), c.agence()));
            }
            if (c.dateDebut() != null) {
                predicates.add(cb.greaterThanOrEqualTo(root.get("creationDate"), c.dateDebut()));
            }
            if (c.recherche() != null) {
                String motif = "%" + c.recherche().toLowerCase() + "%";
                Join<Object, Object> user = root.join("userProfile", JoinType.LEFT);
                predicates.add(cb.or(
                        cb.like(cb.lower(root.get("uniqueRequestRef")), motif),
                        cb.like(cb.lower(user.get("uid")), motif)));
            }
            if (c.operationType() != null || c.operationSens() != null) {
                Join<Object, Object> op = root.join("operations", JoinType.INNER);
                if (c.operationType() != null) {
                    predicates.add(cb.equal(op.get("operationType"), c.operationType()));
                }
                if (c.operationSens() != null) {
                    predicates.add(cb.equal(op.get("operationSens"), c.operationSens()));
                }
                query.distinct(true);
            }
            return cb.and(predicates.toArray(new Predicate[0]));
        };
    }
}

        debounceTime(300),
        map(v => v.trim()),
        distinctUntilChanged()
      )
    ]);

    // tout changement de filtre → retour page 0
    filtres$.pipe(takeUntilDestroyed(this.destroyRef)).subscribe(() => {
      if (this.page$.value !== 0) { this.page$.next(0); }
    });

    combineLatest([filtres$, this.page$])
      .pipe(
        map(([[periode, agence, type, statut, recherche], page]) =>
          this.construireCriteres(periode, agence, type, statut, recherche, page)),
        takeUntilDestroyed(this.destroyRef)
      )
      .subscribe(criteres => this.facade.loadDemandes(criteres));
  }

  rafraichir(): void {
    this.page$.next(this.page$.value);   // ré-émet les mêmes critères
  }

  changerPage(e: PageEvent): void {
    this.page$.next(e.pageIndex);
  }

  // ===== Traduction filtres écran → critères backend =====
  private construireCriteres(periode: string, agence: string, type: string, statut: string,
                             recherche: string, page: number): DemandeCriteria {
    const c: DemandeCriteria = { page, size: this.tailleParDefaut };
    if (statut !== 'TOUS')   { c.statut = statut; }
    if (agence !== 'TOUTES') { c.agence = agence; }
    if (recherche)           { c.recherche = recherche; }
    const limite = this.limitePeriode(periode);
    if (limite)              { c.dateDebut = this.isoLocal(limite); }
    // même sémantique que chipsContenu() : RAMASSAGE = sens COLLECTION, sinon type + SUPPLY
    switch (type) {
      case 'GAB':       c.operationType = 'ATM';       c.operationSens = 'SUPPLY';     break;
      case 'CAISSE':    c.operationType = 'CASH_DESK'; c.operationSens = 'SUPPLY';     break;
      case 'DEVISES':   c.operationType = 'CURRENCY';  c.operationSens = 'SUPPLY';     break;
      case 'RAMASSAGE':                                c.operationSens = 'COLLECTION'; break;
    }
    return c;
  }

  private limitePeriode(code: string): Date | null {
    if (code === '7J')  { const d = new Date(); d.setDate(d.getDate() - 7);  d.setHours(0, 0, 0, 0); return d; }
    if (code === '30J') { const d = new Date(); d.setDate(d.getDate() - 30); d.setHours(0, 0, 0, 0); return d; }
    return null;
  }

  /** LocalDateTime attendu par le backend (@DateTimeFormat ISO.DATE_TIME) — sans décalage UTC de toISOString(). */
  private isoLocal(d: Date): string {
    const p = (n: number) => String(n).padStart(2, '0');
    return `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}T${p(d.getHours())}:${p(d.getMinutes())}:${p(d.getSeconds())}`;
  }

  // ===== Mapping API -> écran (aligné maquette) =====

  reference(d: DemandeFonds): string {
    return d.uniqueRequestRef ?? '—';      // TODO résolu : la vraie référence RT-AAAA-NNNNNN
  }

  dateDemande(d: DemandeFonds): string | undefined {
    return d.createdAt;                    // TODO résolu : date au niveau demande
  }

}
