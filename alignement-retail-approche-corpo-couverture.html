    RETAIL_CONFLICT(409, HttpStatus.CONFLICT, "retail.error.conflict"),

    RETAIL_TECHNICAL_ERROR(500, HttpStatus.INTERNAL_SERVER_ERROR, "retail.error.technical");

    private final int code;
    private final HttpStatus httpStatus;
    private final String messageKey;
}

package com.bmci.workflow.fund.management.application.rest.retail.error;

import com.bmci.q360.domain.retail.exceptions.RetailConflictException;
import com.bmci.q360.domain.retail.exceptions.RetailNotFoundException;
import com.bmci.q360.domain.retail.exceptions.RetailValidationException;
import com.bmci.workflow.fund.management.application.rest.retail.CashDemandController;
import com.bmci.workflow.fund.management.application.rest.retail.enums.RetailHttpStatus;
import com.bmci.workflow.fund.management.common.dto.ErrorDTO;
import com.bmci.workflow.fund.management.common.dto.FieldErrorsDTO;
import com.bmci.workflow.fund.management.service.domain.exceptions.SessionServiceException;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.context.MessageSource;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.HashMap;
import java.util.Locale;
import java.util.Map;

/**
 * Handler retail — MEME pattern que le corpo (BFF : ErrorDTO + MessageSource +
 * registre), scope sur CashDemandController uniquement.
 * PHASE 1 (sans regression) : SessionServiceException conserve son 500 actuel,
 * mais au format ErrorDTO unifie. PHASE 2 (progressive) : au fur et a mesure
 * que l'adapter/use case retail adopte les exceptions typees, les vrais
 * statuts (404/409/422) prennent le relais — methode par methode.
 */
@Slf4j
@RestControllerAdvice(assignableTypes = CashDemandController.class)
@RequiredArgsConstructor
public class RetailDemandExceptionHandler {

    private final MessageSource messageSource;

    // ============ PHASE 2 : erreurs metier typees (a mesure de la migration) ============

    @ExceptionHandler(RetailValidationException.class)
    public ResponseEntity<ErrorDTO> handleValidation(RetailValidationException exception) {
        return build(RetailHttpStatus.RETAIL_VALIDATION_ERROR, exception.getMessage());
    }

    @ExceptionHandler(RetailNotFoundException.class)
    public ResponseEntity<ErrorDTO> handleNotFound(RetailNotFoundException exception) {
        return build(RetailHttpStatus.RETAIL_NOT_FOUND, exception.getMessage());
    }

    @ExceptionHandler(RetailConflictException.class)
    public ResponseEntity<ErrorDTO> handleConflict(RetailConflictException exception) {
        return build(RetailHttpStatus.RETAIL_CONFLICT, exception.getMessage());
    }

    // ============ PHASE 1 : comportement actuel preserve, format unifie ============

    @ExceptionHandler(SessionServiceException.class)
    public ResponseEntity<ErrorDTO> handleSessionService(SessionServiceException exception) {
        log.error("SessionServiceException sur un endpoint retail", exception);
        return build(RetailHttpStatus.RETAIL_TECHNICAL_ERROR, null);
    }

    // ============ Erreurs de surface (des que @Valid sera pose) ============

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorDTO> handleBeanValidation(MethodArgumentNotValidException exception) {
        Map<String, String> fieldErrors = new HashMap<>();
        exception.getBindingResult().getFieldErrors().forEach(fieldError ->
                fieldErrors.put(fieldError.getField(), fieldError.getDefaultMessage()));
        RetailHttpStatus status = RetailHttpStatus.RETAIL_BAD_REQUEST;
        return ResponseEntity.status(status.getHttpStatus())
                .body(new FieldErrorsDTO(String.valueOf(status.getCode()),
                        resolveMessage(status, "des champs sont invalides"), fieldErrors));
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<ErrorDTO> handleUnreadable(HttpMessageNotReadableException exception) {
        return build(RetailHttpStatus.RETAIL_BAD_REQUEST, "le corps de la requête est illisible ou mal formé");
    }

    // ============ Filet final : jamais silencieux ============

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorDTO> handleUnexpected(Exception exception) {
        log.error("Erreur non prévue sur un endpoint retail", exception);
        return build(RetailHttpStatus.RETAIL_TECHNICAL_ERROR, null);
    }

    private ResponseEntity<ErrorDTO> build(RetailHttpStatus status, String detail) {
        return ResponseEntity.status(status.getHttpStatus())
                .body(new ErrorDTO(String.valueOf(status.getCode()), resolveMessage(status, detail)));
    }

    private String resolveMessage(RetailHttpStatus status, String detail) {
        return messageSource.getMessage(status.getMessageKey(),
                new Object[]{detail == null ? "" : detail}, Locale.FRANCE);
    }
}
