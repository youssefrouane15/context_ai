package com.bmci.workflow.fund.management.service.domain.ports.output.repository;

import java.util.UUID;

/**
 * Port de sortie (BPMN 4.2, worker gdf-eclater-besoins-gab) : numérotation des besoins d'une demande GAB hors site.
 * Implémentation côté data-access sur le modèle « besoin » de la demande monétique (AER_LM-264 / 265).
 */
public interface BesoinGabPort {

    /**
     * Attribue à chaque besoin de la demande sa référence technique « referenceDemande-NN » (NN = 01, 02, … dans l'ordre de saisie),
     * persiste, et renvoie le nombre de besoins numérotés. Idempotent : un besoin déjà référencé garde sa référence.
     */
    int numeroterBesoins(UUID demandeId, String referenceDemande);
}

    @Override

        // BPMN 4.2 (AER_LM-265) : pas de cut-off pour une demande GAB hors site — jamais de report, date d'effet = jour de soumission.

        boolean apresCutoff10 = false;
        LocalDateTime maintenant = LocalDateTime.now();
        boolean apresCutoff16 = !monetique && cutoffPolicy.apresCutoff(maintenant);          // AER_LM-267 : le boundary 16h
        // (jour d'effet = aujourd'hui, ou demain si soumise après 16h00 — même règle que CutoffPolicy)

package com.bmci.workflow.fund.management.external.camunda.worker;

import com.bmci.workflow.fund.management.service.domain.ports.output.repository.BesoinGabPort;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.camunda.bpm.client.task.ExternalTask;
import org.camunda.bpm.client.task.ExternalTaskHandler;
import org.camunda.bpm.client.task.ExternalTaskService;
import org.springframework.stereotype.Component;

import java.util.Map;
import java.util.UUID;

/**
 * BPMN 4.2 — topic gdf-eclater-besoins-gab (branche monétique, AER_LM-265).
 * Numérote les besoins de la demande (une référence technique par GAB hors site) puis pose nombreBesoinsGab.
 * Remplace le squelette TacheTraceeHandler de ce topic.
 *
 * @author Youssef ROUANE
 * @date 19/10/2026
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class EclaterBesoinsGabHandler implements ExternalTaskHandler {

    private final BesoinGabPort besoinGabPort;

    @Override
    public void execute(ExternalTask task, ExternalTaskService service) {
        UUID demandeId = UUID.fromString(task.getBusinessKey());
        String referenceDemande = task.getVariable("referenceDemande");

        int nombreBesoins = besoinGabPort.numeroterBesoins(demandeId, referenceDemande);
        log.info("[gdf-eclater-besoins-gab] demande {} ({}) : {} besoin(s) numéroté(s)", demandeId, referenceDemande, nombreBesoins);

        service.complete(task, Map.of("nombreBesoinsGab", nombreBesoins));
    }
}

3. Déploiement du BPMN 4.2 — pas à pas

Postman — collection AER_LM-265_bpmn-4.2 (requêtes 00 → 04), variable camunda = http://localhost:3030

Principe à retenir : une nouvelle version n'affecte jamais les instances en cours ; on ne supprime rien, on ne migre rien. Les deux erreurs du passé — déploiement sans tenant, suppression en cascade — sont rendues impossibles par la requête 02 (tenant imposé) et par l'absence de tout DELETE de déploiement dans la procédure.

# Déploiement BPMN 4.2 — Camunda 7 partagé, tenant QLICKFLOW, moteur local :3030 (adapter l'hôte en recette)
# Règle de base : un nouveau déploiement crée une NOUVELLE VERSION de gestion-des-fonds ; les instances en cours
# restent sur leur version (4.1) et la terminent telles quelles ; fund-management démarre par clé → toujours la dernière version.
# Rien n'est migré, rien n'est supprimé.

## 0. Avant — état des lieux (Postman, GET)
GET {{camunda}}/engine-rest/process-definition?key=gestion-des-fonds&tenantIdIn=QLICKFLOW&sortBy=version&sortOrder=desc
    → noter la version N et le versionTag courant (attendu 4.1). S'il y a plusieurs tenants ou aucun, STOP : c'est le piège
      « déploiement Modeler sans tenant invisible pour fund-management ».
GET {{camunda}}/engine-rest/process-instance/count?processDefinitionKey=gestion-des-fonds&tenantIdIn=QLICKFLOW
    → nombre d'instances actives (information : elles finiront sur 4.1, c'est normal et sans effet sur la 4.2).

## 1. Déployer (Postman — POST multipart, PAS le Modeler : le tenant y est trop facile à oublier)
POST {{camunda}}/engine-rest/deployment/create
  Body form-data :
    deployment-name            gestion-des-fonds-4.2
    tenant-id                  QLICKFLOW
    enable-duplicate-filtering true
    deploy-changed-only        true
    deployment-source          process application
    data (type File)           gestion-des-fonds_v4.2.bpmn
  → 200 : "deployedProcessDefinitions" contient gestion-des-fonds, version N+1, versionTag "4.2", tenantId "QLICKFLOW".
  → si "deployedProcessDefinitions" est vide : le fichier est identique à un déploiement précédent (duplicate filtering) — rien à faire.

## 2. Vérifier
GET {{camunda}}/engine-rest/process-definition?key=gestion-des-fonds&tenantIdIn=QLICKFLOW&latestVersion=true
    → versionTag = 4.2, suspended = false.
GET {{camunda}}/engine-rest/process-definition/key/gestion-des-fonds/tenant-id/QLICKFLOW/xml
    → le XML contient Gateway_CutoffApplicable.

## 3. Redémarrer fund-management APRÈS le déploiement (ordre : BPMN puis service)
    Le worker s'abonne alors à gdf-eclater-besoins-gab avec le vrai handler. Si une instance 4.2 atteint ce topic avant
    le redémarrage, la tâche externe attend simplement (pas d'erreur, pas d'incident).

## 4. Recette (10 min)
  a. Demande agence (j62211) : créer → soumettre → GET /process-instance?processInstanceBusinessKey={id} → processDefinitionId finit par ":N+1:…"
     (4.2) ; comportement identique à avant (report après 16h, verrou 16h).
  b. Demande monétique : POST /cash/init puis /submit avec une demande portée par canalDemande = MONETIQUE (API du collègue → typeDemande MONETIQUE) —

  c. Une instance 4.1 encore ouverte : la traiter depuis la Caisse Centrale → elle se termine normalement.

## 5. Erreurs du passé — et la parade
  - « Déploiement sans tenant » : fund-management ne le voit pas et continue sur l'ancienne version → toujours tenant-id=QLICKFLOW (étape 1), contrôle étape 2.
  - « J'ai supprimé l'ancienne définition / le déploiement avec cascade » : DELETE /deployment/{id}?cascade=true tue toutes les instances
    en cours → ne JAMAIS supprimer un déploiement qui porte des instances actives. On n'y touche pas ; les instances 4.1 finissent seules.
  - « Instances figées sur une version intermédiaire après un ménage partiel » : en local, si tu veux repartir propre, purge d'abord les
    demandes (Purge_demandes_local.html) PUIS DELETE /process-instance/{id} de chaque instance restante
    (GET /process-instance?processDefinitionKey=gestion-des-fonds&tenantIdIn=QLICKFLOW), jamais l'inverse.
  - « Le worker ne prend pas la tâche » : topic non abonné (handler non branché dans GdfWorkerConfig) ou tenant du worker ≠ QLICKFLOW →
    GET /external-task?topicName=gdf-eclater-besoins-gab montre la tâche en attente ; vérifier le log d'abonnement au démarrage.

## 6. Retour arrière (si la 4.2 pose problème en recette)
  Ne pas supprimer la 4.2 si elle porte des instances. Redéployer la 4.1 (fichier gestion-des-fonds_v4.1.bpmn, même POST, deployment-name
  gestion-des-fonds-4.1-rollback, sans deploy-changed-only) → elle devient la dernière version, les nouvelles demandes repartent dessus.
  Sans aucune instance 4.2 : DELETE {{camunda}}/engine-rest/deployment/{idDu4.2}?cascade=false suffit (refusé s'il reste des instances — c'est voulu).
