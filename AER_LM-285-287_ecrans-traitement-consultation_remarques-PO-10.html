/** Coupures d'une alimentation (CASH_DESK ou ATM, sens SUPPLY), cumulées par valeur faciale, triées par valeur décroissante. */
function coupures(ops: OperationDossier[], type: 'CASH_DESK' | 'ATM'): LigneCoupure[] {
  const parValeur = new Map<number, LigneCoupure>();
  for (const op of ops) {
    if (op.operationType !== type || op.operationSens !== 'SUPPLY') { continue; }
    for (const detail of op.currencyDetails ?? []) {
      for (const ligne of detail.operationDenominations ?? []) {
        const code = ligne.denomination?.denominationCode ?? '';
        const faceValue = valeurFaciale(ligne.denomination?.faceValue, code);
        if (faceValue === null) { continue; }
        const quantite = Number(ligne.quantity ?? 0) || 0;
        const existante = parValeur.get(faceValue);
        if (existante) {
          existante.quantite += quantite;
          existante.montant = arrondi(existante.quantite * faceValue);
        } else {
          parValeur.set(faceValue, { code, faceValue, quantite, montant: arrondi(quantite * faceValue) });
        }
      }
    }
  }
  return [...parValeur.values()].sort((a, b) => b.faceValue - a.faceValue);
}

export function valeurFaciale(faceValue: number | string | null | undefined, code: string): number | null {
  const fournie = Number(faceValue);
  if (faceValue !== null && faceValue !== undefined && faceValue !== '' && !Number.isNaN(fournie)) { return fournie; }
  const m = /(\d+)(?:[._](\d+))?\s*$/.exec(code);
  if (!m) { return null; }
  return Number(m[2] ? `${m[1]}.${m[2]}` : m[1]);
}

import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { NgControl } from '@angular/forms';

/**
 * Remarque PO (01/10) — champs de quantité : le 0 est un placeholder, jamais une valeur pré-saisie.
 *  - un champ dont la valeur est 0 (ou absente) se présente vide avec « 0 » en placeholder ;
 *  - au focus, un 0 éventuel disparaît : l'agent tape directement sa quantité ;
 *  - un champ laissé vide vaut 0 pour les totaux et le payload (Number(null) === 0, Number('') === 0).
 * Se pose sur tout <input type="number"> relié à un FormControl : <input type="number" appSaisieQuantite formControlName="…">.
 * Ne change rien aux validateurs : min(0) ignore une valeur vide ; auMoinsUneSectionRenseignee compare des nombres.
 */
@Directive({ selector: 'input[appSaisieQuantite]', standalone: true })
export class SaisieQuantiteDirective implements OnInit {

  private readonly ngControl = inject(NgControl, { optional: true, self: true });
  private readonly destroyRef = inject(DestroyRef);

  @HostBinding('attr.placeholder') readonly placeholder = '0';

  ngOnInit(): void {
    const ctrl = this.ngControl?.control;
    if (!ctrl) { return; }
    this.remplacerZero();
    // un patchValue(0) (chargement d'une demande, « Réinitialiser ») doit lui aussi se présenter vide
    ctrl.valueChanges.pipe(takeUntilDestroyed(this.destroyRef)).subscribe(() => this.remplacerZero());
  }

  @HostListener('focus')
  auFocus(): void { this.remplacerZero(); }

  private remplacerZero(): void {
    const ctrl = this.ngControl?.control;
    if (!ctrl) { return; }
    const v = ctrl.value;
    if (v === 0 || v === '0') {
      // sans émission : la valeur métier ne change pas (0 ⇔ vide), seul l'affichage passe en placeholder
      ctrl.setValue(null, { emitEvent: false });
    }
  }
}

export interface FaitsLectureAgence {
  statut: StatutDemande;
  journeeCloturee: boolean;
  verrouillePar: string | null;     // lockedBy
  verrouDetenu: boolean;            // le verrou est à moi (mode modification en cours)

/** null ⇒ l'écran est modifiable (ou en cours de modification) : pas de bannière. */
export function banniereLectureAgence(f: FaitsLectureAgence): BanniereLecture | null {
  if (f.verrouDetenu) { return null; }
  switch (f.statut) {
    case 'TRAITEE':
      return { ton: 'alerte', titre: 'Consultation seule — Demande traitée et non modifiable',
        texte: `Cette demande a été traitée par la Caisse Centrale (statut : Traitée). La saisie et les ajustements sont définitivement fermés. ${INTERVENTION_CC}` };
    case 'ANNULEE_AGENCE':
      return { ton: 'info', titre: 'Consultation seule — Demande annulée par l\'agence',
        texte: 'Cette demande a été annulée par l\'agence ; elle est conservée pour l\'historique et ne peut plus être modifiée.' };
    case 'ANNULEE_CAISSE_CENTRALE':
      return { ton: 'alerte', titre: 'Consultation seule — Demande annulée par la Caisse Centrale',
        texte: 'Cette demande a été annulée par la Caisse Centrale (voir le motif ci-dessous). Elle ne peut plus être modifiée ; une nouvelle demande doit être initiée si le besoin persiste.' };
  }
  if (!f.demandeDeMonAgence) {
    return { ton: 'info', titre: 'Consultation seule — Demande d\'une autre agence',
      texte: 'Cette demande appartient à une autre agence : elle est consultable mais ne peut être ni modifiée ni annulée depuis votre agence.' };
  }
  if (f.statut === 'EN_MODIFICATION') {
    return { ton: 'alerte', titre: 'Consultation seule — Demande réservée en modification',
      texte: `Cette demande est en cours de modification par ${f.verrouillePar ?? 'un autre agent'}. Elle redeviendra modifiable dès que cette modification sera soumise ou abandonnée.` };
  }
  if (f.journeeCloturee) {
    return { ton: 'alerte', titre: 'Consultation seule — Journée clôturée (cut-off de 16h00 dépassé)',
      texte: `La saisie et les ajustements sont définitivement fermés pour cette journée comptable. ${INTERVENTION_CC}` };
  }
  if (!f.peutModifier) {
    return { ton: 'info', titre: 'Consultation seule — Habilitation de modification absente',
      texte: 'Votre profil permet la consultation des demandes de l\'agence, pas leur mise à jour.' };
  }
  return null;
}

    return this.acteur === 'agence' ? 'Annuler la demande' : 'Motif d\'annulation de la demande';
  }

  get information(): string {
    if (this.type === 'TRAITEE') { return 'La demande est validée et transmise au transporteur ; son statut passera à « Traitée ».'; }
    return this.acteur === 'agence'
      ? 'La demande sort du circuit de traitement ; la Caisse Centrale en sera informée. Une nouvelle demande devra être initiée si le besoin persiste.'
      : 'L\'agence sera informée de l\'annulation ; la demande sort du circuit de traitement.';
  }

  get libelleConfirmer(): string {
    if (this.enCours) { return 'Envoi en cours…'; }
    if (this.type === 'TRAITEE') { return 'Confirmer le traitement'; }
    return this.acteur === 'agence' ? 'Confirmer l\'annulation' : 'Confirmer le rejet';
  }

  choisirMotif(m: string): void {
    this.motif.setValue(m);
    this.motif.markAsTouched();
  }

  valider(): void {
    if (this.type === 'ANNULEE') {
      if (this.motif.invalid) { this.motif.markAsTouched(); return; }
      this.confirmer.emit(this.motif.value.trim());
    } else {
      this.confirmer.emit(undefined);
    }
  }
}

 * AER_LM-285 — Caisse Centrale : écran de traitement d'une demande agence (route caisse-centrale/:id).
 * Lecture complète de la demande (coupures, GAB, ramassages, devises, commentaire) et décision :
 * Traiter / Annuler avec motif, uniquement si la demande est en attente et si l'appelant a GDF_TRAITER_DEMANDE
 * (Support : consultation seule). Le back revérifie les deux (403 / 409).
 */
@Component({
  selector: 'app-demande-traitement',
  standalone: true,
  imports: [CommonModule, MatIconModule, MatSnackBarModule, DemandeEnteteComponent, DemandeBarreActionsComponent, DemandeDecisionModaleComponent],
  templateUrl: './demande-traitement.component.html',
  styleUrl: './demande-traitement.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class DemandeTraitementComponent implements OnInit {

  private readonly route = inject(ActivatedRoute);
  private readonly router = inject(Router);
  private readonly snackBar = inject(MatSnackBar);
  private readonly service = inject(FundHttpService);
  readonly actions = inject(DemandeActionsFacade);
  readonly contexteUtilisateur = inject(ContexteUtilisateurService);

  readonly vue = signal<VueDossier | null>(null);
  readonly chargement = signal(true);
  readonly erreur = signal<string | null>(null);

  /** Décision ouverte dans la modale (null = fermée). */
  readonly decision = signal<TypeDecision | null>(null);
  readonly motifsRapides = MOTIFS_RAPIDES_CAISSE_CENTRALE;

  /** Traiter / Annuler : demande en attente + habilitation TRAITER (Caisse Centrale oui, Support non). */
  readonly peutDecider = computed(() => {
    const v = this.vue();
    return !!v && v.statut === 'SOUMISE' && this.contexteUtilisateur.habilitations().traiter;
  });

  readonly banniere = computed(() => {
    const v = this.vue();
    return v ? banniereLectureCaisseCentrale({
      statut: v.statut, verrouillePar: v.verrouillePar, agence: v.agence,
      peutTraiter: this.contexteUtilisateur.habilitations().traiter
    }) : null;
  });

  readonly montant = formaterMontant;

  ngOnInit(): void {
    this.contexteUtilisateur.charger();
    void this.charger();
  }

  async charger(): Promise<void> {
    const id = this.route.snapshot.paramMap.get('id');
    if (!id) { this.erreur.set('Demande inconnue.'); this.chargement.set(false); return; }
    this.chargement.set(true);
    this.erreur.set(null);
    try {
      this.vue.set(construireVueDossier(await this.service.getDossier(id)));
    } catch (e) {
      this.vue.set(null);
      this.erreur.set(e instanceof ErreurApi ? e.message : 'La demande est indisponible.');
    } finally {
      this.chargement.set(false);
    }
  }

  retour(): void {
    this.router.navigate(['caisse-centrale'], { relativeTo: this.route.parent });
  }

  ouvrir(type: TypeDecision): void {
    if (!this.peutDecider()) { return; }
    this.actions.acquitterErreur();
    this.decision.set(type);
  }

  fermer(): void {
    this.decision.set(null);
  }

  /** Même appel que l'ancienne modale du listing : DemandeActionsFacade.decider (complete-task TRAITEE | ANNULEE). */
  async confirmer(motif: string | undefined): Promise<void> {
    const v = this.vue();
    const type = this.decision();
    if (!v || !type) { return; }
    const ok = await this.actions.decider(v.id, type, motif);
    if (ok) {
      this.decision.set(null);
      this.snackBar.open(
        type === 'TRAITEE'
          ? `Demande ${v.reference} traitée : le statut sera mis à jour dans quelques secondes.`
          : `Demande ${v.reference} annulée : motif consigné.`,
        'Fermer', { duration: 5000 });
      setTimeout(() => void this.charger(), 1500);   // le statut final est écrit par le worker (~1 s)
    } else {
      this.snackBar.open(this.actions.erreur() ?? 'Erreur lors de la décision.', 'Fermer', { duration: 6000 });
    }
  }
}
