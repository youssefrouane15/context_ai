0. Le changement

Avant : un seul template générique mail/gestion-fonds-demande.html pour les 5 évènements.
Après : les 5 maquettes du PO, transcrites à la lettre en Thymeleaf, une par scénario ; fund-management envoie les variables exactes de chaque maquette.
ScénarioDestinatairestemplateCodeTemplate
SoumissionDAGDF_DEMANDE_SOUMISEgestion-fonds-demande-soumise
Mise à jourDAGDF_DEMANDE_MISE_A_JOURgestion-fonds-demande-mise-a-jour

Variables envoyées (toutes en chaîne, jamais nulles) : agence, reference, date, operateurNom, operateurProfil, motif + natureUrgence pour le mail Caisse Centrale. portalUrl et logoCid restent posés par EmailNotificationService (inchangé).

Seuls écarts avec les maquettes (imposés par le dynamique) : logo SVG → <img cid> PNG (Outlook n'affiche pas le SVG) ; civilité « Mme / M. » retirée (absente des données) ; commentaires « NOTE D'INTÉGRATION » / « Variantes » retirés ; bloc motif masqué si vide ; l'annulation reçoit un bloc motif (même style que l'urgence) car la maquette prévoyait l'emplacement « BLOC MOTIF EN CAS DE MAJ OU ANNULATION » sans le dessiner. Les objets des mails (subject) sont inchangés.

/**
 * Variables des 5 templates PO de notifications-ci (mail/gestion-fonds-*.html), une par scénario :
 * agence, reference, date, operateurNom, operateurProfil, motif (+ natureUrgence pour le mail Caisse Centrale).
 * Toutes les valeurs sont des chaînes non nulles, injectées en th:text (échappées par Thymeleaf) ; les th:if testent « vide / non vide ».
 */
@Component
public class VariablesEmailDemande {

    static final String NATURE_URGENCE_SOUMISSION = "Nouvelle demande urgente";
    static final String NATURE_URGENCE_MISE_A_JOUR = "Mise à jour d'une demande urgente";

    private static final DateTimeFormatter DATE_HEURE = DateTimeFormatter.ofPattern("dd/MM/yyyy 'à' HH:mm");

    /** Mail agence : SOUMISSION, MISE_A_JOUR, ANNULATION (Directeur d'agence) ou DECLASSEMENT_URGENCE (DA + Chef de caisse). */
    public Map<String, String> construire(CashRequestDTO d, EvenementDemande evenement) {
        Map<String, String> v = new LinkedHashMap<>();
        v.put("agence", vide(d.getBranchCode()) + (d.getAgenceNom() != null ? " - " + d.getAgenceNom() : ""));
        v.put("reference", vide(d.getUniqueRequestRef()));
        v.put("date", date(dateEvenement(d, evenement)));
        boolean parCaisseCentrale = evenement == EvenementDemande.DECLASSEMENT_URGENCE;
        v.put("operateurNom", vide(parCaisseCentrale ? d.getUrgenceDeclasseePar() : d.getAgentNom()));
        v.put("operateurProfil", vide(parCaisseCentrale ? d.getUrgenceDeclasseeProfil() : d.getAgentProfil()));
        v.put("motif", motif(d, evenement));
        return v;
    }

    /**
     * Mail Caisse Centrale (template GDF_DEMANDE_URGENTE) d'une demande urgente soumise ou mise à jour.
     * evenementOrigine (SOUMISSION / MISE_A_JOUR) fixe la date et la « Nature de l'opération » ; le motif est le commentaire de l'agence.
     */
    public Map<String, String> construireUrgence(CashRequestDTO d, EvenementDemande evenementOrigine) {
        Map<String, String> v = construire(d, evenementOrigine);
        v.put("motif", vide(d.getComment()).trim());
        v.put("natureUrgence", evenementOrigine == EvenementDemande.MISE_A_JOUR ? NATURE_URGENCE_MISE_A_JOUR : NATURE_URGENCE_SOUMISSION);
        return v;
    }

    /** Date affichée « Date et heure » : celle de l'action notifiée. */
    private static LocalDateTime dateEvenement(CashRequestDTO d, EvenementDemande evenement) {
        LocalDateTime derniereModif = d.getUpdatedAt() != null ? d.getUpdatedAt() : d.getCreatedAt();
        return switch (evenement) {
            case SOUMISSION, URGENCE_CAISSE_CENTRALE -> d.getSubmittedAt() != null ? d.getSubmittedAt() : derniereModif;
            case DECLASSEMENT_URGENCE -> d.getUrgenceDeclasseeAt() != null ? d.getUrgenceDeclasseeAt() : derniereModif;
            case MISE_A_JOUR, ANNULATION -> derniereModif;
        };
    }

    /** Motif affiché dans le bloc motif du template (bloc masqué si vide). */
    private static String motif(CashRequestDTO d, EvenementDemande evenement) {
        return switch (evenement) {
            case ANNULATION -> vide(d.getCancellationReason()).trim();
            case DECLASSEMENT_URGENCE -> vide(d.getUrgenceDeclasseeMotif()).trim();
            case URGENCE_CAISSE_CENTRALE -> vide(d.getComment()).trim();
            case SOUMISSION, MISE_A_JOUR -> "";
        };
    }

    private static String date(LocalDateTime date) { return date != null ? DATE_HEURE.format(date) : ""; }

    private static String vide(String s) { return s == null ? "" : s; }
}
