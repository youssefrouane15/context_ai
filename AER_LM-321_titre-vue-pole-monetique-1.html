// app.component.ts — ÉDITION : remplace le bloc « titre » (lignes 42-51) par ce bloc

  /** Pôle opérations monétique : saisie monétique sans périmètre toutes agences (la Caisse Centrale traite le monétique mais a toutesAgences). */
  private readonly estPoleMonetique = computed(() => {
    const h = this.contexteUtilisateur.contexte()?.habilitations;
    return !!h && h.creerMonetique && !h.toutesAgences;
  });

  /**
   * « Vue Caisse Centrale » pour un profil toutes agences sans agence de rattachement (Caisse Centrale, Support, Super Admin)
   * ou dès que l'on est sur l'écran /caisse-centrale ; en-tête de la maquette pour le Pôle opérations monétique ; « Vue Agence » sinon.
   */
  readonly titre = computed(() => {
    const ctx = this.contexteUtilisateur.contexte();
    const profilCentral = !!ctx && !ctx.agenceCode && ctx.habilitations.toutesAgences;
    const ecranCentral = this.url().includes(PREFIXE_CAISSE_CENTRALE);
    if (profilCentral || ecranCentral) {
      return 'Gestion des fonds - Vue Caisse Centrale';
    }
    return this.estPoleMonetique() ? 'BMCI - Opérations Monétiques' : 'Gestion des fonds - Vue Agence';
  });

  /** Sous-titre de la maquette « Pôle Opérations Monétique » ; aucune ligne pour les autres vues. */
  readonly sousTitre = computed(() => this.estPoleMonetique() ? 'Portail de Gestion des Fonds GAB Hors-Site' : null);
