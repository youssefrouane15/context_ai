import { HttpErrorResponse } from '@angular/common/http';
import { catchError, throwError } from 'rxjs';

// 2) À AJOUTER dans la classe, à côté de getDemandes :

  // ===== Soumission — PATCH /cash/{id}/submit → { success, data: DemandeFonds } =====
  soumettreDemande(id: string): Promise<DemandeFonds> {
    return firstValueFrom(
      this.http.patch<ApiResponse<DemandeFonds>>(`${this.apiUrl}/${id}/submit`, {}).pipe(
        map(r => this.data(r)),
        catchError(e => throwError(() => this.enErreurApi(e)))
      )
    );
  }

  /**
   * Convertit une erreur HTTP en ErreurApi en lisant l'enveloppe socle renvoyée par nos handlers
   * (502 workflow indisponible, 409 statut non soumissible, 404 introuvable…).
   * Un ErreurApi déjà levé par data() est renvoyé tel quel.
   */
  private enErreurApi(e: unknown): ErreurApi {
    if (e instanceof ErreurApi) { return e; }
    if (e instanceof HttpErrorResponse) {
      const corps = e.error as Partial<ApiResponse<unknown>> | null;
      const code = corps?.error ?? String(e.status);
      const message = corps?.message
        ?? (e.status === 0 ? 'Serveur injoignable.' : `Erreur ${e.status} lors de l'appel au serveur.`);
      return new ErreurApi(code, message);
    }
    return new ErreurApi('ERR', 'Erreur inattendue.');
  }
apiUrl est déjà la base /cash (c'est celle utilisée par getDemandes) ; data() et ErreurApi existent déjà (doc filtres). Le corps {} est volontaire : le backend n'attend rien, mais un PATCH sans corps est rejeté par certains proxies.

   * Soumet la demande au workflow. En succès, remplace la ligne dans la page courante par la version SUBMITTED.
   * En échec, relance l'ErreurApi (message backend) : le composant l'affiche, la liste n'est pas modifiée
   * (côté serveur la saga a déjà restauré INITIATED).
   */
  async soumettreDemande(id: string): Promise<DemandeFonds> {
    this.store.soumissionEnCours.set(id);
    try {
      const demandeMaj = await this.serviceHttp.soumettreDemande(id);
      this.store.demandes.update(liste => liste.map(d => d.id === demandeMaj.id ? demandeMaj : d));
      return demandeMaj;
    } finally {
      this.store.soumissionEnCours.set(null);
    }
  }

      titre: 'Soumettre la demande',
      message: `Soumettre la demande ${this.reference(d)} au circuit de traitement ? `
             + `Elle ne sera plus modifiable directement depuis cet écran.`,
      libelleConfirmer: 'Soumettre'
    };
    const confirme = await firstValueFrom(
      this.dialog.open(ConfirmationDialogComponent, { data: donnees, width: '440px' }).afterClosed()
    );
    if (!confirme) { return; }

    try {
      await this.facade.soumettreDemande(d.id);
      this.snackBar.open(`Demande ${this.reference(d)} soumise au workflow.`, 'Fermer', { duration: 4000 });
    } catch (e) {
      const message = e instanceof ErreurApi ? e.message : 'La soumission a échoué.';
      this.snackBar.open(message, 'Fermer', { duration: 8000 });
    }
  }

7Pourquoi ces choix

ChoixRaison
Bouton sur la liste (pas à la création)Instruction du tech lead + fidèle au BPMN (brouillon → soumission = deux actes). La création reste possible même moteur éteint.
estSoumissible dans le domaine frontMême garde que ChangerStatutDemandeStep côté back : le bouton n'apparaît que sur INITIATED. Le back reste le juge (409 si contournement).
Signal soumissionEnCours = idDésactive UNIQUEMENT la ligne concernée, pas toute la liste ; empêche le double-clic.
Mise à jour en place du storePas de rechargement de page : la ligne passe « Soumise » instantanément, la pagination et les filtres ne bougent pas.
Message d'erreur = message backendLe 502 dit « la demande est restée dans son état initial », le 409 dit pourquoi : l'utilisateur comprend sans traduction front.
Dialogue de confirmation partagéRéutilisable pour Annuler/Traiter des corbeilles à venir.
