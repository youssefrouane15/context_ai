import { CommonModule } from '@angular/common';
import { ChangeDetectionStrategy, Component, Input, computed, inject, signal } from '@angular/core';
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';
import { LOGOS_TRANSPORTEUR } from './logos-transporteur';

/** Hauteur d'affichage des logos (px) — un seul réglage pour le listing et la carte KPI. */
export const HAUTEUR_LOGO_PX = 18;

/**
 * Impose la taille à la racine <svg> AVANT injection : un SVG sans width/height (ou en %) prend toute la largeur
 * disponible et une règle CSS ne suffit pas toujours à le contenir (régression vue en recette : logo plein écran).
 * - viewBox présent : largeur = hauteur × ratio du viewBox ;
 * - pas de viewBox : on le reconstruit depuis width/height d'origine (sinon ratio 3:1 par défaut).
 */
export function dimensionnerSvg(svg: string, hauteur: number = HAUTEUR_LOGO_PX): string {
  return svg.replace(/<svg\b([^>]*)>/i, (_, attrs: string) => {
    const lire = (nom: string): string | null => new RegExp(`\\s${nom}="([^"]*)"`, 'i').exec(attrs)?.[1] ?? null;
    let viewBox = lire('viewBox');
    if (!viewBox) {
      const w = parseFloat(lire('width') ?? ''), h = parseFloat(lire('height') ?? '');
      viewBox = w > 0 && h > 0 ? `0 0 ${w} ${h}` : `0 0 ${hauteur * 3} ${hauteur}`;
    }
    const [, , vw, vh] = viewBox.trim().split(/[\s,]+/).map(Number);
    const largeur = vw > 0 && vh > 0 ? Math.round(hauteur * vw / vh) : hauteur * 3;
    const nettoyes = attrs.replace(/\s(width|height|style|viewBox)="[^"]*"/gi, '');
    return `<svg${nettoyes} viewBox="${viewBox}" width="${largeur}" height="${hauteur}" preserveAspectRatio="xMinYMid meet" style="display:block">`;
  });
}

/** AER_LM-344 : date ISO (yyyy-MM-dd, heure locale) un mois avant la référence ; 31/03 → 28/02 ou 29/02 (pas de débordement au 03/03). */
export function dateIsoIlYaUnMois(reference: Date = new Date()): string {
  const d = new Date(reference.getFullYear(), reference.getMonth() - 1, reference.getDate());
  if (d.getDate() !== reference.getDate()) { d.setDate(0); }   // débordement → dernier jour du mois précédent
  const mm = String(d.getMonth() + 1).padStart(2, '0');
  const jj = String(d.getDate()).padStart(2, '0');
  return `${d.getFullYear()}-${mm}-${jj}`;
}
