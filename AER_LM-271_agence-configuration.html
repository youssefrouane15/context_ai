Agence — BDD et API : zone de distribution, prestataire, coût de déplacementTechnical Story · 5 pts · Sprint 3 · sert AER_LM-259 (front configuration des agences)

1. Contexte et décisions

AER_LM-271 n’est pas une création : l’agence existe de bout en bout (table, contrôleur, facade, mapper, 3 DTO). Le ticket ajoute trois informations à l’agence — zone de distribution, prestataire, coût fixe de déplacement — et adapte le CRUD existant pour les exposer à l’écran AER_LM-259.

Ce que ce ticket livre
  • Script Flyway V15 : 3 colonnes sur orga_unit (+ FK, + CHECK, + miroir orga_unit_aud).
  • OrganizationUnitEntity : 3 champs. CreateAgenceDTO, UpdateAgenceDTO, AgenceDTO : 3 champs chacun.
  • AgenceMapper réécrit (fichier complet) : résolution et contrôle de la zone, mapping des 3 champs dans les deux sens.
  • GlobalExceptionHandler : le projet n’a aucun @RestControllerAdvice, les doublons sortaient en 500.
  • Zéro changement dans AgenceController, AgenceFacade, AgenceFacadeImpl, OrganizationService, OrganizationUnitRepository.
Dépendances de mergeV15 référence zone_distribution (V14, ticket 270) et l’entité référence l’enum Prestataire (ticket 269). Merger 270 puis 269 avant 271, ou partir de la branche 269 si tu enchaînes.

package com.bnpp.irb.qlickflow.admin.mapper;

import com.bnpp.irb.qlickflow.admin.dto.orga.agence.AgenceDTO;
import com.bnpp.irb.qlickflow.admin.dto.orga.agence.CreateAgenceDTO;
import com.bnpp.irb.qlickflow.admin.dto.orga.agence.UpdateAgenceDTO;
import com.bnpp.irb.qlickflow.admin.entities.UserEntity;
import com.bnpp.irb.qlickflow.admin.entities.orga.OrganizationUnitEntity;
import com.bnpp.irb.qlickflow.admin.entities.orga.OrganizationUnitType;
import com.bnpp.irb.qlickflow.admin.entities.referentiel.ZoneDistributionEntity;
import com.bnpp.irb.qlickflow.admin.service.orga.OrganizationService;
import com.bnpp.irb.qlickflow.admin.service.referentiel.ZoneDistributionService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;

@Component
@RequiredArgsConstructor
public class AgenceMapper extends AbstractOrgaUnitMapper {

    private final OrganizationService organizationService;
    private final OrgaUnitMapper orgaUnitMapper;
    // ---- AER_LM-271
    private final ZoneDistributionService zoneDistributionService;
    private final ZoneDistributionMapper zoneDistributionMapper;
    private final PrestataireMapper prestataireMapper;

    public OrganizationUnitEntity toEntity(CreateAgenceDTO dto) {
        if ( dto == null ) {
            return null;
        }

        var groupe = organizationService.findById(dto.getGroupeId()).orElseThrow(
                () -> new IllegalArgumentException("Groupe not found, id : " + dto.getGroupeId()));

        return OrganizationUnitEntity.builder()
                .actif(true)
                .name(dto.getNom())
                .code(dto.getCode())
                .parent(groupe)
                .type(OrganizationUnitType.AGENCE)
                // ---- AER_LM-271
                .zoneDistribution(resolveZoneDistribution(dto.getZoneDistributionId()))
                .prestataire(dto.getPrestataire())
                .coutDeplacement(dto.getCoutDeplacement())
                .build();
    }

    public AgenceDTO toDto(OrganizationUnitEntity agence) {
        if ( agence == null ) {
            return null;
        }

        AgenceDTO.AgenceDTOBuilder agenceDTO = AgenceDTO.builder();

        UserEntity manager = agence.getManager();
        agenceDTO.manager(toSimpleUser(manager));
        agenceDTO.groupe( orgaUnitMapper.toDto(agence.getParent()));
        agenceDTO.id( agence.getId() );
        agenceDTO.nom( agence.getName() );
        agenceDTO.code( agence.getCode() );
        agenceDTO.actif( agence.isActif() );
        // ---- AER_LM-271
        agenceDTO.zoneDistribution( zoneDistributionMapper.toDto(agence.getZoneDistribution()) );
        agenceDTO.prestataire( prestataireMapper.toDto(agence.getPrestataire()) );
        agenceDTO.coutDeplacement( agence.getCoutDeplacement() );

        return agenceDTO.build();
    }

    public List<AgenceDTO> toDtoList(List<OrganizationUnitEntity> agences) {
        if ( agences == null ) {
            return null;
        }

        List<AgenceDTO> list = new ArrayList<>( agences.size() );
        for ( var agenceEntity : agences ) {
            list.add( toDto( agenceEntity ) );
        }

        return list;
    }

    public void toEntity(OrganizationUnitEntity orgaUnitEntity, UpdateAgenceDTO dto) {

        var groupe = organizationService.findById(dto.getGroupeId()).orElseThrow(
                () -> new IllegalArgumentException("Groupe not found, id : " + dto.getGroupeId()));

        if (groupe.getType() != OrganizationUnitType.GROUPE) {
            throw new IllegalArgumentException("groupeId is not an Orga unit of type GROUPE, groupeId:" + dto.getGroupeId());
        }

        orgaUnitEntity.setId(dto.getId());
        orgaUnitEntity.setName(dto.getNom());
        orgaUnitEntity.setCode(dto.getCode());
        orgaUnitEntity.setParent(groupe);

        // ---- AER_LM-271 : un champ absent du corps (null) laisse la valeur en base inchangée.
        // Compatibilité avec les clients existants du PUT (écran AER_LM-267) qui n'envoient pas ces champs.
        if (dto.getZoneDistributionId() != null) {
            orgaUnitEntity.setZoneDistribution(resolveZoneDistribution(dto.getZoneDistributionId()));
        }
        if (dto.getPrestataire() != null) {
            orgaUnitEntity.setPrestataire(dto.getPrestataire());
        }
        if (dto.getCoutDeplacement() != null) {
            orgaUnitEntity.setCoutDeplacement(dto.getCoutDeplacement());
        }
    }

    /** Résout une zone de distribution par id ; refuse une zone inconnue ou désactivée. */
    private ZoneDistributionEntity resolveZoneDistribution(Long zoneDistributionId) {
        if (zoneDistributionId == null) {
            return null;
        }
        var zone = zoneDistributionService.findById(zoneDistributionId).orElseThrow(
                () -> new IllegalArgumentException("ZoneDistribution not found, id : " + zoneDistributionId));
        if (!zone.isActif()) {
            throw new IllegalArgumentException("ZoneDistribution is inactive, id : " + zoneDistributionId);
        }
        return zone;
    }
}

package com.bnpp.irb.qlickflow.admin.exceptions;

import com.bnpp.irb.qlickflow.admin.common.ApiResponse;
import jakarta.persistence.EntityNotFoundException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.stream.Collectors;

/**
 * Traduction des exceptions métier en codes HTTP, dans l'enveloppe ApiResponse.
 * Volontairement sans handler générique Exception.class : les AccessDeniedException de
 * Spring Security doivent continuer à produire un 403 via ExceptionTranslationFilter.
 */
@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {

    @ExceptionHandler(DuplicateException.class)
    public ResponseEntity<ApiResponse<Void>> handleDuplicate(DuplicateException ex) {
        log.warn("Conflit : {}", ex.getMessage());
        return ResponseEntity.status(HttpStatus.CONFLICT).body(ApiResponse.error(ex.getMessage()));
    }

    @ExceptionHandler({EntityNotFoundException.class, ResourceNotFoundException.class})
    public ResponseEntity<ApiResponse<Void>> handleNotFound(Exception ex) {
        log.warn("Ressource introuvable : {}", ex.getMessage());
        return ResponseEntity.status(HttpStatus.NOT_FOUND).body(ApiResponse.error(ex.getMessage()));
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ApiResponse<Void>> handleIllegalArgument(IllegalArgumentException ex) {
        log.warn("Requête invalide : {}", ex.getMessage());
        return ResponseEntity.badRequest().body(ApiResponse.error(ex.getMessage()));
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Void>> handleValidation(MethodArgumentNotValidException ex) {
        String details = ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> fe.getField() + " : " + fe.getDefaultMessage())
                .collect(Collectors.joining(" ; "));
        log.warn("Validation échouée : {}", details);
        return ResponseEntity.badRequest().body(ApiResponse.error(details, "Validation échouée"));
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<ApiResponse<Void>> handleUnreadable(HttpMessageNotReadableException ex) {
        // Typiquement : valeur d'enum inconnue (ex. prestataire = "BRINK") ou JSON mal formé.
        log.warn("Corps de requête illisible : {}", ex.getMostSpecificCause().getMessage());
        return ResponseEntity.badRequest().body(ApiResponse.error("Corps de requête invalide", "Validation échouée"));
    }
}
