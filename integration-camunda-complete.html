package com.bmci.workflow.fund.management.external.camunda.client;

import com.bmci.workflow.fund.management.external.camunda.config.CamundaProperties;
import lombok.RequiredArgsConstructor;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

import java.time.OffsetDateTime;
import java.time.format.DateTimeFormatter;
import java.util.HashMap;
import java.util.Map;

/**
 * SEULE classe du projet qui connaît les URLs de l'API REST Camunda.
 * Trois opérations : démarrer une instance, la supprimer (compensation), corréler un message.
 */
@Component
@RequiredArgsConstructor
public class CamundaEngineClient {

    /** Format de date attendu par le moteur (variables de type Date, lues par les timers). */
    public static final DateTimeFormatter FORMAT_DATE_CAMUNDA =
            DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss.SSSZ");

    private final RestClient camundaRestClient;
    private final CamundaProperties properties;

    /** POST /process-definition/key/{key}/start — renvoie l'id technique de l'instance. */
    public String demarrerProcessus(String businessKey, Map<String, Object> variables) {
        Map<String, Object> corps = Map.of(
                "businessKey", businessKey,
                "variables", typerVariables(variables));

        Map<?, ?> reponse = camundaRestClient.post()
                .uri("/process-definition/key/{key}/start", properties.getProcessDefinitionKey())
                .contentType(MediaType.APPLICATION_JSON)
                .body(corps)
                .retrieve()
                .body(Map.class);
        return reponse == null ? null : (String) reponse.get("id");
    }

    /** DELETE /process-instance/{id} — compensation de la saga de soumission. */
    public void supprimerInstance(String processInstanceId, String motif) {
        camundaRestClient.delete()
                .uri(uri -> uri.path("/process-instance/{id}")
                        .queryParam("deleteReason", motif)
                        .queryParam("skipCustomListeners", true)
                        .build(processInstanceId))
                .retrieve()
                .toBodilessEntity();
    }

    /** POST /message — corrèle un message BPMN (msgAnnulerDemande…) sur l'instance de la demande. */
    public void correlerMessage(String nomMessage, String businessKey, Map<String, Object> variables) {
        Map<String, Object> corps = Map.of(
                "messageName", nomMessage,
                "businessKey", businessKey,
                "processVariables", typerVariables(variables));
        camundaRestClient.post()
                .uri("/message")
                .contentType(MediaType.APPLICATION_JSON)
                .body(corps)
                .retrieve()
                .toBodilessEntity();
    }

    /** L'API Camunda attend {name: {value, type}} — on type depuis la valeur Java. */
    private static Map<String, Object> typerVariables(Map<String, Object> variables) {
        Map<String, Object> typees = new HashMap<>();
        variables.forEach((nom, valeur) -> {
            String type;
            Object v = valeur;
            if (valeur instanceof Boolean)             { type = "Boolean"; }
            else if (valeur instanceof Integer || valeur instanceof Long) { type = "Long"; }
            else if (valeur instanceof OffsetDateTime date) { type = "Date"; v = FORMAT_DATE_CAMUNDA.format(date); }
            else                                       { type = "String"; v = String.valueOf(valeur); }
            typees.put(nom, Map.of("value", v, "type", type));
        });
        return typees;
    }
}

package com.bmci.workflow.fund.management.service.domain.saga;

import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.List;

/**
 * Exécute les étapes dans l'ordre ; à la première erreur, compense les étapes déjà réussies
 * en ordre inverse. PAS de @Transactional autour d'une saga : chaque étape commit son travail,
 * la saga EST le rollback global.
 */
@Slf4j
@Component
public class SagaOrchestrator {

    public <C> void executer(String nomSaga, C contexte, List<SagaStep<C>> etapes) {
        Deque<SagaStep<C>> reussies = new ArrayDeque<>();
        for (SagaStep<C> etape : etapes) {
            try {
                log.info("Saga [{}] > {}", nomSaga, etape.nom());
                etape.executer(contexte);
                reussies.push(etape);
            } catch (Exception erreur) {
                log.error("Saga [{}] > ÉCHEC de '{}' : {} — compensation de {} étape(s)",
                        nomSaga, etape.nom(), erreur.getMessage(), reussies.size());
                while (!reussies.isEmpty()) {
                    SagaStep<C> aCompenser = reussies.pop();
                    try {
                        log.warn("Saga [{}] > compensation de : {}", nomSaga, aCompenser.nom());
                        aCompenser.compenser(contexte);
                    } catch (Exception erreurCompensation) {
                        // On continue les autres compensations — l'incident est tracé pour intervention manuelle.
                        log.error("Saga [{}] > COMPENSATION EN ÉCHEC pour '{}' : {}",
                                nomSaga, aCompenser.nom(), erreurCompensation.getMessage(), erreurCompensation);
                    }
                }
                throw new SagaExecutionException(
                        "Saga '" + nomSaga + "' en échec à l'étape '" + etape.nom() + "'", erreur);
            }
        }
        log.info("Saga [{}] > terminée ({} étapes)", nomSaga, etapes.size());
    }
}

/** Étape 1 : INITIATED → SUBMITTED. Compensation : restaurer l'ancien statut. */
@Component
@RequiredArgsConstructor
public class ChangerStatutDemandeStep implements SagaStep<SoumissionDemandeContext> {

    private final CashDemandRepository cashDemandRepository;

    @Override public String nom() { return "Passage du statut à SUBMITTED"; }

    @Override
    public void executer(SoumissionDemandeContext contexte) {
        CashRequest demande = cashDemandRepository.getCashRequestById(contexte.getDemandeId());
        if (!RequestStatus.INITIATED.name().equals(demande.getRequestStatus())) {
            throw new RetailConflictException(
                    "Seule une demande INITIATED peut être soumise (statut actuel : " + demande.getRequestStatus() + ")");
        }
        contexte.setAncienStatut(demande.getRequestStatus());
        contexte.setDemande(cashDemandRepository.changerStatut(contexte.getDemandeId(), RequestStatus.SUBMITTED));
    }

    @Override
    public void compenser(SoumissionDemandeContext contexte) {
        cashDemandRepository.changerStatut(contexte.getDemandeId(),
                RequestStatus.valueOf(contexte.getAncienStatut()));
    }
}

    @Override public String nom() { return "Démarrage du processus gestion-des-fonds"; }

    @Override
    public void executer(SoumissionDemandeContext contexte) {
        var demande = contexte.getDemande();

        // Contrat de démarrage documenté dans le BPMN (bloc VARIABLES A FOURNIR AU DEMARRAGE)

        if (contexte.getProcessInstanceId() != null) {

package com.bmci.workflow.fund.management.external.camunda.worker;

import com.bmci.workflow.fund.management.external.camunda.config.CamundaProperties;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.camunda.bpm.client.ExternalTaskClient;
import org.camunda.bpm.client.task.ExternalTaskHandler;
import org.springframework.boot.context.event.ApplicationReadyEvent;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.context.event.EventListener;

/**
 * Ouvre le client external tasks au démarrage et abonne un handler par topic gdf-*.
 * 3 retries espacés de 10 s par tâche, puis incident visible en Cockpit.
 */
@Slf4j
@Configuration
@RequiredArgsConstructor
public class GdfWorkerConfig {

    private static final int RETRIES = 3;
    private static final long RETRY_TIMEOUT_MS = 10_000;

    private final CamundaProperties properties;
    private final InitialiserDemandeHandler initialiserDemandeHandler;
    private final ReporterJourSuivantHandler reporterJourSuivantHandler;
    private final NormaliserUrgenceHandler normaliserUrgenceHandler;
    private final CloturerJourneeHandler cloturerJourneeHandler;
    private final CloturerDemandeHandler cloturerDemandeHandler;
    private final AnnulerDemandeHandler annulerDemandeHandler;

    @Bean(destroyMethod = "stop")
    public ExternalTaskClient externalTaskClient() {
        return ExternalTaskClient.create()
                .baseUrl(properties.getBaseUrl())
                .workerId(properties.getWorker().getWorkerId())
                .lockDuration(properties.getWorker().getLockDurationMs())
                .asyncResponseTimeout(properties.getWorker().getAsyncResponseTimeoutMs())
                .disableBackoffStrategy()
                .build();
    }

    @EventListener(ApplicationReadyEvent.class)
    public void demarrerAbonnements() {
        if (!properties.getWorker().isEnabled()) {
            log.warn("Worker Camunda désactivé (camunda.worker.enabled=false)");
            return;
        }
        ExternalTaskClient client = externalTaskClient();
        abonner(client, GdfTopics.INITIALISER_DEMANDE,    initialiserDemandeHandler);
        abonner(client, GdfTopics.REPORTER_JOUR_SUIVANT,  reporterJourSuivantHandler);
        abonner(client, GdfTopics.NORMALISER_URGENCE,     normaliserUrgenceHandler);
        abonner(client, GdfTopics.CLOTURER_JOURNEE,       cloturerJourneeHandler);
        abonner(client, GdfTopics.CLOTURER_DEMANDE,       cloturerDemandeHandler);
        abonner(client, GdfTopics.ANNULER_DEMANDE,        annulerDemandeHandler);
        abonner(client, GdfTopics.ECLATER_BESOINS_GAB,    new TacheTraceeHandler(GdfTopics.ECLATER_BESOINS_GAB));
        abonner(client, GdfTopics.APPLIQUER_MODIFICATION, new TacheTraceeHandler(GdfTopics.APPLIQUER_MODIFICATION));
        abonner(client, GdfTopics.HISTORISER_COMMENTAIRE, new TacheTraceeHandler(GdfTopics.HISTORISER_COMMENTAIRE));
        log.info("Worker Camunda abonné aux 9 topics gdf-* sur {}", properties.getBaseUrl());
    }

    private void abonner(ExternalTaskClient client, String topic, ExternalTaskHandler handler) {
        client.subscribe(topic)
                .handler((task, service) -> {
                    try {
                        handler.execute(task, service);
                    } catch (Exception erreur) {
                        log.error("[{}] échec pour la demande {} : {}", topic, task.getBusinessKey(), erreur.getMessage(), erreur);
                        int restants = task.getRetries() == null ? RETRIES : task.getRetries() - 1;
                        service.handleFailure(task, erreur.getMessage(),
                                erreur.toString(), restants, RETRY_TIMEOUT_MS);
                    }
                })
                .open();
    }
}

/** Le test qui compte : moteur KO → compensation → statut restauré. */
@ExtendWith(MockitoExtension.class)
class CashDemandSoumissionTest {

    @Mock private CashDemandRepository cashDemandRepository;
    @Mock private ProcessusWorkflowPort processusWorkflowPort;

    private final SagaOrchestrator orchestrator = new SagaOrchestrator();

    private static CashRequest demande(String statut) {
        CashRequest demande = new CashRequest();
        demande.setRequestStatus(statut);
        demande.setUniqueRequestRef("RT-2026-000001");
        demande.setBranchCode("BR-0036");
        return demande;
    }

    @Test
    void happyPath_statutSubmittedEtInstanceDemarree() {
        UUID id = UUID.randomUUID();
        Mockito.when(cashDemandRepository.getCashRequestById(id)).thenReturn(demande("INITIATED"));
        Mockito.when(cashDemandRepository.changerStatut(id, RequestStatus.SUBMITTED)).thenReturn(demande("SUBMITTED"));
        Mockito.when(processusWorkflowPort.demarrerProcessus(Mockito.eq(id.toString()), Mockito.anyMap()))
                .thenReturn("instance-42");

        SoumissionDemandeContext contexte = new SoumissionDemandeContext(id);
        orchestrator.executer("soumission-demande", contexte, List.of(
                new ChangerStatutDemandeStep(cashDemandRepository),
                new DemarrerProcessusStep(processusWorkflowPort)));

        assertEquals("instance-42", contexte.getProcessInstanceId());
        Mockito.verify(processusWorkflowPort).demarrerProcessus(Mockito.eq(id.toString()), Mockito.anyMap());
    }

    @Test
    void moteurEteint_compensationRestaureLeStatut() {
        UUID id = UUID.randomUUID();
        Mockito.when(cashDemandRepository.getCashRequestById(id)).thenReturn(demande("INITIATED"));
        Mockito.when(cashDemandRepository.changerStatut(Mockito.eq(id), Mockito.any())).thenReturn(demande("SUBMITTED"));
        Mockito.when(processusWorkflowPort.demarrerProcessus(Mockito.anyString(), Mockito.anyMap()))
                .thenThrow(new RuntimeException("Connexion refusée"));

        SoumissionDemandeContext contexte = new SoumissionDemandeContext(id);
        assertThrows(SagaExecutionException.class, () ->
                orchestrator.executer("soumission-demande", contexte, List.of(
                        new ChangerStatutDemandeStep(cashDemandRepository),
                        new DemarrerProcessusStep(processusWorkflowPort))));

        // SUBMITTED posé, puis restauré à INITIATED par la compensation
        Mockito.verify(cashDemandRepository).changerStatut(id, RequestStatus.SUBMITTED);
        Mockito.verify(cashDemandRepository).changerStatut(id, RequestStatus.INITIATED);
    }
}
