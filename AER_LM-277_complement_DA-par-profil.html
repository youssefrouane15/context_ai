package com.bnpp.irb.qlickflow.admin.controller.internal;

import com.bnpp.irb.qlickflow.admin.common.ApiResponse;

import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

/**
 * AER_LM-277 — Directeur d'agence d'une agence = utilisateur actif de profil DA rattaché à l'agence de ce code.
 * Déduit de profile.code + orgaUnit.code (pas de orga_unit.manager_id, non géré par l'application).
 * Canal interne (/api/v1/internal, non routé par la gateway), même convention qu'InternalUserController.
 */
@RestController
@RequestMapping("/api/v1/internal/users/agences")
@RequiredArgsConstructor
public class InternalDirecteurAgenceController {

    private static final String PROFIL_DA = "DA";

    private final UserRepository userRepository;

    @GetMapping("/{code}/directeur")
    public ResponseEntity<ApiResponse<DirecteurAgenceDTO>> directeur(@PathVariable String code) {
        return userRepository
                .findFirstByOrganizationUnit_CodeIgnoreCaseAndOrganizationUnit_TypeAndProfile_CodeAndActifTrueOrderByIdAsc(
                        code, OrganizationUnitType.AGENCE, PROFIL_DA)
                .map(this::versDto)
                .map(dto -> ResponseEntity.ok(ApiResponse.success(dto)))
                .orElseGet(() -> ResponseEntity.status(HttpStatus.NOT_FOUND)
                        .body(ApiResponse.error("Aucun Directeur d'agence actif pour l'agence " + code)));
    }

    private DirecteurAgenceDTO versDto(UserEntity u) {
        return new DirecteurAgenceDTO(u.getUid(), u.getEmail(), u.getFirstName(), u.getLastName());
    }
}

package com.bmci.workflow.fund.management.external.users;

import com.bmci.q360.domain.retail.valueobjects.referentiel.DirecteurAgence;
import com.bmci.workflow.fund.management.external.users.dto.DirecteurAgenceDonneesDTO;
import com.bmci.workflow.fund.management.external.users.dto.DirecteurAgenceReponseDTO;
import com.bmci.workflow.fund.management.service.domain.ports.output.referentiel.DirecteurAgencePort;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

import java.util.Optional;

/**
 * AER_LM-277 — Directeur d'agence par code agence : GET /api/v1/internal/users/agences/{code}/directeur (utilisateurs-ci).
 * Même style qu'UtilisateursAgenceAdapter : RestClient.create (pas un bean), tolérant — 404 ou users injoignable → Optional.empty() + warn.
 */
@Component
@Slf4j
public class UtilisateursDirecteurAdapter implements DirecteurAgencePort {

    private final RestClient client;

    public UtilisateursDirecteurAdapter(@Value("${qf.users.internal-base-url}") String baseUrl) {
        this.client = RestClient.create(baseUrl);
    }

    @Override
    public Optional<DirecteurAgence> directeurDe(String branchCode) {
        if (branchCode == null || branchCode.isBlank()) { return Optional.empty(); }
        try {
            DirecteurAgenceReponseDTO reponse = client.get()
                    .uri("/api/v1/internal/users/agences/{code}/directeur", branchCode)
                    .retrieve()
                    .body(DirecteurAgenceReponseDTO.class);
            if (reponse == null || !reponse.success() || reponse.data() == null) { return Optional.empty(); }
            DirecteurAgenceDonneesDTO d = reponse.data();
            String nom = ((d.firstName() == null ? "" : d.firstName()) + " " + (d.lastName() == null ? "" : d.lastName())).trim();
            return Optional.of(new DirecteurAgence(d.uid(), d.email(), nom.isEmpty() ? d.uid() : nom));
        } catch (RuntimeException e) {
            log.warn("[notification] Directeur d'agence de {} introuvable chez utilisateurs-ci : {}", branchCode, e.getMessage());
            return Optional.empty();
        }
    }
}

/**
 * AER_LM-277 — notification email au Directeur d'agence après une soumission, une mise à jour ou une annulation par l'agence.
 * Le DA est résolu par le code agence de la demande. Best effort : aucune exception ne remonte au use case appelant.
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class NotificationDemandeService {

    private final DirecteurAgencePort directeurAgencePort;
    private final NotificationPort notificationPort;
    private final GabaritEmailDemande gabarit;

    public void notifier(CashRequestDTO demande, EvenementDemande evenement) {
        if (demande == null) { return; }
        try {
            Optional<DirecteurAgence> directeur = directeurAgencePort.directeurDe(demande.getBranchCode());
            if (directeur.isEmpty() || directeur.get().email() == null || directeur.get().email().isBlank()) {
                log.warn("[notification] {} {} : aucun Directeur d'agence avec email pour l'agence {} — email non envoyé",
                        evenement, demande.getUniqueRequestRef(), demande.getBranchCode());
                return;
            }
            String sujet = "[QlickFlow] " + evenement.getLibelle() + " " + demande.getUniqueRequestRef()
                    + " — Agence " + demande.getBranchCode();
            String corps = gabarit.construire(demande, evenement, directeur.get().nomComplet());
            notificationPort.envoyer(new EmailNotification(directeur.get().email(), sujet, corps));
            log.info("[notification] {} {} → {}", evenement, demande.getUniqueRequestRef(), directeur.get().email());
        } catch (RuntimeException e) {
            log.error("[notification] {} {} : échec de l'envoi, demande non impactée — {}",
                    evenement, demande.getUniqueRequestRef(), e.getMessage());
        }
    }
}
