import { Injectable } from '@angular/core';
import { HttpClient, HttpErrorResponse } from '@angular/common/http';
import { Observable, throwError } from 'rxjs';
import { catchError, map } from 'rxjs/operators';
import { ApiConfigService } from '../../api-config.service';
import { ApiResponse } from '../../../../shared/interfaces/derogation.interface';
import { AgenceDTO } from '../../../../shared/interfaces/habilitation.interface';
import { PrestataireDTO } from '../../../../shared/interfaces/referentiel.interface';
import {
  CreateGabHorsSiteDTO,
  GabHorsSiteDTO,
  UpdateGabHorsSiteDTO,
  VilleDTO
} from '../../../../shared/interfaces/gab-hors-site.interface';

/**
 * AER_LM-313 — GAB hors site (utilisateurs-ci, /api/v1/public/users/gab/hors-site).
 * Même modèle que ReferentielService : baseUrl console + map(r => r.data) + handleError.
 */
@Injectable({
  providedIn: 'root'
})
export class GabHorsSiteService {

  private readonly baseUrl: string;
  private readonly consoleUrl: string;

  constructor(
    private http: HttpClient,
    private apiConfig: ApiConfigService
  ) {
    this.consoleUrl = this.apiConfig.buildConsoleApiUrl();
    this.baseUrl = `${this.consoleUrl}/gab/hors-site`;
  }

  getGabs(): Observable<GabHorsSiteDTO[]> {
    return this.http.get<ApiResponse<GabHorsSiteDTO[]>>(this.baseUrl)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  getGab(id: number): Observable<GabHorsSiteDTO> {
    return this.http.get<ApiResponse<GabHorsSiteDTO>>(`${this.baseUrl}/${id}`)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  createGab(dto: CreateGabHorsSiteDTO): Observable<GabHorsSiteDTO> {
    return this.http.post<ApiResponse<GabHorsSiteDTO>>(this.baseUrl, dto)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  updateGab(dto: UpdateGabHorsSiteDTO): Observable<string> {
    return this.http.put<ApiResponse<string>>(this.baseUrl, dto)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  /** Liste fermée des villes (enum Ville). */
  getVilles(): Observable<VilleDTO[]> {
    return this.http.get<ApiResponse<VilleDTO[]>>(`${this.baseUrl}/villes`)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  /** Transporteurs = prestataires du référentiel (BRINKS / G4S). */
  getTransporteurs(): Observable<PrestataireDTO[]> {
    return this.http.get<ApiResponse<PrestataireDTO[]>>(`${this.consoleUrl}/referentiels/prestataires`)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  /** Agences de rattachement (GET liste d'AgenceController, non paginé). */
  getAgences(): Observable<AgenceDTO[]> {
    return this.http.get<ApiResponse<AgenceDTO[]>>(`${this.consoleUrl}/orga/agences`)
      .pipe(map(r => r.data), catchError(this.handleError));
  }

  private handleError(error: HttpErrorResponse): Observable<never> {
    let errorMessage: string;

    if (error.status === 0) {
      errorMessage = 'Le serveur est injoignable. Veuillez réessayer plus tard.';
    } else if (error.status === 403) {
      errorMessage = "Vous n'avez pas le droit de gérer les GAB hors site.";
    } else if (error.error?.message) {
      errorMessage = error.error.message;
    } else {
      errorMessage = 'Une erreur est survenue. Veuillez réessayer.';
    }

    return throwError(() => new Error(errorMessage));
  }
}

import { Component, OnInit } from '@angular/core';
import { Router } from '@angular/router';
import { forkJoin, Observable } from 'rxjs';
import { finalize } from 'rxjs/operators';
import { GabHorsSiteService } from '../../../core/services/console/gab/gab-hors-site.service';
import { AuthService } from 'src/app/auth/auth.service';
import { AgenceDTO } from '../../../shared/interfaces/habilitation.interface';
import { PrestataireDTO } from '../../../shared/interfaces/referentiel.interface';
import {
  CreateGabHorsSiteDTO,
  EtatGab,
  GabHorsSiteDTO,
  TypeGab,
  UpdateGabHorsSiteDTO,
  VilleDTO
} from '../../../shared/interfaces/gab-hors-site.interface';

const PERMISSION_CONSULTER = 'GDF_CONSULTER_GAB';
const PERMISSION_GERER = 'GDF_GERER_GAB';
const TAILLE_PAGE = 10;

type OngletGab = 'GAB';
type ModeModale = 'AJOUT' | 'MODIFICATION';

/** Saisie de la modale (prix en texte pour accepter la virgule, comme le coût de déplacement de l'agence). */
interface FormulaireGab {
  id: number | null;
  idGab: string;
  nom: string;
  numeroCompte: string;
  agenceId: number | null;
  ville: string;
  transporteur: string;
  prixUnitaire: string;
  type: TypeGab;
  etat: EtatGab;
}

const FORMULAIRE_VIDE: FormulaireGab = {
  id: null, idGab: '', nom: '', numeroCompte: '', agenceId: null, ville: '',
  transporteur: 'BRINKS', prixUnitaire: '0', type: 'H', etat: 'EN_LIGNE'
};

@Component({
  selector: 'app-console-gab-hors-site',
  templateUrl: './console-gab-hors-site.component.html',
  styleUrls: ['./console-gab-hors-site.component.scss']
})
export class ConsoleGabHorsSiteComponent implements OnInit {

  activeTab: OngletGab = 'GAB';

  gabs: GabHorsSiteDTO[] = [];
  villes: VilleDTO[] = [];
  transporteurs: PrestataireDTO[] = [];
  agences: AgenceDTO[] = [];

  recherche = '';
  pageCourante = 1;
  readonly taillePage = TAILLE_PAGE;

  loading = false;
  errorMessage = '';
  successMessage = '';

  showModal = false;
  modeModale: ModeModale = 'AJOUT';
  formulaire: FormulaireGab = { ...FORMULAIRE_VIDE };
  erreurModale = '';
  enregistrementEnCours = false;

  peutGerer = false;

  readonly types: TypeGab[] = ['H', 'D'];
  readonly etats: { id: EtatGab; nom: string }[] = [
    { id: 'EN_LIGNE', nom: 'En ligne' },
    { id: 'HORS_LIGNE', nom: 'Hors ligne' }
  ];

  constructor(
    private gabService: GabHorsSiteService,
    private authService: AuthService,
    private router: Router
  ) {}

  ngOnInit(): void {
    if (!this.authService.hasPermission([PERMISSION_CONSULTER, PERMISSION_GERER])) {
      this.router.navigate(['/admin/access-denied']);
      return;
    }
    this.peutGerer = this.authService.hasPermission([PERMISSION_GERER]);
    this.charger();
  }

  setActiveTab(tab: OngletGab): void {
    this.activeTab = tab;
  }

  // ----- Chargement

  charger(): void {
    this.loading = true;
    this.errorMessage = '';

    forkJoin({
      gabs: this.gabService.getGabs(),
      villes: this.gabService.getVilles(),
      transporteurs: this.gabService.getTransporteurs(),
      agences: this.gabService.getAgences()
    })
      .pipe(finalize(() => (this.loading = false)))
      .subscribe({
        next: ({ gabs, villes, transporteurs, agences }) => {
          this.gabs = gabs;
          this.villes = villes;
          this.transporteurs = transporteurs;
          this.agences = [...agences].sort((a, b) => a.code.localeCompare(b.code));
        },
        error: (error: Error) => {
          this.errorMessage = error.message || 'Impossible de charger les GAB hors site.';
        }
      });
  }

  // ----- Liste : filtre + pagination côté client (parc de quelques dizaines de GAB)

  get gabsFiltres(): GabHorsSiteDTO[] {
    const terme = this.recherche.trim().toLowerCase();
    if (!terme) {
      return this.gabs;
    }
    return this.gabs.filter(g =>
      g.idGab.toLowerCase().includes(terme)
      || g.nom.toLowerCase().includes(terme)
      || g.numeroCompte.toLowerCase().includes(terme)
      || `${g.agence?.code} ${g.agence?.nom}`.toLowerCase().includes(terme)
      || g.ville?.nom.toLowerCase().includes(terme)
    );
  }

  get nombrePages(): number {
    return Math.max(1, Math.ceil(this.gabsFiltres.length / this.taillePage));
  }

  get pages(): number[] {
    return Array.from({ length: this.nombrePages }, (_, i) => i + 1);
  }

  get gabsPage(): GabHorsSiteDTO[] {
    const debut = (this.pageCourante - 1) * this.taillePage;
    return this.gabsFiltres.slice(debut, debut + this.taillePage);
  }

  onRecherche(): void {
    this.pageCourante = 1;
  }

  allerPage(page: number): void {
    if (page >= 1 && page <= this.nombrePages) {
      this.pageCourante = page;
    }
  }

  trackParId(_: number, gab: GabHorsSiteDTO): number {
    return gab.id;
  }

  formaterPrix(prix: number): string {
    return `${(prix ?? 0).toFixed(2).replace('.', ',')} MAD`;
  }

  // ----- Modale

  ouvrirAjout(): void {
    this.modeModale = 'AJOUT';
    this.formulaire = { ...FORMULAIRE_VIDE };
    this.ouvrirModale();
  }

  ouvrirModification(gab: GabHorsSiteDTO): void {
    this.modeModale = 'MODIFICATION';
    this.formulaire = {
      id: gab.id,
      idGab: gab.idGab,
      nom: gab.nom,
      numeroCompte: gab.numeroCompte,
      agenceId: gab.agence?.id ?? null,
      ville: gab.ville?.id ?? '',
      transporteur: gab.transporteur?.id ?? 'BRINKS',
      prixUnitaire: (gab.prixUnitaire ?? 0).toFixed(2).replace('.', ','),
      type: gab.type,
      etat: gab.etat
    };
    this.ouvrirModale();
  }

  fermerModale(): void {
    this.showModal = false;
    this.erreurModale = '';
  }

  private ouvrirModale(): void {
    this.erreurModale = '';
    this.successMessage = '';
    this.showModal = true;
  }

  get prixValide(): boolean {
    return this.parserMontant(this.formulaire.prixUnitaire) !== null;
  }

  get formulaireValide(): boolean {
    const f = this.formulaire;
    return !!f.idGab.trim() && !!f.nom.trim() && !!f.numeroCompte.trim()
      && f.agenceId !== null && !!f.ville && this.prixValide;
  }

  /** « 350,50 » ou « 350.5 » → 350.5 ; vide → 0 ; invalide → null. */
  parserMontant(saisie: string): number | null {
    const texte = (saisie ?? '').trim().replace(/\s/g, '').replace(',', '.');
    if (texte === '') {
      return 0;
    }
    if (!/^\d+(\.\d{1,2})?$/.test(texte)) {
      return null;
    }
    return Number(texte);
  }

  terminer(): void {
    if (!this.formulaireValide || this.enregistrementEnCours) {
      return;
    }
    this.enregistrementEnCours = true;
    this.erreurModale = '';

    const f = this.formulaire;
    const commun: CreateGabHorsSiteDTO = {
      idGab: f.idGab.trim(),
      nom: f.nom.trim(),
      numeroCompte: f.numeroCompte.trim(),
      agenceId: f.agenceId as number,
      ville: f.ville,
      transporteur: f.transporteur,
      prixUnitaire: this.parserMontant(f.prixUnitaire) ?? 0,
      type: f.type,
      etat: f.etat
    };

    const requete: Observable<unknown> = this.modeModale === 'AJOUT'
      ? this.gabService.createGab(commun)
      : this.gabService.updateGab({ id: f.id as number, ...commun } as UpdateGabHorsSiteDTO);

    requete
      .pipe(finalize(() => (this.enregistrementEnCours = false)))
      .subscribe({
        next: () => {
          this.successMessage = this.modeModale === 'AJOUT'
            ? `Le GAB ${commun.idGab} a été ajouté.`
            : `Le GAB ${commun.idGab} a été mis à jour.`;
          this.fermerModale();
          this.charger();
        },
        error: (error: Error) => {
          this.erreurModale = error.message || "Impossible d'enregistrer le GAB hors site.";
        }
      });
  }
}
