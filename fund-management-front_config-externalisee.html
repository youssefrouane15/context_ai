import { APP_INITIALIZER, Provider } from '@angular/core';
import { ResolveFn } from '@angular/router';
import { Environnement, environment } from '../../../environments/environment';

/** Fixé par Module Federation (publicPath « auto ») à l'emplacement réel de remoteEntry.js ; absent hors webpack (tests). */
declare let __webpack_public_path__: string | undefined;

/**
 * Fichier de configuration d'environnement, écrasé par le CD de l'intégrateur à chaque déploiement
 * (/var/www/html/config/app-config.json) : une seule image pour tous les environnements.
 * Copié dans dist/config par angular.json (assets) ; en local = src/config/app-config.json.
 */
export const FICHIER_CONFIGURATION = 'config/app-config.json';

/**
 * URL absolue du fichier. Sous le shell portail, une URL relative se résoudrait sur l'origine du PORTAIL (404) :
 * on part de l'origine réelle du remote (publicPath), sinon du document (4201 seul, tests).
 */
export function urlConfiguration(): string {
  const base = typeof __webpack_public_path__ === 'string' && /^https?:\/\//.test(__webpack_public_path__)
    ? __webpack_public_path__
    : document.baseURI;
  return new URL(FICHIER_CONFIGURATION, base).toString();
}

/** Clés que le fichier a le droit de surcharger (`production` reste décidé par le build). */
type ConfigurationExterne = Partial<Omit<Environnement, 'production'>>;

let chargement: Promise<void> | null = null;

/**
 * Charge config/app-config.json et fusionne ses valeurs dans `environment` AVANT le premier appel HTTP.
 * fetch() plutôt que HttpClient : rien ne doit passer par les intercepteurs (qui lisent eux-mêmes `environment`).
 * - fichier absent en dev : on garde les valeurs d'environment.ts (avertissement console) ;
 * - fichier absent en prod : erreur bloquante, l'application ne doit pas démarrer avec des URL vides.
 * Idempotent : un seul chargement quel que soit le nombre d'appelants (initialiseur ET résolveur).
 */
export function chargerConfiguration(): Promise<void> {
  return chargement ??= charger();
}

async function charger(): Promise<void> {
  const url = urlConfiguration();
  let reponse: Response;
  try {
    reponse = await fetch(url, { cache: 'no-store' });
  } catch (erreur) {
    return signalerAbsence(`${url} injoignable (${erreur})`);
  }
  if (!reponse.ok) {
    return signalerAbsence(`${url} : HTTP ${reponse.status}`);
  }
  const externe = (await reponse.json()) as ConfigurationExterne;
  const { production: _ignore, ...valeurs } = externe as Record<string, unknown>;   // production jamais surchargée
  Object.assign(environment, valeurs);
}

function signalerAbsence(detail: string): void {
  if (environment.production) {
    throw new Error(`Configuration manquante : ${detail}`);
  }
  console.warn(`[config] ${detail} — valeurs par défaut d'environment.ts conservées.`);
}

/** Remise à zéro pour les tests unitaires. */
export function reinitialiserConfiguration(): void {
  chargement = null;
}

/** Démarrage autonome (4201) : providers de app.config.ts. */
export const CONFIGURATION_INITIALISEUR: Provider = {
  provide: APP_INITIALIZER,
  useFactory: () => chargerConfiguration,
  multi: true
};

/**
 * Sous le shell portail, app.config.ts du remote ne s'exécute pas : le chargement est garanti par ce résolveur
 * sur la route racine exposée (app.routes.ts). Même promesse que l'initialiseur → aucun double appel.
 */
export const CONFIGURATION_RESOLVEUR: ResolveFn<void> = () => chargerConfiguration();
