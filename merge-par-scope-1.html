/**
 * Enveloppe standard de toute réponse HTTP (retail et corpo),
 * format socle QlickFlow : { success, message, data, error }.
 * En erreur, "data" peut porter le détail structuré (ex. erreurs par champ).
 */
@Getter
@NoArgsConstructor
@AllArgsConstructor
@JsonInclude(JsonInclude.Include.NON_NULL)
public class ApiResponse<T> {

    private Boolean success;
    private String message;
    private T data;
    private String error;

    public static <T> ApiResponse<T> success(T data) {
        return new ApiResponse<>(true, null, data, null);
    }

    public static <T> ApiResponse<T> failure(String message, String error) {
        return new ApiResponse<>(false, message, null, error);
    }

    public static <T> ApiResponse<T> failure(String message, String error, T detail) {
        return new ApiResponse<>(false, message, detail, error);
    }
}

        if (RequestStatus.CANCELLED.equals(this.requestStatus)) {

        }

            }
            operation.setOperationStatus(RequestStatus.CANCELLED);
        }
        boolean allCancelled = this.operations.stream()
                .allMatch(op -> RequestStatus.CANCELLED.equals(op.getOperationStatus()));
        if (allCancelled) {
            this.requestStatus = RequestStatus.CANCELLED;
        }
    }

    public ResponseEntity<ApiResponse<Void>> handleValidation(RetailValidationException exception) {
        return build(RetailHttpStatus.RETAIL_VALIDATION_ERROR, exception.getMessage());

    @ExceptionHandler(RetailNotFoundException.class)

    public ResponseEntity<ApiResponse<Map<String, String>>> handleBeanValidation(MethodArgumentNotValidException exception) {
        Map<String, String> fieldErrors = new HashMap<>();
        exception.getBindingResult().getFieldErrors().forEach(fieldError ->
                fieldErrors.put(fieldError.getField(), fieldError.getDefaultMessage()));
        RetailHttpStatus status = RetailHttpStatus.RETAIL_BAD_REQUEST;

                .body(ApiResponse.failure(resolveMessage(status, "des champs sont invalides"),

    public ResponseEntity<ApiResponse<Void>> handleUnreadable(HttpMessageNotReadableException exception) {
        return build(RetailHttpStatus.RETAIL_BAD_REQUEST, "le corps de la requête est illisible ou mal formé");

        log.error("Erreur non prévue sur un endpoint retail", exception);
        return build(RetailHttpStatus.RETAIL_TECHNICAL_ERROR, null);
    }

        return ResponseEntity.status(status.getHttpStatus())
                .body(ApiResponse.failure(resolveMessage(status, detail), String.valueOf(status.getCode())));

import com.bmci.q360.domain.entity.CashRequest;
import com.bmci.q360.domain.entity.enums.RequestStatus;
import com.bmci.q360.domain.valueobjects.CashOperationId;
import org.junit.jupiter.api.Test;

import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import java.util.UUID;

import static org.junit.jupiter.api.Assertions.*;

class CashRequestCancelOperationsTest {

    private CashRequest demande(UUID id1, UUID id2) {
        CashOperation op1 = new CashOperation();
        op1.setId(new CashOperationId(id1));
        op1.setOperationStatus(RequestStatus.INITIATED);
        CashOperation op2 = new CashOperation();
        op2.setId(new CashOperationId(id2));
        op2.setOperationStatus(RequestStatus.INITIATED);
        CashRequest d = new CashRequest();
        d.setRequestStatus(RequestStatus.INITIATED);
        d.setOperations(new ArrayList<>(List.of(op1, op2)));
        return d;
    }

    @Test
    void annuleUneOperationSansToucherLaDemande() {
        UUID id1 = UUID.randomUUID(), id2 = UUID.randomUUID();
        CashRequest d = demande(id1, id2);
        d.cancelOperations(Set.of(id1));
        assertEquals(RequestStatus.CANCELLED, d.getOperations().get(0).getOperationStatus());
        assertEquals(RequestStatus.INITIATED, d.getOperations().get(1).getOperationStatus());
        assertEquals(RequestStatus.INITIATED, d.getRequestStatus());
    }

    @Test
    void annuleLaDemandeQuandToutEstAnnule() {
        UUID id1 = UUID.randomUUID(), id2 = UUID.randomUUID();
        CashRequest d = demande(id1, id2);
        d.cancelOperations(Set.of(id1, id2));
        assertEquals(RequestStatus.CANCELLED, d.getRequestStatus());
    }

    @Test
    void refuseOperationDejaAnnulee() {
        UUID id1 = UUID.randomUUID();
        CashRequest d = demande(id1, UUID.randomUUID());
        d.getOperations().get(0).setOperationStatus(RequestStatus.CANCELLED);
        assertThrows(IllegalStateException.class, () -> d.cancelOperations(Set.of(id1)));
    }

    @Test
    void refuseOperationInconnue() {
        CashRequest d = demande(UUID.randomUUID(), UUID.randomUUID());
        assertThrows(IllegalArgumentException.class, () -> d.cancelOperations(Set.of(UUID.randomUUID())));
    }
}
