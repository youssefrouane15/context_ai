// demandes-list.regles.ts — À AJOUTER après construireCriteres(…)

/** Carte KPI cliquée. null = aucune : la liste suit les filtres de l'écran. */
export type KpiActif = 'EN_ATTENTE' | 'URGENTES' | 'TRAITEES' | null;

/**
 * Règle PO (07/10) : un clic sur une carte montre exactement ce que la carte compte, sans toucher aux filtres visibles.
 * Les cartes ne connaissent aucun filtre d'écran : on repart donc des seuls page / size / agence et on pose le périmètre de la carte.
 *   en attente : SOUMISES, date d'effet ≤ aujourd'hui   ·   urgentes : idem + urgent   ·   traitées : TRAITÉES (toutes dates)
 * Tout changement de filtre, « Actualiser » et l'export ignorent ce périmètre.
 */
export function appliquerKpi(c: DemandeCriteria, kpi: KpiActif): DemandeCriteria {
  if (!kpi) { return c; }
  const base: DemandeCriteria = { page: c.page, size: c.size, ...(c.agence ? { agence: c.agence } : {}) };
  switch (kpi) {
    case 'EN_ATTENTE': return { ...base, statut: 'SUBMITTED', dateFin: `${aujourdHui()}T23:59:59` };
    case 'URGENTES':   return { ...base, statut: 'SUBMITTED', urgent: true, dateFin: `${aujourdHui()}T23:59:59` };
    case 'TRAITEES':   return { ...base, statut: 'VALIDATED' };
  }
}

/** Critères de l'extraction Excel : les filtres de l'écran seulement — ni la recherche libre, ni la carte KPI, ni la pagination. */
export function criteresExport(f: FiltresListing): DemandeCriteria {
  return construireCriteres({ ...f, recherche: '' }, 0, 0);
}
