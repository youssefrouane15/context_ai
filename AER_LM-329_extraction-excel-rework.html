/**
 * Lecture d'une demande pour l'extraction Excel (AER_LM-279, rework AER_LM-329) : tous les accès aux DTO sont isolés ici.
 * Feuille MAD : quantités par coupure (caisse / GAB) + montant du ramassage MAD.
 * Feuille Devise : une ligne par devise (alimentation et ramassage côte à côte), triées par code devise.
 * Feuille GAB Hors Sites (AER_LM-297) : demandes du canal MONETIQUE, une ligne par GAB avec ses 4 cassettes.
 */
final class ExtractionDemande {

    enum Alimentation { CAISSE, GAB }

    /** Une devise d'une demande : montant d'alimentation et/ou montant de ramassage (null = aucun dans ce sens). */
    record LigneDevise(String devise, BigDecimal alimentation, BigDecimal ramassage) {}

    static final String MAD = "MAD";
    static final String CANAL_MONETIQUE = "MONETIQUE";
    static final int NB_CASSETTES = 4;

    /** Un GAB d'une demande monétique : identité du GAB + quantités par cassette (index 0 = cassette 1). */
    record LigneGab(OperationGabDTO gab, Integer[] quantitesCassettes) {}

    private static final Map<String, String> LIBELLES_STATUT = Map.of(
            "SUBMITTED", "En Attente", "IN_MODIFICATION", "En modification",
            "VALIDATED", "Traitée", "CANCELLED", "Annulée", "INITIATED", "Brouillon");

    private final CashRequestDTO d;

    ExtractionDemande(CashRequestDTO d) { this.d = d; }

    String reference()          { return d.getUniqueRequestRef(); }
    boolean urgente()           { return Boolean.TRUE.equals(d.getUrgent())
            || operations().stream().anyMatch(op -> Boolean.TRUE.equals(op.getIsUrgent())); }
    String codeAgence()         { return d.getBranchCode(); }
    String nomAgence()          { return d.getAgenceNom(); }
    String zone()               { return d.getZoneNom() != null ? d.getZoneNom() : d.getZoneCode(); }
    String transporteur()       { return d.getPrestataireNom() != null ? d.getPrestataireNom() : d.getPrestataire(); }
    BigDecimal coutDeplacement(){ return d.getCoutDeplacement(); }
    LocalDateTime dateDemande() { return d.getCreatedAt(); }
    String libelleStatut()      { return LIBELLES_STATUT.getOrDefault(d.getRequestStatus(), d.getRequestStatus()); }

    private List<CashOperationDTO> operations() { return d.getOperations() == null ? List.of() : d.getOperations(); }

    private static List<OperationCurrencyDetailDTO> details(CashOperationDTO op) {
        return op.getCurrencyDetails() == null ? List.of() : op.getCurrencyDetails();
    }

    private static boolean estAlimentation(CashOperationDTO op, Alimentation type) {
        OperationType attendu = type == Alimentation.CAISSE ? OperationType.CASH_DESK : OperationType.ATM;
        return op.getOperationType() == attendu && op.getOperationSens() == RequestType.SUPPLY;
    }

    /** Feuille MAD : la demande y figure si elle porte une alimentation (caisse ou GAB) ou un ramassage en MAD. */
    boolean figureEnMad() {
        return operations().stream().anyMatch(op -> estAlimentation(op, Alimentation.CAISSE) || estAlimentation(op, Alimentation.GAB))
                || ramassageMad() != null;
    }

    /** Quantités par valeur faciale (200, 100, … 0.05), cumulées sur toutes les opérations du même type. */
    Map<BigDecimal, Integer> quantitesParFaceValue(Alimentation type, Map<String, BigDecimal> faceValuesByCode) {
        Map<BigDecimal, Integer> resultat = new HashMap<>();
        for (CashOperationDTO op : operations()) {
            if (!estAlimentation(op, type)) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (detail.getOperationDenominations() == null) continue;
                for (OperationDenominationDTO ligne : detail.getOperationDenominations()) {
                    if (ligne.getDenomination() == null || ligne.getQuantity() == null) continue;
                    BigDecimal faceValue = faceValuesByCode.get(ligne.getDenomination().getDenominationCode());
                    if (faceValue == null) continue;
                    resultat.merge(faceValue.stripTrailingZeros(), ligne.getQuantity(), Integer::sum);
                }
            }
        }
        return resultat;
    }

    /** Montant du ramassage en MAD (somme des détails MAD des opérations COLLECTION), null si aucun. */
    BigDecimal ramassageMad() {
        BigDecimal total = null;
        for (CashOperationDTO op : operations()) {
            if (op.getOperationSens() != RequestType.COLLECTION) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (!MAD.equals(codeDevise(detail)) || detail.getTotalAmountByCurrency() == null) continue;
                total = total == null ? detail.getTotalAmountByCurrency() : total.add(detail.getTotalAmountByCurrency());
            }
        }
        return total;
    }

    /**
     * Feuille Devise : une ligne par devise étrangère (hors MAD), alimentation (CURRENCY / SUPPLY) et ramassage
     * (CURRENCY / COLLECTION) sur la même ligne, triées par code devise.
     */
    List<LigneDevise> lignesDevise() {
        Map<String, BigDecimal[]> parDevise = new HashMap<>();            // [0] alimentation, [1] ramassage
        for (CashOperationDTO op : operations()) {
            if (op.getOperationType() != OperationType.CURRENCY) continue;
            int sens = op.getOperationSens() == RequestType.SUPPLY ? 0 : op.getOperationSens() == RequestType.COLLECTION ? 1 : -1;
            if (sens < 0) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                String devise = codeDevise(detail);
                if (devise == null || MAD.equals(devise) || detail.getTotalAmountByCurrency() == null) continue;
                BigDecimal[] montants = parDevise.computeIfAbsent(devise, k -> new BigDecimal[2]);
                montants[sens] = montants[sens] == null ? detail.getTotalAmountByCurrency() : montants[sens].add(detail.getTotalAmountByCurrency());
            }
        }
        List<LigneDevise> lignes = new ArrayList<>();
        parDevise.forEach((devise, m) -> lignes.add(new LigneDevise(devise, m[0], m[1])));
        lignes.sort(Comparator.comparing(LigneDevise::devise));
        return lignes;
    }

    /** Demande du Pôle opérations monétique (GAB hors site / drive) : canalDemande = MONETIQUE. */
    boolean estMonetique() { return CANAL_MONETIQUE.equals(d.getCanalDemande()); }

    /**
     * Feuille GAB Hors Sites : une ligne par opération GAB de la demande, quantités par numéro de cassette
     * (OperationDenominationDTO.cassetteNumber 1..4), cumulées si plusieurs lignes portent la même cassette.
     */
    List<LigneGab> lignesGab() {
        List<LigneGab> lignes = new ArrayList<>();
        for (CashOperationDTO op : operations()) {
            if (op.getGab() == null) continue;
            Integer[] quantites = new Integer[NB_CASSETTES];
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (detail.getOperationDenominations() == null) continue;
                for (OperationDenominationDTO ligne : detail.getOperationDenominations()) {
                    Integer cassette = ligne.getCassetteNumber();
                    if (cassette == null || cassette < 1 || cassette > NB_CASSETTES || ligne.getQuantity() == null) continue;
                    int i = cassette - 1;
                    quantites[i] = quantites[i] == null ? ligne.getQuantity() : quantites[i] + ligne.getQuantity();
                }
            }
            lignes.add(new LigneGab(op.getGab(), quantites));
        }
        lignes.sort(Comparator.comparing(l -> l.gab().getGabId() == null ? "" : l.gab().getGabId()));
        return lignes;
    }

    private static String codeDevise(OperationCurrencyDetailDTO detail) {
        return detail.getCurrency() == null ? null : detail.getCurrency().getIsoCode();
    }
}

/**
 * Extraction Excel Caisse Centrale (AER_LM-279, rework AER_LM-329) à partir du template
 * « Template_QlickFlow_GesFond_extraction.xlsx » (2 feuilles) :
 *  - « Appels & Ramassages en MAD »   : une ligne par demande ; A-I communes, J-U quantités caisse par coupure,
 *                                        V total caisse (formule du template), W-X quantités GAB, Y total GAB (formule), Z ramassage MAD.
 *  - « Appels & Ramassages en Devise »: une ligne par devise de la demande ; A-I communes, J-K devise / montant d'alimentation,
 *                                        L-M devise / montant de ramassage.
 *  - « GAB Hors Sites » (AER_LM-297)  : demandes MONETIQUE, une ligne par GAB ; A-L (référence, date, agence, ville, transporteur,
 *                                        PU, ID / nom / compte / type GAB, statut), M-P quantités cassettes 1-4, Q total (formule).
 *                                        En-têtes en ligne 3, données à partir de la ligne 4 (structure propre à cette feuille).
 * Les colonnes de total sont recopiées depuis la formule du template (jamais calculées ici). Montants au format français.
 * Le template du PO porte ses formules de total sur ~80 000 lignes (affichent "" tant que la ligne est vide) : on n'en supprime
 * aucune — les demandes s'écrivent à partir de la ligne 3 et les lignes suivantes restent telles que le PO les a conçues.
 */
@Component
public class ExportExcelDemandes {

    public static final String CONTENT_TYPE = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
    static final String TEMPLATE = "export/Template_QlickFlow_GesFond_extraction.xlsx";
    static final DateTimeFormatter HORODATAGE = DateTimeFormatter.ofPattern("yyyyMMdd-HHmmss");
    static final DateTimeFormatter DATE_DEMANDE = DateTimeFormatter.ofPattern("dd/MM/yyyy");

    static final int ONGLET_MAD = 0, ONGLET_DEVISE = 1, ONGLET_GAB = 2;
    static final int LIGNE_GROUPES = 0, LIGNE_ENTETE = 1, PREMIERE_LIGNE = 2;
    static final int NB_COLONNES_COMMUNES = 9;
    /** Feuille GAB Hors Sites : en-têtes en ligne 3, première donnée en ligne 4 ; 12 colonnes d'identité puis 4 cassettes (M-P). */
    static final int GAB_PREMIERE_LIGNE = 3, GAB_NB_COLONNES_IDENTITE = 12, GAB_PREMIERE_CASSETTE = 12;

    /** Formats Excel français : milliers séparés par un espace, décimale virgule. */
    static final String FORMAT_MONTANT = "# ##0,00";
    static final String FORMAT_QUANTITE = "# ##0";

    public static String nomDeFichier(LocalDateTime instant) {
        return "QlickFlow_GesFond_extraction_" + HORODATAGE.format(instant) + ".xlsx";
    }

    public byte[] generer(ExportDemandesDTO export) {
        try (InputStream template = new ClassPathResource(TEMPLATE).getInputStream();
             Workbook classeur = new XSSFWorkbook(template);
             ByteArrayOutputStream sortie = new ByteArrayOutputStream()) {

            classeur.setForceFormulaRecalculation(true);   // Excel recalcule V / Y à l'ouverture

            List<CashRequestDTO> demandes = new ArrayList<>(export.demandes());
            demandes.sort(Comparator.comparing(CashRequestDTO::getUniqueRequestRef, Comparator.nullsLast(Comparator.naturalOrder())));
            // Répartition par canal : AGENCE → feuilles MAD / Devise, MONETIQUE → feuille GAB Hors Sites (même export, mêmes filtres)
            List<CashRequestDTO> agence = demandes.stream().filter(d -> !new ExtractionDemande(d).estMonetique()).toList();
            List<CashRequestDTO> monetique = demandes.stream().filter(d -> new ExtractionDemande(d).estMonetique()).toList();

            Styles styles = new Styles(classeur);
            remplirMad(classeur.getSheetAt(ONGLET_MAD), agence, export.faceValuesByCode(), styles);
            remplirDevise(classeur.getSheetAt(ONGLET_DEVISE), agence, styles);
            remplirGab(classeur.getSheetAt(ONGLET_GAB), monetique, styles);

            classeur.write(sortie);
            return sortie.toByteArray();
        } catch (IOException e) {
            throw new UncheckedIOException("Génération de l'extraction Excel impossible", e);
        }
    }

    // ------------------------------------------------------------------ feuille MAD : une ligne par demande

    private void remplirMad(Sheet onglet, List<CashRequestDTO> demandes, Map<String, BigDecimal> faceValues, Styles styles) {
        ColonnesMad colonnes = ColonnesMad.lire(onglet);
        Row modele = modeleDeLigne(onglet);
        int ligne = PREMIERE_LIGNE;
        for (CashRequestDTO d : demandes) {
            ExtractionDemande x = new ExtractionDemande(d);
            if (!x.figureEnMad()) continue;
            Row r = ligne(onglet, ligne++, modele);
            ecrireColonnesCommunes(r, x, styles);
            Map<BigDecimal, Integer> caisse = x.quantitesParFaceValue(ExtractionDemande.Alimentation.CAISSE, faceValues);
            Map<BigDecimal, Integer> gab = x.quantitesParFaceValue(ExtractionDemande.Alimentation.GAB, faceValues);
            colonnes.caisse.forEach((col, fv) -> quantite(r, col, caisse.get(fv), styles));
            colonnes.gab.forEach((col, fv) -> quantite(r, col, gab.get(fv), styles));
            montant(r, colonnes.ramassageMad, x.ramassageMad(), styles);
        }
    }

    /** Colonnes de la feuille MAD lues dans le template : coupures caisse (J-U), coupures GAB (W-X), ramassage (Z). */
    private record ColonnesMad(Map<Integer, BigDecimal> caisse, Map<Integer, BigDecimal> gab, int ramassageMad) {
        static ColonnesMad lire(Sheet onglet) {
            Row groupes = onglet.getRow(LIGNE_GROUPES);
            Row entete = onglet.getRow(LIGNE_ENTETE);
            int debutGab = -1, debutRamassage = -1;
            for (int c = NB_COLONNES_COMMUNES; c < groupes.getLastCellNum(); c++) {
                String libelle = texteDe(groupes.getCell(c)).toLowerCase();
                if (debutGab < 0 && libelle.contains("gab")) debutGab = c;
                if (debutRamassage < 0 && libelle.contains("ramassage")) debutRamassage = c;
            }
            if (debutGab < 0 || debutRamassage < 0) {
                throw new IllegalStateException("Template Excel : groupes « Alimentation GAB » / « Ramassage » introuvables en ligne 1");
            }
            Map<Integer, BigDecimal> caisse = new HashMap<>(), gab = new HashMap<>();
            for (int c = NB_COLONNES_COMMUNES; c < debutRamassage; c++) {
                Cell cellule = entete.getCell(c);
                if (cellule == null || cellule.getCellType() != CellType.NUMERIC) continue;   // « Total Alim. … » = texte → ignoré
                BigDecimal fv = BigDecimal.valueOf(cellule.getNumericCellValue()).stripTrailingZeros();
                (c < debutGab ? caisse : gab).put(c, fv);
            }
            return new ColonnesMad(caisse, gab, debutRamassage);
        }
    }

    // ------------------------------------------------------------------ feuille Devise : une ligne par devise

    private void remplirDevise(Sheet onglet, List<CashRequestDTO> demandes, Styles styles) {
        Row modele = modeleDeLigne(onglet);
        int ligne = PREMIERE_LIGNE;
        for (CashRequestDTO d : demandes) {
            ExtractionDemande x = new ExtractionDemande(d);
            for (ExtractionDemande.LigneDevise ld : x.lignesDevise()) {
                Row r = ligne(onglet, ligne++, modele);
                ecrireColonnesCommunes(r, x, styles);
                if (ld.alimentation() != null) {
                    texte(r, NB_COLONNES_COMMUNES, ld.devise());
                    montant(r, NB_COLONNES_COMMUNES + 1, ld.alimentation(), styles);
                }
                if (ld.ramassage() != null) {
                    texte(r, NB_COLONNES_COMMUNES + 2, ld.devise());
                    montant(r, NB_COLONNES_COMMUNES + 3, ld.ramassage(), styles);
                }
            }
        }
    }

    // ------------------------------------------------------------------ feuille GAB Hors Sites : une ligne par GAB (AER_LM-297)

    private void remplirGab(Sheet onglet, List<CashRequestDTO> demandes, Styles styles) {
        Row modele = onglet.getRow(GAB_PREMIERE_LIGNE);
        if (modele == null) modele = onglet.createRow(GAB_PREMIERE_LIGNE);
        int ligne = GAB_PREMIERE_LIGNE;
        for (CashRequestDTO d : demandes) {
            ExtractionDemande x = new ExtractionDemande(d);
            for (ExtractionDemande.LigneGab lg : x.lignesGab()) {
                Row r = ligne(onglet, ligne++, modele);
                OperationGabDTO gab = lg.gab();
                texte(r, 0, x.reference());
                texte(r, 1, x.dateDemande() == null ? "" : DATE_DEMANDE.format(x.dateDemande()));
                texte(r, 2, gab.getAgenceCode() != null ? gab.getAgenceCode() : x.codeAgence());
                texte(r, 3, gab.getAgenceLibelle() != null ? gab.getAgenceLibelle() : x.nomAgence());
                texte(r, 4, gab.getVille());
                texte(r, 5, gab.getTransporteur() != null ? gab.getTransporteur() : x.transporteur());
                montant(r, 6, gab.getTarifTransporteur() != null ? gab.getTarifTransporteur() : x.coutDeplacement(), styles);
                texte(r, 7, gab.getGabId());
                texte(r, 8, gab.getGabNom());
                texte(r, 9, gab.getNumeroCompte());
                texte(r, 10, gab.getGabType() == null ? "" : libelleTypeGab(gab.getGabType().name()));
                texte(r, 11, x.libelleStatut());
                for (int i = 0; i < ExtractionDemande.NB_CASSETTES; i++) {
                    quantite(r, GAB_PREMIERE_CASSETTE + i, lg.quantitesCassettes()[i], styles);
                }
            }
        }
    }

    /** Libellé Jira : « Hors site » / « Drive ». */
    private static String libelleTypeGab(String code) {
        return switch (code) {
            case "HORS_SITE" -> "Hors site";
            case "DRIVE" -> "Drive";
            default -> code;
        };
    }

    // ------------------------------------------------------------------ écriture

    private void ecrireColonnesCommunes(Row r, ExtractionDemande x, Styles styles) {
        texte(r, 0, x.reference());
        texte(r, 1, x.urgente() ? "Oui" : "Non");
        texte(r, 2, x.codeAgence());
        texte(r, 3, x.nomAgence());
        texte(r, 4, x.zone());
        texte(r, 5, x.transporteur());
        montant(r, 6, x.coutDeplacement(), styles);
        texte(r, 7, x.dateDemande() == null ? "" : DATE_DEMANDE.format(x.dateDemande()));
        texte(r, 8, x.libelleStatut());
    }

    private static void quantite(Row r, int col, Integer quantite, Styles styles) {
        if (quantite == null || quantite <= 0) return;
        Cell c = r.getCell(col, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
        c.setCellValue(quantite);
        c.setCellStyle(styles.quantite(c.getCellStyle()));
    }

    private static void montant(Row r, int col, BigDecimal valeur, Styles styles) {
        Cell c = r.getCell(col, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
        if (valeur == null) { c.setBlank(); return; }
        c.setCellValue(valeur.doubleValue());
        c.setCellStyle(styles.montant(c.getCellStyle()));
    }

    private static void texte(Row r, int col, String valeur) {
        Cell c = r.getCell(col, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
        c.setCellValue(valeur == null ? "" : valeur);
    }

    private static String texteDe(Cell c) {
        return c != null && c.getCellType() == CellType.STRING ? c.getStringCellValue() : "";
    }

    /** La ligne d'exemple du template (ligne 3) sert de modèle de styles et de formules, puis est réécrite par la première demande
     *  (ses cellules sont vidées par ligne(), y compris le « 1 » de T3). */
    private static Row modeleDeLigne(Sheet onglet) {
        Row r = onglet.getRow(PREMIERE_LIGNE);
        return r != null ? r : onglet.createRow(PREMIERE_LIGNE);
    }

    /**
     * Prépare une ligne : styles du modèle sur toutes les cellules, formules du modèle (totaux V / Y) recopiées
     * avec le numéro de ligne courant, autres cellules vidées.
     */
    private static Row ligne(Sheet onglet, int index, Row modele) {
        Row r = onglet.getRow(index);
        if (r == null) r = onglet.createRow(index);
        int ligneModele = modele.getRowNum() + 1, ligneCible = index + 1;
        for (int c = 0; c < modele.getLastCellNum(); c++) {
            Cell source = modele.getCell(c);
            Cell cible = r.getCell(c, Row.MissingCellPolicy.CREATE_NULL_AS_BLANK);
            if (source != null && source.getCellType() == CellType.FORMULA) {
                // décale les références relatives (M4 → M7) ; les absolues ($M$3 : valeurs faciales de la feuille GAB) restent intactes
                cible.setCellFormula(source.getCellFormula().replaceAll("(?<=[A-Z])(?<!\\$[A-Z])" + ligneModele + "(?![0-9])", String.valueOf(ligneCible)));
            } else {
                cible.setBlank();
            }
            CellStyle style = source != null ? source.getCellStyle() : null;
            if (style != null) cible.setCellStyle(style);
        }
        return r;
    }


    // ------------------------------------------------------------------ styles (format français), un par style source

    private static final class Styles {
        private final Workbook classeur;
        private final DataFormat formats;
        private final Map<CellStyle, CellStyle> montants = new HashMap<>(), quantites = new HashMap<>();

        Styles(Workbook classeur) { this.classeur = classeur; this.formats = classeur.createDataFormat(); }

        CellStyle montant(CellStyle base) { return derive(montants, base, FORMAT_MONTANT); }
        CellStyle quantite(CellStyle base) { return derive(quantites, base, FORMAT_QUANTITE); }

        private CellStyle derive(Map<CellStyle, CellStyle> cache, CellStyle base, String format) {
            return cache.computeIfAbsent(base, b -> {
                CellStyle s = classeur.createCellStyle();
                if (b != null) s.cloneStyleFrom(b);
                s.setDataFormat(formats.getFormat(format));
                return s;
            });
        }
    }
