0. Règle retenue

Une carte = une plateforme (enum Platform d'utilisateurs-ci, 5 valeurs : CONSOLE_ADMIN, DTRP, ML_CHAB, DC, GESTION_FONDS). La carte est visible si l'utilisateur détient au moins une permission de la plateforme. Les permissions de chaque plateforme viennent du référentiel existant GET /api/v1/public/users/habilitations/platforms/active (PlatformDTO.permissions[].code), le test « l'utilisateur les a-t-il ? » est l'existant AuthService.hasPermission(codes) (OU sur extractUserPermissions).

Pourquoi pas une liste de codes en dur par carte (DTRP_CHOIX_WF, GDF_CONSULTER_DEMANDE…) : le Pôle monétique n'a pas GDF_CONSULTER_DEMANDE et doit voir la carte ; les codes ML_*/DC_* appartiennent aux autres squads et bougent ; et tout nouveau droit demanderait une livraison du portail. Avec la plateforme, rien n'est créé dans utilisateurs-ci et un profil qui doit voir Mainlevée se règle dans la console Droits (rattacher un rôle ML_CHAB).

Dégradé : référentiel injoignable → cartes affichées (le back reste la barrière ; un dashboard vide sur un incident serait pire). Un seul appel par chargement, mutualisé entre les 4 cartes.

import { HttpClient } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { firstValueFrom } from 'rxjs';
import { AuthService } from '../../auth/auth.service';
import { ApiResponse } from '../../shared/interfaces/derogation.interface';
import { PlatformDTO } from '../../shared/interfaces/habilitation.interface';
import { ApiConfigService } from './api-config.service';

/** Identifiants des plateformes QlickFlow (utilisateurs-ci : enum Platform, colonne hab_roles.platform). */
export const PLATEFORME = {
  DEROGATION: 'DTRP',
  MAINLEVEE: 'ML_CHAB',
  DROIT_COMMUNICATION: 'DC',
  GESTION_FONDS: 'GESTION_FONDS'
} as const;
export type Plateforme = typeof PLATEFORME[keyof typeof PLATEFORME];

/**
 * AER_LM-355 — accès aux modules du dashboard par permission, sans liste de codes en dur côté portail :
 * le référentiel des plateformes (GET /habilitations/platforms/active) donne les permissions de chaque plateforme,
 * et l'utilisateur accède à un module s'il détient AU MOINS UNE de ses permissions (AuthService.hasPermission).
 * - un nouveau droit ajouté dans utilisateurs-ci est pris en compte sans livraison du portail ;
 * - le Super Admin (permissions DTRP/GDF posées sur un rôle CONSOLE_ADMIN) et le Pôle monétique
 *   (GDF_TRAITER_DEMANDE_MONETIQUE seul) sont couverts, là où un test « rôle de plateforme X » les exclurait.
 * Le référentiel est chargé une fois par session (promesse mutualisée). En cas d'échec : les cartes restent visibles
 * (le back reste la barrière, une panne du référentiel ne doit pas vider le dashboard).
 */
@Injectable({ providedIn: 'root' })
export class PlateformesService {
  private permissionsParPlateforme: Promise<Map<string, string[]>> | null = null;

  constructor(private readonly http: HttpClient,
              private readonly apiConfig: ApiConfigService,
              private readonly authService: AuthService) {}

  /** true si l'utilisateur courant détient au moins une permission de la plateforme (true aussi si le référentiel est injoignable). */
  async utilisateurAccede(plateforme: Plateforme): Promise<boolean> {
    const referentiel = await this.charger();
    if (!referentiel) { return true; }
    const codes = referentiel.get(plateforme) ?? [];
    return codes.length > 0 && this.authService.hasPermission(codes);
  }

  private charger(): Promise<Map<string, string[]> | null> {
    return this.permissionsParPlateforme ??= firstValueFrom(
      this.http.get<ApiResponse<PlatformDTO[]>>(`${this.apiConfig.buildConsoleApiUrl()}/habilitations/platforms/active`))
      .then(reponse => new Map((reponse.data ?? []).map(p => [p.id, (p.permissions ?? []).map(x => x.code)])))
      .catch(err => {
        console.error('[dashboard] référentiel des plateformes injoignable — cartes affichées sans filtrage', err);
        this.permissionsParPlateforme = null;   // nouvel essai au prochain appel
        return null;
      });
  }
}

import { HttpClientTestingModule, HttpTestingController } from '@angular/common/http/testing';
import { TestBed } from '@angular/core/testing';
import { AuthService } from '../../auth/auth.service';
import { ApiConfigService } from './api-config.service';
import { PLATEFORME, PlateformesService } from './plateformes.service';

describe('PlateformesService', () => {
  let service: PlateformesService;
  let http: HttpTestingController;
  const permissionsUtilisateur = ['DTRP_CHOIX_WF', 'GDF_TRAITER_DEMANDE_MONETIQUE'];

  const referentiel = { success: true, message: null, error: null, data: [
    { id: 'DTRP', nom: 'Dérogation', permissions: [{ code: 'DTRP_CHOIX_WF' }, { code: 'DTRP_SUBMIT' }] },
    { id: 'ML_CHAB', nom: 'Mainlevée', permissions: [{ code: 'ML_CHAB_CONSULTER' }] },
    { id: 'GESTION_FONDS', nom: 'Gestion des fonds', permissions: [{ code: 'GDF_CONSULTER_DEMANDE' }, { code: 'GDF_TRAITER_DEMANDE_MONETIQUE' }] }
  ] };

  beforeEach(() => {
    TestBed.configureTestingModule({
      imports: [HttpClientTestingModule],
      providers: [
        { provide: ApiConfigService, useValue: { buildConsoleApiUrl: () => 'http://gw/api/v1/public/users' } },
        { provide: AuthService, useValue: { hasPermission: (codes: string[]) => codes.some(c => permissionsUtilisateur.includes(c)) } }
      ]
    });
    service = TestBed.inject(PlateformesService);
    http = TestBed.inject(HttpTestingController);
  });

  afterEach(() => http.verify());

  it('accède à une plateforme dont il détient une permission, pas aux autres', async () => {
    const derog = service.utilisateurAccede(PLATEFORME.DEROGATION);
    http.expectOne('http://gw/api/v1/public/users/habilitations/platforms/active').flush(referentiel);

    expect(await derog).toBeTrue();
    expect(await service.utilisateurAccede(PLATEFORME.GESTION_FONDS)).toBeTrue();     // monétique : via GDF_TRAITER_DEMANDE_MONETIQUE
    expect(await service.utilisateurAccede(PLATEFORME.MAINLEVEE)).toBeFalse();
    expect(await service.utilisateurAccede(PLATEFORME.DROIT_COMMUNICATION)).toBeFalse();   // plateforme absente du référentiel
  });

  it('ne charge le référentiel qu\'une fois', async () => {
    const a = service.utilisateurAccede(PLATEFORME.DEROGATION);
    const b = service.utilisateurAccede(PLATEFORME.MAINLEVEE);
    http.expectOne('http://gw/api/v1/public/users/habilitations/platforms/active').flush(referentiel);
    await Promise.all([a, b]);
  });

  it('laisse les cartes visibles si le référentiel est injoignable', async () => {
    spyOn(console, 'error');
    const p = service.utilisateurAccede(PLATEFORME.MAINLEVEE);
    http.expectOne('http://gw/api/v1/public/users/habilitations/platforms/active').flush('', { status: 500, statusText: 'KO' });
    expect(await p).toBeTrue();
  });
});

import { Directive, Input, OnInit, TemplateRef, ViewContainerRef } from '@angular/core';
import { Plateforme, PlateformesService } from '../../core/services/plateformes.service';

/**
 * AER_LM-355 — <div *appSiPlateforme="'DTRP'"> : le bloc n'est rendu que si l'utilisateur détient au moins une permission
 * de la plateforme. Structurelle (comme *ngIf), standalone : à ajouter aux imports du NgModule qui déclare le composant.
 */
@Directive({ selector: '[appSiPlateforme]', standalone: true })
export class SiPlateformeDirective implements OnInit {
  @Input({ alias: 'appSiPlateforme', required: true }) plateforme!: Plateforme;

  constructor(private readonly gabarit: TemplateRef<unknown>,
              private readonly conteneur: ViewContainerRef,
              private readonly plateformes: PlateformesService) {}

  async ngOnInit(): Promise<void> {
    const visible = await this.plateformes.utilisateurAccede(this.plateforme);
    this.conteneur.clear();
    if (visible) { this.conteneur.createEmbeddedView(this.gabarit); }
  }
}

FICHIER NOUVEAUqf_admin — permissions par plateforme
recette\355_plateformes.sql
-- qf_admin : ce que le dashboard va utiliser (contrôle avant recette)
set search_path to qf_admin;
select r.platform, rp.permission, count(distinct pr.profile_id) as nb_profils
from hab_roles r
join hab_role_permission rp on rp.role_id = r.id
left join hab_profile_role pr on pr.role_id = r.id
group by r.platform, rp.permission order by 1, 2;
-- Toute permission ML_* / DC_* portée par un rôle de plateforme ML_CHAB / DC donne accès à la carte correspondante.
-- Un profil sans aucune permission d'une plateforme ne voit pas sa carte : si un profil métier doit voir Mainlevée
-- ou Droits de com, on lui rattache un rôle de cette plateforme (console Droits) — aucun code à ajouter dans le portail.
