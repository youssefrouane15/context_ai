Trois permissions sur la plateforme GESTION_FONDS : GDF_CONSULTER_GAB, GDF_GERER_GAB, GDF_TRAITER_DEMANDE_MONETIQUE. Le nouveau rôle reçoit aussi GDF_CONSULTER_DEMANDE et GDF_TOUTES_AGENCES : un profil sans unité d'organisation ne verrait aucune demande sinon (même construction que le rôle 11). Le Super Admin reçoit consultation + gestion pour que h11111 déroule l'écran console en recette (précédent V17) ; à retirer si le PO veut réserver la création au pôle monétique.

Comme en V16 : aucun CHECK sur hab_role_permission.permission, l'enum Java est la seule source de vérité ; le rôle et le profil sont insérés avec max(id) + 1 et référencés par name / code. Le code GDF_TRAITER_DEMANDE_MONETIQUE est à reprendre tel quel par le collègue côté fund-management.

-- V23 — AER_LM-300 : habilitations GAB hors site (plateforme GESTION_FONDS).
-- Permissions (enum Java Permission, seule source de vérité — pas de CHECK) : GDF_CONSULTER_GAB, GDF_GERER_GAB, GDF_TRAITER_DEMANDE_MONETIQUE.
-- Nouveau rôle « GDF Opérations monétique » + nouveau profil « Pôle opérations monétique » (code OPERATIONS_MONETIQUE).
-- Rôles existants complétés : 11 GDF Caisse Centrale (consultation GAB + traitement monétique), 12 GDF Support (consultation GAB),
-- 8 Super Admin (consultation + gestion GAB : h11111 peut dérouler l'écran console en recette, même logique que V17).
-- Idempotent ; ids des nouveautés = max + 1, références par name / code.

-- 1. Rôle
insert into hab_roles (id, name, description, actif, created_at, updated_at, platform)
select (select coalesce(max(id), 0) + 1 from hab_roles),
       'GDF Opérations monétique', 'Gestion des GAB hors site et traitement des demandes monétique', true, now(), now(), 'GESTION_FONDS'
where not exists (select 1 from hab_roles where name = 'GDF Opérations monétique');

insert into hab_role_permission (role_id, permission)
select r.id, p.permission
from hab_roles r,
     (values ('GDF_CONSULTER_DEMANDE'), ('GDF_TOUTES_AGENCES'),
             ('GDF_CONSULTER_GAB'), ('GDF_GERER_GAB'), ('GDF_TRAITER_DEMANDE_MONETIQUE')) as p (permission)
where r.name = 'GDF Opérations monétique'
  and not exists (select 1 from hab_role_permission x where x.role_id = r.id and x.permission = p.permission);

-- 2. Rôles existants
insert into hab_role_permission (role_id, permission)
select v.role_id, v.permission
from (values (11, 'GDF_CONSULTER_GAB'), (11, 'GDF_TRAITER_DEMANDE_MONETIQUE'),   -- GDF Caisse Centrale
             (12, 'GDF_CONSULTER_GAB'),                                          -- GDF Support
             (8,  'GDF_CONSULTER_GAB'), (8,  'GDF_GERER_GAB')) as v (role_id, permission)  -- Super Admin (recette)
where exists (select 1 from hab_roles r where r.id = v.role_id)
  and not exists (select 1 from hab_role_permission x where x.role_id = v.role_id and x.permission = v.permission);

-- 3. Profil
insert into hab_profiles (id, profile_type, name, code, description, actif, created_at, updated_at)
select (select coalesce(max(id), 0) + 1 from hab_profiles),
       'STANDARD_USER', 'Pôle opérations monétique', 'OPERATIONS_MONETIQUE', 'Administrateur monétique — GAB hors site', true, now(), now()
where not exists (select 1 from hab_profiles where code = 'OPERATIONS_MONETIQUE');

insert into hab_profile_role (profile_id, role_id)
select p.id, r.id
from hab_profiles p, hab_roles r
where p.code = 'OPERATIONS_MONETIQUE' and r.name = 'GDF Opérations monétique'
  and not exists (select 1 from hab_profile_role x where x.profile_id = p.id and x.role_id = r.id);

-- 4. Séquences (incrément 50 : les remettre au-delà du dernier id)
select setval('hab_roles_seq',    greatest((select max(id) from hab_roles),    (select last_value from hab_roles_seq)));
select setval('hab_profiles_seq', greatest((select max(id) from hab_profiles), (select last_value from hab_profiles_seq)));

-- Rôle et profil créés, liaisons posées
select r.id, r.name, r.platform, string_agg(p.permission, ', ' order by p.permission) as permissions
from hab_roles r left join hab_role_permission p on p.role_id = r.id
where r.id in (8, 11, 12) or r.name = 'GDF Opérations monétique'
group by r.id, r.name, r.platform order by r.id;

select p.id, p.code, p.name, r.name as role
from hab_profiles p join hab_profile_role pr on pr.profile_id = p.id join hab_roles r on r.id = pr.role_id
where p.code = 'OPERATIONS_MONETIQUE';
