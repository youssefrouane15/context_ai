0. Analyse — ce que change l'architecture

Aujourd'hui (277 / 330 / 333)Avec notifications-ci
fund-management rend le HTML (GabaritEmailDemande + notification-demande.html), calcule l'objet, et envoie via NotificationPort → mock (fichier) ou SMTP (JavaMailSender). fund-management envoie POST /api/v1/notifications/email {to, cc, replyTo, templateCode, variables} ; l'objet et le HTML sont dans notifications-ci (NotificationTemplate + Thymeleaf, logo cid et portalUrl injectés par eux). 202 = accepté, 400 = template inconnu, 502 = SMTP KO.
Décisions :
  1. Aucun changement de contrat chez eux — leur enum est fait pour ça (« ajouter une valeur + un HTML, aucun changement côté appelants »). 5 entrées GDF_* (une par EvenementDemande, l'objet est fixe par entrée) qui pointent sur un seul template mail/gestion-fonds-demande. La référence et l'agence ne sont plus dans l'objet (contrat : objet fixe) ; elles sont dans le texte d'aperçu et le bandeau. Si le PO veut la référence dans l'objet, c'est une évolution de SendEmailRequest (champ subject optionnel) à demander à Mamadou — pas dans ce lot.
  2. EmailNotification garde son nom et devient (destinataire, codeTemplate, variables) : NotificationPort et ses 15 usages ne bougent pas. Un envoi par destinataire (corps personnalisé « Bonjour X »), cc vide.
  3. GabaritEmailDemande → VariablesEmailDemande : même logique métier (totaux, devises, déclassement), sortie = Map<String,String>. Les lignes devises/ramassage restent des <tr> échappées champ par champ (injection th:utext côté template) : une seule structure de variables, compatible avec un Map<String,String> comme avec un Map<String,Object> côté SendEmailRequest.
  4. Mode http (défaut) = NotificationHttpAdapter, RestClient comme UtilisateursDirecteurAdapter, URL interne directe (comme utilisateurs-ci). Mode mock conservé = journal en mémoire seulement (le NotificationJournalPort et son écran de recette restent). SMTP et spring-boot-starter-mail supprimés.
  5. Tolérance aux pannes inchangée : l'échec d'envoi est loggé, la demande n'est jamais impactée (catch RuntimeException existant dans le service).
À confirmer avec Yassine (non bloquant) : en DMZR, fund-management joint notifications-ci en direct (URL interne, comme /internal/users) ou via Apigee ? Si Apigee : ajouter .requestInterceptor(apigeeBearerInterceptor) sur le builder de l'adapter, rien d'autre.

/**
 * Variables du template « mail/gestion-fonds-demande » de notifications-ci (remplace GabaritEmailDemande : plus aucun
 * HTML rendu ici, sauf les lignes de tableau devises / ramassage, échappées champ par champ et injectées en th:utext).
 * Toutes les valeurs sont des chaînes non nulles : le template fait ses th:if sur « vide / non vide ».
 */
@Component
public class VariablesEmailDemande {

    private static final DateTimeFormatter DATE_HEURE = DateTimeFormatter.ofPattern("dd/MM/yyyy 'à' HH:mm");
    private static final DecimalFormat MONTANT = new DecimalFormat("#,##0.00", DecimalFormatSymbols.getInstance(Locale.FRANCE));
    private static final DateTimeFormatter DATE_EFFET = DateTimeFormatter.ofPattern("dd/MM/yyyy");
    private static final String STYLE_TH = "padding:7px 8px; background-color:#f6f8f7; border:1px solid #e8ecea; color:#5c6b63; font-weight:bold;";
    private static final String STYLE_TD = "padding:7px 8px; border:1px solid #e8ecea; text-align:right; font-family:Consolas, monospace; white-space:nowrap;";

    public Map<String, String> construire(CashRequestDTO d, EvenementDemande evenement, String nomDestinataire) {
        List<CashOperationDTO> ops = d.getOperations() == null ? List.of() : d.getOperations();
        boolean urgente = Boolean.TRUE.equals(d.getUrgent())
                || ops.stream().anyMatch(op -> Boolean.TRUE.equals(op.getIsUrgent()));
        LocalDateTime date = d.getUpdatedAt() != null ? d.getUpdatedAt() : d.getCreatedAt();

        Map<String, String> v = new LinkedHashMap<>();
        v.put("evenement", evenement.getLibelle());
        v.put("verbe", evenement.getVerbe());
        v.put("destinataire", vide(nomDestinataire));
        v.put("reference", vide(d.getUniqueRequestRef()));
        v.put("agence", vide(d.getBranchCode()) + (d.getAgenceNom() != null ? " - " + d.getAgenceNom() : ""));
        v.put("zone", d.getZoneNom() != null ? d.getZoneNom() : vide(d.getZoneCode()));
        v.put("transporteur", d.getPrestataireNom() != null ? d.getPrestataireNom() : vide(d.getPrestataire()));
        v.put("date", date != null ? DATE_HEURE.format(date) : "—");
        v.put("dateEffet", d.getDateEffet() != null ? DATE_EFFET.format(d.getDateEffet()) : "—");
        v.put("initiateur", d.getAgentNom() != null ? d.getAgentNom() + (d.getAgentProfil() != null ? " (" + d.getAgentProfil() + ")" : "") : "—");
        v.put("urgent", urgente ? "OUI" : "NON");
        v.put("urgence", urgente ? "OUI" : (d.getUrgenceDeclasseeAt() != null ? "Non — déclassée par la Caisse Centrale" : "Non"));
        v.put("totalCaisse", montant(total(ops, OperationType.CASH_DESK, RequestType.SUPPLY)));
        v.put("totalGab", montant(total(ops, OperationType.ATM, RequestType.SUPPLY)));
        v.put("lignesDevises", lignes(ops, RequestType.SUPPLY));
        v.put("lignesRamassage", lignes(ops, RequestType.COLLECTION));
        v.put("commentaire", d.getComment() != null && !d.getComment().isBlank() ? d.getComment() : "—");
        v.put("motifAnnulation", evenement == EvenementDemande.ANNULATION ? vide(d.getCancellationReason()) : "");
        v.put("declassePar", evenement == EvenementDemande.DECLASSEMENT_URGENCE
                ? vide(d.getUrgenceDeclasseePar())
                  + (d.getUrgenceDeclasseeProfil() != null ? " (" + d.getUrgenceDeclasseeProfil() + ")" : "")
                  + (d.getUrgenceDeclasseeAt() != null ? " le " + DATE_HEURE.format(d.getUrgenceDeclasseeAt()) : "")
                : "");
        v.put("declasseMotif", evenement == EvenementDemande.DECLASSEMENT_URGENCE ? vide(d.getUrgenceDeclasseeMotif()) : "");
        return v;
    }

    private static BigDecimal total(List<CashOperationDTO> ops, OperationType type, RequestType sens) {
        BigDecimal somme = BigDecimal.ZERO;
        for (CashOperationDTO op : ops) {
            if (op.getOperationType() != type || op.getOperationSens() != sens) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                if (detail.getTotalAmountByCurrency() != null) somme = somme.add(detail.getTotalAmountByCurrency());
            }
        }
        return somme;
    }

    /** Une ligne &lt;tr&gt; par devise pour CURRENCY / sens donné (SUPPLY = commande, COLLECTION = ramassage) ; styles inline (email). */
    private static String lignes(List<CashOperationDTO> ops, RequestType sens) {
        StringBuilder sb = new StringBuilder();
        for (CashOperationDTO op : ops) {
            if (op.getOperationType() != OperationType.CURRENCY || op.getOperationSens() != sens) continue;
            for (OperationCurrencyDetailDTO detail : details(op)) {
                String devise = detail.getCurrency() == null ? "—" : vide(detail.getCurrency().getIsoCode());
                sb.append("<tr><td width=\"170\" style=\"").append(STYLE_TH).append("\">").append(echapper(devise))
                  .append("</td><td style=\"").append(STYLE_TD).append("\">")
                  .append(montant(detail.getTotalAmountByCurrency())).append(' ').append(echapper(devise)).append("</td></tr>");
            }
        }
        return sb.length() == 0
                ? "<tr><td colspan=\"2\" style=\"padding:7px 8px; border:1px solid #e8ecea; color:#8a9790; font-style:italic;\">Aucune</td></tr>"
                : sb.toString();
    }

    private static List<OperationCurrencyDetailDTO> details(CashOperationDTO op) {
        return op.getCurrencyDetails() == null ? List.of() : op.getCurrencyDetails();
    }

    private static String montant(BigDecimal m) { return MONTANT.format(m == null ? BigDecimal.ZERO : m); }
    private static String vide(String s) { return s == null ? "" : s; }

package com.bmci.workflow.fund.management.external.notification.http;

import com.bmci.q360.domain.retail.valueobjects.notification.EmailNotification;
import com.bmci.workflow.fund.management.service.domain.retail.ports.output.notification.NotificationPort;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;
import org.springframework.web.client.RestClientResponseException;

import java.util.List;
import java.util.Map;

/**
 * Envoi via le microservice notifications-ci : POST /api/v1/notifications/email, 202 attendu.
 * Le template (objet + HTML) vit côté notifications-ci (NotificationTemplate.GDF_*).
 * Appel direct (URL interne, comme utilisateurs-ci) ; si l'infra impose Apigee, ajouter l'ApigeeBearerInterceptor au builder.
 */
@Component
@ConditionalOnProperty(name = "gdf.notification.mode", havingValue = "http")
@Slf4j
public class NotificationHttpAdapter implements NotificationPort {

    static final String CHEMIN = "/api/v1/notifications/email";

    /** Corps attendu par notifications-ci (SendEmailRequest) : mêmes noms de champs. */
    record EnvoiEmailRequete(String to, List<String> cc, String replyTo, String templateCode, Map<String, String> variables) {}

    /** Réponse de notifications-ci (NotificationResponse). */
    record EnvoiEmailReponse(boolean success, String message) {}

    private final RestClient client;

    public NotificationHttpAdapter(RestClient.Builder builder, @Value("${qf.notifications.base-url}") String baseUrl) {
        this.client = builder.baseUrl(baseUrl).build();
    }

    @Override
    public void envoyer(EmailNotification email) {
        EnvoiEmailRequete requete = new EnvoiEmailRequete(email.destinataire(), List.of(), null, email.codeTemplate(), email.variables());
        try {
            EnvoiEmailReponse reponse = client.post()
                    .uri(CHEMIN)
                    .contentType(MediaType.APPLICATION_JSON)
                    .body(requete)
                    .retrieve()
                    .body(EnvoiEmailReponse.class);
            if (reponse != null && !reponse.success()) {
                throw new IllegalStateException("notifications-ci a refusé l'envoi : " + reponse.message());
            }
            log.info("[notification-http] {} → {} : acceptée", email.codeTemplate(), email.destinataire());
        } catch (RestClientResponseException e) {
            // 400 = template inconnu ou requête invalide, 502 = SMTP en panne côté notifications-ci, 500 = autre
            throw new IllegalStateException("notifications-ci " + e.getStatusCode().value() + " : " + e.getResponseBodyAsString(), e);
        }
    }
}
