Méthode publique (appelée par le template). Pas d'autre changement.

  /** Timeline : le jour d'un évènement n'est affiché que s'il diffère de celui de l'évènement précédent. */
  memeJour(a: Date | string | null | undefined, b: Date | string | null | undefined): boolean {
    if (!a || !b) { return false; }
    const da = new Date(a), db = new Date(b);
    return da.getFullYear() === db.getFullYear() && da.getMonth() === db.getMonth() && da.getDate() === db.getDate();
  }
