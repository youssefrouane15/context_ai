    @Test
    @DisplayName("Extraction GAB Hors Site : demandes MONETIQUE seulement, une ligne par GAB, Vidage seul en M, cassettes N-Q, formule R")
    void feuilleGab() throws Exception {
        CashOperationDTO drive = operationGab("130092026", "CASA OASIS DRIVE", TypeGab.DRIVE, cassette(1, "BILLET_100", 50));
        drive.getGab().setVidage(true);
        CashRequestDTO gab = demandeGab("20261028-9000-01",
                operationGab("130120017", "CASA VAL D'ANFA", TypeGab.HORS_SITE,
                        cassette(1, "BILLET_100", 500), cassette(2, "BILLET_200", 300), cassette(4, "BILLET_100", 100)),
                drive);
        CashRequestDTO agence = demande("20261028-1001-01", operation(OperationType.CASH_DESK, RequestType.SUPPLY, detail("MAD", "200", coupure("BILLET_200", 1))));

        Sheet feuilleGab = lire(export.genererGabHorsSite(List.of(gab, agence))).getSheetAt(0);

        Row r1 = feuilleGab.getRow(ExportExcelDemandes.GAB_PREMIERE_LIGNE), r2 = feuilleGab.getRow(ExportExcelDemandes.GAB_PREMIERE_LIGNE + 1);
        assertThat(texte(r1, 0)).isEqualTo("20261028-9000-01");
        assertThat(texte(r1, 7)).isEqualTo("130092026");                  // tri par ID GAB
        assertThat(texte(r1, 10)).isEqualTo("Drive");
        assertThat(texte(r1, 12)).isEqualTo("Oui");                        // M = Vidage seul
        assertThat(nombre(r1, 13)).isEqualTo(50);                          // N = cassette 1
        assertThat(texte(r2, 7)).isEqualTo("130120017");
        assertThat(texte(r2, 10)).isEqualTo("Hors site");
        assertThat(texte(r2, 12)).isEqualTo("Non");                        // vidage non renseigné
        assertThat(nombre(r2, 6)).isEqualTo(450);                          // PU = tarif transporteur du GAB
        assertThat(nombre(r2, 13)).isEqualTo(500);
        assertThat(nombre(r2, 14)).isEqualTo(300);
        assertThat(r2.getCell(15) == null || r2.getCell(15).getCellType() == CellType.BLANK).isTrue();   // cassette 3 absente
        assertThat(nombre(r2, 16)).isEqualTo(100);
        assertThat(r2.getCell(17).getCellType()).isEqualTo(CellType.FORMULA);
        assertThat(r2.getCell(17).getCellFormula()).contains("N5").contains("$N$3");

        Row suivante = feuilleGab.getRow(ExportExcelDemandes.GAB_PREMIERE_LIGNE + 2);   // la demande agence n'apparaît pas
        assertThat(suivante == null || suivante.getCell(0) == null || suivante.getCell(0).getCellType() == CellType.BLANK).isTrue();
    }
