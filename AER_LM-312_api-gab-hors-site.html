/**
 * @author Youssef ROUANE
 * @date 14/10/2026
 */
@Component
@RequiredArgsConstructor
public class GabHorsSiteMapper {

    private final OrganizationService organizationService;
    private final VilleMapper villeMapper;
    private final FundProviderMapper prestataireMapper;

    public GabHorsSiteEntity toEntity(CreateGabHorsSiteDTO dto) {
        if (dto == null) {
            return null;
        }
        return GabHorsSiteEntity.builder()
                .idGab(dto.getIdGab().trim())
                .nom(dto.getNom().trim())
                .numeroCompte(dto.getNumeroCompte().trim())
                .agence(resolveAgence(dto.getAgenceId()))
                .ville(dto.getVille())
                .transporteur(dto.getTransporteur() != null ? dto.getTransporteur() : FundProvider.BRINKS)
                .prixUnitaire(dto.getPrixUnitaire() != null ? dto.getPrixUnitaire() : BigDecimal.ZERO)
                .type(dto.getType() != null ? dto.getType() : TypeGab.H)
                .etat(dto.getEtat() != null ? dto.getEtat() : EtatGab.EN_LIGNE)
                .build();
    }

    /** Mise à jour : un champ null dans le DTO laisse la valeur en place. */
    public void toEntity(GabHorsSiteEntity entity, UpdateGabHorsSiteDTO dto) {
        if (dto.getIdGab() != null)        { entity.setIdGab(dto.getIdGab().trim()); }
        if (dto.getNom() != null)          { entity.setNom(dto.getNom().trim()); }
        if (dto.getNumeroCompte() != null) { entity.setNumeroCompte(dto.getNumeroCompte().trim()); }
        if (dto.getAgenceId() != null)     { entity.setAgence(resolveAgence(dto.getAgenceId())); }
        if (dto.getVille() != null)        { entity.setVille(dto.getVille()); }
        if (dto.getTransporteur() != null) { entity.setTransporteur(dto.getTransporteur()); }
        if (dto.getPrixUnitaire() != null) { entity.setPrixUnitaire(dto.getPrixUnitaire()); }
        if (dto.getType() != null)         { entity.setType(dto.getType()); }
        if (dto.getEtat() != null)         { entity.setEtat(dto.getEtat()); }
    }

    public GabHorsSiteDTO toDto(GabHorsSiteEntity gab) {
        if (gab == null) {
            return null;
        }
        OrganizationUnitEntity agence = gab.getAgence();
        return GabHorsSiteDTO.builder()
                .id(gab.getId())
                .idGab(gab.getIdGab())
                .nom(gab.getNom())
                .numeroCompte(gab.getNumeroCompte())
                .agence(agence == null ? null : SimpleAgenceDTO.builder()
                        .id(agence.getId()).code(agence.getCode()).nom(agence.getName()).build())
                .ville(villeMapper.toDto(gab.getVille()))
                .transporteur(prestataireMapper.toDto(gab.getTransporteur()))
                .prixUnitaire(gab.getPrixUnitaire())
                .type(gab.getType())
                .etat(gab.getEtat())
                .build();
    }

    public List<GabHorsSiteDTO> toDtoList(List<GabHorsSiteEntity> gabs) {
        if (gabs == null) {
            return null;
        }
        List<GabHorsSiteDTO> list = new ArrayList<>(gabs.size());
        for (var gab : gabs) {
            list.add(toDto(gab));
        }
        return list;
    }

    /** Résout l'agence de rattachement par id ; refuse une unité inconnue ou qui n'est pas une agence (même garde que le groupe dans AgenceMapper). */
    private OrganizationUnitEntity resolveAgence(Long agenceId) {
        var agence = organizationService.findById(agenceId).orElseThrow(
                () -> new IllegalArgumentException("Agence not found, id : " + agenceId));
        if (agence.getType() != OrganizationUnitType.AGENCE) {
            throw new IllegalArgumentException("agenceId is not an Orga unit of type AGENCE, agenceId:" + agenceId);
        }
        return agence;
    }
}

package com.bnpp.irb.qlickflow.admin.facade.impl;

@Component
@RequiredArgsConstructor
public class GabHorsSiteFacadeImpl implements GabHorsSiteFacade {

    private final GabHorsSiteService gabHorsSiteService;
    private final GabHorsSiteMapper gabHorsSiteMapper;
    private final VilleMapper villeMapper;

    @Override
    public Optional<GabHorsSiteDTO> findById(Long id) {
        return gabHorsSiteService.findById(id).map(gabHorsSiteMapper::toDto);
    }

    @Override
    public List<GabHorsSiteDTO> findAll() {
        return gabHorsSiteMapper.toDtoList(gabHorsSiteService.findAll());
    }

    @Override
    public Page<GabHorsSiteDTO> findAll(Pageable pageable) {
        return gabHorsSiteService.findAll(pageable).map(gabHorsSiteMapper::toDto);
    }

    @Override
    public List<GabHorsSiteDTO> findAllByAgenceId(Long agenceId) {
        return gabHorsSiteMapper.toDtoList(gabHorsSiteService.findAllByAgenceId(agenceId));
    }

    @Override
    public List<VilleDTO> findAllVilles() {
        return villeMapper.toDtoList();
    }

    @Override
    @Transactional
    public GabHorsSiteDTO createGabHorsSite(CreateGabHorsSiteDTO dto) {
        // Unicité de l'identifiant GAB
        if (gabHorsSiteService.existsByIdGabIgnoreCase(dto.getIdGab().trim())) {
            throw new DuplicateException("Un GAB hors site avec l'identifiant '" + dto.getIdGab() + "' existe déjà.");
        }
        var entity = gabHorsSiteMapper.toEntity(dto);
        var saved = gabHorsSiteService.save(entity);
        return gabHorsSiteMapper.toDto(saved);
    }

    @Override
    @Transactional
    public void updateGabHorsSite(UpdateGabHorsSiteDTO dto) {
        // Vérifier que le GAB existe
        var gab = gabHorsSiteService.findById(dto.getId())
                .orElseThrow(() -> new EntityNotFoundException("GabHorsSite not found with id: " + dto.getId()));

        // Unicité de l'identifiant GAB sur un autre GAB
        if (dto.getIdGab() != null && gabHorsSiteService.existsByIdGabIgnoreCaseAndIdNot(dto.getIdGab().trim(), dto.getId())) {
            throw new DuplicateException("Un GAB hors site avec l'identifiant '" + dto.getIdGab() + "' existe déjà.");
        }

        gabHorsSiteMapper.toEntity(gab, dto);
        gabHorsSiteService.save(gab);
    }
}
