 *    obligatoire
 *  - totalAmount RESTAURE (la colonne total_amount est NOT NULL en base ;

 */
@AllArgsConstructor
@NoArgsConstructor
@Getter
@Setter
public class CashOperation extends BaseEntity<CashOperationId> {

    private OperationType operationType;
    private Integer institutionCode;
    private String cashDeskNumber;
    private UrgencyFlag isUrgent;
    private String targetCashDeskNumber;
    private RequestType operationSens;
    private BigDecimal totalAmount;
    private String operationStatus;
    private String establishment;
    private LocalDateTime createdAt;
    private LocalDateTime updatedAt;
    private List<OperationCurrencyDetail> currencyDetails = new ArrayList<>();

    public void validateAndComputeTotal(Map<String, BigDecimal> faceValuesByCode) {
        if (operationSens == null) {
            throw new RetailValidationException("Le sens de l'opération (SUPPLY/COLLECTION) est obligatoire");
        }
        // Defauts serveur — garantis par le domaine quel que soit le chemin d'entree
        if (isUrgent == null) isUrgent = UrgencyFlag.NO;
        if (operationStatus == null) operationStatus = "INITIATED";
        if (institutionCode == null) institutionCode = 1;
        if (cashDeskNumber == null) cashDeskNumber = "001";
        if (currencyDetails == null) currencyDetails = new ArrayList<>();

        // Regle retail conservee : alimenter une caisse exige la ventilation
        if (operationSens == RequestType.SUPPLY && operationType == OperationType.CASH_DESK
                && currencyDetails.isEmpty()) {
            throw new RetailValidationException(
                    "Les détails des coupures sont obligatoires pour alimenter un GAB ou une caisse");
        }

        BigDecimal computedTotal = BigDecimal.ZERO;
        for (OperationCurrencyDetail detail : currencyDetails) {
            BigDecimal detailTotal = BigDecimal.ZERO;
            if (detail.getOperationDenominations() == null) {
                detail.setOperationDenominations(new ArrayList<>());
            }
            for (OperationDenomination line : detail.getOperationDenominations()) {
                if (line.getDenomination() == null || line.getDenomination().getDenominationCode() == null) {
                    throw new RetailValidationException("Chaque ligne de ventilation doit référencer une coupure par son code");
                }
                String code = line.getDenomination().getDenominationCode();
                BigDecimal faceValue = faceValuesByCode.get(code);
                if (faceValue == null) {
                    throw new RetailValidationException("La coupure '" + code + "' n'existe pas dans le référentiel");
                }
                int quantity = line.getQuantity() == null ? 0 : line.getQuantity();
                if (quantity < 0) {
                    throw new RetailValidationException("La quantité de la coupure '" + code + "' ne peut pas être négative");
                }
                detailTotal = detailTotal.add(faceValue.multiply(BigDecimal.valueOf(quantity)));
            }
            // Montant par devise : fourni => controle ; absent => calcule
            if (detail.getTotalAmountByCurrency() != null
                    && detail.getTotalAmountByCurrency().compareTo(detailTotal) != 0) {
                throw new RetailValidationException("Le montant par devise fourni ("
                        + detail.getTotalAmountByCurrency() + ") ne correspond pas au calculé (" + detailTotal + ")");
            }
            detail.setTotalAmountByCurrency(detailTotal);
            computedTotal = computedTotal.add(detailTotal);
        }
        // Montant global : fourni => controle ; absent => calcule (RESTAURE)
        if (totalAmount != null && totalAmount.compareTo(computedTotal) != 0) {
            throw new RetailValidationException("Le montant total fourni (" + totalAmount
                    + ") ne correspond pas au calculé (" + computedTotal + ")");
        }
        totalAmount = computedTotal;
    }
}
