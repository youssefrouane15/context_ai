import { Injectable } from "@angular/core";
import { catchError, firstValueFrom, map, of, throwError } from "rxjs";
import { FundService } from "../../domain/domain-application-services/output/fundManagement.service";
import {
  ApiResponse,
  DemandeFonds,
  DemandeFondsPayload,
  Denomination,
  Devise,
  ErreurApi
} from "../../domain/domain-core/entities/operation.model";

@Injectable({ providedIn: 'root' })
export class FundHttpService implements FundService {

  private readonly apiUrl = `http://localhost:8300/cash`;

  constructor(private http: HttpClient) {}

  // ===== Référentiels (mocks de secours tant que le backend n'expose pas d'endpoint) =====

  private readonly mockDevises: Devise[] = [
    { code: 'MAD', libelle: 'Dirham marocain', symbole: 'DH' },
    { code: 'EUR', libelle: 'Euro', symbole: '€' },
    { code: 'USD', libelle: 'Dollar US', symbole: '$' },

  private readonly mockDenominationsGab: Denomination[] = [
    { id: 1, valeur: 200, libelle: '200' },
    { id: 2, valeur: 100, libelle: '100' },
  ];

  private readonly mockDenominationsCaisse: Denomination[] = [
    { id: 1,  valeur: 200,  libelle: '200' },
    { id: 2,  valeur: 100,  libelle: '100' },
    { id: 3,  valeur: 50,   libelle: '50' },
    { id: 4,  valeur: 20,   libelle: '20' },
    { id: 5,  valeur: 10,   libelle: '10' },
    { id: 6,  valeur: 5,    libelle: '5' },
    { id: 7,  valeur: 2,    libelle: '2' },
    { id: 8,  valeur: 1,    libelle: '1' },
    { id: 9,  valeur: 0.5,  libelle: '0.5' },
    { id: 10, valeur: 0.2,  libelle: '0.2' },
    { id: 11, valeur: 0.1,  libelle: '0.1' },
    { id: 12, valeur: 0.05, libelle: '0.05' },
  ];

  getDevises(): Promise<Devise[]> {
    return firstValueFrom(
      this.http.get<Devise[]>(this.apiUrl + '/devises').pipe(catchError(() => of(this.mockDevises)))
    );
  }

  getDenominationsGab(): Promise<Denomination[]> {
    return firstValueFrom(
      this.http.get<Denomination[]>(this.apiUrl + '/operation').pipe(catchError(() => of(this.mockDenominationsGab)))
    );
  }

  getDenominationsCaisse(): Promise<Denomination[]> {
    return firstValueFrom(
      this.http.get<Denomination[]>(this.apiUrl + '/operation').pipe(catchError(() => of(this.mockDenominationsCaisse)))
    );
  }

  // ===== Création — POST /cash → 201 { success, data: CashRequestDTO } =====

  creerDemande(payload: DemandeFondsPayload): Promise<DemandeFonds> {
    return firstValueFrom(
      this.http.post<ApiResponse<DemandeFonds>>(this.apiUrl, payload).pipe(
        map(reponse => {
          if (!reponse.success || !reponse.data) {
            throw new ErreurApi(reponse.error ?? 'ERR', reponse.message ?? 'Réponse invalide du serveur.');
          }
          return reponse.data;
        }),
        catchError(err => throwError(() => this.versErreurApi(err)))
      )
    );
  }

  // ===== Liste — GET /cash → 200 { success, data: CashRequestDTO[] } =====

  getDemandes(): Promise<DemandeFonds[]> {
    return firstValueFrom(
      this.http.get<ApiResponse<DemandeFonds[]>>(this.apiUrl).pipe(
        map(reponse => reponse.data ?? []),
        catchError(err => throwError(() => this.versErreurApi(err)))
      )
    );
  }

  // ===== Export CSV 100% front (inchangé) =====

  telechargerDemande(d: DemandeFonds): void {
    const csv = this.genererCsvDemande(d);
    const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' });
    const url = window.URL.createObjectURL(blob);
    const lien = document.createElement('a');
    lien.href = url;
    lien.download = `demande-fonds-${d.id}.csv`;
    lien.click();
    window.URL.revokeObjectURL(url);
  }

  private genererCsvDemande(d: DemandeFonds): string {
    const operations = d.operations ?? [];
    const types = [...new Set(
      operations.map(op => op.operationType === 'CURRENCY' ? 'DEVISE' : op.operationType)
    )].join(', ');
    const lignes = [
      ['Référence', 'Statut', "Types d'opérations", "Nombre d'opérations"].join(';'),
      [
        this.champCsv(d.id),
        this.champCsv(this.libelleStatutExport(d.requestStatus)),
        this.champCsv(types),
        this.champCsv(operations.length)
      ].join(';')
    ];
    return lignes.join('\r\n');
  }

  private champCsv(valeur: unknown): string {
    if (valeur === null || valeur === undefined) { return ''; }
    const s = String(valeur);
    return /[;"\r\n]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s;
  }

  private libelleStatutExport(s: string): string {
    const map: Record<string, string> = {
      INITIATED: 'En Attente', VALIDATED: 'Validé', CREATED: 'Validé', CANCELLED: 'Annulée'
    };
    return map[s] ?? s;
  }

  // ===== Normalisation des erreurs HTTP en ErreurApi =====

  private versErreurApi(err: unknown): ErreurApi {
    if (err instanceof ErreurApi) { return err; }
    if (err instanceof HttpErrorResponse) {
      const corps = err.error as ApiResponse<Record<string, string>> | null;
      if (corps && typeof corps === 'object' && 'success' in corps) {
        return new ErreurApi(corps.error ?? String(err.status), corps.message ?? err.message, corps.data);
      }
      if (err.status === 0) {
        return new ErreurApi('0', 'Serveur injoignable. Vérifiez votre connexion ou le VPN.');
      }
      return new ErreurApi(String(err.status), err.message);
    }
    return new ErreurApi('ERR', 'Erreur inattendue.');
  }
}

import { Injectable } from "@angular/core";
import { FunManagmentStore } from "./fundMangement.store";
import { FundHttpService } from "../apiAdapters/fundManagement-http.service";
import { FundService } from "../../domain/domain-application-services/output/fundManagement.service";
import { DemandeFonds, DemandeFondsPayload, ErreurApi } from "../../domain/domain-core/entities/operation.model";

@Injectable({ providedIn: 'root' })
export class FundManagementActions {

  private readonly serviceHttp: FundService;

  constructor(private store: FunManagmentStore, serviceHttp: FundHttpService) {
    this.serviceHttp = serviceHttp;   // typé sur le port, injecté par l'implémentation
  }

  async loadDevises() {
    this.store.loading.set(true);
    try {
      this.store.devises.set(await this.serviceHttp.getDevises());
    } catch (e) {
      this.store.error.set("Impossible de charger les devises.");
    } finally {
      this.store.loading.set(false);
    }
  }

  async loadCoupures() {
    this.store.loading.set(true);
    try {
      const caisse = await this.serviceHttp.getDenominationsCaisse();
      const gab = await this.serviceHttp.getDenominationsGab();
      this.store.coupuresGab.set(gab);
      this.store.coupuresCaisse.set(caisse);
    } catch (e) {
      this.store.error.set("Impossible de charger les coupures.");
    } finally {
      this.store.loading.set(false);
    }
  }

  /** Création d'une demande. Réinitialise l'état de création avant l'appel
      pour que le composant réagisse à CETTE soumission et pas à la précédente. */
  async creerDemande(payload: DemandeFondsPayload): Promise<void> {
    this.store.loading.set(true);
    this.store.demandeCreee.set(null);
    this.store.erreurCreation.set(null);
    try {
      const creee: DemandeFonds = await this.serviceHttp.creerDemande(payload);
      this.store.demandeCreee.set(creee);
    } catch (e) {
      this.store.erreurCreation.set(
        e instanceof ErreurApi ? e : new ErreurApi('ERR', "Erreur lors de l'envoi de la demande.")
      );
    } finally {
      this.store.loading.set(false);
    }
  }

  async loadDemandes() {
    this.store.loadingDemandes.set(true);
    this.store.errorDemandes.set(null);
    try {
      this.store.demandes.set(await this.serviceHttp.getDemandes());
    } catch (e) {
      this.store.errorDemandes.set(
        e instanceof ErreurApi ? e.message : "Impossible de charger les demandes."
      );
    } finally {
      this.store.loadingDemandes.set(false);
    }
  }

  telechargerDemande(d: DemandeFonds): void {
    this.serviceHttp.telechargerDemande(d);
  }
}

        operationType: 'CURRENCY',
        operationSens: 'COLLECTION',

        cashDeskNumber: CONTEXTE_AGENCE.cashDeskNumber,

          currency: { isoCode },
          totalAmountByCurrency: total,
          operationDenominations: []
        }))
      });
    }

    return {
      branchCode: CONTEXTE_AGENCE.branchCode,
      user: CONTEXTE_AGENCE.user,
      establishment: CONTEXTE_AGENCE.establishment,
      operations
    };
  }
