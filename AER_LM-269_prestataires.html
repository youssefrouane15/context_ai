1. Contexte et décisions

AER_LM-269 devait créer un référentiel prestataires avec « espace de stockage et API ». Le tech lead a tranché : pas de table, une enum Java exposée via API. Le ticket se réduit donc à une enum et un endpoint de listing — la modélisation est le clone exact du trio Platform / PlatformService / PlatformMapper / PlatformDTO déjà en place pour les habilitations.

Ce que ce ticket livre
  • Enum Prestataire avec libellé (BRINKS → « Brink's », G4S → « G4S »).
  • Service (EnumUtils.isValidEnum, comme PlatformServiceImpl), mapper (id = name()), facade, contrôleur.
  • Aucun script Flyway. La colonne orga_unit.prestataire qui stocke la valeur est livrée par AER_LM-271.
À signaler au tech leadLe ticket vaut 8 points au Jira. Avec cette décision, il vaut 1 point et n’a plus de partie BDD. Le CA « Listing / Consultation / Création / Mise à jour » n’est plus atteignable pour Création et Mise à jour : ajouter un prestataire = modifier l’enum (et, après 271, la contrainte CHECK SQL). C’est le compromis accepté ; il vaut mieux qu’il soit écrit dans le ticket.
Pourquoi id et pas code dans le DTOPour rester identique à PlatformDTO (id = platform.name(), nom = libellé). Le front 259 enverra prestataire: "BRINKS" dans l’agence, c’est-à-dire l’id reçu ici.

package com.bnpp.irb.qlickflow.admin.service.referentiel.impl;

import com.bnpp.irb.qlickflow.admin.entities.referentiel.Prestataire;
import com.bnpp.irb.qlickflow.admin.service.referentiel.PrestataireService;
import lombok.RequiredArgsConstructor;
import org.apache.commons.lang3.EnumUtils;
import org.springframework.stereotype.Service;

import java.util.Arrays;
import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
public class PrestataireServiceImpl implements PrestataireService {

    @Override
    public Optional<Prestataire> findByCode(String code) {
        if (EnumUtils.isValidEnum(Prestataire.class, code)) {
            return Optional.of(Prestataire.valueOf(code));
        }
        return Optional.empty();
    }

    @Override
    public List<Prestataire> findAll() {
        return Arrays.stream(Prestataire.values()).toList();
    }
}

Mapper

Différence volontaire avec PlatformMapper : toDto(null) renvoie null, parce que AgenceMapper (271) l’appellera sur des agences sans prestataire.

FICHIER COMPLETC:\dev\utilisateurs-ci\src\main\java\com\bnpp\irb\qlickflow\admin\mapper\PrestataireMapper.java
package com.bnpp.irb.qlickflow.admin.mapper;

import com.bnpp.irb.qlickflow.admin.dto.referentiel.prestataire.PrestataireDTO;
import com.bnpp.irb.qlickflow.admin.entities.referentiel.Prestataire;
import org.springframework.stereotype.Component;

import java.util.List;

@Component
public class PrestataireMapper {

    public PrestataireDTO toDto(Prestataire prestataire) {
        if (prestataire == null) {
            return null;
        }
        return PrestataireDTO.builder()
                .id(prestataire.name())
                .nom(prestataire.getLibelle())
                .build();
    }

    public List<PrestataireDTO> toDtoList(List<Prestataire> prestataires) {
        if (prestataires == null) {
            return null;
        }
        return prestataires.stream().map(this::toDto).toList();
    }
}

package com.bnpp.irb.qlickflow.admin.facade.impl;

import com.bnpp.irb.qlickflow.admin.dto.referentiel.prestataire.PrestataireDTO;
import com.bnpp.irb.qlickflow.admin.facade.PrestataireFacade;
import com.bnpp.irb.qlickflow.admin.mapper.PrestataireMapper;
import com.bnpp.irb.qlickflow.admin.service.referentiel.PrestataireService;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Optional;

@Component
@RequiredArgsConstructor
public class PrestataireFacadeImpl implements PrestataireFacade {

    private final PrestataireService prestataireService;
    private final PrestataireMapper prestataireMapper;

    @Override
    public Optional<PrestataireDTO> findByCode(String code) {
        return prestataireService.findByCode(code)
                .map(prestataireMapper::toDto);
    }

    @Override
    public List<PrestataireDTO> findAll() {
        return prestataireMapper.toDtoList(prestataireService.findAll());
    }
}

Contrôleur

PlatformController n’a pas de @PreAuthorize ; ici on met le même que les contrôleurs orga, le référentiel étant consommé par l’écran admin agence.

FICHIER COMPLETC:\dev\utilisateurs-ci\src\main\java\com\bnpp\irb\qlickflow\admin\controller\publicapi\referentiel\PrestataireController.java
package com.bnpp.irb.qlickflow.admin.controller.publicapi.referentiel;

import com.bnpp.irb.qlickflow.admin.common.ApiResponse;
import com.bnpp.irb.qlickflow.admin.dto.referentiel.prestataire.PrestataireDTO;
import com.bnpp.irb.qlickflow.admin.facade.PrestataireFacade;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.util.List;
import java.util.Optional;

@RestController
@RequestMapping("/api/v1/public/users/referentiels/prestataires")
@PreAuthorize("hasAnyAuthority('CONSOLE_ADMIN_EDIT_USER', 'CONSOLE_ADMIN_CREATE_USER')")
@RequiredArgsConstructor
@Slf4j
public class PrestataireController {

    private final PrestataireFacade prestataireFacade;

    /** Liste complète des prestataires : alimente la liste déroulante de la fiche agence (AER_LM-259). */
    @GetMapping
    public ResponseEntity<ApiResponse<List<PrestataireDTO>>> getAllPrestataires() {
        List<PrestataireDTO> prestataires = prestataireFacade.findAll();
        return ResponseEntity.ok(ApiResponse.success(prestataires));
    }

    @GetMapping("/{code}")
    public ResponseEntity<ApiResponse<PrestataireDTO>> getPrestataireByCode(@PathVariable String code) {
        Optional<PrestataireDTO> prestataire = prestataireFacade.findByCode(code);
        return prestataire.map(p -> ResponseEntity.ok(ApiResponse.success(p)))
                .orElseGet(() -> ResponseEntity.notFound().build());
    }
}
