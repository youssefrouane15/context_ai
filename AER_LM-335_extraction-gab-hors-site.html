    public List<CashRequest> rechercherTousMonetique(DemandeMonetiqueSearchCriteria criteria) {
        List<CashRequestEntity> entites = cashDemandJpaRepository.findAll(
                DemandeMonetiqueSpecifications.of(criteria), Sort.by(Sort.Direction.DESC, "creationDate"));
        if (entites.isEmpty()) {
            return List.of();
        }
        List<UUID> ids = entites.stream().map(CashRequestEntity::getId).toList();
        Map<UUID, CashRequestEntity> arbre = cashDemandJpaRepository.findAllWithTreeByIdIn(ids).stream()
                .collect(Collectors.toMap(CashRequestEntity::getId, Function.identity()));
        return ids.stream().map(arbre::get).map(mapper::toDomain).toList();
    }

    static final String TEMPLATE_GAB_HORS_SITE = "export/Template_QlickFlow_GesFond_GAB-Hors-Site.xlsx";
    static final DateTimeFormatter JOUR = DateTimeFormatter.ofPattern("yyyyMMdd");

    /** Nom imposé par la Jira : QlickFlow_GesFond_GAB-Hors-Site_AAAAMMJJ.xlsx (sans heure). */
    public static String nomDeFichierGabHorsSite(LocalDate jour) {
        return "QlickFlow_GesFond_GAB-Hors-Site_" + JOUR.format(jour) + ".xlsx";
    }

    /**
     * AER_LM-335 : même périmètre que le listing Monétique (filtres), une ligne par GAB.
     * Le template n'a qu'une feuille, structurée comme la feuille GAB de l'extraction Caisse Centrale
     * (en-têtes ligne 3, ligne 4 = modèle de styles / formule « Total Alim. GAB ») : remplirGab est réutilisé tel quel.
     */
    public byte[] genererGabHorsSite(List<CashRequestDTO> demandes) {
        try (InputStream template = new ClassPathResource(TEMPLATE_GAB_HORS_SITE).getInputStream();
             Workbook classeur = new XSSFWorkbook(template);
             ByteArrayOutputStream sortie = new ByteArrayOutputStream()) {

            classeur.setForceFormulaRecalculation(true);

            List<CashRequestDTO> monetique = new ArrayList<>(demandes.stream()
                    .filter(d -> new ExtractionDemande(d).estMonetique()).toList());
            monetique.sort(Comparator.comparing(CashRequestDTO::getUniqueRequestRef, Comparator.nullsLast(Comparator.naturalOrder())));

            Sheet onglet = classeur.getSheetAt(0);
            if (monetique.isEmpty()) {
                viderLigneModele(onglet);   // aucune demande : on ne laisse pas la ligne d'exemple du template
            } else {
                remplirGab(onglet, monetique, new Styles(classeur));
            }

            classeur.write(sortie);
            return sortie.toByteArray();
        } catch (IOException e) {
            throw new UncheckedIOException("Génération de l'extraction GAB Hors Site impossible", e);
        }
    }

    /** Vide la ligne d'exemple (styles et formule conservés) — ligne() appliquée sur elle-même. */
    private static void viderLigneModele(Sheet onglet) {
        Row modele = onglet.getRow(GAB_PREMIERE_LIGNE);
        if (modele != null) ligne(onglet, GAB_PREMIERE_LIGNE, modele);
    }

function telechargerFichier(contenu: Blob, nom: string): void {
  const url = URL.createObjectURL(contenu);
  const lien = document.createElement('a');
  lien.href = url;
  lien.download = nom;
  lien.click();
  URL.revokeObjectURL(url);
}
